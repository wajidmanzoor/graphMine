184 ag = new bliss::Graph(vertices.size());

186 for(unsigned i = 0; i < number_vertices; ++i)
187 ag->change_color(i, (unsigned)vertices[i + 1]);
188 //read edges
189 assert(qp.get_size() > 1);
190 for(unsigned index = 1; index < qp.get_size(); ++index) {
191 auto element = qp.at(index);
192 vidType from, to;

194 to = element.get_vid();
195 //std::cout << "Adding edge: " << from << " --> " << to << "\n";
196 ag->add_edge(from - 1, to - 1, std::make_pair((unsigned)element.get_his(), index));
197 }

199 const unsigned * cl = ag->canonical_form(stats, &report_aut, stdout); // canonical labeling. This is expensive.
