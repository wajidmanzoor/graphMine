139bool Pattern::is_connected(vidType u, vidType v) const {
140 if (get_degree(u) < get_degree(v)) std::swap(u, v);
141 int begin = 0;
142 int end = get_degree(v)-1;
143 while (begin <= end) {
144 int mid = begin + ((end - begin) >> 1);
145 auto w = get_neighbor(v, mid);
146 if (w == u) return true;
147 else if (w > u) end = mid - 1;
148 else begin = mid + 1;
149 }
150 return false;
151}

195void Pattern::computeKCore() {
196 int nv = size();
197 int md = get_max_degree();
198 std::vector<int> vertices(nv); // Vertices sorted by degree.
199 std::vector<int> position(nv); // The position of vertices in vertices array.
200 std::vector<int> degree_bin(md+1, 0); // Degree from 0 to max_degree.
201 std::vector<int> offset(md+1); // The offset in vertices array according to degree.
202 for (int i = 0; i < nv; ++i) {
203 int degree = get_degree(i);
204 core_table[i] = degree;
205 degree_bin[degree] += 1;
206 }
207 int start = 0;
208 for (int i = 0; i < md+1; ++i) {
209 offset[i] = start;
210 start += degree_bin[i];
211 }
212 for (int i = 0; i < nv; ++i) {
213 int degree = get_degree(i);
214 position[i] = offset[degree];
215 vertices[position[i]] = i;
216 offset[degree] += 1;
217 }
218 for (int i = md; i > 0; --i) {
219 offset[i] = offset[i - 1];
220 }
221 offset[0] = 0;
222 for (int i = 0; i < nv; ++i) {
223 int v = vertices[i];
224 for(int j = 0; j < get_degree(v); ++j) {
225 int u = get_neighbor(v, j);
226 if (core_table[u] > core_table[v]) {
227 // Get the position and vertex which is with the same degree
228 // and at the start position of vertices array.
229 int cur_degree_u = core_table[u];
230 int position_u = position[u];
231 int position_w = offset[cur_degree_u];
232 int w = vertices[position_w];
233 if (u != w) {
234 // Swap u and w.
235 position[u] = position_w;
236 position[w] = position_u;
237 vertices[position_u] = w;
238 vertices[position_w] = u;
239 }
240 offset[cur_degree_u] += 1;
241 core_table[u] -= 1;
242 }
243 }
244 }
