404// Certificate building
405void cert_add(const unsigned int v1, const unsigned int v2, const unsigned int v3) {
406 if(refine_compare_certificate) {
407 if(refine_equal_to_first) {
408 /* So far equivalent to the first path... */
409 unsigned int index = certificate_current_path.size();
410 if(index >= refine_first_path_subcertificate_end) {
411 refine_equal_to_first = false;
412 } else if(certificate_first_path[index] != v1) {
413 refine_equal_to_first = false;
414 } else if(certificate_first_path[++index] != v2) {
415 refine_equal_to_first = false;
416 } else if(certificate_first_path[++index] != v3) {
417 refine_equal_to_first = false;
418 } if(opt_use_failure_recording and !refine_equal_to_first) {
419 /* We just became different from the first path,
420 * remember the deviation point tree-specific invariant
421 * for the use of failure recording */
422 UintSeqHash h;
423 h.update(v1);
424 h.update(v2);
425 h.update(v3);
426 h.update(index);
427 h.update(eqref_hash.get_value());
428 failure_recording_fp_deviation = h.get_value();
429 }
430 }
431 if(refine_cmp_to_best == 0) {
432 /* So far equivalent to the current best path... */
433 unsigned int index = certificate_current_path.size();
434 if(index >= refine_best_path_subcertificate_end) {
435 refine_cmp_to_best = 1;
436 } else if(v1 > certificate_best_path[index]) {
437 refine_cmp_to_best = 1;
438 } else if(v1 < certificate_best_path[index]) {
439 refine_cmp_to_best = -1;
440 } else if(v2 > certificate_best_path[++index]) {
441 refine_cmp_to_best = 1;
442 } else if(v2 < certificate_best_path[index]) {
443 refine_cmp_to_best = -1;
444 } else if(v3 > certificate_best_path[++index]) {
445 refine_cmp_to_best = 1;
446 } else if(v3 < certificate_best_path[index]) {
447 refine_cmp_to_best = -1;
448 }
449 }
450 if((refine_equal_to_first == false) and (refine_cmp_to_best < 0))
451 return;
452 }
453 /* Update the current path certificate */
454 certificate_current_path.push_back(v1);
455 certificate_current_path.push_back(v2);
456 certificate_current_path.push_back(v3);
457}

571void long_prune_add_automorphism(const unsigned int* aut) {
572 if(long_prune_max_stored_autss == 0) return;
573 const unsigned int N = get_nof_vertices();
574 /* If the buffer of stored auts is full, remove the oldest aut */
575 if(long_prune_end - long_prune_begin == long_prune_max_stored_autss) {
576 long_prune_begin++;
577 }
578 long_prune_end++;
579 std::vector<bool>& fixed = long_prune_allocget_fixed(long_prune_end-1);
580 std::vector<bool>& mcrs = long_prune_allocget_mcrs(long_prune_end-1);
581 /* Mark nodes that are (i) fixed or (ii) minimal orbit representatives
582 * under the automorphism 'aut' */
583 for(unsigned int i = 0; i < N; i++) {
584 fixed[i] = (aut[i] == i);
585 if(long_prune_temp[i] == false) {
586 mcrs[i] = true;
587 unsigned int j = aut[i];
588 while(j != i) {
589 long_prune_temp[j] = true;
590 j = aut[j];
591 }
592 } else {
593 mcrs[i] = false;
594 }
595 /* Clear the temp array on-the-fly... */
596 long_prune_temp[i] = false;
597 }

806// Assumes that the elements in the cell are sorted according to their invariant values.
807Partition::Cell* Partition::split_cell(Partition::Cell* const original_cell) {
808 Partition::Cell* cell = original_cell;
809 const bool original_cell_was_in_splitting_queue =
810 original_cell->in_splitting_queue;
811 Partition::Cell* largest_new_cell = 0;
812
813 while(true) {
814 unsigned int* ep = elements + cell->first;
815 const unsigned int* const lp = ep + cell->length;
816 const unsigned int ival = invariant_values[*ep];
817 invariant_values[*ep] = 0;
818 element_to_cell_map[*ep] = cell;
819 in_pos[*ep] = ep;
820 ep++;
821 while(ep < lp) {
822 const unsigned int e = *ep;
823 if(invariant_values[e] != ival)
824 break;
825 invariant_values[e] = 0;
826 in_pos[e] = ep;
827 ep++;
828 element_to_cell_map[e] = cell;
829 }
830 if(ep == lp) break;
831 Partition::Cell* const new_cell = aux_split_in_two(cell, (ep - elements) - cell->first);
832 if(graph and graph->compute_eqref_hash) {
833 graph->eqref_hash.update(new_cell->first);
834 graph->eqref_hash.update(new_cell->length);
835 graph->eqref_hash.update(ival);
836 }
837 /* Add cells in splitting_queue */
838 assert(!new_cell->is_in_splitting_queue());
839 if(original_cell_was_in_splitting_queue)
840 {
841 /* In this case, all new cells are inserted in splitting_queue */
842 assert(cell->is_in_splitting_queue());
843 splitting_queue_add(new_cell);
844 }
845 else
846 {
847 /* Otherwise, we can omit one new cell from splitting_queue */
848 assert(!cell->is_in_splitting_queue());
849 if(largest_new_cell == 0) {
850 largest_new_cell = cell;
851 } else {
852 assert(!largest_new_cell->is_in_splitting_queue());
853 if(cell->length > largest_new_cell->length) {
854 splitting_queue_add(largest_new_cell);
855 largest_new_cell = cell;
856 } else {
857 splitting_queue_add(cell);
858 }
859 }
860 }
861 /* Process the rest of the cell */
862 cell = new_cell;
863 }
864
865
866 if(original_cell == cell) {
867 /* All the elements in cell had the same invariant value */
868 return cell;
869 }
870
871 /* Add cells in splitting_queue */
872 if(!original_cell_was_in_splitting_queue) {
873 /* Also consider the last new cell */
874 assert(largest_new_cell);
875 if(cell->length > largest_new_cell->length) {
876 splitting_queue_add(largest_new_cell);
877 largest_new_cell = cell;
878 } else {
879 splitting_queue_add(cell);
880 }
881 if(largest_new_cell->is_unit()) {
882 /* Needed in certificate computation */
883 splitting_queue_add(largest_new_cell);
884 }
885 }
886 return cell;
887}
