229// partition the graph g according to the cluster id of each vertex
230void PartitionedGraph::partition2D(std::vector<int> cluster_ids) {
231 // degrees of vertices in each partition; degrees[i][j] is the degree of the j-th vertex in the i-th partiton
232 std::vector<std::vector<int>> degrees(num_2D_partitions);
233 std::vector<std::vector<int>> rowptr_partitioned(num_2D_partitions); // row pointers of each partition
234 std::vector<std::vector<int>> colidx_partitioned(num_2D_partitions); // column indices of each partition
235
236 // allocate memory for CSR format
237 for (int i = 0; i < num_vertex_chunks; i++) {
238 auto num = verts_of_clusters[i].size();
239 for (int j = 0; j < num_vertex_chunks; j++) {
240 auto pid = i * num_vertex_chunks + j; // partition id
241 rowptr_partitioned[pid].resize(num+1);
242 degrees[pid].resize(num);
243 std::fill(rowptr_partitioned[pid].begin(), rowptr_partitioned[pid].end(), 0);
244 std::fill(degrees[pid].begin(), degrees[pid].end(), 0);
245 }
246 }
247
248 std::vector<int> nes_of_partitions(num_2D_partitions, 0); // number of edges |E| of each partition
249 auto nv = g->V();
250 // count the degrees and the number of edges in each partition
251 for (vidType v = 0; v < nv; v++) { // for each vertex v in g
252 auto src_cid = cluster_ids[v]; // src cluster id
253 for (auto u : g->N(v)) { // for each neighbor u of vertex v
254 auto dst_cid = cluster_ids[u]; // dst cluster id
255 int pid = src_cid * num_vertex_chunks + dst_cid; // partition id
256 nes_of_partitions[pid] ++; // increase the number of edges in this partition
257 auto r = vertex_rank_in_cluster[v]; // v is the r-th vertex in its cluster
258 degrees[pid][r]++;
259 }
260 }
261
262 // computer row pointers using the degrees
263 for (int i = 0; i < num_2D_partitions; i++) {
264 colidx_partitioned[i].resize(nes_of_partitions[i]); // allocate memory for the edges in each partition
265 for (size_t w = 0; w < degrees[i].size(); w++)
266 rowptr_partitioned[i][w+1] = rowptr_partitioned[i][w] + degrees[i][w];
267 }
268
269 // insert edges to each partition
270 std::vector<int> index(num_2D_partitions, 0);
271 for (int v = 0; v < nv; v++) {
272 auto src_cid = cluster_ids[v]; // src cluster id
273 for (auto u : g->N(v)) {
274 auto dst_cid = cluster_ids[u]; // dst cluster id
275 int pid = src_cid * num_vertex_chunks + dst_cid; // partition id
276 colidx_partitioned[pid][index[pid]++] = u;
277 }
278 }
279
280 // compute the offsets for each partition
281 std::vector<int> vlengths(num_2D_partitions, 0);
282 std::vector<int> elengths(num_2D_partitions, 0);
283 std::vector<int> voffsets(num_2D_partitions+1, 0);
284 std::vector<int> eoffsets(num_2D_partitions+1, 0);
285 for (int i = 0; i < num_2D_partitions; i ++) {
286 vlengths[i] = rowptr_partitioned[i].size();
287 elengths[i] = colidx_partitioned[i].size();
288 //std::cout << "partition " << i << " has " << vlengths[i]-1 << " source vertices and " << elengths[i] << " edges\n";
289 }
290 parallel_prefix_sum<vidType,vidType>(vlengths, voffsets.data());
291 parallel_prefix_sum<vidType,vidType>(elengths, eoffsets.data());
292
293 // write rowptr_partitioned and colidx_partitioned into file, together with their offsets
294 partitioned_file_path = g->get_inputfile_path() + '/';
295 std::cout << "writing to path: " << partitioned_file_path << "\n";
296 ofstream p_meta(partitioned_file_path+"pgraph.meta.txt");
297 assert(p_meta);
298 p_meta << voffsets[num_2D_partitions] << "\n" << eoffsets[num_2D_partitions] << "\n";
299 p_meta.close();
300 ofstream p_voffsets(partitioned_file_path+"pgraph.voffsets.bin", ios::out | ios::binary);
301 ofstream p_eoffsets(partitioned_file_path+"pgraph.eoffsets.bin", ios::out | ios::binary);
302 ofstream p_vertices(partitioned_file_path+"pgraph.vertex.bin", ios::out | ios::binary);
303 ofstream p_edges(partitioned_file_path+"pgraph.edge.bin", ios::out | ios::binary);
304 for (int i = 0; i < num_2D_partitions; i++) {
305 size_t num = rowptr_partitioned[i].size();
306 p_vertices.write((char *) &rowptr_partitioned[i][0], sizeof(int)*num);
307 num = colidx_partitioned[i].size();
308 p_edges.write((char *) &colidx_partitioned[i][0], sizeof(int)*num);
309 }
310 p_voffsets.write((char *) &voffsets[0], sizeof(int)*(num_2D_partitions+1));
311 p_eoffsets.write((char *) &eoffsets[0], sizeof(int)*(num_2D_partitions+1));
312}
