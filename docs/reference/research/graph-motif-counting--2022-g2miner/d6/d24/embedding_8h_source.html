212 void init(Graph& graph, unsigned max_size = 2, bool is_dag = false) {
213 last_level = 1;
214 max_level = max_size;
215 vid_lists.resize(max_level);
216 idx_lists.resize(max_level);
217 #ifdef ENABLE_LABEL
218 his_lists.resize(max_level);
219 #endif
220 for (auto src = 0; src < graph.num_vertices(); src ++) {
221 IndexT row_begin = graph.edge_begin(src);
222 IndexT row_end = graph.edge_end(src);
223 for (IndexT e = row_begin; e < row_end; e++) {
224 auto dst = graph.getEdgeDst(e);
225 if (is_dag || src < dst) {
226 vid_lists[1].push_back(dst);
227 idx_lists[1].push_back(src);
228 #ifdef ENABLE_LABEL
229 his_lists[1].push_back(0);
230 #endif
231 }
232 }
233 }
