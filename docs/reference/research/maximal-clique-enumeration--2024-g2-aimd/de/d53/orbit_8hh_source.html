25/** \internal
26 * \brief A class for representing orbit information.
27 *
28 * Given a set {0,...,N-1} of N elements, represent equivalence
29 * classes (that is, unordered partitions) of the elements.
30 * Supports only equivalence class merging, not splitting.
31 * Merging two classes requires time O(k), where k is the number of
32 * the elements in the smaller of the merged classes.
33 * Getting the smallest representative in a class (and thus testing
34 * whether two elements belong to the same class) is a constant time operation.
35 */
36class Orbit
37{
38 class OrbitEntry
39 {
40 public:
41 unsigned int element;
42 OrbitEntry *next;
43 unsigned int size;
44 };
45
46 OrbitEntry *orbits;
47 OrbitEntry **in_orbit;
48 unsigned int nof_elements;
49 unsigned int _nof_orbits;
50 void merge_orbits(OrbitEntry *o1, OrbitEntry *o2);
51
52public:
53 /**
54 * Create a new orbit information object.
55 * The init() function must be called next to actually initialize
56 * the object.
57 */
58 Orbit();
59 ~Orbit();
60
61 /**
62 * Initialize the orbit information to consider sets of \a N elements.
63 * It is required that \a N > 0.
64 * The orbit information is reset so that each element forms
65 * an orbit of its own.
66 * Time complexity is O(N).
67 * \sa reset()
68 */
69 void init(const unsigned int N);
70
71 /**
72 * Reset the orbits so that each element forms an orbit of its own.
73 * Time complexity is O(N).
74 */
75 void reset();
76
77 /**
78 * Merge the orbits of the elements \a e1 and \a e2.
79 * Time complexity is O(k), where k is the number of elements in
80 * the smaller of the merged orbits.
81 */
82 void merge_orbits(unsigned int e1, unsigned int e2);
83
84 /**
85 * Is the element \a e the smallest element in its orbit?
86 * Time complexity is O(1).
87 */
88 bool is_minimal_representative(unsigned int e) const;
89
90 /**
91 * Get the smallest element in the orbit of the element \a e.
92 * Time complexity is O(1).
93 */
94 unsigned int get_minimal_representative(unsigned int e) const;
95
96 /**
97 * Get the number of elements in the orbit of the element \a e.
98 * Time complexity is O(1).
99 */
100 unsigned int orbit_size(unsigned int e) const;
101
102 /**
103 * Get the number of orbits.
104 * Time complexity is O(1).
105 */
106 unsigned int nof_orbits() const {return _nof_orbits; }
107};
