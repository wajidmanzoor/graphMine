78 * Read an undirected graph from a file in the DIMACS format into a new bliss

80 * Returns 0 if an error occurred.

163/**
164 * Find a set of generators for the automorphism group of the graph.
165 * The hook function \a hook (if non-null) is called each time a new generator
166 * for the automorphism group is found.
167 * The first argument \a user_param for the hook function is
168 * the \a hook_user_param argument,
169 * the second argument \a N is the length of the automorphism (equal to
170 * bliss::bliss_get_nof_vertices(\a graph)) and
171 * the third argument \a aut is the automorphism (a bijection on {0,...,N-1}).
172 * The memory for the automorphism \a aut will be invalidated immediately
173 * after the return from the hook;
174 * if you want to use the automorphism later, you have to take a copy of it.
175 * Do not call bliss_* functions in the hook.
176 * If \a stats is non-null, then some search statistics are copied there.
177 */
178void
179bliss_find_automorphisms(BlissGraph *graph,
180 void (*hook)(void *user_param,
181 unsigned int N,
182 const unsigned int *aut),
183 void *hook_user_param,
184 BlissStats *stats);

187/**
188 * Otherwise the same as bliss_find_automorphisms() except that
189 * a canonical labeling for the graph (a bijection on {0,...,N-1}) is returned.
190 * The returned canonical labeling will remain valid only until
191 * the next call to a bliss_* function with the exception that
192 * bliss_permute() can be called without invalidating the labeling.
193 * To compute the canonical version of a graph, call this function and
194 * then bliss_permute() with the returned canonical labeling.
195 * Note that the computed canonical version may depend on the applied version
196 * of bliss.
197 */
198const unsigned int *
199bliss_find_canonical_labeling(BlissGraph *graph,
200 void (*hook)(void *user_param,
201 unsigned int N,
202 const unsigned int *aut),
203 void *hook_user_param,
204 BlissStats *stats);
