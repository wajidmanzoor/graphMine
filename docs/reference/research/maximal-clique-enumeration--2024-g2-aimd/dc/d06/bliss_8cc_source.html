61
62 fprintf(fp, "bliss version %s (compiled "__DATE__")\n", bliss::version);
63 fprintf(fp, "Copyright 2003-2015 Tommi Junttila\n");

67"\n"
68" -directed the input graph is directed\n"
69" -can compute canonical form\n"
70" -ocan=f compute canonical form and output it in file f\n"
71" -v=N set verbose level to N [N >= 0, default: 1]\n"
72" -sh=X select splitting heuristics, where X is\n"
73" f first non-singleton cell\n"
74" fl first largest non-singleton cell\n"
75" fs first smallest non-singleton cell\n"
76" fm first maximally non-trivially connected\n"

78" flm first largest maximally non-trivially connected\n"
79" non-singleton cell\n"
80" fsm first smallest maximally non-trivially connected\n"
81" non-singleton cell [default]\n"
82" -fr=X use failure recording? [X=y/n, default: y]\n"
83" -cr=X use component recursion? [X=y/n, default: y]\n"
84" -version print the version number and exit\n"
85" -help print this help and exit\n"
86 ,program_name

92static void
93parse_options(const int argc, const char** argv)
94{
95 unsigned int tmp;
96 for(int i = 1; i < argc; i++)
97 {
98 if(strcmp(argv[i], "-can") == 0)
99 opt_canonize = true;
100 else if((strncmp(argv[i], "-ocan=", 6) == 0) and (strlen(argv[i]) > 6))
101 {
102 opt_canonize = true;
103 opt_output_can_file = argv[i]+6;
104 }
105 else if(sscanf(argv[i], "-v=%u", &tmp) == 1)
106 verbose_level = tmp;
107 else if(strcmp(argv[i], "-directed") == 0)
108 opt_directed = true;
109 else if(strcmp(argv[i], "-fr=n") == 0)
110 opt_use_failure_recording = false;
111 else if(strcmp(argv[i], "-fr=y") == 0)
112 opt_use_failure_recording = true;
113 else if(strcmp(argv[i], "-cr=n") == 0)
114 opt_use_component_recursion = false;
115 else if(strcmp(argv[i], "-cr=y") == 0)
116 opt_use_component_recursion = true;
117 else if((strncmp(argv[i], "-sh=", 4) == 0) and (strlen(argv[i]) > 4))
118 {
119 opt_splitting_heuristics = argv[i]+4;
120 }
121 else if(strcmp(argv[i], "-version") == 0)
122 {
123 fprintf(stdout, "bliss version %s\n", bliss::version);
124 exit(0);
125 }
126 else if(strcmp(argv[i], "-help") == 0)
127 {
128 usage(stdout, argv[0]);
129 exit(0);
130 }
131 else if(argv[i][0] == '-')
132 {
133 fprintf(stderr, "Unknown command line argument `%s'\n", argv[i]);
134 usage(stderr, argv[0]);
135 exit(1);
136 }
137 else
138 {
139 if(infilename)
140 {
141 fprintf(stderr, "Too many file arguments\n");
142 usage(stderr, argv[0]);
143 exit(1);
144 }
145 else
146 {
147 infilename = argv[i];
148 }
149 }
150 }
151}

183int
184main(const int argc, const char** argv)
185{
186 bliss::Timer timer;
187 bliss::AbstractGraph* g = 0;
188
189 parse_options(argc, argv);
190
191 /* Parse splitting heuristics */
192 bliss::Digraph::SplittingHeuristic shs_directed = bliss::Digraph::shs_fsm;
193 bliss::Graph::SplittingHeuristic shs_undirected = bliss::Graph::shs_fsm;
194 if(opt_directed)
195 {
196 if(strcmp(opt_splitting_heuristics, "f") == 0)
197 shs_directed = bliss::Digraph::shs_f;
198 else if(strcmp(opt_splitting_heuristics, "fs") == 0)
199 shs_directed = bliss::Digraph::shs_fs;
200 else if(strcmp(opt_splitting_heuristics, "fl") == 0)
201 shs_directed = bliss::Digraph::shs_fl;
202 else if(strcmp(opt_splitting_heuristics, "fm") == 0)
203 shs_directed = bliss::Digraph::shs_fm;
204 else if(strcmp(opt_splitting_heuristics, "fsm") == 0)
205 shs_directed = bliss::Digraph::shs_fsm;
206 else if(strcmp(opt_splitting_heuristics, "flm") == 0)
207 shs_directed = bliss::Digraph::shs_flm;
208 else
209 _fatal("Illegal option -sh=%s, aborting", opt_splitting_heuristics);
210 }
211 else
212 {
213 if(strcmp(opt_splitting_heuristics, "f") == 0)
214 shs_undirected = bliss::Graph::shs_f;
215 else if(strcmp(opt_splitting_heuristics, "fs") == 0)
216 shs_undirected = bliss::Graph::shs_fs;
217 else if(strcmp(opt_splitting_heuristics, "fl") == 0)
218 shs_undirected = bliss::Graph::shs_fl;
219 else if(strcmp(opt_splitting_heuristics, "fm") == 0)
220 shs_undirected = bliss::Graph::shs_fm;
221 else if(strcmp(opt_splitting_heuristics, "fsm") == 0)
222 shs_undirected = bliss::Graph::shs_fsm;
223 else if(strcmp(opt_splitting_heuristics, "flm") == 0)
224 shs_undirected = bliss::Graph::shs_flm;
225 else
226 _fatal("Illegal option -sh=%s, aborting", opt_splitting_heuristics);
227 }
228
229 /* Open the input file */
230 FILE* infile = stdin;
231 if(infilename)
232 {
233 infile = fopen(infilename, "r");
234 if(!infile)
235 _fatal("Cannot not open `%s' for input, aborting", infilename);
236 }
237
238 /* Read the graph from the file */
239 if(opt_directed)
240 {
241 /* Read directed graph in the DIMACS format */
242 g = bliss::Digraph::read_dimacs(infile);
243 }
244 else
245 {
246 /* Read undirected graph in the DIMACS format */
247 g = bliss::Graph::read_dimacs(infile);
248 }
249
250 if(infile != stdin)
251 fclose(infile);
252
253 if(!g)
254 _fatal("Failed to read the graph, aborting");
255
256 if(verbose_level >= 2)
257 {
258 fprintf(verbstr, "Graph read in %.2f seconds\n", timer.get_duration());
259 fflush(verbstr);
260 }
261
262
263 bliss::Stats stats;
264
265 /* Set splitting heuristics and verbose level */
266 if(opt_directed)
267 ((bliss::Digraph*)g)->set_splitting_heuristic(shs_directed);
268 else
269 ((bliss::Graph*)g)->set_splitting_heuristic(shs_undirected);
270 g->set_verbose_level(verbose_level);
271 g->set_verbose_file(verbstr);
272 g->set_failure_recording(opt_use_failure_recording);
273 g->set_component_recursion(opt_use_component_recursion);
274
275
276 if(opt_canonize == false)
277 {
278 /* No canonical labeling, only automorphism group */
279 g->find_automorphisms(stats, &report_aut, stdout);
280 }
281 else
282 {
283 /* Canonical labeling and automorphism group */
284 const unsigned int* cl = g->canonical_form(stats, &report_aut, stdout);
285
286 fprintf(stdout, "Canonical labeling: ");
287 bliss::print_permutation(stdout, g->get_nof_vertices(), cl, 1);
288 fprintf(stdout, "\n");
289
290 if(opt_output_can_file)
291 {
292 bliss::AbstractGraph* cf = g->permute(cl);
293 FILE* const fp = fopen(opt_output_can_file, "w");
294 if(!fp)
295 _fatal("Cannot open '%s' for outputting the canonical form, aborting", opt_output_can_file);
296 cf->write_dimacs(fp);
297 fclose(fp);
298 delete cf;
299 }
300 }
301
302 /* Output search statistics */
303 if(verbose_level > 0 and verbstr)
304 stats.print(verbstr);
305
306 if(verbose_level > 0)
307 {
308 fprintf(verbstr, "Total time:\t%.2f seconds\n", timer.get_duration());
309 fflush(verbstr);
310 }
311
312
313 delete g; g = 0;
314
315 return 0;
316}
