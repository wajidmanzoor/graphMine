80void Orbit::merge_orbits(OrbitEntry *orbit1, OrbitEntry *orbit2)
81{
82
83 if(orbit1 != orbit2)
84 {
85 _nof_orbits--;
86 /* Only update the elements in the smaller orbit */
87 if(orbit1->size > orbit2->size)
88 {
89 OrbitEntry * const temp = orbit2;
90 orbit2 = orbit1;
91 orbit1 = temp;
92 }
93 /* Link the elements of orbit1 to the almost beginning of orbit2 */
94 OrbitEntry *e = orbit1;
95 while(e->next)
96 {
97 in_orbit[e->element] = orbit2;
98 e = e->next;
99 }
100 in_orbit[e->element] = orbit2;
101 e->next = orbit2->next;
102 orbit2->next = orbit1;
103 /* Keep the minimal orbit representative in the beginning */
104 if(orbit1->element < orbit2->element)
105 {
106 const unsigned int temp = orbit1->element;
107 orbit1->element = orbit2->element;
108 orbit2->element = temp;
109 }
110 orbit2->size += orbit1->size;
111 }
