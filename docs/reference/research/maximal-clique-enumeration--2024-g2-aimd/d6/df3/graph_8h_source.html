115Graph::Graph(const std::string &filename, PresetPatternType queryType = P0, GraphType gt = GraphType::DATA)
116{
117 if (gt == GraphType::DATA)
118 {
119 std::string suffix = filename.substr(filename.rfind(".") + 1);
120 if (suffix == "bin")
121 readBinFile(filename);
122 else if (suffix == "graph")
123 readGraphFile(filename);
124 else if (suffix == "txt")
125 readSnapFile(filename);
126 else if (suffix == "lvid")
127 readLVIDFile(filename);
128 else {
129 std::cout << "Cannot read graph file based on its suffix ..." << std::endl;
130 assert(false);
131 }
132 }
133 else if (gt == GraphType::QUERY)
134 {
135 std::vector<std::vector<uintV>> conn = selectPresetPatterns(queryType);
136 edge_count_ *= 2;
137 row_ptrs_ = new uintE[vertex_count_ + 1];
138 cols_ = new uintV[edge_count_];
139 std::vector<uintE> offsets(vertex_count_ + 1, 0);
140
141 for (size_t i = 0; i < vertex_count_; ++i)
142 {
143 offsets[i] = conn[i].size();
144 }
145 uintE prefix = 0;
146 for (size_t i = 0; i < vertex_count_ + 1; ++i) {
147 row_ptrs_[i] = prefix;
148 prefix += offsets[i];
149 offsets[i] = row_ptrs_[i];
150 }
151 for (size_t i = 0; i < vertex_count_; ++i)
152 {
153 for (const auto &j: conn[i])
154 {
155 cols_[offsets[j]++] = i;
156 }
157 }
158 for (uintV u = 0; u < vertex_count_; ++u) {
159 std::sort(cols_ + row_ptrs_[u], cols_ + row_ptrs_[u + 1]);
160 }
161 ReMapVertexId();
162 }
163 else
164 assert(false);
165}

177void Graph::ReMapVertexId()
178{
179 // 1. find the root vertex with largest degree
180 size_t max_degree = 0;
181 uintV root = 0;
182 for (uintV i = 0; i < vertex_count_; i++) {
183 if (row_ptrs_[i + 1] - row_ptrs_[i] > max_degree) {
184 max_degree = row_ptrs_[i + 1] - row_ptrs_[i];
185 root = i;
186 }
187 }
188 // 2. bfs from the root vertex, make sure connected
189 // order: higher degree, more connections to the visited vertices
190 std::queue<uintV> queue;
191 std::vector<bool> visited(vertex_count_, false);
192 queue.push(root);
193 visited[root] = true;
194 uintV new_vid = 0;
195 std::vector<uintV> old_to_new(vertex_count_);
196 std::vector<uintV> new_to_old(vertex_count_);
197 uintE *new_row_ptrs_ = new uintE[vertex_count_ + 1];
198 uintV *new_cols_ = new uintV[edge_count_];
199
200 while (!queue.empty()) {
201 size_t size = queue.size();
202 std::vector<uintV> same_level_vertices;
203 for (size_t i = 0; i < size; i++) {
204 uintV front = queue.front();
205 same_level_vertices.push_back(front);
206 queue.pop();
207 for (size_t j = row_ptrs_[front]; j < row_ptrs_[front + 1]; ++j)
208 {
209 uintV ne = cols_[j];
210 if (!visited[ne]) {
211 visited[ne] = true;
212 queue.push(ne);
213 }
214 }
215 }
216 std::vector<std::tuple<size_t, size_t, uintV>> weights; // degree, connections, vid
217 for (size_t i = 0; i < size; i++) {
218 uintV v = same_level_vertices[i];
219 size_t connections = 0;
220 for (size_t j = row_ptrs_[v]; j < row_ptrs_[v + 1]; ++j)
221 {
222 uintV ne = cols_[j];
223 if (visited[ne])
224 connections++;
225 }
226 weights.emplace_back(row_ptrs_[v + 1] - row_ptrs_[v], connections, v);
227 }
228 std::sort(weights.begin(), weights.end(), [](const auto& a, const auto& b) {
229 if (std::get<0>(a) != std::get<0>(b))
230 return std::get<0>(a) > std::get<0>(b);
231 else if (std::get<1>(a) != std::get<1>(b))
232 return std::get<1>(a) > std::get<1>(b);
233 else if (std::get<2>(a) != std::get<2>(b))
234 return std::get<2>(a) < std::get<2>(b);
235 return false;
236 });
237 for (const auto& w : weights) {
238 old_to_new[std::get<2>(w)] = new_vid;
239 new_to_old[new_vid] = std::get<2>(w);
240 new_vid++;
241 }
242 }
243 auto offsets = new uintE[vertex_count_ + 1];
244 memset(offsets, 0, sizeof(uintE) * (vertex_count_ + 1));
245
246 for (size_t i = 0; i < vertex_count_; ++i)
247 {
248 offsets[i] = row_ptrs_[new_to_old[i] + 1] - row_ptrs_[new_to_old[i]];
249 }
250 uintE prefix = 0;
251 for (size_t i = 0; i < vertex_count_ + 1; ++i) {
252 new_row_ptrs_[i] = prefix;
253 prefix += offsets[i];
254 offsets[i] = new_row_ptrs_[i];
255 }
256 for (size_t i = 0; i < vertex_count_; ++i)
257 {
258 for (size_t j = row_ptrs_[i]; j < row_ptrs_[i + 1] ; ++j)
259 {
260 uintV ne = cols_[j];
261 new_cols_[offsets[old_to_new[i]]++] = old_to_new[ne];
262 }
263 }
264 for (uintV u = 0; u < vertex_count_; ++u) {
265 std::sort(new_cols_ + new_row_ptrs_[u], new_cols_ + new_row_ptrs_[u + 1]);
266 }
267 delete[] offsets;
268 delete[] row_ptrs_;
269 delete[] cols_;
270 row_ptrs_ = new_row_ptrs_;
271 cols_ = new_cols_;
272}

274void Graph::readSnapFile(const std::string &filename)
275{
276 Timer timer;
277 timer.StartTimer();
278
279 vertex_count_ = 0;
280 edge_count_ = 0;
281 row_ptrs_ = NULL;
282 cols_ = NULL;
283
284 uintV min_vertex_id = std::numeric_limits<uintV>::max();
285 uintV max_vertex_id = std::numeric_limits<uintV>::min();
286 std::ifstream file(filename.c_str(), std::fstream::in);
287 std::string line;
288 uintV vids[2];
289 while (getline(file, line)) {
290 if (line.length() == 0 || !std::isdigit(line[0]))
291 continue;
292 std::istringstream iss(line);
293 for (int i = 0; i < 2; ++i) {
294 iss >> vids[i];
295 min_vertex_id = std::min(min_vertex_id, vids[i]);
296 max_vertex_id = std::max(max_vertex_id, vids[i]);
297 }
298 edge_count_++;
299 }
300 file.close();
301
302 vertex_count_ = max_vertex_id - min_vertex_id + 1;
303 edge_count_ *= 2;
304 std::cout << "vertex_count=" << vertex_count_ << ", edge_count=" << edge_count_ << std::endl;
305
306 row_ptrs_ = new uintE[vertex_count_ + 1];
307 cols_ = new uintV[edge_count_];
308 auto offsets = new uintE[vertex_count_ + 1];
309 memset(offsets, 0, sizeof(uintE) * (vertex_count_ + 1));
310
311 {
312 std::ifstream file(filename.c_str(), std::fstream::in);
313 std::string line;
314 uintV vids[2];
315 while (getline(file, line)) {
316 if (line.length() == 0 || !std::isdigit(line[0]))
317 continue;
318 std::istringstream iss(line);
319 for (int i = 0; i < 2; ++i)
320 {
321 iss >> vids[i];
322 vids[i] -= min_vertex_id;
323 }
324 offsets[vids[0]]++;
325 offsets[vids[1]]++;
326 }
327 file.close();
328 }
329
330 uintE prefix = 0;
331 for (size_t i = 0; i < vertex_count_ + 1; ++i) {
332 row_ptrs_[i] = prefix;
333 prefix += offsets[i];
334 offsets[i] = row_ptrs_[i];
335 }
336
337 {
338 std::ifstream file(filename.c_str(), std::fstream::in);
339 std::string line;
340 uintV vids[2];
341 while (getline(file, line)) {
342 if (line.length() == 0 || !std::isdigit(line[0]))
343 continue;
344 std::istringstream iss(line);
345 for (int i = 0; i < 2; ++i)
346 {
347 iss >> vids[i];
348 vids[i] -= min_vertex_id;
349 }
350 cols_[offsets[vids[0]]++] = vids[1];
351 cols_[offsets[vids[1]]++] = vids[0];
352 }
353 file.close();
354 }
355 delete[] offsets;
356 offsets = NULL;
357
358 for (uintV u = 0; u < vertex_count_; ++u) {
359 std::sort(cols_ + row_ptrs_[u], cols_ + row_ptrs_[u + 1]);
360 }
361
362 timer.EndTimer();
363 timer.PrintElapsedMicroSeconds("reading CSR Snap file");
364}

406void Graph::readGraphFile(const std::string &filename)
407{
408 vertex_count_ = 0;
409 edge_count_ = 0;
410 row_ptrs_ = NULL;
411 cols_ = NULL;
412
413 std::ifstream file_in(filename);
414 if (!file_in.is_open())
415 {
416 std::cout << "Unable to read the graph " << filename << std::endl;
417 }
418
419 char type;
420 file_in >> type >> vertex_count_ >> edge_count_;
421 edge_count_ *= 2;
422 std::cout << "vertex_count=" << vertex_count_ << ", edge_count=" << edge_count_ << std::endl;
423
424 row_ptrs_ = new uintE[vertex_count_ + 1];
425 cols_ = new uintV[edge_count_];
426 row_ptrs_[0] = 0;
427
428 std::vector<uintV> neighbors_offset(vertex_count_, 0);
429
430 while (file_in >> type) {
431 if (type == 'v') { // Read vertex.
432 ui id;
433 ui label;
434 ui degree;
435 file_in >> id >> label >> degree;
436 row_ptrs_[id + 1] = row_ptrs_[id] + degree;
437 }
438 else if (type == 'e') { // Read edge.
439 uintV begin;
440 uintV end;
441 file_in >> begin >> end;
442
443 uintV offset = row_ptrs_[begin] + neighbors_offset[begin];
444 cols_[offset] = end;
445 offset = row_ptrs_[end] + neighbors_offset[end];
446 cols_[offset] = begin;
447 neighbors_offset[begin] += 1;
448 neighbors_offset[end] += 1;
449 }
450 }
451
452 file_in.close();
453
454 for (ui i = 0; i < vertex_count_; ++i) {
455 std::sort(cols_ + row_ptrs_[i], cols_ + row_ptrs_[i + 1]);
456 }
457}

459void Graph::readLVIDFile(const std::string &filename)
460{
461 vertex_count_ = 0;
462 edge_count_ = 0;
463 row_ptrs_ = NULL;
464 cols_ = NULL;
465
466 std::cout << "start build csr..." << std::endl;
467 // const char* kDelimiters = " ,;\t";
468 const char* kDelimiters = "0123456789";
469 std::unordered_map<std::string, uintV> ids;
470 std::vector<uintV> edge_pairs;
471 {
472 std::ifstream file(filename.c_str(), std::fstream::in);
473 std::string line;
474 while (getline(file, line)) {
475 if (line.length() == 0 || !std::isdigit(line[0]))
476 continue;
477
478 std::vector<std::string> num_strs;
479 size_t cur_pos = 0;
480 while (cur_pos < line.length()) {
481 cur_pos = line.find_first_of(kDelimiters, cur_pos, strlen(kDelimiters));
482 if (cur_pos < line.length()) {
483 size_t next_pos = line.find_first_not_of(kDelimiters, cur_pos, strlen(kDelimiters));
484 num_strs.push_back(line.substr(cur_pos, next_pos - cur_pos));
485 assert(next_pos > cur_pos);
486 cur_pos = next_pos;
487 }
488 }
489
490 for (auto& str : num_strs) {
491 assert(str.length());
492 for (auto ch : str) {
493 assert(std::isdigit(ch));
494 }
495 }
496
497 for (auto& str : num_strs) {
498 if (ids.find(str) == ids.end()) {
499 ids.insert(std::make_pair(str, vertex_count_++));
500 }
501 edge_pairs.push_back(ids[str]);
502 }
503 }
504 file.close();
505 }
506 ids.clear();
507
508 std::cout << "edge pairs size=" << edge_pairs.size() << std::endl;
509 assert(edge_pairs.size() % 2 == 0);
510 edge_count_ = edge_pairs.size(); // / 2;
511
512 std::vector<uintE> offsets(vertex_count_ + 1, 0);
513 for (size_t i = 0; i < edge_pairs.size(); i += 2) {
514 offsets[edge_pairs[i]]++;
515 offsets[edge_pairs[i + 1]]++;
516 }
517
518 row_ptrs_ = new uintE[vertex_count_ + 1];
519 cols_ = new uintV[edge_count_];
520
521 uintE prefix = 0;
522 for (uintV i = 0; i <= vertex_count_; ++i) {
523 row_ptrs_[i] = prefix;
524 prefix += offsets[i];
525 offsets[i] = row_ptrs_[i];
526 }
527
528 for (size_t i = 0; i < edge_pairs.size(); i += 2)
529 {
530 cols_[offsets[edge_pairs[i]]++] = edge_pairs[i + 1];
531 cols_[offsets[edge_pairs[i + 1]]++] = edge_pairs[i];
532 }
533
534 offsets.clear();
535 edge_pairs.clear();
536
537#pragma omp parallel for schedule(dynamic)
538 for (uintV u = 0; u < vertex_count_; ++u)
539 {
540 std::sort(cols_ + row_ptrs_[u], cols_ + row_ptrs_[u + 1]);
541 }
542 std::cout << "finish building CSR" << std::endl;
543}

616void Graph::CompleteAutomorphisms(std::vector<std::vector<uint32_t>>& perm_group)
617{
618 // multiplying std::vector<uint32_t>s is just function composition: (p1*p2)[i] = p1[p2[i]]
619 std::vector<std::vector<uint32_t>> products;
620 // for filtering duplicates
621 std::unordered_set<std::string> dups;
622 for (auto f : perm_group)
623 dups.insert(OrderToString(f));
624
625 for (auto k = perm_group.begin(); k != perm_group.end(); k++)
626 {
627 for (auto l = perm_group.begin(); l != perm_group.end(); l++)
628 {
629 std::vector<uint32_t> p1 = *k;
630 std::vector<uint32_t> p2 = *l;
631
632 std::vector<uint32_t> product;
633 product.resize(p1.size());
634 for (unsigned i = 0; i < product.size(); i++)
635 product[i] = p1[p2[i]];
636
637 // don't count duplicates
638 if (dups.count(OrderToString(product)) == 0)
639 {
640 dups.insert(OrderToString(product));
641 products.push_back(product);
642 }
643 }
644 }
645
646 for (auto p : products)
647 perm_group.push_back(p);
648}

651{
652 std::map<uintV, std::set<uintV>> eclasses;
653 for (size_t i = 0; i < vertex_count_; i++)

655 std::set<uintV> eclass;
656 for (auto&& perm : aut)
657 eclass.insert(perm[i]);

659 eclasses[rep].insert(eclass.cbegin(), eclass.cend());
660 }
661 return eclasses;

664std::vector<std::pair<uintV, uintV>> Graph::GetConditions(G2AimdBlissGraph* bg)
665{
666 std::vector<std::vector<uintV>> aut = GetAutomorphisms(bg);
667 std::map<uintV, std::set<uintV>> eclasses = GetAEquivalenceClasses(aut);
668
669 std::vector<std::pair<uintV, uintV>> result;
670 auto eclass_it = std::find_if(eclasses.cbegin(), eclasses.cend(), [](auto&& e) { return e.second.size() > 1; });
671 while (eclass_it != eclasses.cend() && eclass_it->second.size() > 1)
672 {
673 const auto& eclass = eclass_it->second;
674 uintV n0 = *eclass.cbegin();
675
676 for (auto&& perm : aut)
677 {
678 uintV min = *std::min_element(std::next(eclass.cbegin()), eclass.cend(), [perm](uint32_t n, uint32_t m) { return perm[n] < perm[m]; });
679 result.emplace_back(n0, min);
680 }
681 aut.erase(std::remove_if(aut.begin(), aut.end(), [n0](auto&& perm) { return perm[n0] != n0; }), aut.end());
682
683 eclasses = GetAEquivalenceClasses(aut);
684 eclass_it = std::find_if(eclasses.cbegin(), eclasses.cend(), [](auto&& e) { return e.second.size() > 1; });
685 }
686
687 // remove duplicate conditions
688 std::sort(result.begin(), result.end());
689 result.erase(std::unique(result.begin(), result.end()), result.end());
690
691 return result;
692}

726void Graph::Preprocess() {
727 // remove dangling nodes
728 // self loops
729 // parallel edges
730 Timer timer;
731 timer.StartTimer();
732 std::cout << "start preprocess..." << std::endl;
733 size_t vertex_count = GetVertexCount();
734 size_t edge_count = GetEdgeCount();
735 auto row_ptrs = GetRowPtrs();
736 auto cols = GetCols();
737
738 auto vertex_ecnt = new uintE[vertex_count + 1];
739 memset(vertex_ecnt, 0, sizeof(uintE) * (vertex_count + 1));
740 for (uintV u = 0; u < vertex_count; ++u)
741 {
742 for (auto j = row_ptrs[u]; j < row_ptrs[u + 1]; ++j)
743 {
744 auto v = cols[j];
745 bool parallel_edge = (j > row_ptrs[u] && v == cols[j - 1]);
746 bool self_loop = u == v;
747 if (!parallel_edge && !self_loop)
748 {
749 vertex_ecnt[u]++;
750 }
751 }
752 }
753 auto nrow_ptrs = new uintE[vertex_count + 1];
754 uintE nedge_count = 0;
755 for (uintV u = 0; u < vertex_count; ++u)
756 {
757 nrow_ptrs[u] = nedge_count;
758 nedge_count += vertex_ecnt[u];
759 }
760 nrow_ptrs[vertex_count] = nedge_count;
761 delete[] vertex_ecnt;
762 vertex_ecnt = NULL;
763
764 auto ncols = new uintV[nedge_count];
765 for (uintV u = 0; u < vertex_count; ++u)
766 {
767 auto uoff = nrow_ptrs[u];
768 for (uintE j = row_ptrs[u]; j < row_ptrs[u + 1]; ++j)
769 {
770 auto v = cols[j];
771 bool parallel_edge = j > row_ptrs[u] && v == cols[j - 1];
772 bool self_loop = u == v;
773 if (!parallel_edge && !self_loop)
774 {
775 ncols[uoff++] = v;
776 }
777 }
778 }
779 edge_count = nedge_count;
780 std::swap(row_ptrs, nrow_ptrs);
781 std::swap(cols, ncols);
782 delete[] nrow_ptrs;
783 nrow_ptrs = NULL;
784 delete[] ncols;
785 ncols = NULL;
786
787 auto new_vertex_ids = new uintV[vertex_count];
788 uintV max_vertex_id = 0;
789 for (uintV u = 0; u < vertex_count; ++u)
790 {
791 if (row_ptrs[u] == row_ptrs[u + 1]) {
792 new_vertex_ids[u] = vertex_count;
793 }
794 else
795 {
796 new_vertex_ids[u] = max_vertex_id++;
797 row_ptrs[new_vertex_ids[u]] = row_ptrs[u];
798 }
799 }
800 for (uintE j = 0; j < edge_count; ++j)
801 {
802 cols[j] = new_vertex_ids[cols[j]];
803 }
804 delete[] new_vertex_ids;
805 new_vertex_ids = NULL;
806 vertex_count = max_vertex_id;
807 row_ptrs[vertex_count] = edge_count;
808
809 timer.EndTimer();
810 std::cout << "finish preprocess, time=" << timer.GetElapsedMicroSeconds() / 1000.0 << "ms"
811 << ", now vertex_count=" << vertex_count << ",edge_count=" << edge_count << std::endl;
812
813 SetVertexCount(vertex_count);
814 SetEdgeCount(edge_count);
815 SetRowPtrs(row_ptrs);
816 SetCols(cols);
817}
