44/**
45 * \brief Statistics returned by the bliss search algorithm.
46 */
47class Stats
48{
49 friend class AbstractGraph;
50 /** \internal The size of the automorphism group. */
51 BigNum group_size;
52 /** \internal An approximation (due to possible overflows) of
53 * the size of the automorphism group. */
54 long double group_size_approx;
55 /** \internal The number of nodes in the search tree. */
56 long unsigned int nof_nodes;
57 /** \internal The number of leaf nodes in the search tree. */
58 long unsigned int nof_leaf_nodes;
59 /** \internal The number of bad nodes in the search tree. */
60 long unsigned int nof_bad_nodes;
61 /** \internal The number of canonical representative updates. */
62 long unsigned int nof_canupdates;
63 /** \internal The number of generator permutations. */
64 long unsigned int nof_generators;
65 /** \internal The maximal depth of the search tree. */
66 unsigned long int max_level;
67 /** */
68 void reset()
69 {
70 group_size.assign(1);
71 group_size_approx = 1.0;
72 nof_nodes = 0;
73 nof_leaf_nodes = 0;
74 nof_bad_nodes = 0;
75 nof_canupdates = 0;
76 nof_generators = 0;
77 max_level = 0;
78 }
79public:
80 Stats() { reset(); }
81 /** Print the statistics. */
82 size_t print(FILE* const fp) const
83 {
84 size_t r = 0;
85 r += fprintf(fp, "Nodes: %lu\n", nof_nodes);
86 r += fprintf(fp, "Leaf nodes: %lu\n", nof_leaf_nodes);
87 r += fprintf(fp, "Bad nodes: %lu\n", nof_bad_nodes);
88 r += fprintf(fp, "Canrep updates: %lu\n", nof_canupdates);
89 r += fprintf(fp, "Generators: %lu\n", nof_generators);
90 r += fprintf(fp, "Max level: %lu\n", max_level);
91 r += fprintf(fp, "|Aut|: ")+group_size.print(fp)+fprintf(fp, "\n");
92 fflush(fp);
93 return r;
94 }
95 /** An approximation (due to possible overflows/rounding errors) of
96 * the size of the automorphism group. */
97 long double get_group_size_approx() const {return group_size_approx;}
98 /** The number of nodes in the search tree. */
99 long unsigned int get_nof_nodes() const {return nof_nodes;}
100 /** The number of leaf nodes in the search tree. */
101 long unsigned int get_nof_leaf_nodes() const {return nof_leaf_nodes;}
102 /** The number of bad nodes in the search tree. */
103 long unsigned int get_nof_bad_nodes() const {return nof_bad_nodes;}
104 /** The number of canonical representative updates. */
105 long unsigned int get_nof_canupdates() const {return nof_canupdates;}
106 /** The number of generator permutations. */
107 long unsigned int get_nof_generators() const {return nof_generators;}
108 /** The maximal depth of the search tree. */
109 unsigned long int get_max_level() const {return max_level;}
110};

535 * The possible splitting heuristics.
536 * The selected splitting heuristics affects the computed canonical
537 * labelings; therefore, if you want to compare whether two graphs
538 * are isomorphic by computing and comparing (for equality) their
539 * canonical versions, be sure to use the same splitting heuristics
540 * for both graphs.

544 * Very fast but may result in large search spaces on difficult graphs.
545 * Use for large but easy graphs. */

548 * Fast, should usually produce smaller search spaces than shs_f. */
549 shs_fs,

551 * Fast, should usually produce smaller search spaces than shs_f. */
552 shs_fl,
553 /** First maximally non-trivially connected non-unit cell.
554 * Not so fast, should usually produce smaller search spaces than shs_f,
555 * shs_fs, and shs_fl. */
556 shs_fm,
557 /** First smallest maximally non-trivially connected non-unit cell.
558 * Not so fast, should usually produce smaller search spaces than shs_f,
559 * shs_fs, and shs_fl. */
560 shs_fsm,
561 /** First largest maximally non-trivially connected non-unit cell.
562 * Not so fast, should usually produce smaller search spaces than shs_f,
563 * shs_fs, and shs_fl. */
564 shs_flm
