96__device__ inline bool emb_validation_check(KeyT *emb, CSRGraph g, int level, expand_constraint ec,

100 for (int i = 0; i <= level; i ++)
101 if (emb[i] == dst)

105 return false;

113 case ID:

119 uint32_t nbr_deg = g.getDegree(cur_nbr);

121 //if (g.getDegree(dst) < g.getDegree(emb[(nbrs>>(8*i))&0xff])) return false;
122 break;
123 }
124 }
125 //adjacency check

135 }
