170{

172 * the clique list. */

180 //final vertex is stored in the sublist ID for the tail node

226{
227 /* Returns the number of 2-cliques (edges) for each source vertex, keeping only
228 * one edge from each undirected pair, based on either index or degree orientation, and
229 * where both vertices' threshold values are greater than the current lower bound max clique size. */
230 long long unsigned int idx = blockIdx.x * blockDim.x + threadIdx.x;
231 if (idx >= numVertices) return;
232
233 unsigned int src = d_vertexLabels[idx];
234 if (d_filterThresholds[src] < (w - 1)) return;
235 int listStart = graph.CsrT::GetNeighborListOffset(src);
236 int numNeighbors = d_vertexDegrees[src];
237 int listEnd = listStart + numNeighbors;
238 unsigned int validCount = 0;
239 if (!byDegree) { //select edge from each undirected pair based on vertices' indices
240 for (int e = listStart; e < listEnd; e++) {
241 unsigned int dest = graph.CsrT::GetEdgeDest(e);
242 if (src < dest) { //index orientation
243 if (!(d_filterThresholds[dest] < (w - 1))) {
244 validCount++;
245 }
246 }
247 }
248 }
249 if (byDegree) { //select edge from each undirected pair based on vertices' degrees
250 for (int e = listStart; e < listEnd; e++) {
251 unsigned int dest = graph.CsrT::GetEdgeDest(e);
252 int numNeighborsDest = d_vertexDegrees[dest];
253 if ((numNeighbors < numNeighborsDest) || ((numNeighbors == numNeighborsDest) && (src < dest))) { //degree orientation
254 if (!(d_filterThresholds[dest] < (w - 1))) {
255 validCount++;
256 }
257 }
258 }
259 }
260
261 if (validCount >= (w - 1)) {
262 counts[idx] = validCount;
263 flags[idx] = 1;
264 }
265}

457__global__ void findWindowTail(struct clique_node currCliques, unsigned int windowSize, long long unsigned int start, long long unsigned int* tail)
458{
459 /* Returns the end of the window for windowed version of clique merging algorithm.
460 * Threads are assigned to one sublistID each, and check if it is the beginning of a sublist.
461 * If yes, use atomic minimum to find the first location where this happens. Return in tail.
462 * Loop over another windowSize chunk of vertices until a sublist start is found. */
463 long long unsigned int idx = blockIdx.x * blockDim.x + threadIdx.x;
464 if (idx >= (windowSize - 1)) return;
465 idx += start;
466
467 unsigned int iter = 0;
468 while(tail[0] > (start + (windowSize * iter))) {
469
470 if (idx >= (currCliques.numVertices - 1)) return;
471
472 if(currCliques.sublistIDs[idx] != currCliques.sublistIDs[idx + 1]) {
473 atomicMin(tail, idx);
474 }
475
476 iter++;
477 idx += iter * windowSize;
478 }
479}

482{
483 /* Returns the number of cliques that can be formed from each clique
484 * by combining it with other cliques in its sublist. Used for finding
485 * the amount of memory to allocate for new clique list node. */

487 if (idx >= (currCliques.numVertices - 1)) return;
488

490 unsigned int vertex = currCliques.vertexIDs[idx];
491 //get the neighbor list start & length for src
492 int currentStart = graph.CsrT::GetNeighborListOffset(vertex);
493 int listEnd = graph.CsrT::GetNeighborListLength(vertex) + currentStart - 1;
494 //need to search within that list for dest (binary search)
495 long long unsigned int numNew = 0llu;
496 long long unsigned int j = idx + 1;
497 while((j < currCliques.numVertices) && (currCliques.sublistIDs[j] == sublist)) {
498 int destIndex = gunrock::util::BinarySearch(currCliques.vertexIDs[j], graph.column_indices, currentStart, listEnd);

501 }
502 j++;
503 }
504
505 if ((numNew + currCliques.k) < w) {
506 numNew = 0llu;
507 }

511__global__ void mergeCliques(struct clique_node newCliques, struct clique_node currCliques, CsrT graph, long long unsigned int* offsets)
512{
513 /* Merges k-cliques from newCliques.previous with any matching
514 * k-cliques in its sublist into k+1-cliques and stores them in newCliques.
515 * (newCliques should have an appropriately sized empty node as head)
516 * offsets = number of new cliques formed in preceding cliques
517 * Sizes of new sublists, used to compute offsets, are found by the countCliques kernel. */
518 long long unsigned int idx = blockIdx.x * blockDim.x + threadIdx.x;
519 if (idx >= (currCliques.numVertices - 1)) return;
520
521 long long unsigned int cliqueOffset = offsets[idx];
522 if (cliqueOffset == offsets[idx + 1]) return; //if count=0
523 unsigned int sublist = currCliques.sublistIDs[idx];
524 unsigned int vertex = currCliques.vertexIDs[idx];
525 //get the neighbor list start & length for src
526 int currentStart = graph.CsrT::GetNeighborListOffset(vertex);
527 int listEnd = graph.CsrT::GetNeighborListLength(vertex) + currentStart - 1;
528 //need to search within that list for dest (binary search)
529 long long unsigned int j = idx + 1;
530 long long unsigned int count = 0;
531 while((j < currCliques.numVertices) && (currCliques.sublistIDs[j] == sublist)) {
532 int destIndex = gunrock::util::BinarySearch(currCliques.vertexIDs[j], graph.column_indices, currentStart, listEnd);
533 if ((destIndex <= listEnd) && (graph.column_indices[destIndex] == currCliques.vertexIDs[j])) {
534 newCliques.vertexIDs[cliqueOffset + count] = currCliques.vertexIDs[j];
535 newCliques.sublistIDs[cliqueOffset + count] = idx;
536 count++;
537 }
538 j++;
539 }
540}

552__host__ unsigned int greedyHeuristic(CsrT graph, unsigned int* d_vertexList, unsigned int numVertices)
553{
554 /* Basic greedy heuristic to find a large clique to use for the initial lower bound
555 * for pruning. This is the "single-run heuristic". In each iteration, the vertex with
556 * the largest degree or core number is chosen to be added to the heuristic clique, and
557 * the vertex list is filtered to remove vertices not connected to this new vertex.
558 * The filtering operation is performed in parallel on the GPU with the CUB Select operation.
559 * Input vertex list must be pre-sorted by either k-core or degree. */
560 cudaError_t retval = cudaSuccess;
561
562 void* d_temp_storage = NULL;
563 size_t temp_storage_bytes = 0;
564
565 unsigned int cliqueSize = 0;
566 unsigned int* d_vertexList_temp;
567 cudaMalloc((void**) &d_vertexList_temp, numVertices * sizeof(unsigned int));
568 cudaMemcpy(d_vertexList_temp, d_vertexList, numVertices * sizeof(unsigned int), cudaMemcpyDeviceToDevice);
569 unsigned int h_newVertex;
570 cudaMemcpy(&h_newVertex, d_vertexList, sizeof(unsigned int), cudaMemcpyDeviceToHost);
571
572 //Filter out vertices not connected to first vertex
573 unsigned int* d_numRemaining;
574 cudaMalloc((void**) &d_numRemaining, sizeof(unsigned int));
575 unsigned int* d_filteredVertices;
576 cudaMalloc((void**) &d_filteredVertices, numVertices * sizeof(unsigned int));
577 unsigned int h_numRemaining = numVertices;
578
579 while(h_numRemaining > 0) {
580 cliqueSize++;
581 IsNeighbor select_op(graph, h_newVertex);
582 d_temp_storage = NULL;
583 temp_storage_bytes = 0;
584 GUARD_CU(CUDAErrorCheck());
585 cub::DeviceSelect::If(d_temp_storage, temp_storage_bytes, d_vertexList_temp, d_filteredVertices, d_numRemaining, h_numRemaining, select_op);
586 cudaMalloc(&d_temp_storage, temp_storage_bytes);
587 cub::DeviceSelect::If(d_temp_storage, temp_storage_bytes, d_vertexList_temp, d_filteredVertices, d_numRemaining, h_numRemaining, select_op);
588 GUARD_CU(CUDAErrorCheck());
589
590 cudaMemcpy(&h_numRemaining, d_numRemaining, sizeof(unsigned int), cudaMemcpyDeviceToHost);
591
592 //Select next vertex from candidate list
593 cudaMemcpy(&h_newVertex, d_filteredVertices, sizeof(unsigned int), cudaMemcpyDeviceToHost);
594 unsigned int* d_pointerHolder = d_filteredVertices;
595 d_filteredVertices = d_vertexList_temp;
596 d_vertexList_temp = d_pointerHolder;
597 }
598
599 cudaFree(d_vertexList_temp);
600 cudaFree(d_numRemaining);
601 cudaFree(d_filteredVertices);
602
603 return cliqueSize;
604}

608{
609 /* First set up step in parallel heuristic:
610 * Get number of neighbors for each vertex to allocate memory. */
611 unsigned int idx = blockIdx.x * blockDim.x + threadIdx.x;
612 if (idx >= numVertices) return;

619__global__ void setUpNeighborsThresholds(CsrT graph, unsigned int* vertexList, unsigned int* neighborCounts, unsigned int* thresholds_in, unsigned int* indices, unsigned int numVertices, unsigned int* neighbors, unsigned int* neighborThresholds)
620{
621 /* Second set up step in parallel heuristic:
622 * Gather threshold values for all neighbors to enable the selection
623 * of the neighbor with highest threshold value to add to the heuristic
624 * clique in each iteration of the main loop of the heuristic. */
625 unsigned int idx = blockIdx.x * blockDim.x + threadIdx.x;
626 if (idx >= numVertices) return;
627
628 unsigned int vertex = vertexList[idx];
629 unsigned int n = neighborCounts[idx];
630 unsigned int offset = indices[idx];
631 int eStart = graph.CsrT::GetNeighborListOffset(vertex);
632 int eEnd = eStart + n;
633 int count = 0;
634 for (int e = eStart; e < eEnd; e++) {
635 int dst = graph.CsrT::GetEdgeDest(e);
636 neighbors[offset + count] = dst;
637 neighborThresholds[offset + count] = thresholds_in[dst];
638 count++;
639 }
640}

642__global__ void checkConnections(CsrT graph, unsigned int* candidates, unsigned int* indices, cub::KeyValuePair<int, unsigned int>* bestVertices, unsigned int numSegments, unsigned int numCandidates, char* keepFlags, unsigned int* connectedCounts)
643{
644 /* Step in main loop of parallel heuristic:
645 * Uses the vertices identified as having the highest threshold value
646 * (k-core or degree) as the newly-added vertex for each clique.
647 * Each thread checks other vertices in segment to see if they are connected
648 * to the newly-added vertex. Outputs flags for which vertices to keep and
649 * the total count for vertices remaining in segment. */
650 unsigned int idx = blockIdx.x * blockDim.x + threadIdx.x;
651 if (idx >= numSegments) return;
652
653 unsigned int startIndex = indices[idx];
654 unsigned int nextStart = indices[idx + 1];
655 if (bestVertices[idx].value == 0) {
656 connectedCounts[idx] = 0;
657 return;
658 }
659 unsigned int newVertex = candidates[bestVertices[idx].key + startIndex];
660 //get the neighbor list start & length for src
661 int currentStart = graph.CsrT::GetNeighborListOffset(newVertex);
662 int listEnd = graph.CsrT::GetNeighborListLength(newVertex) + currentStart - 1;
663 unsigned int numConnected = 0;
664 //check all vertices in segment to see if connected to chosen vertex
665 for (int u = startIndex; u < nextStart; u++) {
666 if(u >= numCandidates) {
667 printf("vertex index (%u) greater than number of candidates (%u)\n", u, numCandidates);
668 return;
669 }
670 unsigned int currentVertex = candidates[u];
671 if (currentVertex == newVertex) {
672 keepFlags[u] = 0;
673 continue;
674 }
675 int destIndex = gunrock::util::BinarySearch(currentVertex, graph.column_indices, currentStart, listEnd);
676 if (graph.column_indices[destIndex] == currentVertex) {
677 keepFlags[u] = 1;
678 numConnected++;
679 currentStart = destIndex;
680 }
681 else {
682 keepFlags[u] = 0;
683 }
684 }
685
686 connectedCounts[idx] = numConnected;
687}

689__host__ unsigned int parallelGreedyHeuristic(CsrT graph, unsigned int* d_vertexList, unsigned int* d_thresholds_in, unsigned int numVertices)
690{
691 cudaError_t retval = cudaSuccess;
692
693 //Setup:
694 unsigned int* d_neighborCounts;
695 cudaMalloc((void**) &d_neighborCounts, numVertices * sizeof(unsigned int));
696 cudaMemset(d_neighborCounts, 0, numVertices * sizeof(unsigned int));
697 //Kernel: one vertex per thread; return number of neighbors
698 GUARD_CU(CUDAErrorCheck());
699 getNeighborCounts<<<(numVertices + (BLOCK_SIZE - 1)) / BLOCK_SIZE, BLOCK_SIZE>>>(graph, d_vertexList, d_neighborCounts, numVertices);
700 GUARD_CU(CUDAErrorCheck());

712 //Allocate memory for neighbor lists and their threshold values
713 unsigned int numCandidates_nextLast;
714 cudaMemcpy(&numCandidates_nextLast, d_indices + (numVertices - 1), sizeof(unsigned int), cudaMemcpyDeviceToHost);
715 unsigned int numNeighborsLast;
716 cudaMemcpy(&numNeighborsLast, d_neighborCounts + (numVertices - 1), sizeof(unsigned int), cudaMemcpyDeviceToHost);
717 unsigned int numCandidates = numCandidates_nextLast + numNeighborsLast;
718 //printf("numCandidates: %u\n", numCandidates);
719 cudaMemcpy((d_indices + numVertices), &numCandidates, sizeof(unsigned int), cudaMemcpyHostToDevice);

721 cudaMalloc((void**) &d_candidates, numCandidates * sizeof(unsigned int));
722 cudaMemset(d_candidates, 0, numCandidates * sizeof(unsigned int));
723 unsigned int* d_neighborThresholds;
724 cudaMalloc((void**) &d_neighborThresholds, numCandidates * sizeof(unsigned int));

730 GUARD_CU(CUDAErrorCheck());

743 cudaMalloc((void**) &d_candidates_out, numCandidates * sizeof(unsigned int));

750 unsigned int* d_numSegments_out;
751 cudaMalloc((void**) &d_numSegments_out, sizeof(unsigned int));
752
753 while(numSegments > 0) {
754 //Segmented max reduce over threshold values to find next vertex to add in each segment
755 cudaMemset(d_maxIndices, 0, numSegments * sizeof(cub::KeyValuePair<int, unsigned int>));

765 GUARD_CU(CUDAErrorCheck());

767 GUARD_CU(CUDAErrorCheck());
768
769 //Keep only flagged for both candidate lists and threshold lists
770 cudaMemset(d_candidates_out, 0, numCandidates * sizeof(unsigned int));
771 cudaMemset(d_neighborThresholds_out, 0, numCandidates * sizeof(unsigned int));
772 cudaMemset(d_numCandidates_out, 0, sizeof(unsigned int));
773 d_temp_storage = NULL;
774 temp_storage_bytes = 0;
775 CubDebugExit(cub::DeviceSelect::Flagged(d_temp_storage, temp_storage_bytes, d_candidates, d_keepFlags, d_candidates_out, d_numCandidates_out, numCandidates));

777 CubDebugExit(cub::DeviceSelect::Flagged(d_temp_storage, temp_storage_bytes, d_candidates, d_keepFlags, d_candidates_out, d_numCandidates_out, numCandidates));

779 temp_storage_bytes = 0;

787 //Compact out segments with count = 0
788 cudaMemset(d_nonzeroCounts, 0, numSegments * sizeof(unsigned int));
789 cudaMemset(d_numSegments_out, 0, sizeof(unsigned int));
790 GreaterThan select_op(0);
791 d_temp_storage = NULL;
792 temp_storage_bytes = 0;
793 CubDebugExit(cub::DeviceSelect::If (d_temp_storage, temp_storage_bytes, d_connectedCounts, d_nonzeroCounts, d_numSegments_out, numSegments, select_op));
794 cudaMalloc(&d_temp_storage, temp_storage_bytes);
795 CubDebugExit(cub::DeviceSelect::If (d_temp_storage, temp_storage_bytes, d_connectedCounts, d_nonzeroCounts, d_numSegments_out, numSegments, select_op));
796 cudaMemcpy(&numSegments, d_numSegments_out, sizeof(unsigned int), cudaMemcpyDeviceToHost);
797
798 //Scan on number of vertices remaining to get new indices for segments
799 cudaMemset(d_indices, 0, (numSegments + 1) * sizeof(unsigned int));
800 d_temp_storage = NULL;
801 temp_storage_bytes = 0;
802 CubDebugExit(cub::DeviceScan::ExclusiveSum(d_temp_storage, temp_storage_bytes, d_nonzeroCounts, d_indices, (numSegments + 1)));
803 cudaMalloc(&d_temp_storage, temp_storage_bytes);
804 CubDebugExit(cub::DeviceScan::ExclusiveSum(d_temp_storage, temp_storage_bytes, d_nonzeroCounts, d_indices, (numSegments + 1)));

806 //Swap pointers
807 unsigned int* temp = d_neighborThresholds;
808 d_neighborThresholds = d_neighborThresholds_out;
809 d_neighborThresholds_out = temp;
810 temp = d_candidates;
811 d_candidates = d_candidates_out;
812 d_candidates_out = temp;
813 }
814
815 cudaFree(d_indices);
816 cudaFree(d_candidates);
817 cudaFree(d_neighborThresholds);
818 cudaFree(d_maxIndices);
819 cudaFree(d_keepFlags);
820 cudaFree(d_connectedCounts);
821 cudaFree(d_candidates_out);
822 cudaFree(d_neighborThresholds_out);
823 cudaFree(d_numCandidates_out);
824 cudaFree(d_nonzeroCounts);
825 cudaFree(d_numSegments_out);
826
827 return cliqueSize;
828}
