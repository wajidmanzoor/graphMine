32int get_pattern_edge_num(const Pattern& p)
33{
34 int edge_num = 0;
35 const int* ptr = p.get_adj_mat_ptr();
36 int size = p.get_size();
37 for (int i = 0; i < size; ++i)
38 for (int j = i + 1; j < size; ++j)
39 if (ptr[i * size + j] != 0)
40 edge_num += 1;
41 return edge_num;
42}
