57void process_message(uint64_t recv_buf[], uint64_t send_buf[][MSG_BUF_LEN], int node, int sender) {
58 MPI_Request send_req;
59 switch (recv_buf[0]) {
60 case MSG_REQUEST_WORK: { // me: master
61 // recv: NODE_TASK_GRANULARUTY
62 // notice: avoid collision
63 send_buf[1][0] = MSG_DISPATCH_WORK;
64 send_buf[1][1] = global_cur_task;
65 send_buf[1][2] = global_cur_task = std::min(global_cur_task + recv_buf[1], nr_tasks);
66 MPI_Isend(&send_buf[1][0], 3, MPI_UINT64_T, sender, 0, MPI_COMM_WORLD, &send_req);
67 log("master got work request from node %d , replying [%ld, %ld)\n", sender, send_buf[1][1], send_buf[1][2]);
68 break;
69 }
70 case MSG_DISPATCH_WORK: { // me: slave
71 // recv: start, end
72 uint64_t new_task_cur = recv_buf[1];
73 uint64_t new_task_end = recv_buf[2];
74 task_status = std::make_tuple(new_task_cur, new_task_end);
75 task_ready = true;
76 log("slave node %d got task [%llu, %llu)\n", node, new_task_cur, new_task_end);
77 break;
78 }
79 case MSG_REPORT_ANS: { // me: master
80 finished_number += 1;
81 log("slave node %d finished its work. Finished number: %d\n", sender, finished_number);
82 break;
83 }
84 }
85}
