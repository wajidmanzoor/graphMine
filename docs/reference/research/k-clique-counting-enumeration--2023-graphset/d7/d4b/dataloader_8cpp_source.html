28static void calculate_checksum(GraphHeader& h)
29{
30 h.checksum = 0;
31 uint32_t sum = 0;
32 uint32_t *p = reinterpret_cast<uint32_t*>(&h);
33 int len = sizeof(h) / sizeof(uint32_t);
34 for (int i = 0; i < len; ++i)
35 sum += p[i];
36 h.checksum = -sum;
37}

99 g.tri_cnt = h.tri_cnt;

110 printf("load_graph: failed to load edges.\n");
