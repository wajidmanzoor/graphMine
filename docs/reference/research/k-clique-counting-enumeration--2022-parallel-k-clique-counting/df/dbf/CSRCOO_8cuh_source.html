37 static CSRCOO<Index, Vector> from_edgelist(const EdgeList& es, bool (*edgeFilter)(const Edge&) = nullptr)
38 {
39 CSRCOO csr;
40
41 if (es.size() == 0) {
42 Log(warn, "constructing from empty edge list");
43 return csr;
44 }
45
46 for (const auto& edge : es) {
47
48 const Index src = static_cast<Index>(edge.first);
49 const Index dst = static_cast<Index>(edge.second);
50
51 if (src == dst)
52 continue;
53
54 // edge has a new src and should be in a new row
55 // even if the edge is filtered out, we need to add empty rows
56 while (csr.rowPtr_.size() != size_t(src + 1)) {
57 // expecting inputs to be sorted by src, so it should be at least
58 // as big as the current largest row we have recored
59 assert(src >= csr.rowPtr_.size());
60 // SPDLOG_TRACE(logger::console(), "node {} edges start at {}", edge.src_,
61 // csr.edgeSrc_.size());
62 csr.rowPtr_.push_back(csr.colInd_.size());
63 }
64
65 // filter or add the edge
66 if (nullptr != edgeFilter && edgeFilter(edge)) {
67 continue;
68 }
69 else {
70 csr.rowInd_.push_back(src);
71 csr.colInd_.push_back(dst);
72 }
73 }
74
75 // add the final length of the non-zeros to the offset array
76 csr.rowPtr_.push_back(csr.colInd_.size());
77
78 assert(csr.rowInd_.size() == csr.colInd_.size());
79 return csr;
