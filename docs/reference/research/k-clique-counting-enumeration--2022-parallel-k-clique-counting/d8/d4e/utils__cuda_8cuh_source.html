17/*! Binary search
18
19 \tparam arr Pointer to the array
20 \tparam l Left boundary of arr
21 \tparam r Right boundary of arr
22 \tparam x Value to search for
23*/
24__device__ inline uint binarySearch_b(const uint* arr, uint l, uint r, uint x)
25{
26 size_t left = l;
27 size_t right = r;
28 while (left < right) {
29 const size_t mid = (left + right) / 2;
30 uint val = arr[mid];
31 bool pred = val < x;
32 if (pred) {
33 left = mid + 1;
34 }
35 else {
36 right = mid;
37 }
38 }
39 return left;
40}
