57 nodeDegree.gdata(), node_num, bucket.mark.gdata(), level, bucket_level_end_ + KCL_NODE_LEVEL_SKIP_SIZE);
58
59 multi++;
60
61 bucket.count.gdata()[0] = CUBSelect(asc.gdata(), bucket.queue.gdata(), bucket.mark.gdata(), node_num, dev_);
62 bucket_level_end_ += KCL_NODE_LEVEL_SKIP_SIZE;

64 // SCAN the window.
65 if (bucket.count.gdata()[0] != 0)
66 {
67 current.count.gdata()[0] = 0;

70 bucket.queue.gdata(), bucket.count.gdata()[0], nodeDegree.gdata(), current.mark.gdata(), current.queue.gdata(), &(current.count.gdata()[0]), level, span);

72 else

74 current.count.gdata()[0] = 0;

183 cpn.setAll(0, true);

187 current_q.count.gdata()[0] = 0;
188 level = kcount - 1;
189 bucket_level_end_ = level;

191 const T partitionSize = PSIZE; //PART_SIZE;
192 T factor = (block_size / partitionSize);

194 const uint dv = 32;
195 const uint max_level = 9;

201 GPUArray<T> node_be("Binary Encoding Array", gpu, encode_size, dev_);
202 current_level2.setAll(0, true);
203 node_be.setAll(0, true);

205 const T numPartitions = block_size/partitionSize;
206 cudaMemcpyToSymbol(KCCOUNT, &kcount, sizeof(KCCOUNT));
207 cudaMemcpyToSymbol(PARTSIZE, &partitionSize, sizeof(PARTSIZE));

212 cudaMemcpyToSymbol(CBPSM, &(conc_blocks_per_SM), sizeof(CBPSM));
213
214 while (todo > 0)
215 {
216 CUDA_RUNTIME(cudaGetLastError());
217 cudaDeviceSynchronize();
218
219 //1 bucket fill
220 bucket_scan(nodeDegree, g.numNodes, level, span, current_q, identity_arr_asc, bucket_q, bucket_level_end_);
221 CUDA_RUNTIME(cudaGetLastError());
222 cudaDeviceSynchronize();

224 todo -= current_q.count.gdata()[0];
225 if (current_q.count.gdata()[0] > 0)
226 {
227 auto grid_block_size = current_q.count.gdata()[0];
228 execKernel((kckernel_node_block_warp_binary_count_local_sharedmem_lazy_loop<T, block_size, partitionSize>), grid_block_size, block_size, dev_, false,
229 counter.gdata(),
230 g,
231 current_q.device_queue->gdata()[0],
232 current_level2.gdata(), cpn.gdata(),
233 d_bitmap_states.gdata(), node_be.gdata());
234 }
235 level += span;
236 }
237
238 std::cout.imbue(std::locale(""));
239 std::cout << "Nodes = " << g.numNodes << " Counter = " << counter.gdata()[0] << "\n";
240
241 current_level2.freeGPU();
242 counter.freeGPU();
243 node_be.freeGPU();
244 d_bitmap_states.freeGPU();
245 maxDegree.freeGPU();
246 cpn.copytocpu(0);
247 cpn.freeGPU();
