89__device__ INT_t linear_search_GPU(const UINT_t* list, const UINT_t start, const UINT_t end, const UINT_t target) {
90 for (UINT_t i=start; i<end; i++) {
91 if (list[i] == target) {
92 return i;
93 } else if (list[i] > target) {
94 break;
95 }
96 }
97
98 return -1;
99}

101__device__ INT_t binary_search_GPU(const UINT_t* list, const UINT_t start, const UINT_t end, const UINT_t target) {
102 UINT_t s=start, e=end, mid;
103 while (s < e) {
104 mid = (s + e) >> 1;
105 if (list[mid] == target)
106 return mid;
107
108 if (list[mid] < target)
109 s = mid + 1;
110 else
111 e = mid;
112 }
113 return -1;
114}

117__device__ UINT_t binary_search_closest_ULONG_GPU(const ULONG_t* list, const UINT_t start, const UINT_t end, const ULONG_t target) {
118 /* Finds the index of the rightmost closest value smaller or equal than target, e.g.,
119 * for target 1 and list=[0,0,0,2,2,2] it returns 2,
120 * for target 2 and list=[0,0,0,2,2,2] it returns 5.
121 * Assumes list[0]=0
122 * Assumes end-1 <= UINT_MAX/2
123 */
124
125 UINT_t s=start, e=end, mid;
126 while (s < e) {
127 mid = (s + e) >> 1;
128
129 if (list[mid] < target+1) {
130 s = mid + 1;
131 } else {
132 e = mid;
133 }
134 }
135
136 return max2(start, (s > 0) ? s-1 : 0);
137}

139#if BINSEARCH_CONSTANT
140__device__ UINT_t binary_search_closest_ULONG_constant_GPU(const ULONG_t *list, const UINT_t start, const UINT_t end, const ULONG_t target) {
141 /* Finds the index of the rightmost closest value smaller or equal than target.
142 * Uses constant memory for the first BINSEARCH_CONSTANT_LEVELS levels.
143 */
144 ULONG_t mid;
145
146 UINT_t g_s = start;
147 UINT_t g_e = end;
148 UINT_t g_mid;
149
150 UINT_t c_index = 0;
151
152 #pragma unroll
153 for (UINT_t iter=0; iter<BINSEARCH_CONSTANT_LEVELS; iter++) {
154 mid = c_binary_search_cache[c_index];
155 g_mid = (g_s+g_e) >> 1;
156
157 c_index *= 2;
158 c_index += 1;
159
160 if (mid < target+1) {
161 c_index += 1;
162 g_s = g_mid+1;
163 } else {
164 g_e = g_mid;
165 }
166 }
167
168 g_s = max2(start, (g_s > 0) ? g_s-1 : 0);
169 return binary_search_closest_ULONG_GPU(list, g_s, g_e, target);
170}

173__global__ void tc_GPU_kernel(const UINT_t *g_Ap, const UINT_t *g_Ai, const ULONG_t *g_wedgeSum, const ULONG_t wedgeSum_total, const UINT_t num_vertices, ULONG_t *g_total_count, const UINT_t spread, const UINT_t *g_adjacency_matrix, const UINT_t adjacency_matrix_len, const ULONG_t adjacency_matrix_size) {
174 const ULONG_t i_start = ((ULONG_t) blockIdx.x * blockDim.x + threadIdx.x) * spread;
175
176 extern __shared__ UINT_t sdata[];
177 UINT_t *shared_src = sdata; // Size: blockDim.x X spread
178 UINT_t *shared_dst = &sdata[blockDim.x*spread]; // Size: blockDim.x X spread
179 UINT_t *shared_count = &sdata[2*blockDim.x*spread]; // Size: 1
180
181 UINT_t v;
182 UINT_t w;
183 UINT_t u;
184 UINT_t vb; // Start index of adj(v)
185 UINT_t ve; // End index of adj(v)
186 UINT_t d_v; // Degree of v
187 UINT_t w_i; // Index of w in adj(v)
188 UINT_t u_i; // Index of u in adj(v)
189 UINT_t wedges; // Number of wedges of v
190 UINT_t i_v; // Index of current wedge in wedges(v) (i.e., 0...(d_v*(d_v-1)/2)-1)
191
192 UINT_t s_i = threadIdx.x*spread;
193
194 for (ULONG_t i=i_start; i<min2(i_start+spread, wedgeSum_total); i++, i_v++, s_i++) {
195 if (i == i_start) {
196 /* First wedge. */
197#if BINSEARCH_CONSTANT
198 v = binary_search_closest_ULONG_constant_GPU(g_wedgeSum, 0, num_vertices, i_start);
199#else
200 v = binary_search_closest_ULONG_GPU(g_wedgeSum, 0, num_vertices, i_start);
201#endif
202
203 vb = g_Ap[v];
204 ve = g_Ap[v+1];
205 d_v = ve - vb;
206 wedges = (d_v*(d_v-1)) >> 1;
207 i_v = i_start - g_wedgeSum[v];
208
209 /* Known formulas to create cartesian indices for a (d_v,d_v) (top right) triangular matrix from a linear index. */
210 /* Note: not tested for limits. Uses UINT_t cast instead of floor(). */
211 w_i = d_v - 2 - (UINT_t) (sqrt((double (wedges-i_v) - 0.875) * 2) - 0.5);
212 u_i = i_v + w_i + 1 - wedges + (((d_v-w_i)*((d_v-w_i)-1)) >> 1);
213
214 w = g_Ai[vb + w_i];
215 } else if (i_v >= wedges) {
216 /* Next wedge, new vertex (new v,w,u). */
217 do {
218 v++;
219 vb = ve;
220 ve = g_Ap[v+1];
221 d_v = ve-vb;
222 } while (d_v < 2);
223
224 wedges = (d_v*(d_v-1)) >> 1;
225 i_v = 0;
226 w_i = 0;
227 u_i = 1;
228
229 w = g_Ai[vb];
230 } else {
231 /* Next wedge, same row (new u). */
232 u_i++;
233
234 if (u_i >= d_v) {
235 /* Next wedge, next row (new w,u). */
236 w_i++;
237 u_i = w_i+1;
238 w = g_Ai[vb + w_i];
239 }
240 }
241
242 shared_src[s_i] = w;
243 shared_dst[s_i] = g_Ai[vb + u_i];
244 }
245
246 if (threadIdx.x == 0)
247 *shared_count = 0;
248
249 __syncthreads();
250
251 /* Index into the shared 'transposed' matrix (spread X blockDim.x) */
252 for (s_i=threadIdx.x; s_i<(blockDim.x*spread); s_i+=blockDim.x) {
253 /* Check bounds. */
254#if CHECK_BOUNDS
255 if (s_i >= (wedgeSum_total - (((ULONG_t) blockIdx.x * blockDim.x) * spread)))
256 break;
257#endif
258
259 w = shared_src[s_i];
260 u = shared_dst[s_i];
261
262 if (w >= (max2(num_vertices, adjacency_matrix_len) - adjacency_matrix_len)) {
263 ULONG_t adjacency_i = (adjacency_matrix_size - (((ULONG_t) (num_vertices-w) * (ULONG_t) ((num_vertices-w)-1)) >> 1)) + u - w - 1;
264
265#if UINT_WIDTH == 32
266 bool found = (g_adjacency_matrix[adjacency_i >> 5] & (1 << (adjacency_i & 31))) > 0;
267#else
268 bool found = (g_adjacency_matrix[adjacency_i / UINT_WIDTH] & (1 << (adjacency_i % UINT_WIDTH))) > 0;
269#endif
270 if (found) {
271 atomicAdd_block(shared_count, 1);
272 }
273 } else {
274 UINT_t wb = g_Ap[w];
275 UINT_t we = g_Ap[w+1];
276
277 if (we-wb < 2) {
278 if (linear_search_GPU(g_Ai, wb, we, u) >= 0) {
279 atomicAdd_block(shared_count, 1);
280 }
281 } else {
282 if (binary_search_GPU(g_Ai, wb, we, u) >= 0) {
283 atomicAdd_block(shared_count, 1);
284 }
285 }
286 }
287 }
288
289 __syncthreads();
290
291 if (threadIdx.x == 0)
292 atomicAdd((unsigned long long int *) g_total_count, *shared_count);
293}

315ULONG_t tc_GPU(const GRAPH_TYPE *graph, UINT_t spread, UINT_t adjacency_matrix_len, GPU_time *t) {
316 UINT_t *d_Ap;
317 UINT_t *d_Ai;
318 ULONG_t *d_wedgeSum;
319 UINT_t *d_adjacency_matrix;
320 ULONG_t *d_total_count;
321
322 cudaEvent_t GPU_copy_start, GPU_copy_stop, GPU_exec_start, GPU_exec_stop;
323 float GPU_copy_elapsed, GPU_exec_elapsed;
324 checkCudaErrors(cudaEventCreate(&GPU_copy_start));
325 checkCudaErrors(cudaEventCreate(&GPU_copy_stop));
326 checkCudaErrors(cudaEventCreate(&GPU_exec_start));
327 checkCudaErrors(cudaEventCreate(&GPU_exec_stop));
328
329 /* Store all existing edges of the vertices from n-adjacency_matrix_len to n in an upper triangluar adjacency matrix. */
330 ULONG_t adjacency_matrix_size = (((ULONG_t) adjacency_matrix_len) * ((ULONG_t) adjacency_matrix_len - 1)) / 2;
331 UINT_t *h_adjacency_matrix = (UINT_t *) calloc(adjacency_matrix_size/UINT_WIDTH, sizeof(UINT_t));
332 assert_malloc(h_adjacency_matrix);
333
334 for (UINT_t v=graph->numVertices - min2(graph->numVertices, adjacency_matrix_len); v<graph->numVertices; v++) {
335 for (UINT_t i=graph->rowPtr[v]; i<graph->rowPtr[v+1]; i++) {
336 UINT_t w = graph->colInd[i];
337
338 ULONG_t adjacency_i = (adjacency_matrix_size - (((ULONG_t) (graph->numVertices-v) * (ULONG_t) ((graph->numVertices-v)-1)) / 2)) + w - v - 1;
339
340#if UINT_WIDTH == 32
341 h_adjacency_matrix[adjacency_i >> 5] |= (1 << (adjacency_i & 31));
342#else
343 h_adjacency_matrix[adjacency_i / UINT_WIDTH] |= (1 << (adjacency_i % UINT_WIDTH));
344#endif
345 }
346 }
347
348 ULONG_t *h_wedgeSum = (ULONG_t *) malloc((graph->numVertices + 1) * sizeof(ULONG_t));
349 assert_malloc(h_wedgeSum);
350 h_wedgeSum[0] = 0;
351
352 for (UINT_t v=0; v<graph->numVertices; v++) {
353 UINT_t d_v = graph->rowPtr[(v+1)] - graph->rowPtr[v];
354 if (d_v < 2) {
355 h_wedgeSum[v+1] = h_wedgeSum[v];
356 } else {
357 h_wedgeSum[v+1] = h_wedgeSum[v] + ((d_v*(d_v-1))/2);
358 }
359 }
360
361 ULONG_t wedgeSum_total = h_wedgeSum[graph->numVertices];
362 // printf("wedgeSum_total=%lu\n", wedgeSum_total);
363
364#if BINSEARCH_CONSTANT
365 ULONG_t *h_wedgeSum_cache = (ULONG_t *) malloc(BINSEARCH_CONSTANT_CACHE_SIZE * sizeof(ULONG_t));
366 assert_malloc(h_wedgeSum_cache);
367 build_binary_search_cache(h_wedgeSum, h_wedgeSum_cache, 0, BINSEARCH_CONSTANT_LEVELS, 0, 0, graph->numVertices);
368#endif
369
370 checkCudaErrors(cudaEventRecord(GPU_copy_start));
371
372 checkCudaErrors(cudaMalloc((void **)&d_Ap, (graph->numVertices + 1) * sizeof(UINT_t)));
373 checkCudaErrors(cudaMalloc((void **)&d_Ai, graph->numEdges * sizeof(UINT_t)));
374 checkCudaErrors(cudaMalloc((void **)&d_wedgeSum, (graph->numVertices+1) * sizeof(ULONG_t)));
375 checkCudaErrors(cudaMalloc((void **)&d_adjacency_matrix, (adjacency_matrix_size/32) * sizeof(UINT_t)));
376 checkCudaErrors(cudaMalloc((void **)&d_total_count, 1 * sizeof(ULONG_t)));
377
378 checkCudaErrors(cudaMemcpy(d_Ap, graph->rowPtr, (graph->numVertices + 1) * sizeof(UINT_t), cudaMemcpyHostToDevice));
379 checkCudaErrors(cudaMemcpy(d_Ai, graph->colInd, graph->numEdges * sizeof(UINT_t), cudaMemcpyHostToDevice));
380 checkCudaErrors(cudaMemcpy(d_wedgeSum, h_wedgeSum, (graph->numVertices+1) * sizeof(ULONG_t), cudaMemcpyHostToDevice));
381 checkCudaErrors(cudaMemcpy(d_adjacency_matrix, h_adjacency_matrix, (adjacency_matrix_size/32) * sizeof(UINT_t), cudaMemcpyHostToDevice));
382
383#if BINSEARCH_CONSTANT
384 checkCudaErrors(cudaMemcpyToSymbol(c_binary_search_cache, h_wedgeSum_cache, BINSEARCH_CONSTANT_CACHE_SIZE * sizeof(ULONG_t)));
385#endif
386
387 checkCudaErrors(cudaMemset(d_total_count, 0, 1 * sizeof(ULONG_t)));
388
389 checkCudaErrors(cudaEventRecord(GPU_copy_stop));
390 checkCudaErrors(cudaEventSynchronize(GPU_copy_stop));
391 checkCudaErrors(cudaEventElapsedTime(&GPU_copy_elapsed, GPU_copy_start, GPU_copy_stop));
392 t->copy += GPU_copy_elapsed;
393
394 UINT_t num_threads = 128;
395 ULONG_t num_blocks = (wedgeSum_total / (spread * num_threads)) + 1;
396
397 if (num_blocks > (((ULONG_t) 1 << 31)-1)) {
398 fprintf(stderr, "ERROR: maximum grid size reached.\n");
399 exit(EXIT_FAILURE);
400 }
401
402 dim3 grid(num_blocks, 1, 1);
403 dim3 threads(num_threads, 1, 1);
404
405 int smem_size = (2 * num_threads * spread + 1) * sizeof(UINT_t);
406
407 checkCudaErrors(cudaEventRecord(GPU_exec_start));
408
409 tc_GPU_kernel<<<grid, threads, smem_size>>>(d_Ap, d_Ai, d_wedgeSum, wedgeSum_total, graph->numVertices, d_total_count, spread, d_adjacency_matrix, adjacency_matrix_len, adjacency_matrix_size);
410
411 checkCudaErrors(cudaEventRecord(GPU_exec_stop));
412 checkCudaErrors(cudaEventSynchronize(GPU_exec_stop));
413 checkCudaErrors(cudaEventElapsedTime(&GPU_exec_elapsed, GPU_exec_start, GPU_exec_stop));
414 t->exec += GPU_exec_elapsed;
415
416 ULONG_t h_total_count = 0;
417 checkCudaErrors(cudaMemcpy(&h_total_count, d_total_count, 1 * sizeof(ULONG_t), cudaMemcpyDeviceToHost));
418
419 checkCudaErrors(cudaFree(d_Ap));
420 checkCudaErrors(cudaFree(d_Ai));
421 checkCudaErrors(cudaFree(d_wedgeSum));
422 checkCudaErrors(cudaFree(d_adjacency_matrix));
423 checkCudaErrors(cudaFree(d_total_count));
424
425 checkCudaErrors(cudaEventDestroy(GPU_copy_start));
426 checkCudaErrors(cudaEventDestroy(GPU_copy_stop));
427 checkCudaErrors(cudaEventDestroy(GPU_exec_start));
428 checkCudaErrors(cudaEventDestroy(GPU_exec_stop));
429
430 free(h_wedgeSum);
431 free(h_adjacency_matrix);
432
433#if BINSEARCH_CONSTANT
434 free(h_wedgeSum_cache);
435#endif
436
437#if RESET_DEVICE
438 checkCudaErrors(cudaDeviceReset());
439#endif
440 return h_total_count;
441}

470static int compareEdge_t(const void *a, const void *b) {
471 edge_t arg1 = *(const edge_t *) a;
472 edge_t arg2 = *(const edge_t *) b;
473 if (arg1.src < arg2.src) return -1;
474 if (arg1.src > arg2.src) return 1;
475 if ((arg1.src == arg2.src) && (arg1.dst < arg2.dst)) return -1;
476 if ((arg1.src == arg2.src) && (arg1.dst > arg2.dst)) return 1;
477 return 0;
478}

561GRAPH_TYPE *read_graph(char *filename, bool matrix_market, bool zero_indexed, preprocess_t preprocess_style) {
562 FILE *infile = fopen(filename, "r");
563 if (infile == NULL) {
564 fprintf(stderr, "ERROR: unable to open graph file.\n");
565 usage();
566 }
567
568 GRAPH_TYPE *graph = (GRAPH_TYPE *) malloc(sizeof(GRAPH_TYPE));
569 char line[256];
570
571 /* Skip any header lines */
572 do {
573 if (fgets(line, sizeof(line), infile) == NULL) usage();
574 } while (line[0] < '0' || line[0] > '9');
575
576 /* Skip line if the file is in Matrix Market format. We do not use the given vertex/edge counts. */
577 if (matrix_market) {
578 if (fgets(line, sizeof(line), infile) == NULL) usage();
579 }
580
581 UINT_t vertex_count = 0;
582 UINT_t edge_count = 0;
583 size_t size = 10240;
584 edge_t* edges = (edge_t*) malloc(size * sizeof(edge_t));
585 assert_malloc(edges);
586
587 UINT_t max_vertex = 0;
588 UINT_t v, w;
589
590 if (sscanf(line, "%d %d\n", &v, &w) == 2) {
591 do {
592 if (edge_count >= size) {
593 size += 10240;
594 edge_t *new_edges = (edge_t*) realloc(edges, size * sizeof(edge_t));
595 assert_malloc(new_edges);
596 edges = new_edges;
597 }
598
599 if ((!zero_indexed) && (v == 0 || w == 0)) {
600 fprintf(stderr, "ERROR: zero vertex id detected but -z was not set.\n");
601 usage();
602 }
603
604 v -= (zero_indexed ? 0 : 1);
605 w -= (zero_indexed ? 0 : 1);
606
607 /* Remove self-loops. */
608 if (v != w) {
609 max_vertex = max2(max_vertex, max2(v, w));
610
611 /* v->w */
612 edges[edge_count].src = v;
613 edges[edge_count].dst = w;
614 edge_count++;
615 /* w->v */
616 edges[edge_count].src = w;
617 edges[edge_count].dst = v;
618 edge_count++;
619 }
620 } while (fscanf(infile, "%d %d\n", &v, &w) == 2);
621 }
622
623 fclose(infile);
624
625 vertex_count = max_vertex + 1;
626
627 /* Sort edges (in order to remove duplicates). */
628 if (preprocess_style != PREPROCESS_CPU) {
629 edge_t *d_edges;
630 edge_t *d_edges_alt;
631 edge_t *d_out;
632
633 checkCudaErrors(cudaMalloc((void **)&d_edges, edge_count * sizeof(edge_t)));
634 checkCudaErrors(cudaMalloc((void **)&d_edges_alt, edge_count * sizeof(edge_t)));
635 checkCudaErrors(cudaMemcpy(d_edges, edges, edge_count * sizeof(edge_t), cudaMemcpyHostToDevice));
636
637 if (preprocess_style == PREPROCESS_GPU_CONSTRAINED)
638 d_out = sort_edges_GPU(d_edges, d_edges_alt, edge_count, true);
639 else
640 d_out = sort_edges_GPU(d_edges, d_edges_alt, edge_count, false);
641
642 checkCudaErrors(cudaMemcpy(edges, d_out, edge_count * sizeof(edge_t), cudaMemcpyDeviceToHost));
643 checkCudaErrors(cudaFree(d_edges));
644 checkCudaErrors(cudaFree(d_edges_alt));
645 } else {
646 qsort(edges, edge_count, sizeof(edge_t), compareEdge_t);
647 }
648
649 UINT_t *rowPtr = (UINT_t *) calloc(vertex_count+1, sizeof(UINT_t));
650 assert_malloc(rowPtr);
651
652 UINT_t edge_count_no_dup = 1;
653
654 edge_t lastedge;
655 lastedge.src = edges[0].src;
656 lastedge.dst = edges[0].dst;
657
658 UINT_t *colInd = (UINT_t *) edges; /* colInd overwrites the edges array. Possible because sizeof(edge_t) > sizeof(UINT_t). */
659 colInd[0] = lastedge.dst;
660 rowPtr[lastedge.src + 1]++;
661
662 /* Remove duplicate edges. */
663 for (UINT_t i=1; i<edge_count; i++) {
664 if (compareEdge_t(&lastedge, &edges[i]) != 0) {
665 colInd[edge_count_no_dup++] = edges[i].dst;
666 rowPtr[edges[i].src + 1]++;
667 lastedge.src = edges[i].src;
668 lastedge.dst = edges[i].dst;
669 }
670 }
671
672 /* Free excess memory from the colInd/edges array. */
673 UINT_t *new_colInd = (UINT_t *) realloc(colInd, edge_count_no_dup * sizeof(UINT_t));
674
675 for (UINT_t v=1; v<=vertex_count; v++) {
676 rowPtr[v] += rowPtr[v-1];
677 }
678
679 graph->numVertices = vertex_count;
680 graph->numEdges = edge_count_no_dup;
681 graph->rowPtr = rowPtr;
682 graph->colInd = new_colInd;
683
684 return graph;
685}

687GRAPH_TYPE *preprocess(const GRAPH_TYPE *original_graph, preprocess_t preprocess_style) {
688 preprocess_vertex_t *vertices = (preprocess_vertex_t *) malloc(original_graph->numVertices * sizeof(preprocess_vertex_t));
689 assert_malloc(vertices);
690
691 for (UINT_t v=0; v<original_graph->numVertices; v++) {
692 vertices[v].id = v;
693 vertices[v].edges = &original_graph->colInd[original_graph->rowPtr[v]];
694 vertices[v].num_edges = original_graph->rowPtr[v+1] - original_graph->rowPtr[v];
695 }
696
697 if (preprocess_style != PREPROCESS_CPU) {
698 preprocess_vertex_t *d_vertices;
699 preprocess_vertex_t *d_vertices_alt;
700 preprocess_vertex_t *d_out;
701
702 checkCudaErrors(cudaMalloc((void **)&d_vertices, original_graph->numVertices * sizeof(preprocess_vertex_t)));
703 checkCudaErrors(cudaMalloc((void **)&d_vertices_alt, original_graph->numVertices * sizeof(preprocess_vertex_t)));
704 checkCudaErrors(cudaMemcpy(d_vertices, vertices, original_graph->numVertices * sizeof(preprocess_vertex_t), cudaMemcpyHostToDevice));
705
706 if (preprocess_style == PREPROCESS_GPU_CONSTRAINED)
707 d_out = sort_vertices_GPU(d_vertices, d_vertices_alt, original_graph->numVertices, true);
708 else
709 d_out = sort_vertices_GPU(d_vertices, d_vertices_alt, original_graph->numVertices, false);
710
711 checkCudaErrors(cudaMemcpy(vertices, d_out, original_graph->numVertices * sizeof(preprocess_vertex_t), cudaMemcpyDeviceToHost));
712 checkCudaErrors(cudaFree(d_vertices));
713 checkCudaErrors(cudaFree(d_vertices_alt));
714 } else {
715 qsort(vertices, original_graph->numVertices, sizeof(preprocess_vertex_t), compare_vertex_degree_ascending);
716 }
717
718 UINT_t *reverse = (UINT_t *) malloc(original_graph->numVertices * sizeof(UINT_t));
719 assert_malloc(reverse);
720
721 for (UINT_t v=0; v<original_graph->numVertices; v++) {
722 reverse[vertices[v].id] = v;
723 }
724
725 GRAPH_TYPE *graph = (GRAPH_TYPE *) malloc(sizeof(GRAPH_TYPE));
726 assert_malloc(graph);
727
728 graph->numVertices = original_graph->numVertices;
729 graph->numEdges = original_graph->numEdges/2;
730
731 graph->rowPtr = (UINT_t*) malloc((graph->numVertices + 1) * sizeof(UINT_t));
732 assert_malloc(graph->rowPtr);
733 graph->colInd = (UINT_t*) malloc(graph->numEdges * sizeof(UINT_t));
734 assert_malloc(graph->colInd);
735
736 UINT_t edge_count = 0;
737
738 graph->rowPtr[0] = 0;
739
740 for (UINT_t v=0; v<original_graph->numVertices; v++) {
741 UINT_t new_degree = 0;
742
743 for (INT_t j=0; j<vertices[v].num_edges; j++) {
744 UINT_t w = vertices[v].edges[j];
745 UINT_t w_new = reverse[w];
746
747 if (w_new > v) {
748 graph->colInd[edge_count++] = w_new;
749 new_degree++;
750 }
751 }
752
753 graph->rowPtr[v+1] = graph->rowPtr[v] + new_degree;
754
755 if (preprocess_style == PREPROCESS_CPU) {
756 qsort(&graph->colInd[graph->rowPtr[v]], new_degree, sizeof(UINT_t), compareInt_t);
757 }
758 }
759
760 free(vertices);
761 free(reverse);
762
763 if (preprocess_style != PREPROCESS_CPU) {
764 UINT_t *d_rowPtr;
765 UINT_t *d_colInd;
766 UINT_t *d_colInd_alt;
767 UINT_t *d_colInd_out;
768
769 checkCudaErrors(cudaMalloc((void **)&d_rowPtr, (graph->numVertices+1) * sizeof(UINT_t)));
770 checkCudaErrors(cudaMalloc((void **)&d_colInd, graph->numEdges * sizeof(UINT_t)));
771 checkCudaErrors(cudaMalloc((void **)&d_colInd_alt, graph->numEdges * sizeof(UINT_t)));
772 checkCudaErrors(cudaMemcpy(d_rowPtr, graph->rowPtr, (graph->numVertices+1) * sizeof(UINT_t), cudaMemcpyHostToDevice));
773 checkCudaErrors(cudaMemcpy(d_colInd, graph->colInd, graph->numEdges * sizeof(UINT_t), cudaMemcpyHostToDevice));
774
775 if (preprocess_style == PREPROCESS_GPU_CONSTRAINED)
776 d_colInd_out = sort_colInd_GPU(d_rowPtr, d_colInd, d_colInd_alt, graph->numVertices, graph->numEdges, true);
777 else
778 d_colInd_out = sort_colInd_GPU(d_rowPtr, d_colInd, d_colInd_alt, graph->numVertices, graph->numEdges, false);
779
780 checkCudaErrors(cudaMemcpy(graph->colInd, d_colInd_out, graph->numEdges * sizeof(UINT_t), cudaMemcpyDeviceToHost));
781 checkCudaErrors(cudaFree(d_rowPtr));
782 checkCudaErrors(cudaFree(d_colInd));
783 checkCudaErrors(cudaFree(d_colInd_alt));
784 }
785
786 return graph;
787}
