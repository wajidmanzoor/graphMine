16void MotifSolver(Graph &g, int k, std::vector<uint64_t> &total, int, int) {
17 int num_threads = 1;
18 #pragma omp parallel
19 {
20 num_threads = omp_get_num_threads();
21 }
22 std::cout << "OpenMP Motif solver (" << num_threads << " threads) ...\n";

24 int num_patterns = num_possible_patterns[k];
25 std::vector<std::vector<uint64_t>> global_counters(num_threads);
26 std::vector<EmbList> emb_lists(num_threads);
27 std::vector<std::vector<uint8_t>> ccodes(num_threads);
28 auto max_degree = g.get_max_degree();
29 for (int tid = 0; tid < num_threads; tid ++) {
30 auto &local_counters = global_counters[tid];
31 local_counters.resize(num_patterns);
32 std::fill(local_counters.begin(), local_counters.end(), 0);
33 auto &local_ccodes = ccodes[tid];
34 local_ccodes.resize(g.size()); // the connectivity code
35 std::fill(local_ccodes.begin(), local_ccodes.end(), 0);
36 auto &emb_list = emb_lists[tid];
37 emb_list.init(k, max_degree, num_patterns);
38 }
39
40 Timer t;
41 t.Start();
42 //kmotif(g, k, global_counters, emb_lists, ccodes);
43 ccode_kmotif(g, k, global_counters, ccodes);
44 for (int tid = 0; tid < num_threads; tid++)
45 for (int pid = 0; pid < num_patterns; pid++)
46 total[pid] += global_counters[tid][pid];
47 t.Stop();
48 std::cout << "runtime [omp_cmap] = " << t.Seconds() << "\n";

51void extend_motif(unsigned level, unsigned k, Graph &g, EmbList &emb_list,
52 std::vector<uint8_t> &ccodes, std::vector<uint64_t> &counter) {
53 if (level == k - 2) {
54 for (int pcode = 0; pcode < num_possible_patterns[level+1]; pcode++) {
55 for (vidType emb_id = 0; emb_id < emb_list.size(level, pcode); emb_id++) {
56 auto v = emb_list.get_vertex(level, emb_id, pcode);
57 emb_list.push_history(v);
58 update_ccodes(level, g, v, ccodes);
59 uint8_t src_idx = 0;
60 if (k > 3) src_idx = emb_list.get_src(level, emb_id, pcode);
61 for (unsigned id = 0; id < level+1; id++) {
62 auto src = emb_list.get_history(id);
63 auto begin = g.edge_begin(src);
64 auto end = g.edge_end(src);
65 for (auto e = begin; e < end; e++) {
66 auto dst = g.getEdgeDst(e);
67 auto emb_ptr = emb_list.get_history_ptr();
68 if (do_early_break(k, level, dst, id, emb_ptr)) break;
69 auto ccode = ccodes[dst];
70 if (is_canonical(k, level, dst, id, ccode, emb_ptr)) {
71 auto pid = get_pattern_id(level, ccode, pcode, src_idx);
72 counter[pid] ++;
73 }
74 }
75 }
76 resume_ccodes(level, g, v, ccodes);
77 emb_list.pop_history();
78 }
79 }
80 return;
81 }
82 for (int pcode = 0; pcode < num_possible_patterns[level+1]; pcode++) {
83 for (vidType emb_id = 0; emb_id < emb_list.size(level, pcode); emb_id++) {
84 auto v = emb_list.get_vertex(level, emb_id, pcode);
85 emb_list.push_history(v);
86 update_ccodes(level, g, v, ccodes);
87 emb_list.clear_size(level+1);
88 for (unsigned idx = 0; idx < level+1; idx++) {
89 auto src = emb_list.get_history(idx);
90 auto begin = g.edge_begin(src);
91 auto end = g.edge_end(src);
92 for (auto e = begin; e < end; e++) {
93 auto dst = g.getEdgeDst(e);
94 auto emb_ptr = emb_list.get_history_ptr();
95 if (do_early_break(k, level, dst, idx, emb_ptr)) break;
96 uint8_t pcode = 0;
97 auto ccode = ccodes[dst];
98 if (is_canonical(k, level, dst, idx, ccode, emb_ptr)) {
99 auto pid = get_pattern_id(level, ccode, pcode, idx);
100 emb_list.add_emb(level+1, dst, pid, idx);
101 }
102 }
103 }
104 extend_motif(level+1, k, g, emb_list, ccodes, counter);

111void kmotif(Graph &g, unsigned k, std::vector<std::vector<uint64_t>> &counters,
112 std::vector<EmbList> &emb_lists,
113 std::vector<std::vector<uint8_t>> &ccodes) {
114 #pragma omp parallel
115 {
116 auto tid = omp_get_thread_num();
117 auto &counter = counters.at(tid);
118 auto &local_ccodes = ccodes[tid];
119 auto &emb_list = emb_lists[tid];
120 #pragma omp for schedule(dynamic, 1) nowait
121 for (vidType v = 0; v < g.size(); v ++) {
122 emb_list.clear_size(1);
123 auto begin = g.edge_begin(v);
124 auto end = g.edge_end(v);
125 for (auto e = begin; e < end; e++) {
126 auto dst = g.getEdgeDst(e);
127 local_ccodes[dst] = 1;
128 if (dst < v) emb_list.add_emb(1, dst);
129 }
130 emb_list.push_history(v);
131 extend_motif(1, k, g, emb_list, local_ccodes, counter);
132 for (auto e = begin; e < end; e++) {
133 auto dst = g.getEdgeDst(e);
134 local_ccodes[dst] = 0;
135 }
136 emb_list.pop_history();
137 }
138 }
139}

141static inline bool do_early_break(unsigned k, unsigned level, vidType dst, unsigned src_idx, const VertexList *emb) {
142 if (k == 3) {
143 if (src_idx == 0 && dst >= (*emb)[1]) return true;
144 if (src_idx == 1 && dst >= (*emb)[0]) return true;
145 } else {
146 if (dst >= (*emb)[0]) return true;

149 }

153static inline bool is_connected(unsigned level, unsigned id, uint8_t ccode) {
154 if (level == 1) {
155 return (ccode == id + 1) || (ccode == 3);
156 } else if (level == 2) {
157 return (ccode & (1 << id));
158 } else {
159 return false;
160 }

163static inline bool is_canonical(unsigned k, unsigned level, vidType dst, unsigned src_idx, uint8_t ccode, const VertexList *emb) {
164 if (k == 3) {
165 if (src_idx == 1 && ccode == 3) return false;
166 } else {
167 for (unsigned i = 1; i < level+1; ++i)
168 if (dst == (*emb)[i]) return false;
169 for (unsigned i = 0; i < src_idx; ++i)
170 if (is_connected(level, i, ccode))
171 return false;
172 }
173 return true;
174}

176static inline unsigned get_pattern_id(unsigned level, uint8_t ccode,
177 unsigned pcode, unsigned src_idx) {
178 unsigned pid = 0;
179 if (level == 1) { // count 3-motifs
180 if (ccode == 3) {
181 pid = 0; // triangle
182 } else {
183 pid = 1; // wedge
184 }
185 } else if (level == 2) { // count 4-motifs
186 if (pcode == 0) { // extending a triangle
187 if (ccode == 7) {
188 pid = 5; // clique
189 } else if (ccode == 3 || ccode == 5 || ccode == 6) {
190 pid = 4; // diamond
191 } else
192 pid = 3; // tailed-triangle
193 } else {
194 if (ccode == 7) {
195 pid = 4; // diamond
196 } else if (src_idx == 0) {
197 if (ccode == 6)
198 pid = 2; // 4-cycle
199 else if (ccode == 3 || ccode == 5)
200 pid = 3; // tailed-triangle
201 else if (ccode == 1)
202 pid = 1; // 3-star
203 else
204 pid = 0; // 4-chain
205 } else {
206 if (ccode == 5)
207 pid = 2; // 4-cycle
208 else if (ccode == 3 || ccode == 6)
209 pid = 3; // tailed-triangle
210 else if (ccode == 2)
211 pid = 1; // 3-star
212 else
213 pid = 0; // 4-chain
214 }
215 }
216 }
217 return pid;
