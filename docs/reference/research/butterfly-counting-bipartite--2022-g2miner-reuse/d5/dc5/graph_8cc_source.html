256void Graph::orientation() {
257 std::cout << "Orientation enabled, using DAG\n";
258 Timer t;
259 t.Start();
260 std::vector<vidType> degrees(n_vertices, 0);
261 #pragma omp parallel for
262 for (vidType v = 0; v < n_vertices; v++) {
263 degrees[v] = get_degree(v);
264 }
265 std::vector<vidType> new_degrees(n_vertices, 0);
266 #pragma omp parallel for
267 for (vidType src = 0; src < n_vertices; src ++) {
268 for (auto dst : N(src)) {
269 if (degrees[dst] > degrees[src] ||
270 (degrees[dst] == degrees[src] && dst > src)) {
271 new_degrees[src]++;
272 }
273 }
274 }
275 max_degree = *(std::max_element(new_degrees.begin(), new_degrees.end()));
276 eidType *old_vertices = vertices;
277 vidType *old_edges = edges;
278 eidType *new_vertices = custom_alloc_global<eidType>(n_vertices+1);
279 //prefix_sum<vidType,eidType>(new_degrees, new_vertices);
280 parallel_prefix_sum<vidType,eidType>(new_degrees, new_vertices);
281 auto num_edges = new_vertices[n_vertices];
282 vidType *new_edges = custom_alloc_global<vidType>(num_edges);
283 #pragma omp parallel for
284 for (vidType src = 0; src < n_vertices; src ++) {
285 auto begin = new_vertices[src];
286 eidType offset = 0;
287 for (auto dst : N(src)) {
288 if (degrees[dst] > degrees[src] ||
289 (degrees[dst] == degrees[src] && dst > src)) {
290 new_edges[begin+offset] = dst;
291 offset ++;
292 }
293 }
294 }
295 vertices = new_vertices;
296 edges = new_edges;
297 custom_free<eidType>(old_vertices, n_vertices);
298 custom_free<vidType>(old_edges, n_edges);
299 n_edges = num_edges;
300 t.Stop();
301 std::cout << "Time on generating the DAG: " << t.Seconds() << " sec\n";

320eidType Graph::init_edgelist(bool sym_break, bool ascend) {
321 Timer t;
322 t.Start();
323 if (nnz != 0) return nnz; // already initialized
324 nnz = E();
325 if (sym_break) nnz = nnz/2;
326 sizes.resize(V());
327 src_list = new vidType[nnz];
328 if (sym_break) dst_list = new vidType[nnz];
329 else dst_list = edges;
330 size_t i = 0;
331 for (vidType v = 0; v < V(); v ++) {
332 for (auto u : N(v)) {
333 if (u == v) continue; // no selfloops
334 if (ascend) {
335 if (sym_break && v > u) continue;
336 } else {
337 if (sym_break && v < u) break;
338 }
339 src_list[i] = v;
340 if (sym_break) dst_list[i] = u;
341 sizes[v] ++;
342 i ++;
343 }
344 }
345 //assert(i == nnz);
346 t.Stop();
347 std::cout << "Time on generating the edgelist: " << t.Seconds() << " sec\n";
348 return nnz;
349}

367bool Graph::binary_search(vidType key, eidType begin, eidType end) const {
368 auto l = begin;
369 auto r = end-1;
370 while (r >= l) {
371 auto mid = l + (r - l) / 2;
372 auto value = getEdgeDst(mid);
373 if (value == key) return true;
374 if (value < key) l = mid + 1;
375 else r = mid - 1;
376 }
377 return false;
378}

702void Graph::computeKCore() {
703 int nv = size();
704 int md = get_max_degree();
705 std::vector<int> vertices(nv); // Vertices sorted by degree.
706 std::vector<int> position(nv); // The position of vertices in vertices array.
707 std::vector<int> degree_bin(md+1, 0); // Degree from 0 to max_degree.
708 std::vector<int> offset(md+1); // The offset in vertices array according to degree.
709 for (int i = 0; i < nv; ++i) {
710 int degree = get_degree(i);
711 core_table[i] = degree;
712 degree_bin[degree] += 1;
713 }
714 int start = 0;
715 for (int i = 0; i < md+1; ++i) {
716 offset[i] = start;
717 start += degree_bin[i];
718 }
719 for (int i = 0; i < nv; ++i) {
720 int degree = get_degree(i);
721 position[i] = offset[degree];
722 vertices[position[i]] = i;
723 offset[degree] += 1;
724 }
725 for (int i = md; i > 0; --i) {
726 offset[i] = offset[i - 1];
727 }
728 offset[0] = 0;
729 for (int i = 0; i < nv; ++i) {
730 int v = vertices[i];
731 for(int j = 0; j < get_degree(v); ++j) {
732 int u = N(v, j);
733 if (core_table[u] > core_table[v]) {
734 // Get the position and vertex which is with the same degree
735 // and at the start position of vertices array.
736 int cur_degree_u = core_table[u];
737 int position_u = position[u];
738 int position_w = offset[cur_degree_u];
739 int w = vertices[position_w];
740 if (u != w) {
741 // Swap u and w.
742 position[u] = position_w;
743 position[w] = position_u;
744 vertices[position_u] = w;
745 vertices[position_w] = u;
746 }
747 offset[cur_degree_u] += 1;
748 core_table[u] -= 1;
749 }
750 }
751 }
