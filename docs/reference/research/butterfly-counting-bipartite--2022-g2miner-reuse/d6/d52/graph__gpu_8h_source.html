226 inline __device__ vidType warp_intersect(vidType src, vidType dst) {
227 int thread_lane = threadIdx.x & (WARP_SIZE-1); // thread index within the warp
228 vidType count = 0;
229 assert(src != dst);
230 vidType src_size = getOutDegree(src);
231 vidType dst_size = getOutDegree(dst);
232 if (src_size == 0 || dst_size == 0) return 0;
233 vidType* lookup = d_colidx + edge_begin(src);
234 vidType* search = d_colidx + edge_begin(dst);
235 vidType lookup_size = src_size;
236 vidType search_size = dst_size;
237 if (src_size > dst_size) {
238 auto temp = lookup;
239 lookup = search;
240 search = temp;
241 search_size = src_size;
242 lookup_size = dst_size;
243 }
244 for (vidType i = thread_lane; i < lookup_size; i += WARP_SIZE) {
245 auto key = lookup[i];
246 if (binary_search(search, key, search_size))
247 count += 1;
248 }
249 return count;
250 }
