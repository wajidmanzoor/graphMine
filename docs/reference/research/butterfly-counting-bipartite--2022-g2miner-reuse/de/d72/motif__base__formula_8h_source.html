6void automine_3motif(Graph &g, std::vector<std::vector<uint64_t>> &global_counters) {
7 #pragma omp parallel
8 {
9 auto &counter = global_counters.at(omp_get_thread_num());
10 #pragma omp for schedule(dynamic,1) nowait
11 for(vidType v0 = 0; v0 < g.V(); v0++) {
12 VertexSet y0 = g.N(v0);
13 uint64_t n = (uint64_t)y0.size();
14 counter[0] += n * (n - 1);
15 VertexSet y0f0 = bounded(y0,v0);

17 vidType v1 = y0f0.begin()[idx1];

25void automine_4motif(Graph &g, std::vector<std::vector<uint64_t>> &global_counters) {
26 #pragma omp parallel
27 {
28 auto &counter = global_counters.at(omp_get_thread_num());
29 #pragma omp for schedule(dynamic,1) nowait
30 for(vidType v0 = 0; v0 < g.V(); v0++) {
31 VertexSet y0 = g.N(v0);
32 VertexSet y0f0 = bounded(y0,v0);
33 for(vidType idx1 = 0; idx1 < y0f0.size(); idx1++) {
34 vidType v1 = y0f0.begin()[idx1];
35 VertexSet y1 = g.N(v1);
36 uint64_t tri = intersection_num(y0, y1);
37 counter[4] += tri * (tri - 1);
38 uint64_t staru = y0.size() - tri - 1;
39 uint64_t starv = y1.size() - tri - 1;
40 counter[2] += tri * (staru + starv);
41 counter[1] += staru * starv;
42 counter[0] += staru * (staru - 1);
43 counter[0] += starv * (starv - 1);
44 VertexSet y0f0y1f1 = intersection_set(y0f0, y1, v1);
45 VertexSet n0f0y1; difference_set(n0f0y1,y1, y0);
46 VertexSet y0f0n1f1 = difference_set(y0f0, y1, v1);
47 for(vidType idx2 = 0; idx2 < y0f0y1f1.size(); idx2++) {
48 vidType v2 = y0f0y1f1.begin()[idx2];
49 VertexSet y2 = g.N(v2);
50 counter[5] += intersection_num(y0f0y1f1, y2, v2);
51 }
52 for(vidType idx2 = 0; idx2 < y0f0n1f1.size(); idx2++) {
53 vidType v2 = y0f0n1f1.begin()[idx2];
54 VertexSet y2 = g.N(v2);
55 counter[3] += intersection_num(n0f0y1, y2, v0);
56 }
57 }
58 }
59 }
60}
