255 inline bool is_vertexInduced_automorphism(const VertexEmbedding& emb, unsigned idx, vidType src, vidType dst) {
256 unsigned n = emb.size();
257 // the new vertex id should be larger than the first vertex id
258 if (dst <= emb.get_vertex(0)) return true;
259 // the new vertex should not already exist in the embedding
260 for (unsigned i = 1; i < n; ++i)
261 if (dst == emb.get_vertex(i)) return true;
262 // the new vertex should not already be extended by any previous vertex in the embedding
263 for (unsigned i = 0; i < idx; ++i)
264 if (is_connected(emb.get_vertex(i), dst)) return true;
265 // the new vertex id should be larger than any vertex id after its source vertex in the embedding
266 for (unsigned i = idx+1; i < n; ++i)
267 if (dst < emb.get_vertex(i)) return true;
268 return false;
269 }

271 inline unsigned find_motif_pattern_id(unsigned n, unsigned idx, vidType dst, const VertexEmbedding& emb, unsigned pos = 0) {
272 unsigned pid = 0;
273 if (n == 2) { // count 3-motifs
274 pid = 1; // 3-chain
275 if (idx == 0) {
276 if (is_connected(emb.get_vertex(1), dst)) pid = 0; // triangle
277#ifdef USE_WEDGE
278 else if (max_size == 4) is_wedge[pos] = 1; // wedge; used for 4-motif
279#endif
280 }
281 } else if (n == 3) { // count 4-motifs
282 unsigned num_edges = 1;
283 pid = emb.get_pid();
284 if (pid == 0) { // extending a triangle
285 for (unsigned j = idx+1; j < n; j ++)
286 if (is_connected(emb.get_vertex(j), dst)) num_edges ++;
287 pid = num_edges + 2; // p3: tailed-triangle; p4: diamond; p5: 4-clique
288 } else { // extending a 3-chain
289 assert(pid == 1);
290 std::vector<bool> connected(3, false);
291 connected[idx] = true;
292 for (unsigned j = idx+1; j < n; j ++) {
293 if (is_connected(emb.get_vertex(j), dst)) {
294 num_edges ++;
295 connected[j] = true;
296 }
297 }
298 if (num_edges == 1) {
299 pid = 0; // p0: 3-path
300 unsigned center = 1;
301#ifdef USE_WEDGE
302 if (is_wedge[pos]) center = 0;
303#else
304 center = is_connected(emb.get_vertex(1), emb.get_vertex(2)) ? 1 : 0;
305#endif
306 if (idx == center) pid = 1; // p1: 3-star
307 } else if (num_edges == 2) {
308 pid = 2; // p2: 4-cycle
309 unsigned center = 1;
310#ifdef USE_WEDGE
311 if (is_wedge[pos]) center = 0;
312#else
313 center = is_connected(emb.get_vertex(1), emb.get_vertex(2)) ? 1 : 0;
314#endif
315 if (connected[center]) pid = 3; // p3: tailed-triangle
316 } else {
317 pid = 4; // p4: diamond
318 }
319 }
320 } else { // count 5-motif and beyond
321 std::vector<bool> connected;
322 get_connectivity(n, idx, dst, emb, connected);
323 Matrix A(n+1, std::vector<MatType>(n+1, 0));
324 gen_adj_matrix(n+1, connected, A);
325 std::vector<MatType> c(n+1, 0);
326 char_polynomial(n+1, A, c);
327 bliss::UintSeqHash h;
328 for (unsigned i = 0; i < n+1; ++i)
329 h.update((unsigned)c[i]);
330 pid = h.get_value();
331 }
332 return pid;
333 }
