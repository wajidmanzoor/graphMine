25 __global__ void getNodeDegree_kernel(T *node_degree, graph::COOCSRGraph_d<T> g, T *max_degree)
26 {
27 T gtid = threadIdx.x + blockIdx.x * blockDim.x;
28 typedef cub::BlockReduce<T, BLOCK_DIM_X> BlockReduce;
29 __shared__ typename BlockReduce::TempStorage temp_storage;
30 T degree = 0;
31 if (gtid < g.numNodes)
32 {
33 degree = g.rowPtr[gtid + 1] - g.rowPtr[gtid];
34 node_degree[gtid] = degree;
35 }
36
37 T aggregate = BlockReduce(temp_storage).Reduce(degree, cub::Max());
38 if (threadIdx.x == 0)
39 atomicMax(max_degree, aggregate);
40 }

643 {
644 for (T j = i + 1; j < Cmax_size && is_clique; j++)
645 {

648 for (T k = 0, index = gsplit.colInd[gsplit.rowPtr[src1]];
649 k < gsplit.rowPtr[src1 + 1] - gsplit.rowPtr[src1];
650 index = gsplit.colInd[gsplit.rowPtr[src1] + ++k])
651 {
652 if (index == src2)
653 {
654 connected = true;

656 }
657 }
658
659 if (!connected)
660 {
661 printf("%u and %u not connected.\n", src1, src2);
662 is_clique = false;
663 }
664
665 }
666 }
667
668 if (!is_clique)
669 printf("Not a Clique.\n");
670
671 free(Max_Clique);
672 }
