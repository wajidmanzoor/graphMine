51static std::vector<int> parseDevice(const string &s)
52{
53 std::vector<int> res;
54 const int lim = s.size();
55 for (int i = 0; i < lim; i++)
56 {
57 if (!isdigit(s[i]))
58 continue;
59 int cur = s[i] - '0', j = i;
60 while (j + 1 < lim && isdigit(s[j + 1]))
61 cur = cur * 10 + s[++j] - '0';
62 res.emplace_back(cur);
63 i = j + 1;
64 }
65 return res;
66}
