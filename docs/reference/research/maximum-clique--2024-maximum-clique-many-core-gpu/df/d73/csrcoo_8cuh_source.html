42 static CSRCOO<Index, Vector> from_edgelist(const vector<edge_type> &es, bool (*edgeFilter)(const edge_type &) = nullptr)
43 {
44 CSRCOO csr;
45
46 if (es.size() == 0)
47 {
48 Log(warn, "constructing from empty edge list");
49 return csr;
50 }
51
52 for (const auto &edge : es)
53 {
54 const Index src = static_cast<Index>(edge.first);
55 const Index dst = static_cast<Index>(edge.second);
56
57 // edge has a new src and should be in a new row
58 // even if the edge is filtered out, we need to add empty rows
59 while (csr.rowPtr_.size() != size_t(src + 1))
60 {
61 // expecting inputs to be sorted by src, so it should be at least
62 // as big as the current largest row we have recorded
63 assert(src >= csr.rowPtr_.size());
64 csr.rowPtr_.push_back(csr.colInd_.size());
65 }
66
67 // filter or add the edge
68 if (nullptr != edgeFilter && edgeFilter(edge))
69 {
70 continue;
71 }
72 else
73 {
74 csr.rowInd_.push_back(src);
75 csr.colInd_.push_back(dst);
76 }
77 }
78
79 // add the final length of the non-zeros to the offset array
80 csr.rowPtr_.push_back(csr.colInd_.size());
81 return csr;
82 }
