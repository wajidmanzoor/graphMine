51static __global__
52void globalInit(const ECLgraph g, int2* const __restrict__ wl1, int* const __restrict__ wl1size, int2* const __restrict__ iomax)
53{
54 const int thread = threadIdx.x + blockIdx.x * ThreadsPerBlock;
55 const int threads = gridDim.x * ThreadsPerBlock;
56 for (int v = thread; v < g.nodes; v += threads) {
57 const int beg = g.nindex[v];
58 const int end = g.nindex[v + 1];
59 int y = v;
60 for (int j = beg; j < end; j++) {
61 const int w = g.nlist[j];
62 wl1[j] = int2{v, w};
63 y = max(y, w);
64 }
65 iomax[v] = int2{v, y};
66 }
67 if (thread == 0) *wl1size = g.edges;
68}

71static __global__
72void propagateMax(const int2* const __restrict__ wl1, const int wl1size, int2* const __restrict__ iomax, volatile bool* const __restrict__ goagain)
73{
74 const int thread = threadIdx.x + blockIdx.x * ThreadsPerBlock;
75 const int threads = gridDim.x * ThreadsPerBlock;
76 bool updated, again = false;
77 do { // iterate locally
78 updated = false;
79 for (int i = thread; i < wl1size; i += threads) {
80 const int2 el = wl1[i];
81 const int v = el.x;
82 const int w = el.y;
83
84 const int2 iov = iomax[v];
85 const int2 iow = iomax[w];
86 int im = iov.x;
87 int om = iow.y;
88 if (im > v) im = iomax[im].x; // 'path compress'
89 if (om > w) om = iomax[om].y; // 'path compress'
90
91 // propagate
92 if (iov.x < im) {iomax[v].x = im; updated = true;}
93 if (iov.y < om) {iomax[v].y = om; updated = true;}
94 if (iow.x < im) {iomax[w].x = im; updated = true;}
95 if (iow.y < om) {iomax[w].y = om; updated = true;}
96
97 // update other vertices on path
98 if ((iov.x < om) && (iomax[iov.x].y < om)) {iomax[iov.x].y = om; updated = true;}
99 if ((iov.x != iow.x) && (iow.x < om) && (iomax[iow.x].y < om)) {iomax[iow.x].y = om; updated = true;}
100 if ((iov.y < im) && (iomax[iov.y].x < im)) {iomax[iov.y].x = im; updated = true;}
101 if ((iov.y != iow.y) && (iow.y < im) && (iomax[iow.y].x < im)) {iomax[iow.y].x = im; updated = true;}
102 }
103 again |= updated;
104 } while (__syncthreads_or(updated));
105 again = __syncthreads_or(again);
106 if ((threadIdx.x == 0) && again) *goagain = true;
107}

110static __global__
111void removeEdges(const int2* const __restrict__ wl1, const int wl1size, int2* const __restrict__ wl2, int* const __restrict__ wl2size, const int2* const __restrict__ iomax)
112{
113 const int thread = threadIdx.x + blockIdx.x * ThreadsPerBlock;
114 const int threads = gridDim.x * ThreadsPerBlock;
115 for (int i = thread; i < wl1size; i += threads) {
116 const int2 el = wl1[i];
117 const int v = el.x;
118 const int w = el.y;
119 const int2 iov = iomax[v];
120 const int2 iow = iomax[w];
121 if (iov.x != iov.y) {
122 if ((iow.x == iov.x) && (iow.y == iov.y)) {
123 const int k = atomicAdd(wl2size, 1);
124 wl2[k] = el;
125 }
126 }
127 }
128}

131static __global__
132void localInit(const int nodes, int2* const __restrict__ iomax, volatile bool* const __restrict__ goagain)
133{
134 const int thread = threadIdx.x + blockIdx.x * ThreadsPerBlock;
135 const int threads = gridDim.x * ThreadsPerBlock;
136 bool again = false;
137 for (int v = thread; v < nodes; v += threads) {
138 const int2 iov = iomax[v];
139 if (iov.x != iov.y) {
140 iomax[v] = int2{v, v};
141 again = true;
142 }
143 }
144 again = __syncthreads_or(again);
145 if ((threadIdx.x == 0) && again) *goagain = true;
146}

170int main(int argc, char* argv [])
171{
172 printf("ECL-SCC v1.0\n\n"); fflush(stdout);
173
174 // process command line
175 if (argc != 2) {printf("USAGE: %s input_file_name\n\n", argv[0]); exit(-1);}
176
177 ECLgraph g = readECLgraph(argv[1]);
178 printf("input: %s\n", argv[1]);
179 printf("nodes: %d\n", g.nodes);
180 printf("edges: %d\n", g.edges);
181 printf("avg degree: %.2f\n\n", 1.0 * g.edges / g.nodes);
182
183 // get GPU info
184 cudaSetDevice(Device);
185 cudaDeviceProp deviceProp;
186 cudaGetDeviceProperties(&deviceProp, Device);
187 if ((deviceProp.major == 9999) && (deviceProp.minor == 9999)) {printf("ERROR: there is no CUDA capable device\n\n"); exit(-1);}
188 const int SMs = deviceProp.multiProcessorCount;
189 const int mTpSM = deviceProp.maxThreadsPerMultiProcessor;
190 printf("GPU: %s with %d SMs and %d mTpSM (%.1f MHz and %.1f MHz)\n\n", deviceProp.name, SMs, mTpSM, deviceProp.clockRate * 0.001, deviceProp.memoryClockRate * 0.001); fflush(stdout);
191 const int blocks = SMs * (mTpSM / ThreadsPerBlock);
192
193 // allocate GPU memory
194 int2 *d_wl1, *d_wl2, *d_iomax;
195 int *d_wl2size;
196 bool *d_goagain;
197 cudaMalloc((void **)&d_iomax, g.nodes * sizeof(int2));
198 cudaMalloc((void **)&d_wl1, g.edges * sizeof(int2));
199 cudaMalloc((void **)&d_wl2, g.edges * sizeof(int2));
200 cudaMalloc((void **)&d_wl2size, sizeof(int));
201 cudaMalloc((void **)&d_goagain, sizeof(bool));
202
203 // copy graph to GPU
204 ECLgraph d_g = g;
205 cudaMalloc((void **)&d_g.nindex, (g.nodes + 1) * sizeof(int));
206 cudaMemcpy(d_g.nindex, g.nindex, (g.nodes + 1) * sizeof(int), cudaMemcpyHostToDevice);
207 cudaMalloc((void **)&d_g.nlist, g.edges * sizeof(int));
208 cudaMemcpy(d_g.nlist, g.nlist, g.edges * sizeof(int), cudaMemcpyHostToDevice);
209 CheckCuda(__LINE__);
210
211/*************************************************************************/
212
213 // start time
214 printf("running algorithm\n"); fflush(stdout);
215 GPUTimer timer;
216 timer.start();
217
218 // global initialization
219 int wl1size;
220 globalInit<<<blocks, ThreadsPerBlock>>>(d_g, d_wl1, d_wl2size, d_iomax);
221 cudaMemcpy(&wl1size, d_wl2size, sizeof(int), cudaMemcpyDeviceToHost);
222
223 bool goagain;
224 do {
225 // propagate max values
226 do {
227 cudaMemsetAsync(d_goagain, 0, sizeof(bool));
228 propagateMax<<<blocks, ThreadsPerBlock>>>(d_wl1, wl1size, d_iomax, d_goagain);
229 cudaMemcpy(&goagain, d_goagain, sizeof(bool), cudaMemcpyDeviceToHost);
230 } while (goagain);
231
232 // remove edges that cannot be part of an SCC
233 cudaMemsetAsync(d_wl2size, 0, sizeof(int));
234 removeEdges<<<blocks, ThreadsPerBlock>>>(d_wl1, wl1size, d_wl2, d_wl2size, d_iomax);
235 std::swap(d_wl1, d_wl2);
236 cudaMemcpyAsync(&wl1size, d_wl2size, sizeof(int), cudaMemcpyDeviceToHost);
237
238 // local re-initialization
239 cudaMemsetAsync(d_goagain, 0, sizeof(bool));
240 localInit<<<blocks, ThreadsPerBlock>>>(g.nodes, d_iomax, d_goagain);
241 cudaMemcpy(&goagain, d_goagain, sizeof(bool), cudaMemcpyDeviceToHost);
242 } while (goagain);
243
244 // end time
245 const double runtime = timer.stop();
246 printf("compute time: %.6f s\n\n", runtime);
247 CheckCuda(__LINE__);
248
249/*************************************************************************/
250
251 // copy result to host
252 const int n = g.nodes;
253 int2* iomax = new int2 [n];
254 cudaMemcpy(iomax, d_iomax, n * sizeof(int2), cudaMemcpyDeviceToHost);
255 CheckCuda(__LINE__);
256
257 // output SCC sizes and frequency
258 printf("result:\n");
259 int* count = new int [n];
260 for (int v = 0; v < n; v++) count[v] = 0;
261 for (int v = 0; v < n; v++) count[iomax[v].x]++;
262 std::sort(count, count + n, std::greater<int>());
263 int cnt = 1;
264 int SCC_cnt = 0;
265 for (int v = 1; v < n; v++) {
266 if (count[v] != count[v - 1]) {
267 printf("%d SCCs of size %d\n", cnt, count[v - 1]);
268 SCC_cnt += cnt;
269 cnt = 1;
270 } else {
271 cnt++;
272 }
273 }
274 if (count[n - 1] > 0) {
275 printf("%d SCCs of size %d\n", cnt, count[n - 1]);
276 SCC_cnt += cnt;
277 }
278 printf("number of SCCs: %d\n", SCC_cnt);
279
280 // clean up
281 cudaFree(d_g.nindex); cudaFree(d_g.nlist); cudaFree(d_iomax); cudaFree(d_wl1); cudaFree(d_wl2); cudaFree(d_wl2size); cudaFree(d_goagain);
282 delete [] iomax;
283 delete [] count;
284 freeECLgraph(g);
285 return 0;
286}
