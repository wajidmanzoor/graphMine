80void process_message(uint64_t recv_buf[], uint64_t send_buf[][MSG_BUF_LEN], int node, int sender) {
81 MPI_Request send_req;
82 switch (recv_buf[0]) {
83 case MSG_REQUEST_WORK: { // me: master
84 // try to avoid colliding with other worker
85 // though this isn't guanrtee correctness
86 int worker = recv_buf[1];
87 send_buf[DEVICE_PER_NODE + worker][0] = MSG_DISPATCH_WORK;
88 send_buf[DEVICE_PER_NODE + worker][1] = worker;
89 send_buf[DEVICE_PER_NODE + worker][2] = global_cur_task;
90 send_buf[DEVICE_PER_NODE + worker][3] = global_cur_task = std::min(global_cur_task + recv_buf[2], nr_tasks);
91 MPI_Isend(&send_buf[DEVICE_PER_NODE + worker][0], 4, MPI_UINT64_T, sender, 0, MPI_COMM_WORLD, &send_req);
92 log("master got work request from node %d (worker %lld), replying [%ld, %ld)\n", sender, worker, send_buf[DEVICE_PER_NODE + recv_buf[1]][2], send_buf[DEVICE_PER_NODE + recv_buf[1]][3]);
93 break;
94 }
95 case MSG_DISPATCH_WORK: { // me: slave
96 int worker = int(recv_buf[1]);
97 uint64_t new_task_cur = recv_buf[2];
98 uint64_t new_task_end = recv_buf[3];
99 task_status[worker] = std::make_tuple(worker, new_task_cur, new_task_end);
100 task_ready[worker] = true;
101 log("slave node %d (worker %d) got task [%llu, %llu)\n", node, worker, new_task_cur, new_task_end);
102 break;
103 }
104 case MSG_REPORT_ANS: { // me: master
105 finished_number += 1;
106 log("slave node %d finished its work. Finished number: %d\n", sender, finished_number);
107 break;
108 }
109 }

164 msg_received = 0;

166 MPI_Test(&recv_req, &msg_received, &mpi_status);
167 }

169 // for all worker
170 // #pragma omp parallel for
171 ForallDevice(i, node_devices,
172 if(is_working[i]) {
173 // working
174 auto result = cudaEventQuery(event[i]);
175 // Case 1: result isn't available yet
176 if(result == cudaErrorNotReady) continue;
177 // Case 2: result is already available, ask for another job
178 // use workers' own send buffer[]
179 send_buf[i][0] = MSG_REQUEST_WORK;
180 send_buf[i][1] = i;
181 send_buf[i][2] = NODE_TASK_GRANULARUTY;
182 MPI_Isend(&send_buf[i][0], 3, MPI_UINT64_T, 0, 0, MPI_COMM_WORLD, &send_req[i]);
183 log("node %d worker %d ask for task.\n", node, i);
184 is_working[i] = false;
185 task_ready[i] = false;
186 } else {
187 // idle
188 // Case 1: task_not ready
189 if(!task_ready[i]) continue;
190 // Case 2: task is ready, launch kernel
191 // Case 2.1: start at final part, there won't be any other task
192 if(std::get<1>(task_status[i]) == nr_tasks) {
193 if(finished[i] == false){

197 }

199 continue;
200 }
201 // do its job anyway
202 launch_pattern_matching_kernel(context[i], task_status[i]);
203 cudaEventRecord(event[i]);
204 is_working[i] = true;
205 task_ready[i] = false;
206 }
207 )
208 // Is all devices in a node finished all job?
209 bool all_finished = true;
210 for(int i = 0; i < node_devices; i++) all_finished &= finished[i];
211 // scheduler cannot break, because other nodes may don't know that it's already finished
212 if(node != 0 && all_finished) break;
213 // scheduler node finish its own work, and all other nodes have finished their work
214 if(node == 0 && finished_number == comm_sz - 1 && all_finished) break;
215 }

218 send_buf[node_devices][0] = MSG_REPORT_ANS;

220 }

223
224 unsigned long long sum[node_devices];
225 #pragma omp parallel for
226 ForallDevice(i, node_devices,
227 gpuErrchk(cudaDeviceSynchronize());

229 )
230 for(int i = 0; i < node_devices; i++){
231 gpu_ans += sum[i];
232 }
233 fprintf(stderr, "node %d receive answer: %lld\n", node, gpu_ans);
234 // Step2: reuduce to root 0
235 MPI_Reduce(&gpu_ans, &global_ans, 1, MPI_UINT64_T, MPI_SUM, 0, MPI_COMM_WORLD);
