30
31 if (na > nb) {
32 swap(a, b);

35

37 out_size = 0;
38

41 uint32_t u = a[num_done + lid]; // u: an element in set a
42 int mid, l = 0, r = int(nb) - 1;
43 while (l <= r) {
44 mid = (l + r) >> 1;
45 if (b[mid] < u) {

48 r = mid - 1;
