114__device__ void GPU_pattern_matching_final_in_exclusion(const GPUSchedule *schedule, GPUVertexSet *vertex_set, GPUVertexSet &subtraction_set,
115 GPUVertexSet &tmp_set, unsigned long long &local_ans, uint32_t *edge, e_index_t *vertex) {
116 int last_pos = -1;
117 long long val;
118
119 extern __shared__ char ans_array[];
120 int *ans = ((int *)(ans_array + schedule->ans_array_offset)) + schedule->in_exclusion_optimize_vertex_id_size * (threadIdx.x / THREADS_PER_WARP);
121
122 for (int i = 0; i < schedule->in_exclusion_optimize_vertex_id_size; ++i) {
123 if (schedule->in_exclusion_optimize_vertex_flag[i]) {
124 ans[i] = vertex_set[schedule->in_exclusion_optimize_vertex_id[i]].get_size() - schedule->in_exclusion_optimize_vertex_coef[i];
125 } else {
126 ans[i] = unordered_subtraction_size(vertex_set[schedule->in_exclusion_optimize_vertex_id[i]], subtraction_set);
127 }
128 }
129
130 for (int pos = 0; pos < schedule->in_exclusion_optimize_array_size; ++pos) {
131
132 if (pos == last_pos + 1)
133 val = ans[schedule->in_exclusion_optimize_ans_pos[pos]];
134 else {
135 if (val != 0)
136 val = val * ans[schedule->in_exclusion_optimize_ans_pos[pos]];
137 }
138 if (schedule->in_exclusion_optimize_flag[pos]) {
139 last_pos = pos;
140 local_ans += val * schedule->in_exclusion_optimize_coef[pos];
141 }
142 }
143}

149__device__ void remove_anti_edge_vertices(GPUVertexSet &out_buf, const GPUVertexSet &in_buf, const GPUSchedule &sched,
150 const GPUVertexSet &partial_embedding, int vp, const uint32_t *edge, const e_index_t *vertex) {
151
152 __shared__ uint32_t block_out_offset[THREADS_PER_BLOCK];
153 __shared__ uint32_t block_out_size[WARPS_PER_BLOCK];
154
155 auto d_out = out_buf.get_data_ptr();
156 auto d_in = in_buf.get_data_ptr();
157 int n_in = in_buf.get_size();
158
159 int warp = threadIdx.x / THREADS_PER_WARP;
160 int lane = threadIdx.x % THREADS_PER_WARP;
161 auto out_offset = block_out_offset + warp * THREADS_PER_WARP;
162 auto &out_size = block_out_size[warp];
163
164 if (lane == 0)
165 out_size = 0;
166
167 for (int nr_done = 0; nr_done < n_in; nr_done += THREADS_PER_WARP) {
168 int i = nr_done + lane;
169 bool produce_output = false;
170
171 if (i < n_in) {
172 produce_output = true;
173 for (int u = 0; u < vp; ++u) {
174 if (sched.adj_mat[u * sched.get_size() + vp] != 0)
175 continue;
176
177 auto v = partial_embedding.get_data(u);
178 e_index_t v_neighbor_begin, v_neighbor_end;
179 get_edge_index(v, v_neighbor_begin, v_neighbor_end);
180 int m = v_neighbor_end - v_neighbor_begin; // m = |N(v)|
181
182 if (binary_search(&edge[v_neighbor_begin], m, d_in[i])) {
183 produce_output = false;
184 break;
185 }
186 }
187 }
188 out_offset[lane] = produce_output;
189 __threadfence_block();
190
191#pragma unroll
192 for (int s = 1; s < THREADS_PER_WARP; s *= 2) {
193 uint32_t v = lane >= s ? out_offset[lane - s] : 0;
194 out_offset[lane] += v;
195 __threadfence_block();
196 }
197
198 if (produce_output) {
199 uint32_t offset = out_offset[lane] - 1;
200 d_out[out_size + offset] = d_in[i];
201 }
202
203 if (lane == THREADS_PER_WARP - 1)
204 out_size += out_offset[THREADS_PER_WARP - 1];
205 }
206
207 if (lane == 0)
208 out_buf.init(out_size, d_out);
209 __threadfence_block();
210}
