130/**
131 * search-based intersection
132 *
133 * returns the size of the intersection set
134 *
135 * @note：a和b并不是地位相等的。如果要进行in-place操作，请把输入放在a而不是b。
136 * @todo：shared memory缓存优化
137 */
138__device__ uint32_t do_intersection(uint32_t* out, const uint32_t* a, const uint32_t* b, uint32_t na, uint32_t nb)
139{
140 // __shared__ uint32_t block_out_offset[THREADS_PER_BLOCK];
141 __shared__ uint32_t block_out_size[WARPS_PER_BLOCK];
142 typedef cub::WarpScan<uint8_t> WarpScan;
143 __shared__ typename WarpScan::TempStorage temp_storage[WARPS_PER_BLOCK];
144
145 int wid = threadIdx.x / THREADS_PER_WARP; // warp id
146 int lid = threadIdx.x % THREADS_PER_WARP; // lane id
147 // uint32_t *out_offset = block_out_offset + wid * THREADS_PER_WARP;
148 uint32_t &out_size = block_out_size[wid];
149
150 if (lid == 0)
151 out_size = 0;
152
153 for(int num_done = 0; num_done < na; num_done += THREADS_PER_WARP) {
154 bool found = 0;
155 uint32_t u = 0;
156 if (num_done + lid < na) {
157 u = a[num_done + lid]; // u: an element in set a
158 int mid, l = 0, r = int(nb) - 1;
159 while (l <= r) {
160 mid = (l + r) >> 1;
161 if (b[mid] < u) {
162 l = mid + 1;
163 } else if (b[mid] > u) {
164 r = mid - 1;
165 } else {
166 found = 1;
167 break;
168 }
169 }
170 }
171
172 uint8_t thread_data = found;
173
174 WarpScan(temp_storage[wid]).InclusiveSum(thread_data, thread_data);
175
176 if (found) {
177 out[out_size + thread_data - 1] = u;
178 }
179
180 // __syncwarp();
181
182 if (lid == THREADS_PER_WARP - 1)
183 out_size += thread_data;
184 }
185
186 __threadfence_block();
187 return out_size;
188}

193__device__ void intersection2(uint32_t *tmp, const uint32_t *lbases, const uint32_t *rbases, uint32_t ln, uint32_t rn, uint32_t* p_tmp_size)
194{
195 // make sure ln <= rn
196 if (ln > rn) {
197 swap(ln, rn);
198 swap(lbases, rbases);
199 }
200 /**
201 * @todo 考虑ln < rn <= 32时，每个线程在lbases里面找rbases的一个元素可能会更快
202 */
203
204 uint32_t intersection_size = do_intersection(tmp, lbases, rbases, ln, rn);
205
206 if (threadIdx.x % THREADS_PER_WARP == 0)
207 *p_tmp_size = intersection_size;
208 __threadfence_block();
209}

216__device__ void GPU_pattern_matching_func(const GPUSchedule* schedule, GPUVertexSet* vertex_set, unsigned long long& local_ans, uint32_t *edge, e_index_t *vertex)
217{
218
219 if (depth == schedule->get_size() - schedule->get_in_exclusion_optimize_num()) {
220 assert(false);
221 // GPU_pattern_matching_final_in_exclusion(schedule, vertex_set, subtraction_set, tmp_set, local_ans, edge, vertex);
222 return;

224
225 int loop_set_prefix_id = schedule->get_loop_set_prefix_id(depth);
226 int loop_size = vertex_set[loop_set_prefix_id].get_size();
227
228 uint32_t* loop_data_ptr = vertex_set[loop_set_prefix_id].get_data_ptr();
229 if (depth == schedule->get_size() - 1 && schedule->get_in_exclusion_optimize_num() == 0) {
230 local_ans += loop_size;
231 return;
232 }
233
234 for (int i = 0; i < loop_size; ++i)
235 {
236 uint32_t v = loop_data_ptr[i];
237 long long l, r;
238 get_edge_index(v, l, r);
239 bool is_zero = false;
240 for (int prefix_id = schedule->get_last(depth); prefix_id != -1; prefix_id = schedule->get_next(prefix_id))
241 {
242 vertex_set[prefix_id].build_vertex_set(schedule, vertex_set, &edge[l], (int)(r - l), prefix_id);
243 if (vertex_set[prefix_id].get_size() == schedule->get_break_size(prefix_id)) {
244 is_zero = true;
245 break;
246 }
247 }
248 if (is_zero)
249 continue;
250 GPU_pattern_matching_func<depth + 1>(schedule, vertex_set, local_ans, edge, vertex);
251 }
