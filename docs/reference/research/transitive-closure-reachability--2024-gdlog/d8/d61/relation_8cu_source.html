12__global__ void calculate_index_hash(GHashRelContainer *target,
13 tuple_indexed_less cmp) {
14 tuple_size_t index = (blockIdx.x * blockDim.x) + threadIdx.x;
15 if (index >= target->tuple_counts)
16 return;
17
18 tuple_size_t stride = blockDim.x * gridDim.x;
19
20 for (tuple_size_t i = index; i < target->tuple_counts; i += stride) {
21 tuple_type cur_tuple = target->tuples[i];
22
23 u64 hash_val = prefix_hash(cur_tuple, target->index_column_size);
24 u64 request_hash_index = hash_val % target->index_map_size;
25 tuple_size_t position = request_hash_index;
26 // insert into data container
27 while (true) {
28 // critical condition!
29 u64 existing_key = atomicCAS(&(target->index_map[position].key),
30 EMPTY_HASH_ENTRY, hash_val);
31 tuple_size_t existing_value = target->index_map[position].value;
32 if (existing_key == EMPTY_HASH_ENTRY || existing_key == hash_val) {
33 bool collison_flag = false;
34 while (true) {
35 if (existing_value < i) {
36 // occupied entry, but no need for swap, just check if
37 // collision
38 if (!tuple_eq(target->tuples[existing_value], cur_tuple,
39 target->index_column_size)) {
40 // collision, find nex available entry
41 collison_flag = true;
42 break;
43 } else {
44 // no collision but existing tuple is smaller, in
45 // this case, not need to swap, just return(break;
46 // break)
47 break;
48 }
49 }
50 if (existing_value > i &&
51 existing_value != EMPTY_HASH_ENTRY) {
52 // occupied entry, may need for swap
53 if (!tuple_eq(target->tuples[existing_value], cur_tuple,
54 target->index_column_size)) {
55 // collision, find nex available entry
56 collison_flag = true;
57 break;
58 }
59 // else, swap
60 }
61 // swap value
62 if (existing_value == i) {
63 // swap success return
64 break;
65 } else {
66 // need swap
67 existing_value =
68 atomicCAS(&(target->index_map[position].value),
69 existing_value, i);
70 }
71 }
72 if (!collison_flag) {
73 break;
74 }
75 }
76
77 position = (position + 1) % target->index_map_size;
78 }
79 }
80}

96__global__ void shrink_index_map(GHashRelContainer *target,
97 MEntity *old_index_map,
98 tuple_size_t old_index_map_size) {
99 tuple_size_t index = (blockIdx.x * blockDim.x) + threadIdx.x;
100 if (index >= old_index_map_size)
101 return;
102
103 tuple_size_t stride = blockDim.x * gridDim.x;
104 for (tuple_size_t i = index; i < old_index_map_size; i += stride) {
105 if (target->index_map[i].value != EMPTY_HASH_ENTRY) {
106 u64 hash_val = target->index_map[i].key;
107 tuple_size_t position = hash_val % target->index_map_size;
108 while (true) {
109 u64 existing_key = atomicCAS(&target->index_map[position].key,
110 EMPTY_HASH_ENTRY, hash_val);
111 if (existing_key == EMPTY_HASH_ENTRY) {
112 target->index_map[position].key = hash_val;
113 break;
114 } else if (existing_key == hash_val) {
115 // hash for tuple's index column has already been recorded
116 break;
117 }
118 position = (position + 1) % target->index_map_size;
119 }
120 }
121 }

292 int join_column_counts, tuple_generator_hook tp_gen,
293 tuple_predicate tp_pred, int output_arity,

343 output_raw_data +

405void Relation::flush_delta(int grid_size, int block_size, float *detail_time) {
406 if (delta->tuple_counts == 0) {
407 return;
408 }
409 KernelTimer timer;
410 timer.start_timer();
411 tuple_type *tuple_full_buf;
412 tuple_size_t new_full_size = current_full_size + delta->tuple_counts;
413
414 bool extened_mem = false;
415
416 tuple_size_t total_mem_size = get_total_memory();
417 tuple_size_t free_mem = get_free_memory();
418 u64 delta_mem_size = delta->tuple_counts * sizeof(tuple_type);
419 int multiplier = FULL_BUFFER_VEC_MULTIPLIER;
420 if (!pre_allocated_merge_buffer_flag && !fully_disable_merge_buffer_flag &&
421 delta_mem_size * multiplier <= 0.1 * free_mem) {
422 // std::cout << "reenable pre-allocated merge buffer" << std::endl;
423 pre_allocated_merge_buffer_flag = true;
424 }
425
426 if (!fully_disable_merge_buffer_flag && pre_allocated_merge_buffer_flag) {
427 if (tuple_merge_buffer_size <= new_full_size) {
428 if (tuple_merge_buffer != nullptr) {
429 checkCuda(cudaFree(tuple_merge_buffer));
430 tuple_merge_buffer = nullptr;
431 }
432 // std::cout << "extend mem" << std::endl;
433 extened_mem = true;
434 tuple_merge_buffer_size =
435 current_full_size + (delta->tuple_counts * multiplier);
436 u64 tuple_full_buf_mem_size =
437 tuple_merge_buffer_size * sizeof(tuple_type);
438
439 while (((free_mem - tuple_full_buf_mem_size) * 1.0 /
440 total_mem_size) < 0.4 &&
441 delta_mem_size * multiplier > 0.1 * free_mem) {
442 // std::cout << "multiplier : " << multiplier << std::endl;
443 multiplier--;
444 tuple_merge_buffer_size =
445 current_full_size + (delta->tuple_counts * multiplier);
446 tuple_full_buf_mem_size =
447 tuple_merge_buffer_size * sizeof(tuple_type);
448 if (multiplier == 2) {
449 // std::cout << "not enough memory for merge buffer"
450 // << std::endl;
451 // not enough space for pre-allocated buffer
452 pre_allocated_merge_buffer_flag = false;
453 tuple_merge_buffer_size = 0;
454 // cudaFree(tuple_merge_buffer);
455 checkCuda(cudaMalloc((void **)&tuple_full_buf,
456 tuple_full_buf_mem_size));
457 break;
458 }
459 }
460 if (pre_allocated_merge_buffer_flag) {
461 checkCuda(cudaMalloc((void **)&tuple_merge_buffer,
462 tuple_full_buf_mem_size));
463 tuple_full_buf = tuple_merge_buffer;
464 }
465 } else {
466 tuple_full_buf = tuple_merge_buffer;
467 }
468 } else {
469 tuple_merge_buffer_size = current_full_size + delta->tuple_counts;
470 u64 tuple_full_buf_mem_size =
471 tuple_merge_buffer_size * sizeof(tuple_type);
472 checkCuda(
473 cudaMalloc((void **)&tuple_full_buf, tuple_full_buf_mem_size));
474 // checkCuda(cudaMemset(tuple_full_buf, 0, tuple_full_buf_mem_size));

478

480 // std::cout << "malloc time : " << timer.get_spent_time() << std::endl;
481 detail_time[0] = timer.get_spent_time();
482
483 timer.start_timer();
484 tuple_type *end_tuple_full_buf = thrust::merge(
485 thrust::device, tuple_full, tuple_full + current_full_size,
486 delta->tuples, delta->tuples + delta->tuple_counts, tuple_full_buf,

489 // std::cout << "merge time : " << timer.get_spent_time() << std::endl;
490 detail_time[1] = timer.get_spent_time();
491 // checkCuda(cudaDeviceSynchronize());
492 current_full_size = new_full_size;
493
494 timer.start_timer();
495 if (!fully_disable_merge_buffer_flag && pre_allocated_merge_buffer_flag) {
496 auto old_full = tuple_full;
497 tuple_full = tuple_merge_buffer;
498 tuple_merge_buffer = old_full;
499 if (extened_mem) {
500 checkCuda(cudaFree(tuple_merge_buffer));
501 tuple_merge_buffer = nullptr;
502 u64 tuple_full_buf_mem_size =
503 tuple_merge_buffer_size * sizeof(tuple_type);
504 checkCuda(cudaMalloc((void **)&tuple_merge_buffer,
505 tuple_full_buf_mem_size));
506 }
507 } else {
508 checkCuda(cudaFree(tuple_full));
509 tuple_full = tuple_full_buf;
510 }
511 timer.stop_timer();
512 detail_time[2] = timer.get_spent_time();
513 buffered_delta_vectors.push_back(delta);
514 full->tuples = tuple_full;

516 if (index_flag) {

532 target->arity = arity;

567 checkCuda(cudaDeviceSynchronize());

572 if (arity <= RADIX_SORT_THRESHOLD) {

578 tuple_indexed_less(index_column_size, arity));
579 // checkCuda(cudaDeviceSynchronize());

588 target->tuples + data_row_size, t_equal(arity));
589 // checkCuda(cudaDeviceSynchronize());

591 timer.stop_timer();

593 }

596 // print_tuple_rows(target, "after dedup");
597
598 if (build_index_flag) {
599 timer.start_timer();
600 // init the index map
601 // set the size of index map same as data, (this should give us almost

616 cudaMemcpyHostToDevice));

618 checkCuda(cudaGetLastError());
619 checkCuda(cudaDeviceSynchronize());
620 // std::cout << "finish init index map" << std::endl;
621 // print_hashes(target, "after construct index map");
622 // calculate hash

624 target_device,

709 checkCuda(cudaDeviceSynchronize());
710 // std::cout << "finish init index map" << std::endl;
711 // print_hashes(target, "after construct index map");
712 // calculate hash
713 calculate_index_hash<<<grid_size, block_size>>>(
714 target_device,
715 tuple_indexed_less(target->index_column_size, target->arity));
716 checkCuda(cudaGetLastError());
717 checkCuda(cudaDeviceSynchronize());
718 checkCuda(cudaFree(target_device));
719 timer.stop_timer();
720 detail_time[2] = timer.get_spent_time();
