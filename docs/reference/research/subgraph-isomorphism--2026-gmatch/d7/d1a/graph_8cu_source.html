385Graph::find_automorphisms(std::vector<std::vector<uint32_t>> &embeddings) const {
386 // Note that this method is working on small graphs (tens of vertices) only.
387
388 // This method conducts subgraph matching from Q to Q.
389 std::vector<bool> visited(vcount_, false);
390 std::vector<uint32_t> idx(vcount_);
391 std::vector<uint32_t> mapping(vcount_);
392 std::vector<std::vector<uint32_t>> local_candidates(vcount_);
393 std::vector<std::vector<uint32_t>> global_candidates(vcount_);
394 std::vector<std::vector<uint32_t>> backward_neighbors(vcount_);
395
396 // Initialize global candidates.
397 for (uint32_t u = 0; u < vcount_; u++) {
398 uint32_t u_label = label(u);
399 uint32_t u_degree = degree(u);
400
401 for (uint32_t v = 0; v < vcount_; v++) {
402 uint32_t v_label = label(v);
403 uint32_t v_degree = degree(v);
404
405 if (v_label == u_label && v_degree == u_degree) {
406 global_candidates[u].push_back(v);
407 }
408 }
409 }
410
411 // Set backward neighbors to compute local candidates.
412 for (uint32_t i = 1; i < vcount_; i++) {
413 for (uint32_t j = 0; j < i; j++) {
414 if (is_adjacent(i, j)) {
415 backward_neighbors[i].push_back(j);
416 }
417 }
418 }
419
420 // Recursive search along the matching order (0, 1, ..., n).
421 int cur_level = 0;
422 local_candidates[cur_level] = global_candidates[0];
423
424 while (true) {
425 while (idx[cur_level] < local_candidates[cur_level].size()) {
426 uint32_t u = cur_level;
427 uint32_t v = local_candidates[cur_level][idx[cur_level]];
428 idx[cur_level] += 1;
429
430 if (cur_level == vcount_ - 1) {
431 // Find an embedding
432 mapping[u] = v;
433 embeddings.push_back(mapping);
434 }
435 else {
436 mapping[u] = v;
437 visited[v] = true;
438 cur_level += 1;
439 idx[cur_level] = 0;
440
441 {
442 // Compute local candidates.
443 u = cur_level;
444 for (auto temp_v : global_candidates[u]) {
445 if (!visited[temp_v]) {
446 bool is_feasible = true;
447 for (auto uu : backward_neighbors[u]) {
448 uint32_t temp_vv = mapping[uu];
449 if (!is_adjacent(temp_v, temp_vv)) {
450 is_feasible = false;
451 continue;
452 }
453 }
454 if (is_feasible) {
455 local_candidates[cur_level].push_back(temp_v);
456 }
457 }
458 }
459 }
460 }
461 }
462
463 local_candidates[cur_level].clear();
464 cur_level -= 1;
465 if (cur_level < 0) {
466 break;
467 }
468 visited[mapping[cur_level]] = false;
469 }
470
471 std::cout << "Found " << embeddings.size() << " automorphisms of the query graph." << std::endl;
472 return embeddings.size();
473}

508 for (int v = 0; v < vcount_; v++) {

514 }

516 }

527 M[u][v] = 1;

531 }
532 }
