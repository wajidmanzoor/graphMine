151 get_partial(int partial_id, int *partial_matching_len = nullptr) {
152 partial_props *props = _props_array[current_props_array_id];
153 int props_len = _props_array_len[current_props_array_id];
154
155 if (partial_id >= tot_partial_cnt[current_props_array_id]) {
156 return nullptr;
157 }
158
159 partial_id += 1;
160
161 // Single thread binary search
162 // Find the first cnt_prefix_sum[i] that is no less than partial_id
163 int low = 0;
164 int high = props_len - 1;
165 int mid = ((low + high) >> 1);
166
167 while (low < high) {
168 if (cnt_prefix_sum[mid] < partial_id) {
169 low = mid + 1;
170 }
171 else if (cnt_prefix_sum[mid] == partial_id) {
172 break;
173 }
174 else {
175 high = mid;
176 }
177 mid = ((low + high) >> 1);
178 }
179
180 if (cnt_prefix_sum[mid] < partial_id) {
181 return nullptr;
182 }
183
184 int inner_idx = partial_id - 1;
185 if (mid >= 1) {
186 inner_idx -= cnt_prefix_sum[mid - 1];
187 }
188
189 if (partial_matching_len != nullptr) {
190 *partial_matching_len = props[mid].partial_len;
191 }
192 int *ret = props[mid].start_addr + inner_idx * props[mid].partial_len;
193
194 if (ret[0] == -1) {
195 return nullptr;
196 }
197
198 return ret;
