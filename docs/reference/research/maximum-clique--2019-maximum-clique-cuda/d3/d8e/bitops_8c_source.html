49int bits_32(unsigned int v)
50{
51 int c;
52
53// printf("32:\n");
54 v = v - ((v >> 1) & 0x55555555); // reuse input as temporary
55// printf("%llx\n", v);
56 v = (v & 0x33333333) + ((v >> 2) & 0x33333333); // temp
57// printf("%llx\n", v);
58 v =((v + (v >> 4)) & 0x0F0F0F0F); // count
59// printf("%llx\n", v);

196int mask_comp(t_bitmask mask1, t_bitmask mask2, int len)
197{
198 int res=MASK_SUBSET | MASK_SUPERSET | MASK_EQUAL;
199
200 for(int i=0; i<=len/MASK_CELL_SIZE; i++) {
201 if(mask1[i] & ~mask2[i]) res=res&~(MASK_EQUAL | MASK_SUBSET);
202 if(~mask1[i] & mask2[i]) res=res&~(MASK_EQUAL | MASK_SUPERSET);
203 }
204
205
206 return res;
207}
