102static error_t parse_opt (int key, char *arg, struct argp_state *state)
103{
104 struct arguments *arguments = state->input;
105
106 switch (key)
107 {
108 case 1: // DIMACSbin
109 if(arguments->file_format>0)
110 argp_usage (state);
111 arguments->file_format=1;
112 break;
113
114 case 2: // DIMACSascii
115 if(arguments->file_format>0)
116 argp_usage (state);
117 arguments->file_format=2;
118 break;
119
120 case 't': // atten
121 if(arguments->mode>1)
122 argp_usage (state);
123 arguments->mode=3;
124 arguments->max_masks = arg ? integer_option(arg, state) : 10;
125 break;
126
127 case 'n': // anneal
128 if(arguments->mode>1)
129 argp_usage (state);
130 arguments->mode=2;
131 break;
132
133 case 'a': // alpha
134 if(strcmp(arg, "auto")) {
135 arguments->alpha = float_option(arg, state);
136 arguments->alpha_auto=0;
137 } else arguments->alpha_auto=1;
138 break;
139
140 case 'c': // cpu
141 arguments->cpu = 1;
142 break;
143
144 case 'g': // gpu
145 arguments->gpu = 1;
146 break;
147
148 case 'm': // max-res
149 arguments->max_res = integer_option(arg, state);
150 break;
151
152 case 'u': // max-unsolved
153 arguments->max_unsolved = integer_option(arg, state);
154 break;
155
156 case 'z': // zero
157 arguments->zero = float_option(arg, state);
158 break;
159
160 case 's': // alpha-step
161 arguments->alpha_step = float_option(arg, state);
162 if(arguments->alpha_step<0)
163 argp_usage(state);
164 break;
165
166 case ARGP_KEY_ARG:
167 if(!arguments->mode)
168 arguments->mode=1;
169 if((arguments->max_res>1 && arguments->mode!=3) || state->arg_num >= 1)
170 argp_usage (state);
171 if(!arguments->alpha_auto && arguments->alpha_step!=0)
172 argp_usage (state);
173 arguments->graph_file = arg;
174 break;
175
176 case ARGP_KEY_END:
177 if(state->arg_num < 1)
178 argp_usage (state);
179 break;
180
181 default:
182 return ARGP_ERR_UNKNOWN;
183 }
184 return 0;
185}

327 void print_res(t_bitmask res) {
328 for(int i=0; i<N; i++) if(BIT_TEST(res, i)) printf("%d ", i);
329 }
330
331 if(arguments.max_res>1) for(int i=0; i<n_res; i++) {
332 int order[n_res];
333
334 for(int i=0; i<n_res; i++) order[i]=i;
335
336
337 for (int a = 1; a < n_res; ++a) {
338 int value = order[a], b = a;
339 while (b > 0 && mask_size(res_all[order[b-1]], N) < mask_size(res_all[value], N)) {
340 order[b] = order[b-1];
341 --b;
342 }
343 order[b] = value;
344 }
345
346 printf("clique %d size: %d nodes:\n", i+1, mask_size(res_all[order[i]], N));
347 print_res(res_all[order[i]]);
348 printf("\n");
349 } else {
350 printf("clique size: %d nodes:\n", mask_size(res, N));
351 print_res(res);
352 printf("\n");
353 }
354
355 printf("%s time (us): %ld\n", cuda ? "GPU" : "CPU", end-start);
356 printf("%s time (s): %.3f\n", cuda ? "GPU" : "CPU", (end-start)*0.000001);
357 mask_free(res);
358 mask_free(res_upper);
359 if (arguments.max_res > 1)
360 for (int i = 0; i < n_res; ++i) mask_free(res_all[i]);
361 }
362
363 if(arguments.cpu) run(0);
364 if(arguments.gpu) run(1);
365
366 if (arr) destroy_array((void**)arr);
367 return 0;
