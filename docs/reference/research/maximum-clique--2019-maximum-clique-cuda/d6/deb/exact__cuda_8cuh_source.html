16__device__ int ms_first(const ms_word* bits, int words) {
17 for (int w = 0; w < words; ++w)

22// Greedy independent color classes. The first i vertices use at most
23// bounds[i-1] colors, including after later vertices have been branched away.
24__device__ int ms_color(ms_word* todo, ms_word* available,
25 const ms_word* adjacency, int words,
26 int* order, int* bounds) {
27 int count = 0, color = 0;
28 while (ms_first(todo, words) >= 0) {
29 ++color;
30 for (int w = 0; w < words; ++w) available[w] = todo[w];
31 for (int v = ms_first(available, words); v >= 0; v = ms_first(available, words)) {
32 order[count] = v;
33 bounds[count++] = color;
34 todo[v/64] &= ~(1ULL << (v%64));
35 available[v/64] &= ~(1ULL << (v%64));
36 for (int w = 0; w < words; ++w)
37 available[w] &= ~adjacency[static_cast<size_t>(v)*words+w];
38 }
39 }
40 return count;
41}

43__global__ void ms_exact_search(const ms_word* adjacency, const ms_word* allowed,
44 int n, int words, int* scratch, size_t stride,
45 int* global_best, int* sizes, int* witnesses) {
46 // One independent depth-first traversal per worker. No shared scratch or
47 // stack limits from device recursion; the stack has n explicitly allocated levels.
48 int* base = scratch + static_cast<size_t>(blockIdx.x)*stride;
49 int* order = base;
50 int* bounds = order + static_cast<size_t>(n)*n;
51 int* cursor = bounds + static_cast<size_t>(n)*n;
52 int* chosen = cursor + n;
53 ms_word* todo = reinterpret_cast<ms_word*>(chosen + n);
54 ms_word* available = todo + words;
55 int* witness = witnesses + static_cast<size_t>(blockIdx.x)*n;
56 int local_best = 0;
57 sizes[blockIdx.x] = 0;
58
59 for (int root = blockIdx.x; root < n; root += gridDim.x) {
60 if (!(allowed[root/64] & (1ULL << (root%64)))) continue;
61 chosen[0] = root;
62 if (atomicAdd(global_best, 0) == 0) {
63 witness[0] = root;
64 local_best = 1;
65 sizes[blockIdx.x] = 1;
66 atomicMax(global_best, 1);
67 }
68 for (int w = 0; w < words; ++w) {
69 const ms_word before = w < root/64 ? ~0ULL :
70 (w == root/64 ? ((1ULL << (root%64))-1) : 0ULL);
71 todo[w] = adjacency[static_cast<size_t>(root)*words+w] & allowed[w] & before;
72 }
73 cursor[0] = ms_color(todo, available, adjacency, words, order, bounds)-1;
74 int level = 0;
75 while (level >= 0) {
76 const size_t offset = static_cast<size_t>(level)*n;
77 const int at = cursor[level];
78 if (at < 0 || level+1+bounds[offset+at] <= atomicAdd(global_best, 0)) {
79 --level;
80 continue;
81 }
82 const int v = order[offset+at];
83 --cursor[level];
84 chosen[level+1] = v;
85 const int size = level+2;
86 if (size > local_best) {
87 for (int i = 0; i < size; ++i) witness[i] = chosen[i];
88 local_best = size;
89 sizes[blockIdx.x] = size;
90 atomicMax(global_best, size);
91 }
92 for (int w = 0; w < words; ++w) todo[w] = 0;
93 for (int i = 0; i < at; ++i) {
94 const int u = order[offset+i];
95 if (adjacency[static_cast<size_t>(v)*words+u/64] & (1ULL << (u%64)))
96 todo[u/64] |= 1ULL << (u%64);
97 }
98 // With n chosen vertices no candidates can remain (no self loops).
99 if (ms_first(todo, words) >= 0) {
100 ++level;
101 const size_t child = static_cast<size_t>(level)*n;
102 cursor[level] = ms_color(todo, available, adjacency, words,
103 order+child, bounds+child)-1;
104 }
105 }
106 }
107}

116extern "C" int complete_cuda_clique(char** graph, int n, t_bitmask allowed,
117 t_bitmask result) {
118 if (n < 0 || !result) return -1;
119 if (n == 0) { mask_zeroall(result, 0); return 0; }
120 try {
121 const int words = (n-1)/64+1;
122 const size_t nn = static_cast<size_t>(n);
123 if (nn > (std::numeric_limits<size_t>::max()/sizeof(int)-4*static_cast<size_t>(words)-2*nn-1)/nn/2)
124 return -1;
125 std::vector<ms_word> matrix(nn*words, 0), permitted(words, 0);
126 std::vector<int> seed;
127 for (int v = 0; v < n; ++v) {
128 if (!graph || !graph[v]) return -1;
129 const bool valid = !allowed || (allowed[v/64] & (1ULL << (v%64)));
130 if (valid) permitted[v/64] |= 1ULL << (v%64);
131 if (valid && (result[v/64] & (1ULL << (v%64)))) {
132 bool adjacent = true;
133 for (int u : seed) adjacent &= graph[u][v] && graph[v][u];
134 if (adjacent) seed.push_back(v);
135 }
136 for (int u = 0; u < n; ++u)
137 if (v != u && graph[v][u]) matrix[static_cast<size_t>(v)*words+u/64] |= 1ULL << (u%64);
138 }
139 mask_zeroall(result, n);
140 for (int v : seed) result[v/64] |= 1ULL << (v%64);
141 int best = static_cast<int>(seed.size());
142 size_t free_bytes, total_bytes;
143 if (cudaMemGetInfo(&free_bytes, &total_bytes) != cudaSuccess) return -1;
144 // Round to an even number of ints so bitset scratch stays 8-byte aligned.
145 const size_t stride = (2*nn*nn+2*nn+4*words+1) & ~size_t(1);
146 const size_t budget = std::min(free_bytes/2,
147 std::max(size_t(512)*1024*1024, stride*sizeof(int)));
148 const int workers = static_cast<int>(std::min(std::min(nn, size_t(32)), budget/(stride*sizeof(int))));
149 if (workers < 1) return -1;
150 ms_device_buffer d_matrix, d_allowed, d_scratch, d_best, d_sizes, d_witnesses;
151 if (!d_matrix.allocate(matrix.size()*sizeof(ms_word)) ||
152 !d_allowed.allocate(permitted.size()*sizeof(ms_word)) ||
153 !d_scratch.allocate(workers*stride*sizeof(int)) ||
154 !d_best.allocate(sizeof(int)) || !d_sizes.allocate(workers*sizeof(int)) ||
155 !d_witnesses.allocate(workers*nn*sizeof(int))) return -1;
156 if (cudaMemcpy(d_matrix.data, matrix.data(), matrix.size()*sizeof(ms_word), cudaMemcpyHostToDevice) != cudaSuccess ||
157 cudaMemcpy(d_allowed.data, permitted.data(), permitted.size()*sizeof(ms_word), cudaMemcpyHostToDevice) != cudaSuccess ||
158 cudaMemcpy(d_best.data, &best, sizeof(int), cudaMemcpyHostToDevice) != cudaSuccess) return -1;
159 ms_exact_search<<<workers, 1>>>(static_cast<ms_word*>(d_matrix.data),
160 static_cast<ms_word*>(d_allowed.data), n, words, static_cast<int*>(d_scratch.data),
161 stride, static_cast<int*>(d_best.data), static_cast<int*>(d_sizes.data),
162 static_cast<int*>(d_witnesses.data));
163 if (cudaGetLastError() != cudaSuccess || cudaDeviceSynchronize() != cudaSuccess) return -1;
164 std::vector<int> sizes(workers), witness(n);
165 if (cudaMemcpy(sizes.data(), d_sizes.data, workers*sizeof(int), cudaMemcpyDeviceToHost) != cudaSuccess) return -1;
166 const int owner = static_cast<int>(std::max_element(sizes.begin(), sizes.end())-sizes.begin());
167 if (sizes[owner] > best) {
168 best = sizes[owner];
169 if (cudaMemcpy(witness.data(), static_cast<int*>(d_witnesses.data)+owner*nn,
170 best*sizeof(int), cudaMemcpyDeviceToHost) != cudaSuccess) return -1;
171 mask_zeroall(result, n);
172 for (int i = 0; i < best; ++i) result[witness[i]/64] |= 1ULL << (witness[i]%64);
173 }
174 return best;
175 } catch (const std::bad_alloc&) { return -1; }
176}
