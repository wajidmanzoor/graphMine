117template <unsigned int blockSize, int biased> __device__ __forceinline__ void normiterGPU_dev(struct cuda_clique_status *d_x0_status, struct cuda_clique_status *d_x1_status, float *d_x1, int n, float alpha=0)
118{
119 extern __shared__ float sdata[];
120
121 unsigned int i=threadIdx.x;
122
123 sdata[threadIdx.x]=0;
124
125 if(biased) sdata[blockSize+threadIdx.x]=0;
126 while (i < n) {
127 sdata[threadIdx.x] += d_x1[i];
128
129 if(biased) sdata[blockSize+threadIdx.x] += d_x1[i]*d_x1[i];
130 i += blockSize;
131 }
132
133 __syncthreads();
134 reduce_sum<blockSize, float>(sdata, threadIdx.x);
135 if(biased) reduce_sum<blockSize, float>(&(sdata[blockSize]), threadIdx.x);
136
137 if(threadIdx.x == 0) {
138 float norm;
139 float normsq;
140
141 if(sdata[0]>0) {
142 norm=sdata[0]/d_x0_status->norm2;
143 if(biased) normsq=sdata[blockSize]/(d_x0_status->norm2 * d_x0_status->norm2);
144 } else {
145 norm=0.0;
146 }
147 d_x1_status->norm=norm;
148 d_x1_status->norm2=norm*norm;
149 if(biased) {
150 d_x1_status->csize=1/(1-norm+alpha*normsq/(norm*norm)); // ***** BIASED
151 } else {
152 d_x1_status->csize=1/(1-norm); // ***** not BIASED
153 }
154 }
155}

386template <unsigned int blockSizeX, unsigned int blockSizeY, int biased, int final_mult> __device__ __forceinline__ void iter1GPU_dev(float *d_x1, float *d_x2, unsigned char *d_mat, size_t mat_pitch, size_t x_pitch, int *d_map, float alpha, float omega)
387{
388 extern __shared__ float sdata[];
389
390 float *d_x=sdata+blockSizeY;
391
392 float *pows=d_x+blockSizeY;
393
394 unsigned int tid = threadIdx.x;
395 int n;
396 int n1;
397
398 n=MIN(d_map[-1], (blockIdx.x+1)*blockSizeX);
399 n1=MIN(d_map[-1], (blockIdx.y+1)*blockSizeY);
400
401 int i=tid+blockIdx.x*blockSizeX;
402
403 if(i<n) {
404 d_x[tid]=d_x1[i];
405 }
406
407 if(biased==3) {
408 for(int i=tid; i<256; i+=blockDim.x) {
409 pows[i] = powf(omega, (i-1)*(i-1));
410 }
411 }
412
413 __syncthreads();
414
415 n=MIN((int)(d_map[-1]- blockIdx.x*blockSizeX), blockSizeX);
416 int row = blockIdx.y*blockSizeY + tid;
417
418 if(row < n1) {
419 int j=blockIdx.x*blockSizeX;
420
421 sdata[tid]=0;
422
423 for(i=0; i<n; i++) {
424 int pos = row + d_map[j]*mat_pitch;
425
426 if(biased<=1 && d_mat[pos]) {
427 sdata[tid] += d_x[i];
428 } else if(biased==2) {
429 if(d_mat[pos]>=2) {
430 sdata[tid] += omega * d_x[i];
431 } else if(d_mat[pos]) {
432 sdata[tid] += d_x[i];
433 }
434 } else if(biased==3 && d_mat[pos]) {
435 sdata[tid] += pows[d_mat[pos]] * d_x[i];
436 }
437 j++;
438 }
439
440 if(final_mult) {
441 d_x2[row+x_pitch*blockIdx.x] = d_x1[row] * sdata[tid];
442 } else {
443 d_x2[row+x_pitch*blockIdx.x] = sdata[tid];
444 }
445 }
446}

465

502{
503 extern __shared__ float sdata[];

506
507 unsigned int tid = threadIdx.x;
508 int row = blockIdx.x;
509

511 if(final_mult) {
512 norm=norm*norm;
513 }
514

516 for(int i=tid; i<256; i+=blockDim.x) {

681template <unsigned int blockSize> __global__ void unmapResGPU(float *d_x0, float *d_x1, struct cuda_clique_status *d_x0_status, int *d_ones, int *d_map, int n)
682{
683 extern __shared__ int idata[];
684
685 unsigned int tid = threadIdx.x;
686
687 float one=1.0/MAX(1.0,d_x0_status->csize);
688
689 for(int i= threadIdx.x; i < n; i+= blockDim.x){
690 d_x1[i]=0;
691 }
692
693 __syncthreads();
694 int cnt=0;
695
696 for(int i= threadIdx.x; i < d_map[-1]; i+= blockDim.x){
697 if(d_map[i]>=0) {
698 d_x1[d_map[i]]=d_x0[i];
699 cnt++;
700 }
701 }

704
705 __syncthreads();
706

709 __syncthreads();

712 for(int i= threadIdx.x; i < d_ones[-1]; i+= blockDim.x) {
713 d_x1[d_ones[i]]=one;
714 }
715 } else {
716 one=1/(float)n;

719 }
