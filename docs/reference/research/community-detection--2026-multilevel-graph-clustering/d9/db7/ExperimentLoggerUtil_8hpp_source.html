151 void log(const char* filename, bool first, bool last) {
152 std::ofstream f;
153 f.open(filename, std::ios::app);
154
155 if (f.is_open()) {
156 if (first) {
157 f << "[";
158 }
159 f << "{";
160 f << "\"edge-cut\":" << std::fixed << edge_cut << ",";
161 f << "\"modularity\":" << modularity << ",";
162 f << "\"total-nnz\":" << t_nnz << ",";
163 f << "\"level-count\":" << level_count << ",";
164 for (int i = 0; i < static_cast<int>(Measurement::END); i++) {
165 f << "\"" << measurementNames[i] << "-duration-seconds\":" << measurements[i] << ",";
166 }
167 for(size_t i = 0; i < iter_times.size(); i++){
168 f << "\"iter-times-" << i << "\":" << iter_times[i] << ",";
169 }
170 for(size_t i = 0; i < iter_obj.size(); i++){
171 f << "\"iter-obj-" << i << "\":" << iter_obj[i] << ",";
172 }
173 f << "\"number-coarse-levels\":" << numCoarseLevels;
174 f << "}";
175 if (!last) {
176 f << ",";
177 }
178 else {
179 f << "]";
180 }
181 f.close();
182 }
183 else {
184 std::cerr << "Could not open " << filename << std::endl;
185 }
186 }

188 void refinementReport(){
189 std::cout << std::setprecision(6);
190 std::cout << std::left << std::setw(6) << "Level" << std::setw(16) << "Edge Cut" << std::setw(10) << "Imbalance";
191 std::cout << std::setw(13) << "Vertices" << std::setw(16) << "Edges" << std::setw(22) << "Total Refinement Time";
192 std::cout << std::setw(17) << "Total Iterations" << std::setw(14) << "LP Iterations" << std::setw(23) << "Average Iteration Time" << std::endl;
193 for(size_t i = 0; i < coarseLevels.size(); i++){
194 CoarseLevel cl = coarseLevels[i];
195 int level = coarseLevels.size() - 1 - i;
196 std::cout << std::fixed << std::left << std::setw(6) << level << std::setw(16) << cl.edge_cut << std::setw(10) << cl.imb;
197 std::cout << std::setw(13) << cl.numVertices << std::setw(16) << cl.numEdges << std::setw(22) << cl.totalRefTime;
198 std::cout << std::setw(17) << cl.totalIterations << std::setw(14) << cl.lpIterations << std::setw(23) << (cl.iterationsTime / cl.totalIterations) << std::endl;
199 }
200 }
