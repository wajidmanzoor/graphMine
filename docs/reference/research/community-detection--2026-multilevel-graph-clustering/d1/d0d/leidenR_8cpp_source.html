41 template <bool uniform>
42 void ensure_gamma_connectivity(const wg_t wg, vtx_vt vcmap, vtx_vt constraint, vtx_vt order, const wgt_vt total_deg, mem_t& mem, const refine_data& rfd) {
43
44 const matrix_t g = wg.mtx;
45 const wgt_vt vtx_w = wg.vtx_w;
46 const ordinal_t n = g.numRows();
47 edge_vt row_map = Kokkos::subview(mem.p_mem.row_map, std::make_pair(static_cast<ordinal_t>(0), n + 1));
48 vtx_vt store_atom = Kokkos::subview(mem.p_mem.entries, std::make_pair(static_cast<ordinal_t>(0), n));
49 vtx_vt ids = Kokkos::subview(mem.s_mem.vtx1, std::make_pair(static_cast<ordinal_t>(0), n));
50 vtx_vt sort_order = Kokkos::subview(mem.s_mem.vtx2, std::make_pair(static_cast<ordinal_t>(0), n));
51 Kokkos::deep_copy(exec_space(), row_map, 0);
52
53 Kokkos::parallel_for("compute cluster cardinality", policy_t(0, n), KOKKOS_LAMBDA(ordinal_t i) {
54 ordinal_t c = vcmap(i);
55 store_atom(i) = Kokkos::atomic_fetch_add(&row_map(c), 1);
56 });
57
58 Kokkos::parallel_scan("scan offsets", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t i, ordinal_t& update, const bool final){
59 ordinal_t val = row_map(i);
60 if(final) row_map(i) = update;
61 update += val;
62 if(final && i + 1 == n){
63 row_map(n) = update;
64 }
65 });
66
67 Kokkos::parallel_for("insert", policy_t(0, n), KOKKOS_LAMBDA(ordinal_t i) {
68 ordinal_t c = vcmap(i);
69 ordinal_t insert = row_map(c) + store_atom(i);
70 ids(insert) = i;
71 sort_order(insert) = order(i);
72 });
73
74 // sort_order and row_map are unneeded after this
75 KokkosSparse::sort_crs_matrix<exec_space, edge_vt, vtx_vt, vtx_vt>(exec_space(), row_map, sort_order, ids);
76
77 wgt_vt total_size = mem.p_mem.pvals;
78 wgt_vt inner_conn = Kokkos::subview(mem.p_mem.vals, std::make_pair(static_cast<ordinal_t>(0), n));
79 wgt_vt pvals = Kokkos::subview(mem.p_mem.pvals_clone, std::make_pair(static_cast<ordinal_t>(0), n));
80 // gets the total size of each cluster during construction by order of ids view
81 Kokkos::parallel_scan("scan total_size", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t x, scalar_t& update, const bool final){
82 ordinal_t i = ids(x);
83 scalar_t val = vtx_w(i);
84 if(final){
85 total_size(x) = update;
86 }
87 update += val;
88 });
89 float gamma = rfd.get_penalty_modifier();
90 vtx_vt order1 = mem.o_mem.order1;
91 ordinal_t big_begin = mem.o_mem.offset_large;
92 ordinal_t biggest_begin = mem.o_mem.offset_large2;
93 vtx_vt small_vtx = Kokkos::subview(order1, std::make_pair(static_cast<ordinal_t>(0), big_begin));
94 vtx_vt large_vtx = Kokkos::subview(order1, std::make_pair(big_begin, biggest_begin));
95 vtx_vt largest_vtx = Kokkos::subview(order1, std::make_pair(biggest_begin, n));
96 Kokkos::parallel_for("compute inner_conn", policy_t(0, big_begin), KOKKOS_LAMBDA(const ordinal_t x) {
97 ordinal_t i = small_vtx(x);
98 edge_offset_t end = g.graph.row_map(i + 1);
99 edge_offset_t start = g.graph.row_map(i);
100 scalar_t result = 0;
101 for(edge_offset_t idx = start; idx < end; idx++) {
102 ordinal_t v = g.graph.entries(idx);
103 scalar_t wgt;
104 if constexpr(uniform) wgt = 1;
105 else wgt = g.values(idx);
106 if(vcmap(i) == vcmap(v) && order(v) < order(i)) result += wgt;
107 }
108 inner_conn(i) = result;
109 });
110 Kokkos::parallel_for("compute inner_conn", team_policy_t(biggest_begin - big_begin, Kokkos::AUTO), KOKKOS_LAMBDA(const member & thread) {
111 ordinal_t i = large_vtx(thread.league_rank());
112 edge_offset_t end = g.graph.row_map(i + 1);
113 edge_offset_t start = g.graph.row_map(i);
114 Kokkos::parallel_reduce(Kokkos::TeamThreadRange(thread, start, end), [=](const edge_offset_t idx, scalar_t& update) {
115 ordinal_t v = g.graph.entries(idx);
116 scalar_t wgt;
117 if constexpr(uniform) wgt = 1;
118 else wgt = g.values(idx);
119 if(vcmap(i) == vcmap(v) && order(v) < order(i)) update += wgt;
120 }, inner_conn(i));
121 });
122 Kokkos::parallel_for("compute inner_conn", team_policy_t(n - biggest_begin, 1024), KOKKOS_LAMBDA(const member & thread) {
123 ordinal_t i = largest_vtx(thread.league_rank());
124 edge_offset_t end = g.graph.row_map(i + 1);
125 edge_offset_t start = g.graph.row_map(i);
126 Kokkos::parallel_reduce(Kokkos::TeamThreadRange(thread, start, end), [=](const edge_offset_t idx, scalar_t& update) {
127 ordinal_t v = g.graph.entries(idx);
128 scalar_t wgt;
129 if constexpr(uniform) wgt = 1;
130 else wgt = g.values(idx);
131 if(vcmap(i) == vcmap(v) && order(v) < order(i)) update += wgt;
132 }, inner_conn(i));
133 });
134
135 // alias and reuse
136 vtx_vt breakers = order;
137 Kokkos::deep_copy(exec_space(), breakers, n);
138
139 // break clusters if vertex i not well connected to cluster
140 Kokkos::parallel_for("find breakpoints (part 1)", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t x){
141 ordinal_t i = ids(x);
142 ordinal_t c = vcmap(i);
143 ordinal_t c_begin = row_map(c);
144 scalar_t prev_size = total_size(x) - total_size(c_begin);
145 if(inner_conn(i) < gamma*vtx_w(i)*prev_size){
146 // break cluster on x
147 Kokkos::atomic_min(&breakers(c), x);
148 }
149 });
150
151 // aliasing
152 wgt_vt outer_conn = inner_conn;
153 // gets the total outward connectivity of each cluster during construction by order of ids view
154 Kokkos::parallel_scan("scan outward connectivity (inclusive)", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t x, scalar_t& update, const bool final){
155 ordinal_t i = ids(x);
156 // *2 to account for edges previously outside cluster that are now within it
157 scalar_t val = pvals(i) - 2*inner_conn(i);
158 update += val;
159 if(final){
160 // use i instead of x so that the aliasing works
161 outer_conn(i) = update;
162 }
163 });
164
165 // break cluster if cluster up to i not well connected to rest of constraint cluster
166 Kokkos::parallel_for("find breakpoints (part 2)", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t x){
167 ordinal_t i = ids(x);
168 ordinal_t c = vcmap(i);
169 ordinal_t c_begin = row_map(c);
170 scalar_t curr_size = total_size(x) + vtx_w(i) - total_size(c_begin);
171 scalar_t other_size = total_deg(constraint(i)) - curr_size;
172 scalar_t outer = outer_conn(i);
173 if(c_begin > 0) outer -= outer_conn(ids(c_begin - 1));
174 if(outer < gamma*curr_size*other_size){
175 // break cluster after x
176 Kokkos::atomic_min(&breakers(c), x + 1);
177 }
178 });
179
180 Kokkos::parallel_for("break clusters", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t x){
181 ordinal_t i = ids(x);
182 ordinal_t c = vcmap(i);
183 if(x >= breakers(c)){
184 vcmap(i) = i;
185 }
186 });
187 }

189 // reassigns cluster labels into a contiguous range beginning from 0
190 ordinal_t contigitize_clusters(vtx_vt vcmap, const ordinal_t n) {
191 ordinal_t nc = 0;
192 Kokkos::parallel_scan("assign contiguous id", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t u, ordinal_t& update, const bool final){
193 // every cluster with label "u" must contain vertex "u"
194 if(vcmap(u) == u){
195 if(final){
196 vcmap(u) = update;
197 }
198 update++;
199 } else if(final){
200 vcmap(u) = vcmap(u) + n;
201 }
202 }, nc);
203 Kokkos::parallel_for("propagate ids", policy_t(0, n), KOKKOS_LAMBDA(ordinal_t u) {
204 if(vcmap(u) >= n) {
205 ordinal_t c_id = vcmap(u) - n;
206 vcmap(u) = vcmap(c_id);
207 }
208 });
209 return nc;
210 }

212 //hn is a list of vertices such that vertex i wants to aggregate with vertex hn(i)
213 void find_trees(vtx_vt vcmap, const ordinal_t n, const vtx_vt hn, vtx_vt order) {
214
215 // compute connected components on the forest induced by hn
216 // in this kernel we ignore edges that go towards a higher ordinal vertex
217 Kokkos::parallel_for("connected components part 1", policy_t(0, n), KOKKOS_LAMBDA(ordinal_t i) {
218 ordinal_t now = i;
219 ordinal_t then = hn(i);
220
221 if(now == then){
222 vcmap(i) = i;
223 return;
224 }
225
226 // pointer-chasing towards lowest order
227 while(order(now) > order(then)){
228 now = then;
229 then = hn(then);
230 }
231
232 if(now != i){
233 vcmap(i) = now;
234 // other vertices in path will get here, except for now
235 vcmap(now) = now;
236 }
237
238 });
239
240 // flip the ordering for remaining unclustered vertices
241 // this also makes them higher ordered then all previously clustered vertices
242 Kokkos::parallel_for("update order", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t i){
243 if(vcmap(i) == ORD_MAX) order(i) = 2*split - order(i);
244 });
245
246 // in this kernel we consider the edges ignored by the previous kernel
247 // in order to connect vertices left unassigned by previous kernel
248 Kokkos::parallel_for("connected components part 2", policy_t(0, n), KOKKOS_LAMBDA(ordinal_t i) {
249 if(vcmap(i) != ORD_MAX) return;
250 ordinal_t now = i;
251 ordinal_t then = hn(i);
252
253 while(vcmap(now) == ORD_MAX && order(now) > order(then)){
254 now = then;
255 then = hn(then);
256 }
257
258 vcmap(i) = vcmap(now);
259 });
260 }

262 // uniform is true only if top is true
263 template <bool top, bool uniform>
264 vtx_vt coarsen_leidenR(const wg_t wg,
265 const vtx_vt& constraint,
266 mem_t& mem,
267 const refine_data& rfd,
268 ordinal_t& coarse_vtx_count) {
269
270 matrix_t g = wg.mtx;
271 wgt_vt vtx_w = wg.vtx_w;
272 ordinal_t n = g.numRows();
273 float gamma = rfd.get_penalty_modifier();
274 vtx_vt hn = Kokkos::subview(mem.s_mem.vtx1, std::make_pair(static_cast<ordinal_t>(0), n));
275 vtx_vt vcmap("vcmap", n);
276 Kokkos::deep_copy(exec_space(), vcmap, ORD_MAX);
277
278 vtx_vt well_conn = Kokkos::subview(mem.p_mem.cluster_sizes, std::make_pair(static_cast<ordinal_t>(0), n));
279 const wgt_vt pvals = Kokkos::subview(mem.p_mem.pvals_clone, std::make_pair(static_cast<ordinal_t>(0), n));
280 const wgt_vt total_deg = rfd.total_deg;
281 Kokkos::parallel_for("determine well connected", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t i){
282 ordinal_t c = constraint(i);
283 scalar_t wd = vtx_w(i);
284 // is vertex 'i' well connected to the rest of constraint cluster 'c'?
285 if(pvals(i) >= gamma*wd*(total_deg(c) - wd)) well_conn(i) = 1;
286 else well_conn(i) = 0;
287 });
288
289 std::random_device rd;
290 if(uniform){
291 ordinal_t seed = rd();
292 vtx_vt order1 = mem.o_mem.order1;
293 ordinal_t biggest_begin = mem.o_mem.offset_large2;
294 vtx_vt smaller_vtx = Kokkos::subview(order1, std::make_pair(static_cast<ordinal_t>(0), biggest_begin));
295 vtx_vt largest_vtx = Kokkos::subview(order1, std::make_pair(biggest_begin, n));
296 Kokkos::parallel_for("select random edge", policy_t(0, biggest_begin), KOKKOS_LAMBDA(const ordinal_t x) {
297 ordinal_t i = smaller_vtx(x);
298 edge_offset_t end = g.graph.row_map(i + 1);
299 edge_offset_t start = g.graph.row_map(i);
300 ordinal_t width = end - start;
301 float multi = gamma*vtx_w(i);
302 hasher_t hash;
303 edge_offset_t jx = hash(seed + i);
304 // 0.001 chance to self-aggregate
305 if(jx % 1000 == 0 || well_conn(i) == 0){
306 hn(i) = i;
307 return;
308 }
309 jx = start + (jx % static_cast<uint32_t>(width));
310 // choose random satisfactory neighbor
311 for(edge_offset_t j = jx; j < end; j++){
312 ordinal_t v = g.graph.entries(j);
313 if(constraint(i) != constraint(v)) continue;
314 if(well_conn(v) == 0) continue;
315 scalar_t wgt;
316 if constexpr(uniform) wgt = 1;
317 else wgt = g.values(j);
318 if(wgt >= multi*vtx_w(v)){
319 hn(i) = v;
320 return;
321 }
322 }
323 for(edge_offset_t j = start; j < jx; j++){
324 ordinal_t v = g.graph.entries(j);
325 if(constraint(i) != constraint(v)) continue;
326 if(well_conn(v) == 0) continue;
327 scalar_t wgt;
328 if constexpr(uniform) wgt = 1;
329 else wgt = g.values(j);
330 if(wgt >= multi*vtx_w(v)){
331 hn(i) = v;
332 return;
333 }
334 }
335 hn(i) = i;
336 });
337 // I didn't think this would ever be necessary
338 // I thought very high degree vertices would have many viable choices such that it would be quick to find one
339 // If good clusterings are correlated with the vertex ordering,
340 // then choosing a random search starting point could drop us in the middle of a chunk of vertices in different constraint clusters
341 // This would lead to traversing a large number of vertices before finding a viable edge
342 Kokkos::parallel_for("select random edge", team_policy_t(n - biggest_begin, 1024), KOKKOS_LAMBDA(const member& t) {
343 ordinal_t i = largest_vtx(t.league_rank());
344 edge_offset_t end = g.graph.row_map(i + 1);
345 edge_offset_t start = g.graph.row_map(i);
346 ordinal_t width = end - start;
347 float multi = gamma*vtx_w(i);
348 hasher_t hash;
349 edge_offset_t jx = hash(seed + i);
350 // 0.001 chance to self-aggregate
351 if(jx % 1000 == 0 || well_conn(i) == 0){
352 hn(i) = i;
353 return;
354 }
355 jx = start + (jx % static_cast<uint32_t>(width));
356 // choose random satisfactory neighbor
357 edge_offset_t result = end;
358 Kokkos::parallel_reduce(Kokkos::TeamThreadRange(t, jx, end), [=](const edge_offset_t j, edge_offset_t& local) {
359 ordinal_t v = g.graph.entries(j);
360 if(constraint(i) != constraint(v)) return;
361 if(well_conn(v) == 0) return;
362 scalar_t wgt;
363 if constexpr(uniform) wgt = 1;
364 else wgt = g.values(j);
365 if(wgt >= multi*vtx_w(v)){
366 if(j < local){
367 local = j;
368 }
369 }
370 }, min_reducer_t(result));
371 if(result < end) {
372 hn(i) = g.graph.entries(result);
373 return;
374 }
375 Kokkos::parallel_reduce(Kokkos::TeamThreadRange(t, start, jx), [=](const edge_offset_t j, edge_offset_t& local) {
376 ordinal_t v = g.graph.entries(j);
377 if(constraint(i) != constraint(v)) return;
378 if(well_conn(v) == 0) return;
379 scalar_t wgt;
380 if constexpr(uniform) wgt = 1;
381 else wgt = g.values(j);
382 if(wgt >= multi*vtx_w(v)){
383 if(j < local){
384 local = j;
385 }
386 }
387 }, min_reducer_t(result));
388 if(result < end) {
389 hn(i) = g.graph.entries(result);
390 } else {
391 hn(i) = i;
392 }
393 });
394 } else {
395 vtx_vt order1 = mem.o_mem.order1;
396 ordinal_t big_begin = mem.o_mem.offset_large;
397 vtx_vt small_vtx = Kokkos::subview(order1, std::make_pair(static_cast<ordinal_t>(0), big_begin));
398 vtx_vt large_vtx = Kokkos::subview(order1, std::make_pair(big_begin, n));
399 ordinal_t seed = rd();
400 Kokkos::parallel_for("select heaviest edge (low degree)", policy_t(0, big_begin), KOKKOS_LAMBDA(const ordinal_t x){
401 ordinal_t i = small_vtx(x);
402 edge_offset_t end = g.graph.row_map(i + 1);
403 edge_offset_t start = g.graph.row_map(i);
404 float multi = gamma*vtx_w(i);
405 hasher_t hash;
406 // 0.0001 chance to self-aggregate
407 if(hash(seed + i) % 10000 == 0 || well_conn(i) == 0){
408 hn(i) = i;
409 return;
410 }
411 float m = 0;
412 edge_offset_t am = end;
413 // get neighbor maximizing objective
414 for(edge_offset_t j = start; j < end; j++) {
415 ordinal_t v = g.graph.entries(j);
416 if(constraint(i) != constraint(v)) continue;
417 if(well_conn(v) == 0) continue;
418 scalar_t wgt = g.values(j);
419 float val = static_cast<float>(wgt) - multi*vtx_w(v);
420 if(val >= 0){
421 float sample = Kokkos::abs(static_cast<float>(hash(seed + j)) / static_cast<float>(ORD_MAX));
422 sample = sample*Kokkos::pow(val, 4);
423 if(sample >= m){
424 m = sample;
425 am = j;
426 }
427 }
428
429 }
430 if(am >= start && am < end && m >= 0){
431 ordinal_t h = g.graph.entries(am);
432 hn(i) = h;
433 } else {
434 hn(i) = i;
435 }
436 });
437 Kokkos::parallel_for("select heaviest edge (high degree)", team_policy_t(n - big_begin, Kokkos::AUTO), KOKKOS_LAMBDA(const member & thread) {
438 ordinal_t i = large_vtx(thread.league_rank());
439 edge_offset_t end = g.graph.row_map(i + 1);
440 edge_offset_t start = g.graph.row_map(i);
441 float multi = gamma*vtx_w(i);
442 hasher_t hash;
443 // 0.0001 chance to self-aggregate
444 if(hash(seed + i) % 10000 == 0 || well_conn(i) == 0){
445 hn(i) = i;
446 return;
447 }
448 argmax_t argmax{0, end};
449 // get neighbor maximizing objective
450 Kokkos::parallel_reduce(Kokkos::TeamThreadRange(thread, start, end), [=](const edge_offset_t idx, argmax_t& local) {
451 ordinal_t v = g.graph.entries(idx);
452 if(constraint(i) != constraint(v)) return;
453 if(well_conn(v) == 0) return;
454 scalar_t wgt = g.values(idx);
455 float val = static_cast<float>(wgt) - multi*vtx_w(v);
456 if(val >= 0){
457 float sample = Kokkos::abs(static_cast<float>(hash(seed + idx)) / static_cast<float>(ORD_MAX));
458 sample = sample*Kokkos::pow(val, 4);
459 if(sample >= local.val){
460 local.val = sample;
461 local.loc = idx;
462 }
463 }
464
465 }, argmax_reducer_t(argmax));
466 Kokkos::single(Kokkos::PerTeam(thread), [=](){
467 if(argmax.loc >= start && argmax.loc < end && argmax.val >= 0){
468 ordinal_t h = g.graph.entries(argmax.loc);
469 hn(i) = h;
470 } else {
471 hn(i) = i;
472 }
473 });
474 });
475 }
476
477 // alias and reuse
478 vtx_vt order = well_conn;
479 ordinal_t seed = rd();
480 // this random ordering determines the spanning trees
481 // and the insertion order into each cluster
482 Kokkos::parallel_for("set order", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t i){
483 hasher_t hash;
484 ordinal_t o = hash(i + seed);
485 order(i) = Kokkos::abs(o % split);
486 });
487 Kokkos::parallel_for("unselect equal order edges", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t i){
488 ordinal_t h = hn(i);
489 // if two vertices have the same ordering and one points to the other
490 // just unselect that edge
491 // sidenote: if two vertices have the same ordering but neither points to the other
492 // than the sort which comes later will be the tie-breaker
493 if(order(i) == order(h)) hn(i) = i;
494 });
495 find_trees(vcmap, n, hn, order);
496 ensure_gamma_connectivity<uniform>(wg, vcmap, constraint, order, rfd.total_deg, mem, rfd);
497 coarse_vtx_count = contigitize_clusters(vcmap, n);
498
499 return vcmap;
500 }
