91KOKKOS_INLINE_FUNCTION ordinal_t xorshiftHash(ordinal_t key) {
92 ordinal_t x = key;
93 x ^= x << 13;
94 x ^= x >> 17;
95 x ^= x << 5;
96 return x;
97}

99template <bool pval_clone_valid>
100struct countingFunctor {
101
102 matrix_t g;
103 vtx_view_t vcmap;
104 edge_view_t degree_initial;
105 wgt_view_t pvals;
106 ordinal_t workLength;
107
108 countingFunctor(matrix_t _g,
109 vtx_view_t _vcmap,
110 edge_view_t _degree_initial,
111 wgt_view_t _pvals) :
112 g(_g),
113 vcmap(_vcmap),
114 degree_initial(_degree_initial),
115 pvals(_pvals),
116 workLength(_g.numRows()) {}
117
118 KOKKOS_INLINE_FUNCTION
119 void operator()(const ordinal_t& i) const
120 {
121 ordinal_t u = vcmap(i);
122 edge_offset_t start = g.graph.row_map(i);
123 edge_offset_t end = g.graph.row_map(i + 1);
124 ordinal_t nonLoopEdgesTotal = end - start;
125 // for uniformly edge weighted graphs
126 // pvals(i) is equal to the number of self loops
127 // which fine vertex i would contribute to coarse vertex u
128 // this optimization allows the memory initialization and stream compaction
129 // operations to do less work, and makes the cache utilization of
130 // deduplication a bit better
131 if constexpr(pval_clone_valid) nonLoopEdgesTotal -= pvals(i);
132 Kokkos::atomic_add(&degree_initial(u), nonLoopEdgesTotal);
133 }
134};

161 // uses linear probing to resolve hash conflicts
162 KOKKOS_INLINE_FUNCTION
163 edge_offset_t insert(const edge_offset_t& hash_start, const edge_offset_t& size, const ordinal_t& u, const ordinal_t& i) const {
164 edge_offset_t offset = xorshiftHash(u) % static_cast<uint32_t>(size);
165 while(true){
166 if(htable(hash_start + offset) == HASH_NULL){
167 if(Kokkos::atomic_compare_exchange(&htable(hash_start + offset), HASH_NULL, u) == HASH_NULL){
168 Kokkos::atomic_add(&counts(i), 1);
169 }
170 }
171 if(htable(hash_start + offset) == u){
172 return offset;
173 } else {
174 offset++;
175 if(offset >= size) offset -= size;
176 }
177 }
178 }
179
180 KOKKOS_INLINE_FUNCTION
181 void operator()(const member& thread) const
182 {
183 const ordinal_t x = vtx(thread.league_rank());
184 const ordinal_t i = vcmap(x);
185 const edge_offset_t start = g.graph.row_map(x);
186 const edge_offset_t end = g.graph.row_map(x + 1);
187 const edge_offset_t hash_start = hrow_map(i);
188 const edge_offset_t size = hrow_map(i + 1) - hash_start;
189 Kokkos::parallel_for(Kokkos::TeamThreadRange(thread, start, end), [&](const edge_offset_t j){
190 ordinal_t u = vcmap(g.graph.entries(j));
191 if(i == u) return;
192 edge_offset_t offset = insert(hash_start, size, u, i);
193 if constexpr(uniform) Kokkos::atomic_add(&hvals(hash_start + offset), 1);
194 else Kokkos::atomic_add(&hvals(hash_start + offset), g.values(j));
195 });
196 }
197
198 KOKKOS_INLINE_FUNCTION
199 void operator()(const ordinal_t& xx) const
200 {
201 const ordinal_t x = vtx(xx);
202 const ordinal_t i = vcmap(x);
203 const edge_offset_t start = g.graph.row_map(x);
204 const edge_offset_t end = g.graph.row_map(x + 1);
205 const edge_offset_t hash_start = hrow_map(i);
206 const edge_offset_t size = hrow_map(i + 1) - hash_start;
207 for(edge_offset_t j = start; j < end; j++){
208 ordinal_t u = vcmap(g.graph.entries(j));
209 if(i == u) continue;
210 edge_offset_t offset = insert(hash_start, size, u, i);
211 if constexpr(uniform) Kokkos::atomic_add(&hvals(hash_start + offset), 1);
212 else Kokkos::atomic_add(&hvals(hash_start + offset), g.values(j));
213 }
214 }

217void fast_fill(vtx_view_t a, ordinal_t V){
218 edge_offset_t width = a.extent(0);
219 edge_offset_t w8 = (width + 7)/8;
220 Kokkos::parallel_for("fast fill", policy_t(0, w8), KOKKOS_LAMBDA(const edge_offset_t i){
221 a(i) = V;
222 a(i + w8) = V;
223 a(i + 2*w8) = V;
224 a(i + 3*w8) = V;
225 a(i + 4*w8) = V;
226 a(i + 5*w8) = V;
227 a(i + 6*w8) = V;
228 // this one may be out of bounds for even arbitrarily large width
229 // technically some of the other ones can also be out of bounds for width < 42
230 if(i + 7*w8 < width) a(i + 7*w8) = V;
231 });
232}

234template <bool uniform, bool pval_clone_valid>
235wg_t build_coarse_graph(const wg_t curr_level,
236 const vtx_view_t vcmap,
237 const ordinal_t nc,
238 mem_t& mem) {
239
240 matrix_t g = curr_level.mtx;
241 ordinal_t n = g.numRows();
242 double multiplier = 1.0;
243 if(nc > n * 0.5) multiplier = 1.5;
244 if(g.nnz() * multiplier > mem.p_mem.entries.extent(0)){
245 // rare edge-case
246 multiplier = 1.0;
247 }
248
249 Kokkos::Timer timer;
250
251 // determine upper bound on coarse vertex sizes
252 edge_view_t hrow_map = Kokkos::subview(mem.p_mem.row_map, std::make_pair((ordinal_t)0, nc + 1));
253 wgt_view_t pvals_clone = mem.p_mem.pvals_clone;
254 Kokkos::deep_copy(exec_space(), hrow_map, 0);
255 countingFunctor<pval_clone_valid> countF(g, vcmap, hrow_map, pvals_clone);
256 Kokkos::parallel_for("count edges per coarse vertex (also compute coarse vertex weights)", policy_t(0, n), countF);
257
258 // allocate hash tables for each coarse vertex
259 edge_offset_t hash_size = 0;
260 Kokkos::parallel_scan("scan offsets", policy_t(0, nc + 1), KOKKOS_LAMBDA(const ordinal_t i, edge_offset_t& update, const bool final){
261 edge_offset_t val = hrow_map(i) * multiplier;
262 if(final){
263 hrow_map(i) = update;
264 }
265 update += val;
266 }, mem.s_mem.edge_scan_host);
267 exec_space().fence();
268 hash_size = mem.s_mem.edge_scan_host();
269 vtx_view_t htable = Kokkos::subview(mem.p_mem.entries, std::make_pair((edge_offset_t)0, hash_size));
270 fast_fill(htable, HASH_NULL);
271 // Kokkos::deep_copy(exec_space(), htable, HASH_NULL);
272 wgt_view_t hvals = Kokkos::subview(mem.p_mem.vals, std::make_pair((edge_offset_t)0, hash_size));
273 Kokkos::deep_copy(exec_space(), hvals, 0);
274
275 // accumulates edge totals per coarse vertex
276 edge_view_t coarse_row_map_f("edges_per_source", nc + 1);
277 ordinal_t small = mem.o_mem.offset_mid;
278 ordinal_t large = mem.o_mem.offset_mid2 - small;
279 ordinal_t largest = n - mem.o_mem.offset_mid2;
280 vtx_view_t vtx_small = Kokkos::subview(mem.o_mem.order2, std::make_pair(static_cast<ordinal_t>(0), small));
281 vtx_view_t vtx_large = Kokkos::subview(mem.o_mem.order2, std::make_pair(small, mem.o_mem.offset_mid2));
282 vtx_view_t vtx_largest = Kokkos::subview(mem.o_mem.order2, std::make_pair(mem.o_mem.offset_mid2, n));
283 // identifies unique coarse edges
284 combineAndDedupe<uniform> cnd_small(g, vcmap, htable, hvals, hrow_map, coarse_row_map_f, vtx_small);
285 combineAndDedupe<uniform> cnd_large(g, vcmap, htable, hvals, hrow_map, coarse_row_map_f, vtx_large);
286 combineAndDedupe<uniform> cnd_largest(g, vcmap, htable, hvals, hrow_map, coarse_row_map_f, vtx_largest);
287 Kokkos::parallel_for("deduplicate", team_policy_t(large, Kokkos::AUTO), cnd_large);
288 Kokkos::parallel_for("deduplicate", team_policy_t(largest, 1024), cnd_largest);
289 Kokkos::parallel_for("deduplicate", policy_t(0, small), cnd_small);
290 edge_offset_t old_size = hash_size;
291 // build row map of coarse graph
292 Kokkos::parallel_scan("scan offsets", policy_t(0, nc + 1), KOKKOS_LAMBDA(const ordinal_t i, edge_offset_t& update, const bool final){
293 edge_offset_t val = coarse_row_map_f(i);
294 if(final){
295 coarse_row_map_f(i) = update;
296 }
297 update += val;
298 }, mem.s_mem.edge_scan_host);
299 exec_space().fence();
300 hash_size = mem.s_mem.edge_scan_host();
301
302 // segmented stream compaction
303 // segments have size equal to max representable number of ordinal_t
304 // this allows storing offsets directly into entries_coarse, even if edge_offset_t is a larger type
305 vtx_view_t entries_coarse(Kokkos::ViewAllocateWithoutInitializing("coarse entries"), hash_size);
306 wgt_view_t wgts_coarse(Kokkos::ViewAllocateWithoutInitializing("coarse weights"), hash_size);
307 Kokkos::fence();
308 edge_offset_t stream_compact_start = 0;
309 edge_offset_t coarse_start = 0;
310 while(stream_compact_start < old_size) {
311 ordinal_t* htb = htable.data() + stream_compact_start;
312 ordinal_t width = std::numeric_limits<ordinal_t>::max();
313 if(stream_compact_start + width >= old_size) width = old_size - stream_compact_start;
314 thrust::counting_iterator<ordinal_t> iter(0);
315 ordinal_t* ec = entries_coarse.data() + coarse_start;
316 ordinal_t* ec_end;
317 if(!is_host_space) ec_end = thrust::copy_if(thrust::device, iter, iter + width, htb, ec, is_nonnegative());
318 else ec_end = thrust::copy_if(thrust::host, iter, iter + width, htb, ec, is_nonnegative());
319 ordinal_t compacted_width = ec_end - ec;
320 Kokkos::parallel_for("read", policy_t(0, compacted_width), KOKKOS_LAMBDA(const ordinal_t dst_base){
321 edge_offset_t dst_offset = coarse_start + dst_base;
322 edge_offset_t src_offset = entries_coarse(dst_offset) + stream_compact_start;
323 entries_coarse(dst_offset) = htable(src_offset);
324 wgts_coarse(dst_offset) = hvals(src_offset);
325 });
326 coarse_start += compacted_width;
327 stream_compact_start += width;
328 }
329
330 graph_type gc_graph(entries_coarse, coarse_row_map_f);
331 matrix_t gc("gc", nc, wgts_coarse, gc_graph);
332 wg_t next_level;
333 next_level.mtx = gc;
334 next_level.edge_uniform = false;
335 return next_level;
336}
