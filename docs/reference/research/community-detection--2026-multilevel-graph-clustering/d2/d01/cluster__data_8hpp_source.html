30 static edge_offset_t sum(const wgt_vt wdeg){
31 edge_offset_t result = 0;
32 Kokkos::parallel_reduce("sum view", policy_t(0, wdeg.size()), KOKKOS_LAMBDA(const ordinal_t i, edge_offset_t& update){
33 update += wdeg(i);
34 }, result);

38 cluster_data(const matrix_t g, const wgt_vt wdeg, double _lambda) {
39 total_deg = wgt_vt("total degree of clusters", g.numRows());
40 top_nnz = g.nnz();
41 Kokkos::deep_copy(exec_space(), total_deg, wdeg);
42 uncut = 0;
43 label_count = g.numRows();
44 lambda = _lambda;
45 update_objective();
46 }

48 void reset(const matrix_t g, const wgt_vt wdeg) {
49 wgt_vt td_lhs = Kokkos::subview(total_deg, std::make_pair((ordinal_t)0, g.numRows()));
50 Kokkos::deep_copy(td_lhs, wdeg);
51 uncut = 0;
52 label_count = g.numRows();
53 update_objective();
54 }

56 void copy(const cluster_data& rhs){
57 wgt_vt td_lhs = Kokkos::subview(total_deg, std::make_pair((ordinal_t)0, rhs.label_count));
58 wgt_vt td_rhs = Kokkos::subview(rhs.total_deg, std::make_pair((ordinal_t)0, rhs.label_count));
59 Kokkos::deep_copy(exec_space(), td_lhs, td_rhs);

76 void update_objective() {
77 // avoid implicit capture of "this"
78 wgt_vt total = total_deg;
79 int64_t square_sum = 0;
80 Kokkos::parallel_reduce("sum of squares", policy_t(0, label_count), KOKKOS_LAMBDA(const ordinal_t l, int64_t& update){
81 int64_t c_size = total(l);
82 update += c_size*c_size;
83 }, square_sum);
84 double m = uncut;
85 m -= lambda * static_cast<double>(square_sum);
86 obj = m;
87 }

109struct modularity : public cluster_data {
110 using Device = typename matrix_t::device_type;
111 using scalar_t = typename matrix_t::value_type;
112 using wgt_vt = Kokkos::View<scalar_t*, Device>;
113
114 double inv_gdeg = 0;
115
116 modularity(const matrix_t g, const wgt_vt wdeg, double _penalty_scale, bool uniform) : cluster_data(g, wdeg, 1.0) {
117 edge_offset_t g_deg = 0;
118 if(uniform) g_deg = g.nnz();
119 else g_deg = cluster_data::sum(wdeg);
120 inv_gdeg = 1.0 / static_cast<double>(g_deg);
121 cluster_data::lambda = _penalty_scale * inv_gdeg;
122 }
