103 KOKKOS_INLINE_FUNCTION
104 void operator()(const ordinal_t i, wrapper& update, const bool final) const {
105 ordinal_t degree = g.graph.row_map(i + 1) - g.graph.row_map(i);
106 if(degree < LARGE_CUTOFF){
107 if(final) order1(offsets.value[0] + update.value[0]) = i;
108 update.value[0]++;
109 } else if(degree < MASSIVE_CUTOFF){
110 if(final) order1(offsets.value[1] + update.value[1]) = i;
111 update.value[1]++;
112 } else {
113 if(final) {
114 order1(offsets.value[4] + update.value[4]) = i;
115 order2(offsets.value[4] + update.value[4]) = i;
116 }
117 update.value[4]++;
118 }
119 if(degree < MID_CUTOFF){
120 if(final) order2(offsets.value[2] + update.value[2]) = i;
121 update.value[2]++;
122 } else if(degree < MASSIVE_CUTOFF){
123 if(final) order2(offsets.value[3] + update.value[3]) = i;
124 update.value[3]++;
125 }
126 }

129// buckets vertices by degree, whilst maintaining natural order within each bucket
130// generates two such orderings, only differing in the cutoff separating the first and second buckets
131void generate_orderings(mem_t& mem, const matrix_t& g) {
132 ordinal_t n = g.numRows();
133 vtx_vt order1 = mem.o_mem.order1;
134 vtx_vt order2 = mem.o_mem.order2;
135 wrapper bucket_sizes;
136 typedef SumMyArray<Kokkos::HostSpace> ArraySumResult;
137
138 Kokkos::parallel_reduce("sum bucket sizes", policy_t(0, n), KOKKOS_LAMBDA(const ordinal_t i, wrapper& update) {
139 ordinal_t degree = g.graph.row_map(i + 1) - g.graph.row_map(i);
140 if(degree < LARGE_CUTOFF){
141 update.value[0]++;
142 } else if(degree < MASSIVE_CUTOFF){
143 update.value[1]++;
144 } else {
145 update.value[4]++;
146 }
147 if(degree < MID_CUTOFF){
148 update.value[2]++;
149 } else if(degree < MASSIVE_CUTOFF){
150 update.value[3]++;
151 }
152 }, ArraySumResult(bucket_sizes));
153 wrapper offsets(bucket_sizes);
154 offsets.value[0] = 0;
155 offsets.value[1] = bucket_sizes.value[0];
156 offsets.value[2] = 0;
157 offsets.value[3] = bucket_sizes.value[2];
158 offsets.value[4] = bucket_sizes.value[0] + bucket_sizes.value[1];
159 mem.o_mem.offset_large = offsets.value[1];
160 mem.o_mem.offset_mid = offsets.value[3];
161 mem.o_mem.offset_large2 = offsets.value[4];
162 mem.o_mem.offset_mid2 = offsets.value[4];
163 Kokkos::parallel_scan("generate orders", policy_t(0, n), ScanMyArray(g, order1, order2, offsets));
164}
