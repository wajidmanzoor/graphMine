47 template <bool plus, bool improve>
48 vtx_vt leiden_part(mem_t& mem, wg_t top, rfd_t& rfd, ExperimentLoggerUtil<value_t>& experiment, vtx_vt input){
49 std::vector<wg_t> levels;
50 std::vector<vtx_vt> parts;
51 levels.push_back(top);
52 double aggregate = 0;
53 vtx_vt part("cluster assignments", top.mtx.numRows());
54 if(!improve){
55 Kokkos::parallel_for("set initial assignments", policy_t(0, top.mtx.numRows()), KOKKOS_LAMBDA(const ordinal_t x){
56 part(x) = x;
57 });
58 } else Kokkos::deep_copy(part, input);
59 while(true) {
60 wg_t c = levels[levels.size() - 1];
61 double old_obj = rfd.obj;
62 // orderings must be generated for use in local_move and build_coarse_graph
63 order::generate_orderings(mem, c.mtx);
64 lm_t::local_move<false>(c, part, rfd, !improve && (levels.size() == 1), mem, part, true);
65 if(c.mtx.nnz() > 100000 && old_obj == rfd.obj){
66 // it is usually worth trying this first
67 lm_t::local_move<false>(c, part, rfd, !improve && (levels.size() == 1), mem, part, false);
68 }
69 if(old_obj == rfd.obj){
70 lm_t::local_move_strict<false>(c, part, rfd, !improve && (levels.size() == 1), mem, part);
71 }
72 if(rfd.label_count == c.mtx.numRows()){
73 parts.push_back(part);
74 break;
75 }
76 vtx_vt louv = part;
77 int coarse_vtx_count = 0;
78#ifdef EXTRA_TIMING
79 Kokkos::fence();
80 Kokkos::Timer lr_time;
81#endif
82 vtx_vt coarse_map;
83 if(levels.size() == 1 && c.edge_uniform) coarse_map = lr_t::template coarsen_leidenR<true, true>(c, louv, mem, rfd, coarse_vtx_count);
84 else if(levels.size() == 1 && !(c.edge_uniform)) coarse_map = lr_t::template coarsen_leidenR<true, false>(c, louv, mem, rfd, coarse_vtx_count);
85 else coarse_map = lr_t::template coarsen_leidenR<false, false>(c, louv, mem, rfd, coarse_vtx_count);
86 parts.push_back(coarse_map);
87#ifdef EXTRA_TIMING
88 Kokkos::fence();
89 experiment.addMeasurement(Measurement::LeidenRefine, lr_time.seconds());
90#endif
91 if(coarse_vtx_count < c.mtx.numRows()){
92#ifdef EXTRA_TIMING
93 Kokkos::Timer t;
94#endif
95 wg_t next_level;
96 if(c.edge_uniform) next_level = contract_t::build_coarse_graph<true, false>(c, coarse_map, coarse_vtx_count, mem);
97 else next_level = contract_t::build_coarse_graph<false, false>(c, coarse_map, coarse_vtx_count, mem);
98 next_level.vtx_w = wgt_vt("weighted degree 2", coarse_vtx_count);
99 coarsen_vtx_w(c.vtx_w, next_level.vtx_w, coarse_map);
100
101 part = vtx_vt("cluster assignments coarse", coarse_vtx_count);
102 downsample(louv, part, coarse_map);
103 levels.push_back(next_level);
104
105#ifdef EXTRA_TIMING
106 Kokkos::fence();
107 aggregate += t.seconds();
108#endif
109 } else {
110 // avoid creating new graph if leidenR didn't contract any vertices
111 // which may happen with astronomically low probability for any input clustering
112 // or if input clustering is very bad
113 levels.push_back(c);
114 }
115 }
116
117 experiment.addMeasurement(Measurement::Contract, aggregate);
118 int64_t t_nnz = 0;
119 for(const wg_t& level : levels){
120 t_nnz += level.mtx.nnz();
121 }
122 experiment.setTotalNnz(t_nnz);
123 experiment.setLevelCount(levels.size());
124 for(int i = levels.size() - 2; i >= 0; i--){
125 wg_t c = levels[i];
126 vtx_vt coarse_part = parts[i + 1];
127 vtx_vt fine_part = parts[i];
128 Kokkos::parallel_for("update top level assignments", policy_t(0, c.mtx.numRows()), KOKKOS_LAMBDA(const ordinal_t x){
129 fine_part(x) = coarse_part(fine_part(x));
130 });
131 if constexpr(plus){
132 order::generate_orderings(mem, c.mtx);
133 lm_t::local_move<false>(c, fine_part, rfd, false, mem, part, true);
134 }
135 }
136 return parts[0];
137 }

139 template <bool constrained>
140 vtx_vt louvain_part(mem_t& mem, wg_t top, rfd_t& rfd, ExperimentLoggerUtil<value_t>& experiment, vtx_vt constraint){
141 if(constrained) rfd.reset(top.mtx, top.vtx_w);
142 std::vector<wg_t> levels;
143 std::vector<vtx_vt> parts;
144 levels.push_back(top);
145 double aggregate = 0;
146 bool drop_constraint = constrained;
147 while(true) {
148 wg_t c = levels[levels.size() - 1];
149 vtx_vt part("cluster assignments", c.mtx.numRows());
150 Kokkos::parallel_for("set initial assignments", policy_t(0, c.mtx.numRows()), KOKKOS_LAMBDA(const ordinal_t x){
151 part(x) = x;
152 });
153 // orderings must be generated for use in local_move and build_coarse_graph
154 order::generate_orderings(mem, c.mtx);
155 lm_t::local_move<constrained>(c, part, rfd, true, mem, constraint, true);
156 // the user clearly cares about quality if they are doing multiple iterations
157 if(constrained && rfd.label_count == c.mtx.numRows()){
158 lm_t::local_move_strict<constrained>(c, part, rfd, true, mem, constraint);
159 }
160 parts.push_back(part);
161 if(rfd.label_count < c.mtx.numRows()){
162#ifdef EXTRA_TIMING
163 Kokkos::fence();
164 Kokkos::Timer t;
165#endif
166 wg_t next_level;
167 if(c.edge_uniform) next_level = contract_t::build_coarse_graph<true, true>(c, part, rfd.label_count, mem);
168 else next_level = contract_t::build_coarse_graph<false, false>(c, part, rfd.label_count, mem);
169 wgt_vt td_rfd = Kokkos::subview(rfd.total_deg, std::make_pair((ordinal_t)0, rfd.label_count));
170 next_level.vtx_w = wgt_vt("next level vtx weights", rfd.label_count);
171 Kokkos::deep_copy(next_level.vtx_w, td_rfd);
172 levels.push_back(next_level);
173
174 if(constrained) {
175 vtx_vt next_constraint("next constraint", rfd.label_count);
176 downsample(constraint, next_constraint, part);
177 constraint = next_constraint;
178 }
179
180#ifdef EXTRA_TIMING
181 Kokkos::fence();
182 aggregate += t.seconds();
183#endif
184 } else if(drop_constraint) {
185 Kokkos::deep_copy(constraint, 0);
186 parts.pop_back();
187 drop_constraint = false;
188 } else {
189 break;
190 }
191 }
192
193 int64_t t_nnz = 0;
194 for(const wg_t& level : levels){
195 t_nnz += level.mtx.nnz();
196 }
197 experiment.setTotalNnz(t_nnz);
198 experiment.setLevelCount(levels.size());
199
200 if(levels.size() > 1){
201 // last level has the same partition as previous level
202 // so refining this level on the uncoarsening pass
203 // would not integrate any coarse information
204 levels.pop_back();
205 parts.pop_back();
206 }
207
208 // levels.size()-2 so that (i+1) is in bounds
209 for(int i = levels.size() - 2; i >= 0; i--){
210 wg_t c = levels[i];
211 vtx_vt coarse_part = parts[i + 1];
212 vtx_vt part = parts[i];
213 Kokkos::parallel_for("update top level assignments", policy_t(0, c.mtx.numRows()), KOKKOS_LAMBDA(const ordinal_t x){
214 part(x) = coarse_part(part(x));
215 });
216 order::generate_orderings(mem, c.mtx);
217 lm_t::local_move<false>(c, part, rfd, false, mem, constraint, true);
218 }
219
220 experiment.addMeasurement(Measurement::Contract, aggregate);
221 return parts[0];
222 }
