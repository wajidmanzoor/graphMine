28template<typename COO, typename BitmapCOO>
29void convert_coo2bmp(COO mat_input, BitmapCOO& mat_output)
30{
31 using IndexType = typename BitmapCOO::index_type;
32 using BitmapType = typename BitmapCOO::bitmap_type;
33 using InValueType = typename COO::value_type;
34 using OutValueType = typename BitmapCOO::value_type;
35
36 constexpr auto num_bmp64 = BitmapCOO::bmp64_count;
37 // Use thrust::device directly for simplicity and readability.
38
39 auto exec = thrust::device;
40 const auto nnz = mat_input.num_entries;
41 const auto nrow = mat_input.num_rows;
42 const auto ncol = mat_input.num_cols;
43 const auto nrow_tile = div_up(nrow, FRAG_DIM);
44 const auto ncol_tile = div_up(ncol, FRAG_DIM);
45 ASSERT(nrow_tile * ncol_tile < std::numeric_limits<BitmapType>::max()
46 && "BitmapType is not large enough to represent the number of tiles");
47
48 thrust::sort_by_key(mat_input.column_indices.begin(),
49 mat_input.column_indices.end(),
50 thrust::make_zip_iterator(thrust::make_tuple(mat_input.row_indices.begin(), mat_input.values.begin())));
51 thrust::stable_sort_by_key(
52 mat_input.row_indices.begin(),
53 mat_input.row_indices.end(),
54 thrust::make_zip_iterator(thrust::make_tuple(mat_input.column_indices.begin(), mat_input.values.begin())));
55
56 thrust::device_vector<BitmapType> tile_indices(nnz);
57 thrust::device_vector<BitmapType> pos_in_tile(nnz);
58
59 // Calculate tile indices and pos_in_tiles with a single pass.
60
61 thrust::transform(
62 exec,
63 thrust::make_zip_iterator(thrust::make_tuple(mat_input.row_indices.begin(), mat_input.column_indices.begin())),
64 thrust::make_zip_iterator(thrust::make_tuple(mat_input.row_indices.end(), mat_input.column_indices.end())),
65 thrust::make_zip_iterator(thrust::make_tuple(tile_indices.begin(), pos_in_tile.begin())),
66 LocateTile256<BitmapType>(ncol_tile));
67
68 // print_vec(tile_indices, "tile_indices: ");
69 // print_vec(pos_in_tile, "pos_in_tile: ");
70
71 // Sort based on tile indices. This operation affects the original matrices
72 // in-place.
73 //! due to this step, we have to utilize a vector of row_indices
74 thrust::stable_sort_by_key(
75 exec,
76 tile_indices.begin(),
77 tile_indices.end(),
78 thrust::make_zip_iterator(thrust::make_tuple(
79 mat_input.row_indices.begin(), mat_input.column_indices.begin(), mat_input.values.begin(), pos_in_tile.begin())));
80
81 // Perform reduction by key in-place where possible.
82 // Using Thrust's reduce_by_key to compact and aggregate bitmap
83 // pos_in_tiles.
84 thrust::device_vector<BitmapType> unique_tile_indices = tile_indices;
85 auto tile_indices_end = thrust::unique(exec, unique_tile_indices.begin(), unique_tile_indices.end());
86 auto num_tiles = tile_indices_end - unique_tile_indices.begin();
87 unique_tile_indices.erase(tile_indices_end, unique_tile_indices.end());
88
89 thrust::device_vector<BitmapType> bitmaps(num_tiles * num_bmp64);
90 thrust::device_vector<BitmapType> tile_positions(nnz);
91
92 thrust::lower_bound(unique_tile_indices.begin(),
93 unique_tile_indices.end(),
94 tile_indices.begin(),
95 tile_indices.end(),
96 tile_positions.begin());
97
98 thrust::for_each(exec,
99 thrust::make_zip_iterator(thrust::make_tuple(tile_positions.begin(), pos_in_tile.begin())),
100 thrust::make_zip_iterator(thrust::make_tuple(tile_positions.end(), pos_in_tile.end())),
101 CombineToBMP256<BitmapType>(thrust::raw_pointer_cast(bitmaps.data())));
102
103 // free vector
104 tile_indices.resize(0);
105 tile_positions.resize(0);
106 pos_in_tile.resize(0);
107 tile_indices.shrink_to_fit();
108 tile_positions.shrink_to_fit();
109 pos_in_tile.shrink_to_fit();
110
111 // mat_input.row_indices.resize(num_tiles);
112 // mat_input.row_indices.shrink_to_fit();
113 // Setup output matrix dimensions based on FRAG_DIM.
114 mat_output.resize(nrow_tile, ncol_tile, nnz, num_tiles);
115
116 // Transform tile indices to row and column indices for the output matrix.
117 thrust::transform(
118 exec,
119 unique_tile_indices.begin(),
120 unique_tile_indices.end(),
121 thrust::make_zip_iterator(thrust::make_tuple(mat_output.row_indices.begin(), mat_output.column_indices.begin())),
122 COOIndices<IndexType, BitmapType>(ncol_tile));
123
124 // Copying values and computing bmp_offsets is already efficient.
125 mat_output.values = std::move(mat_input.values);
126
127 thrust::transform(
128 exec, bitmaps.begin(), bitmaps.end(), mat_output.tile_offsets.begin(), BmpPopcount<IndexType, BitmapType>());
129
130 // Convert bit counts to offsets for the bitmap.
131 thrust::exclusive_scan(
132 exec, mat_output.tile_offsets.begin(), mat_output.tile_offsets.end(), mat_output.tile_offsets.begin(), 0);
133
134 // Copy the final bitmap values to the output matrix.
135 mat_output.bitmaps = std::move(bitmaps);
136}
