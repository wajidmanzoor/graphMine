16template<typename T>
17void exclusive_scan(T* input, int length)
18{
19 if (length == 0 || length == 1)
20 return;
21
22 T old_val, new_val;
23
24 old_val = input[0];
25 input[0] = 0;
26 for (int i = 1; i < length; i++) {
27 new_val = input[i];
28 input[i] = old_val + input[i - 1];
29 old_val = new_val;
30 }
31}
