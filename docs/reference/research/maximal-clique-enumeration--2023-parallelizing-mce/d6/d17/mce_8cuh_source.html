26 __global__ void getNodeDegree_kernel(T *node_degree, graph::COOCSRGraph_d<T> g, T *max_degree)
27 {
28 T gtid = threadIdx.x + blockIdx.x * blockDim.x;
29 typedef cub::BlockReduce<T, BLOCK_DIM_X> BlockReduce;
30 __shared__ typename BlockReduce::TempStorage temp_storage;
31 T degree = 0;
32 if (gtid < g.numNodes)
33 {
34 degree = g.rowPtr[gtid + 1] - g.rowPtr[gtid];
35 node_degree[gtid] = degree;
36 }
37
38 T aggregate = BlockReduce(temp_storage).Reduce(degree, cub::Max());
39 if (threadIdx.x == 0)
40 atomicMax(max_degree, aggregate);
41 }
