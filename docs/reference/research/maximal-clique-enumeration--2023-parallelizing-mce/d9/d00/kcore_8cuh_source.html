107template <typename T, typename PeelT, int BD, int P>
108__global__ void
109kernel_partition_level_next(
110 graph::COOCSRGraph_d<T> g,
111 int level, bool *processed, PeelT *nodeDegree,
112 graph::GraphQueue_d<T, bool> current,
113 graph::GraphQueue_d<T, bool> &next,
114 graph::GraphQueue_d<T, bool> &bucket,
115 int bucket_level_end_,
116 T priority,
117 T *nodePriority)
118{
119 const size_t partitionsPerBlock = BD / P;
120 const size_t lx = threadIdx.x % P;
121 const int warpIdx = threadIdx.x / P; // which warp in thread block
122 const size_t gwx = (blockDim.x * blockIdx.x + threadIdx.x) / P;
123
124 for (auto i = gwx; i < current.count[0]; i += blockDim.x * gridDim.x / P)
125 {
126 T nodeId = current.queue[i];
127 T srcStart = g.rowPtr[nodeId];
128 T srcStop = g.rowPtr[nodeId + 1];
129
130 nodePriority[nodeId] = priority;
131
132 for (auto j = srcStart + lx; j < (srcStop + P - 1) / P * P; j += P)
133 {
134 __syncwarp();
135 if (j < srcStop)
136 {
137 T affectedNode = g.colInd[j];
138 if (!current.mark[affectedNode])
139 process_degree<T>(affectedNode, level, nodeDegree, next, bucket, bucket_level_end_);
140 }
141 }
