28std::pair<NeighborIterator, NeighborIterator> neighbors(
29 const CsrGraph& graph, VertexIndex vertex) {
30 const auto begin = graph.offsets[vertex];
31 const auto end = graph.offsets[vertex + 1];
32 return {graph.neighbors.begin() + static_cast<std::ptrdiff_t>(begin),
33 graph.neighbors.begin() + static_cast<std::ptrdiff_t>(end)};

36bool adjacent(const CsrGraph& graph, VertexIndex lhs, VertexIndex rhs) {
37 const auto [begin, end] = neighbors(graph, lhs);
38 return std::binary_search(begin, end, rhs);

41std::vector<VertexIndex> intersect_neighbors(
42 const CsrGraph& graph, const std::vector<VertexIndex>& vertices,
43 VertexIndex pivot) {
44 std::vector<VertexIndex> result;
45 result.reserve(vertices.size());
46 const auto [begin, end] = neighbors(graph, pivot);
47 std::set_intersection(vertices.begin(), vertices.end(), begin, end,
48 std::back_inserter(result));

52VertexIndex choose_pivot(const CsrGraph& graph,
53 const std::vector<VertexIndex>& possible,
54 const std::vector<VertexIndex>& excluded) {
55 VertexIndex best = possible.empty() ? excluded.front() : possible.front();
56 std::size_t best_score = 0;
57 const auto score = [&](VertexIndex candidate) {
58 std::size_t count = 0;
59 const auto [begin, end] = neighbors(graph, candidate);
60 auto p = possible.begin();
61 auto n = begin;
62 while (p != possible.end() && n != end) {
63 if (*p < *n) {
64 ++p;
65 } else if (*n < *p) {
66 ++n;
67 } else {
68 ++count;
69 ++p;
70 ++n;
71 }
72 }
73 return count;
74 };
75
76 for (const auto vertex : possible) {
77 const auto candidate_score = score(vertex);
78 if (candidate_score > best_score) {
79 best = vertex;
80 best_score = candidate_score;
81 }
82 }
83 for (const auto vertex : excluded) {
84 const auto candidate_score = score(vertex);
85 if (candidate_score > best_score) {
86 best = vertex;
87 best_score = candidate_score;
88 }
89 }
90 return best;
91}

93void collect_bron_kerbosch(const CsrGraph& graph, InternalClique& current,
94 std::vector<VertexIndex> possible,
95 std::vector<VertexIndex> excluded,
96 CollectionState& state) {
97 if (state.stopped_early) {
98 return;
99 }
100 if (possible.empty() && excluded.empty()) {
101 if (current.size() < state.minimum_size) {
102 return;
103 }
104 ++state.qualifying_count;
105 if (!state.result_limit || state.cliques.size() < *state.result_limit) {
106 auto clique = current;
107 std::sort(clique.begin(), clique.end());
108 state.cliques.push_back(std::move(clique));
109 }
110 if (state.result_limit && state.cliques.size() >= *state.result_limit &&
111 !state.must_finish) {
112 state.stopped_early = true;
113 }
114 return;
115 }
116
117 const auto pivot = choose_pivot(graph, possible, excluded);
118 std::vector<VertexIndex> candidates;
119 candidates.reserve(possible.size());
120 for (const auto vertex : possible) {
121 if (!adjacent(graph, pivot, vertex)) {
122 candidates.push_back(vertex);
123 }
124 }
125
126 for (const auto vertex : candidates) {
127 if (state.stopped_early) {
128 return;
129 }
130 const auto in_possible =
131 std::lower_bound(possible.begin(), possible.end(), vertex);
132 if (in_possible == possible.end() || *in_possible != vertex) {
133 continue;
134 }
135
136 current.push_back(vertex);
137 auto next_possible = intersect_neighbors(graph, possible, vertex);
138 auto next_excluded = intersect_neighbors(graph, excluded, vertex);
139 collect_bron_kerbosch(graph, current, std::move(next_possible),
140 std::move(next_excluded), state);
141 current.pop_back();
142
143 possible.erase(std::lower_bound(possible.begin(), possible.end(), vertex));
144 excluded.insert(std::lower_bound(excluded.begin(), excluded.end(), vertex),
145 vertex);
146 }
147}

149CollectionState collect_cliques(const CsrGraph& graph,
150 const MaximalCliqueOptions& options,
151 bool backend_total_is_qualifying) {
152 CollectionState state;
153 state.minimum_size = options.minimum_clique_size;
154 state.result_limit = options.result_limit;
155 state.must_finish =
156 has_output(options.optional_outputs,
157 MaximalCliqueOptionalOutput::total_count) &&
158 !backend_total_is_qualifying;
159
160 std::vector<VertexIndex> possible(graph.vertex_count());
161 std::iota(possible.begin(), possible.end(), VertexIndex{0});
162 InternalClique current;
163 collect_bron_kerbosch(graph, current, std::move(possible), {}, state);
164 std::sort(state.cliques.begin(), state.cliques.end());
165 return state;
166}

168CsrGraph without_isolated_vertices(const CsrGraph& graph,
169 std::uint64_t& isolated_count) {
170 isolated_count = 0;
171 std::vector<VertexIndex> old_to_new(
172 graph.vertex_count(), std::numeric_limits<VertexIndex>::max());
173 VertexIndex next = 0;
174 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
175 if (graph.offsets[vertex] == graph.offsets[vertex + 1]) {
176 ++isolated_count;
177 } else {
178 old_to_new[vertex] = next++;
179 }
180 }
181 if (isolated_count == 0) {
182 return graph;
183 }
184
185 CsrGraph compact;
186 compact.offsets.resize(static_cast<std::size_t>(next) + 1, 0);
187 for (VertexIndex old_vertex = 0; old_vertex < graph.vertex_count();
188 ++old_vertex) {
189 const auto new_vertex = old_to_new[old_vertex];
190 if (new_vertex == std::numeric_limits<VertexIndex>::max()) {
191 continue;
192 }
193 const auto [begin, end] = neighbors(graph, old_vertex);
194 for (auto it = begin; it != end; ++it) {
195 compact.neighbors.push_back(old_to_new[*it]);
196 }
197 compact.offsets[new_vertex + 1] = compact.neighbors.size();
198 }
199 return compact;
200}

249SupportReport MaximalCliques::supports(const Graph& graph) const {
250 if (options_.minimum_clique_size == 0) {
251 return {false, "minimum_clique_size must be at least one"};
252 }
253 if (options_.result_limit && *options_.result_limit == 0) {
254 return {false, "result_limit must be null or at least one"};
255 }
256 if (graph.directed() && !options_.allow_directed_projection) {
257 return {false,
258 "directed input requires allow_directed_projection=true"};
259 }
260 if (options_.execution.device_ids.empty()) {

262 }
263 const auto selected = options_.backend == MaximalCliqueBackend::automatic
264 ? MaximalCliqueBackend::rdmce
265 : options_.backend;
266 if (selected == MaximalCliqueBackend::mce_gpu &&
267 !detail::mce_gpu_backend_compiled()) {
268 return {false, "the mce-gpu backend is not compiled in this build"};
269 }
270 if (selected == MaximalCliqueBackend::g2_aimd &&
271 !detail::g2_aimd_backend_compiled()) {
272 return {false, "the G2-AIMD backend is not compiled in this build"};
273 }
274 if (selected == MaximalCliqueBackend::rdmce &&
275 !detail::rdmce_backend_compiled()) {
276 return {false, "the RDMCE backend is not compiled in this build"};
277 }
278 return {true, {}};

281ExecutionResult<MaximalCliqueOutput> MaximalCliques::run(
282 const Graph& graph) const {
283 const auto started = std::chrono::steady_clock::now();
284 const auto support = supports(graph);
285 if (!support.supported) {
286 return ExecutionResult<MaximalCliqueOutput>::failure(
287 Status{StatusCode::unsupported, support.reason});
288 }
289
290 const auto selected = options_.backend == MaximalCliqueBackend::automatic
291 ? MaximalCliqueBackend::rdmce
292 : options_.backend;
293 auto provenance = provenance_for(selected);
294
295 UndirectedGraphView normalized;
296 try {
297 normalized = graph.simple_undirected(
298 {options_.allow_directed_projection});
299 } catch (const std::exception& error) {
300 return ExecutionResult<MaximalCliqueOutput>::failure(
301 Status{StatusCode::invalid_argument, error.what()}, provenance);
302 }
303
304 std::vector<std::string> warnings;
305 if (normalized.normalization.directed_projection_applied) {
306 warnings.emplace_back("directed input was projected to an undirected graph");
307 }
308 if (normalized.normalization.self_loops_removed > 0) {
309 warnings.emplace_back("self-loops were removed during normalization");
310 }
311 if (normalized.normalization.parallel_edges_collapsed > 0) {
312 warnings.emplace_back("parallel edges were collapsed during normalization");
313 }
314
315 std::uint64_t backend_count = 0;
316 double backend_ms = 0.0;
317 std::uint64_t isolated_count = 0;
318 auto backend_graph =
319 without_isolated_vertices(normalized.csr, isolated_count);
320 if (backend_graph.vertex_count() > 0) {
321 detail::BackendCountResult backend_result;
322 switch (selected) {
323 case MaximalCliqueBackend::mce_gpu:
324 backend_result = detail::run_mce_gpu_count(
325 backend_graph, options_.execution.device_ids.front());
326 break;
327 case MaximalCliqueBackend::g2_aimd:
328 backend_result = detail::run_g2_aimd_count(
329 backend_graph, options_.execution.device_ids.front());
330 break;
331 case MaximalCliqueBackend::rdmce:
332 backend_result = detail::run_rdmce_count(
333 backend_graph, options_.execution.device_ids.front());
334 break;
335 case MaximalCliqueBackend::automatic:
336 return ExecutionResult<MaximalCliqueOutput>::failure(
337 {StatusCode::internal_error,
338 "automatic maximal-clique backend was not resolved"},
339 provenance, std::move(warnings));
340 }
341 if (!backend_result.status.ok()) {
342 return ExecutionResult<MaximalCliqueOutput>::failure(
343 backend_result.status, provenance, std::move(warnings));
344 }
345 backend_count = backend_result.count + isolated_count;
346 backend_ms = backend_result.elapsed_ms;
347 } else {
348 backend_count = isolated_count;
349 }
350
351 const bool backend_total_is_qualifying = options_.minimum_clique_size == 1;
352 const auto materialization_started = std::chrono::steady_clock::now();
353 auto collected = collect_cliques(normalized.csr, options_,
354 backend_total_is_qualifying);
355 const auto materialization_finished = std::chrono::steady_clock::now();
356
357 const bool collector_finished = !collected.stopped_early;
358 if (collector_finished && options_.minimum_clique_size == 1 &&
359 collected.qualifying_count != backend_count) {
360 return ExecutionResult<MaximalCliqueOutput>::failure(
361 Status{StatusCode::correctness_mismatch,
362 std::string(to_string(selected)) +
363 " count disagreed with the exact output materializer"},
364 provenance, std::move(warnings));
365 }
366
367 MaximalCliqueOutput output;
368 output.cliques.reserve(collected.cliques.size());
369 for (const auto& internal_clique : collected.cliques) {
370 std::vector<ExternalId> clique;
371 clique.reserve(internal_clique.size());
372 for (const auto vertex : internal_clique) {
373 clique.push_back(graph.external_id(vertex));
374 }
375 output.cliques.push_back(std::move(clique));
376 }
377 output.returned_count = output.cliques.size();
378
379 std::uint64_t known_total = collected.qualifying_count;
380 bool total_known = collector_finished;
381 if (backend_total_is_qualifying) {
382 known_total = backend_count;
383 total_known = true;
384 }
385 output.complete = total_known && output.returned_count == known_total;
386 if (has_output(options_.optional_outputs,
387 MaximalCliqueOptionalOutput::total_count)) {
388 output.total_count = known_total;
389 }
390
391 warnings.emplace_back(
392 std::string(to_string(selected)) +
393 " supplies the validated GPU total; clique sets are materialized by "
394 "GraphMine's exact Bron-Kerbosch collector");
395
396 const auto finished = std::chrono::steady_clock::now();
397 ExecutionStatistics statistics;
398 statistics.backend_ms = backend_ms;
399 statistics.output_materialization_ms =
400 std::chrono::duration<double, std::milli>(materialization_finished -
401 materialization_started)
402 .count();
403 statistics.end_to_end_ms =
404 std::chrono::duration<double, std::milli>(finished - started).count();
405
406 return ExecutionResult<MaximalCliqueOutput>::success(
407 std::move(output), std::move(provenance), statistics,
408 std::move(warnings));
409}
