76struct PartitionMetrics {
77 double modularity = 0.0;
78 std::uint64_t edge_cut = 0;

81PartitionMetrics partition_metrics(const CsrGraph& graph,
82 const std::vector<std::uint32_t>& labels,
83 std::uint32_t community_count) {
84 PartitionMetrics result;
85 if (graph.neighbors.empty()) return result;
86 std::vector<std::uint64_t> degree_sum(community_count, 0);
87 std::uint64_t internal_arcs = 0;
88 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
89 degree_sum[labels[vertex]] +=
90 graph.offsets[vertex + 1U] - graph.offsets[vertex];
91 for (auto edge = graph.offsets[vertex];
92 edge < graph.offsets[vertex + 1U]; ++edge) {
93 const auto neighbor = graph.neighbors[edge];
94 if (labels[vertex] == labels[neighbor]) {
95 ++internal_arcs;
96 } else if (vertex < neighbor) {
97 ++result.edge_cut;
98 }
99 }
100 }
101 const auto arc_count = static_cast<long double>(graph.neighbors.size());
102 long double quality =
103 static_cast<long double>(internal_arcs) / arc_count;
104 for (const auto degree : degree_sum) {
105 const auto fraction = static_cast<long double>(degree) / arc_count;
106 quality -= fraction * fraction;
107 }
108 result.modularity = static_cast<double>(quality);
109 return result;
110}

133SupportReport CommunityDetection::supports(const Graph& graph) const {
134 const auto backend = selected_backend(options_);
135 if (graph.directed() && !options_.allow_directed_projection) {
136 return {false,
137 "directed input requires allow_directed_projection=true"};
138 }
139 if (options_.execution.device_ids.empty()) {
140 return {false, "at least one CUDA device id must be provided"};
141 }
142 for (const auto& edge : graph.canonical().edges) {
143 if (edge.weight &&
144 (!std::isfinite(*edge.weight) || *edge.weight != 1.0)) {
145 return {false,
146 "the validated community paths accept unweighted graphs only"};
147 }
148 }
149 if (!compiled(backend) && graph.edge_count() != 0) {
150 return {false, std::string("the ") + to_string(backend) +
151 " backend is not compiled in this build"};
152 }
153 return {true, {}};
154}

156ExecutionResult<CommunityDetectionOutput> CommunityDetection::run(
157 const Graph& graph) const {
158 const auto started = std::chrono::steady_clock::now();
159 const auto backend = selected_backend(options_);
160 auto provenance = provenance_for(backend);
161 const auto support = supports(graph);
162 if (!support.supported) {
163 return ExecutionResult<CommunityDetectionOutput>::failure(
164 {StatusCode::unsupported, support.reason}, provenance);
165 }
166
167 UndirectedGraphView normalized;
168 try {
169 normalized = graph.simple_undirected(
170 {options_.allow_directed_projection});
171 } catch (const std::exception& error) {
172 return ExecutionResult<CommunityDetectionOutput>::failure(
173 {StatusCode::invalid_argument, error.what()}, provenance);
174 }
175
176 std::vector<std::string> warnings;
177 if (normalized.normalization.directed_projection_applied) {
178 warnings.emplace_back("directed input was projected to an undirected graph");
179 }
180 if (normalized.normalization.self_loops_removed > 0) {
181 warnings.emplace_back("self-loops were removed during normalization");
182 }
183 if (normalized.normalization.parallel_edges_collapsed > 0) {
184 warnings.emplace_back("parallel edges were collapsed during normalization");
185 }
186
187 detail::CommunityDetectionBackendResult backend_result;
188 if (normalized.csr.vertex_count() == 0) {
189 backend_result.status = Status::success();
190 provenance.execution_path += " (trivial empty-graph result)";
191 } else if (normalized.csr.neighbors.empty()) {
192 backend_result.status = Status::success();
193 backend_result.labels.resize(normalized.csr.vertex_count());
194 for (std::size_t vertex = 0; vertex < backend_result.labels.size();
195 ++vertex) {
196 backend_result.labels[vertex] = static_cast<std::uint32_t>(vertex);
197 }
198 backend_result.native_modularity = 0.0;
199 provenance.execution_path += " (trivial edgeless-graph result)";
200 } else if (backend == CommunityDetectionBackend::gleiden) {
201 backend_result = detail::run_gleiden(
202 normalized.csr, options_.execution.device_ids.front());
203 } else {
204 backend_result = detail::run_pggc(
205 normalized.csr, options_.execution.device_ids.front(),
206 pggc_algorithm(backend));
207 }
208 if (!backend_result.status.ok()) {
209 return ExecutionResult<CommunityDetectionOutput>::failure(
210 backend_result.status, provenance, std::move(warnings));
211 }
212 if (backend_result.labels.size() != normalized.csr.vertex_count()) {
213 return ExecutionResult<CommunityDetectionOutput>::failure(
214 {StatusCode::correctness_mismatch,
215 std::string(to_string(backend)) +
216 " did not return one label per input vertex"},
217 provenance, std::move(warnings));
218 }
219
220 // Community IDs are semantically arbitrary. Densify by first appearance
221 // to give callers stable, compact IDs without changing the partition.
222 std::unordered_map<std::uint32_t, std::uint32_t> dense_labels;
223 std::vector<std::uint32_t> labels;
224 labels.reserve(backend_result.labels.size());
225 for (const auto raw : backend_result.labels) {
226 const auto inserted = dense_labels.emplace(
227 raw, static_cast<std::uint32_t>(dense_labels.size()));
228 labels.push_back(inserted.first->second);
229 }
230 if (dense_labels.size() >
231 static_cast<std::size_t>(std::numeric_limits<std::uint32_t>::max())) {
232 return ExecutionResult<CommunityDetectionOutput>::failure(
233 {StatusCode::resource_exhausted,
234 "community count exceeds the public label type"},
235 provenance, std::move(warnings));
236 }
237 const auto community_count =
238 static_cast<std::uint32_t>(dense_labels.size());
239 const auto metrics =
240 partition_metrics(normalized.csr, labels, community_count);
241 if (backend_result.native_modularity &&
242 std::abs(*backend_result.native_modularity - metrics.modularity) >
243 1e-7) {
244 return ExecutionResult<CommunityDetectionOutput>::failure(
245 {StatusCode::correctness_mismatch,
246 std::string(to_string(backend)) + " reported modularity " +
247 std::to_string(*backend_result.native_modularity) +
248 ", but the returned partition has modularity " +
249 std::to_string(metrics.modularity)},
250 provenance, std::move(warnings));
251 }
252
253 const auto materialization_started = std::chrono::steady_clock::now();
254 CommunityDetectionOutput output;
255 output.community_count = community_count;
256 output.modularity = metrics.modularity;
257 output.assignment_by_vertex.reserve(graph.vertex_count());
258 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
259 output.assignment_by_vertex.push_back(
260 {graph.external_id(vertex), labels[vertex]});
261 }
262 if (has_output(options_.optional_outputs,
263 CommunityDetectionOptionalOutput::communities)) {
264 output.communities.emplace(community_count);
265 for (std::uint32_t community = 0; community < community_count;
266 ++community) {
267 (*output.communities)[community].id = community;
268 }
269 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
270 (*output.communities)[labels[vertex]].vertices.push_back(
271 graph.external_id(vertex));
272 }
273 }
274 if (has_output(options_.optional_outputs,
275 CommunityDetectionOptionalOutput::edge_cut)) {
276 output.edge_cut = metrics.edge_cut;
277 }
278 const auto materialization_finished = std::chrono::steady_clock::now();
279 const auto finished = std::chrono::steady_clock::now();
280
281 ExecutionStatistics statistics;
282 statistics.backend_ms = backend_result.elapsed_ms;
283 statistics.output_materialization_ms =
284 std::chrono::duration<double, std::milli>(materialization_finished -
285 materialization_started)
286 .count();
287 statistics.end_to_end_ms =
288 std::chrono::duration<double, std::milli>(finished - started).count();
289 return ExecutionResult<CommunityDetectionOutput>::success(
290 std::move(output), std::move(provenance), statistics,
291 std::move(warnings));
292}
