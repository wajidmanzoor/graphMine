56std::uint64_t count_triangles(const CsrGraph& graph) {
57 std::uint64_t triangles = 0;
58 for (VertexIndex source = 0; source < graph.vertex_count(); ++source) {
59 for (auto offset = graph.offsets[source];
60 offset < graph.offsets[source + 1]; ++offset) {
61 const auto middle = graph.neighbors[offset];
62 if (middle <= source) continue;
63 for (auto other = graph.offsets[middle];
64 other < graph.offsets[middle + 1]; ++other) {
65 const auto target = graph.neighbors[other];
66 if (target <= middle) continue;
67 const auto first = graph.neighbors.begin() + graph.offsets[source];
68 const auto last = graph.neighbors.begin() + graph.offsets[source + 1];
69 if (std::binary_search(first, last, target)) ++triangles;
70 }
71 }
72 }
73 return triangles;
74}

76std::vector<std::uint64_t> common_order(
77 std::uint32_t motif_size,
78 const std::vector<unsigned long long>& graphset_counts) {
79 if (motif_size == 3 && graphset_counts.size() == 2) {
80 return {graphset_counts[0], graphset_counts[1]};
81 }
82 if (motif_size == 4 && graphset_counts.size() == 6) {
83 // MotifGenerator emits clique, diamond, tailed triangle, 3-star,
84 // 4-cycle, path. The facade's common order is star, path, tailed
85 // triangle, cycle, diamond, clique.
86 return {graphset_counts[3], graphset_counts[5], graphset_counts[2],
87 graphset_counts[4], graphset_counts[1], graphset_counts[0]};
88 }
89 return {};
90}

96GraphMotifBackendResult run_graphset_motifs(const CsrGraph& graph,
97 std::uint32_t motif_size,
98 int device_id) {
99 const auto result_size = motif_size == 3 ? 2U : 6U;
100 if (motif_size < 3 || motif_size > 4) {
101 return {{StatusCode::unsupported,
102 "the validated GraphSet motif path supports sizes 3 and 4"},
103 {}, 0.0};
104 }
105 if (graph.vertex_count() < motif_size || graph.neighbors.empty()) {
106 return {Status::success(), std::vector<std::uint64_t>(result_size, 0),
107 0.0};
108 }
109 if (graph.vertex_count() > static_cast<std::size_t>(INT_MAX) ||
110 graph.neighbors.size() >
111 static_cast<std::size_t>(std::numeric_limits<std::int64_t>::max())) {
112 return {{StatusCode::unsupported,
113 "GraphSet uses signed 32-bit vertices and signed 64-bit offsets"},
114 {}, 0.0};
115 }
116
117 std::size_t maximum_degree = 0;
118 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
119 maximum_degree = std::max(
120 maximum_degree,
121 static_cast<std::size_t>(graph.offsets[vertex + 1] -
122 graph.offsets[vertex]));
123 }
124 if (maximum_degree > static_cast<std::size_t>(INT_MAX)) {
125 return {{StatusCode::unsupported,
126 "GraphSet uses signed 32-bit neighborhood sizes"},
127 {}, 0.0};
128 }
129
130 int device_count = 0;
131 if (device_id < 0 || cudaGetDeviceCount(&device_count) != cudaSuccess ||
132 device_id >= device_count || cudaSetDevice(device_id) != cudaSuccess) {
133 return {{StatusCode::backend_unavailable,
134 "requested CUDA device is unavailable for GraphSet motifs"},
135 {}, 0.0};
136 }
137
138 try {
139 const auto triangle_count = count_triangles(graph);
140 if (triangle_count >
141 static_cast<std::uint64_t>(std::numeric_limits<long long>::max())) {
142 return {{StatusCode::unsupported,
143 "GraphSet's triangle statistic exceeds signed 64-bit range"},
144 {}, 0.0};
145 }
146
147 auto original = std::make_unique<GraphMineGraphsetMotifGraph>();
148 original->v_cnt = static_cast<std::int32_t>(graph.vertex_count());
149 original->e_cnt = static_cast<std::int64_t>(graph.neighbors.size());
150 original->tri_cnt = static_cast<long long>(triangle_count);
151 original->vertex = new std::int64_t[graph.offsets.size()];
152 original->edge = new std::int32_t[graph.neighbors.size()];
153 std::copy(graph.offsets.begin(), graph.offsets.end(), original->vertex);
154 std::copy(graph.neighbors.begin(), graph.neighbors.end(), original->edge);
155
156 const std::lock_guard<std::mutex> lock(graphset_motif_mutex);
157 GraphMineGraphsetMotifVertexSet::max_intersection_size =
158 static_cast<int>(maximum_degree);
159 std::vector<unsigned long long> raw_counts;
160 const auto started = std::chrono::steady_clock::now();
161 graphmine_graphset_motif_counts(original.get(),
162 static_cast<int>(motif_size), raw_counts);
163 const auto cuda_status = cudaDeviceSynchronize();
164 const auto finished = std::chrono::steady_clock::now();
165 if (cuda_status != cudaSuccess) {
166 return {{StatusCode::execution_failed,
167 std::string("GraphSet motif execution failed: ") +
168 cudaGetErrorString(cuda_status)},
169 {}, 0.0};
170 }
171 auto counts = common_order(motif_size, raw_counts);
172 if (counts.size() != result_size) {
173 return {{StatusCode::internal_error,
174 "GraphSet returned an unexpected motif count vector"},
175 {}, 0.0};
176 }
177 return {Status::success(), std::move(counts),
178 std::chrono::duration<double, std::milli>(finished - started)
179 .count()};
180 } catch (const std::bad_alloc&) {
181 return {{StatusCode::resource_exhausted,
182 "GraphSet motif host allocation failed"},
183 {}, 0.0};
184 } catch (const std::exception& error) {
185 return {{StatusCode::execution_failed,
186 std::string("GraphSet motif execution failed: ") + error.what()},
187 {}, 0.0};
188 }
189}
