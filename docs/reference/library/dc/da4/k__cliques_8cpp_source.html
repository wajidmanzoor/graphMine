22void enumerate_k_cliques(const CsrGraph& graph, std::uint32_t k,
23 const KCliqueOptions& options,
24 std::vector<VertexIndex>& current,
25 const std::vector<VertexIndex>& candidates,
26 KCliqueMaterialization& output) {
27 const auto needed = static_cast<std::size_t>(k) - current.size();
28 if (needed == 0) {
29 if (output.count == std::numeric_limits<std::uint64_t>::max()) {
30 throw std::overflow_error("k-clique count exceeds uint64_t");
31 }
32 ++output.count;
33 if (options.enumerate &&
34 (!options.result_limit ||
35 output.instances.size() < *options.result_limit)) {
36 output.instances.push_back(current);
37 }
38 if (options.include_per_vertex_counts) {
39 for (const auto vertex : current) {
40 ++output.per_vertex[vertex];
41 }
42 }
43 return;
44 }
45 if (candidates.size() < needed) {
46 return;
47 }
48
49 for (std::size_t index = 0;
50 index + needed <= candidates.size(); ++index) {
51 const auto vertex = candidates[index];
52 current.push_back(vertex);
53 std::vector<VertexIndex> next;
54 if (needed > 1) {
55 const auto neighbor_begin =
56 graph.neighbors.begin() +
57 static_cast<std::ptrdiff_t>(graph.offsets[vertex]);
58 const auto neighbor_end =
59 graph.neighbors.begin() +
60 static_cast<std::ptrdiff_t>(graph.offsets[vertex + 1]);
61 std::set_intersection(candidates.begin() +
62 static_cast<std::ptrdiff_t>(index + 1),
63 candidates.end(), neighbor_begin, neighbor_end,
64 std::back_inserter(next));
65 }
66 enumerate_k_cliques(graph, k, options, current, next, output);
67 current.pop_back();
68 }
69}

71KCliqueMaterialization materialize_k_cliques(const CsrGraph& graph,
72 const KCliqueOptions& options) {
73 KCliqueMaterialization output;
74 if (options.include_per_vertex_counts) {
75 output.per_vertex.resize(graph.vertex_count(), 0);
76 }
77 std::vector<VertexIndex> candidates(graph.vertex_count());
78 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
79 candidates[vertex] = vertex;
80 }
81 std::vector<VertexIndex> current;
82 current.reserve(options.k);
83 enumerate_k_cliques(graph, options.k, options, current, candidates, output);
84 return output;
85}

137SupportReport KCliques::supports(const Graph& graph) const {
138 if (options_.k < 2) {
139 return {false, "k must be at least two"};
140 }
141 if (graph.directed() && !options_.allow_directed_projection) {
142 return {false,
143 "directed input requires allow_directed_projection=true"};
144 }
145 if (options_.execution.device_ids.empty()) {
146 return {false, "at least one CUDA device id must be provided"};
147 }
148 if (options_.result_limit && *options_.result_limit == 0) {
149 return {false, "result_limit must be null or at least one"};
150 }
151 if (options_.result_limit && !options_.enumerate) {
152 return {false, "result_limit is meaningful only when enumerate=true"};
153 }
154 const auto selected = options_.backend == KCliqueBackend::automatic
155 ? KCliqueBackend::graphset
156 : options_.backend;
157 if (selected == KCliqueBackend::gamma && options_.k > 5 &&
158 graph.vertex_count() >= options_.k) {
159 return {false, "the validated GAMMA path supports k up to five"};
160 }
161 if (selected == KCliqueBackend::kcgpu && options_.k > 5 &&
162 graph.vertex_count() >= options_.k) {
163 return {false, "the validated KCGPU path supports k up to five"};
164 }
165 if (selected == KCliqueBackend::graphset && options_.k > 6 &&
166 graph.vertex_count() >= options_.k) {
167 return {false, "the validated GraphSet path supports k up to six"};
168 }
169 if (selected == KCliqueBackend::kcgpu &&
170 !detail::kcgpu_backend_compiled() && options_.k > 2 &&
171 graph.vertex_count() >= options_.k) {
172 return {false, "the KCGPU backend is not compiled"};
173 }
174 if (selected == KCliqueBackend::graphset &&
175 !detail::graphset_kclique_backend_compiled() && options_.k > 2 &&
176 graph.vertex_count() >= options_.k) {
177 return {false, "the GraphSet k-clique backend is not compiled"};
178 }
179 if (selected == KCliqueBackend::gamma &&
180 !detail::gamma_kclique_backend_compiled() && options_.k > 2 &&
181 graph.vertex_count() >= options_.k) {
182 return {false, "the GAMMA backend is not compiled"};
183 }
184 return {true, {}};
185}

187ExecutionResult<KCliqueOutput> KCliques::run(const Graph& graph) const {
188 const auto started = std::chrono::steady_clock::now();
189 const auto selected = options_.backend == KCliqueBackend::automatic
190 ? KCliqueBackend::graphset
191 : options_.backend;
192 auto provenance = kclique_provenance(selected);
193 const auto support = supports(graph);
194 if (!support.supported) {
195 return ExecutionResult<KCliqueOutput>::failure(
196 {StatusCode::unsupported, support.reason}, provenance);
197 }
198
199 UndirectedGraphView normalized;
200 try {
201 normalized = graph.simple_undirected(
202 {options_.allow_directed_projection});
203 } catch (const std::exception& error) {
204 return ExecutionResult<KCliqueOutput>::failure(
205 {StatusCode::invalid_argument, error.what()}, provenance);
206 }
207
208 std::vector<std::string> warnings;
209 if (normalized.normalization.directed_projection_applied) {
210 warnings.emplace_back("directed input was projected to an undirected graph");
211 }
212 if (normalized.normalization.self_loops_removed > 0) {
213 warnings.emplace_back("self-loops were removed during normalization");
214 }
215 if (normalized.normalization.parallel_edges_collapsed > 0) {
216 warnings.emplace_back("parallel edges were collapsed during normalization");
217 }
218
219 detail::KCliqueBackendResult backend;
220 if (options_.k > normalized.csr.vertex_count()) {
221 backend.status = Status::success();
222 provenance.execution_path += " (trivial k-greater-than-n result)";
223 } else if (options_.k == 2) {
224 backend.status = Status::success();
225 backend.count = normalized.csr.undirected_edge_count();
226 provenance.execution_path += " (common exact k=2 path)";
227 } else if (normalized.csr.undirected_edge_count() <
228 (static_cast<std::uint64_t>(options_.k) *
229 (options_.k - 1U)) /
230 2U) {
231 backend.status = Status::success();
232 provenance.execution_path += " (trivial insufficient-edge result)";
233 } else {
234 switch (selected) {
235 case KCliqueBackend::kcgpu:
236 backend = detail::run_kcgpu(normalized.csr, options_.k,
237 options_.execution.device_ids.front());
238 break;
239 case KCliqueBackend::graphset:
240 backend = detail::run_graphset_kclique(
241 normalized.csr, options_.k,
242 options_.execution.device_ids.front());
243 break;
244 case KCliqueBackend::gamma:
245 backend = detail::run_gamma_kclique(
246 normalized.csr, options_.k,
247 options_.execution.device_ids.front());
248 break;
249 case KCliqueBackend::automatic:
250 return ExecutionResult<KCliqueOutput>::failure(
251 {StatusCode::internal_error,
252 "k-clique backend was not resolved"},
253 provenance, std::move(warnings));
254 }
255 }
256 if (!backend.status.ok()) {
257 return ExecutionResult<KCliqueOutput>::failure(
258 backend.status, provenance, std::move(warnings));
259 }
260
261 KCliqueOutput output;
262 output.count = backend.count;
263 const auto materialization_started = std::chrono::steady_clock::now();
264 if (options_.enumerate || options_.include_per_vertex_counts) {
265 KCliqueMaterialization materialized;
266 try {
267 materialized = materialize_k_cliques(normalized.csr, options_);
268 } catch (const std::exception& error) {
269 return ExecutionResult<KCliqueOutput>::failure(
270 {StatusCode::resource_exhausted, error.what()}, provenance,
271 std::move(warnings));
272 }
273 if (materialized.count != backend.count) {
274 return ExecutionResult<KCliqueOutput>::failure(
275 {StatusCode::correctness_mismatch,
276 std::string(to_string(selected)) + " count " +
277 std::to_string(backend.count) +
278 " disagreed with exact materializer count " +
279 std::to_string(materialized.count)},
280 provenance, std::move(warnings));
281 }
282 if (options_.enumerate) {
283 output.cliques.emplace();
284 output.cliques->reserve(materialized.instances.size());
285 for (const auto& clique : materialized.instances) {
286 output.cliques->emplace_back();
287 output.cliques->back().reserve(clique.size());
288 for (const auto vertex : clique) {
289 output.cliques->back().push_back(graph.external_id(vertex));
290 }
291 }
292 output.complete = output.cliques->size() == output.count;
293 }
294 if (options_.include_per_vertex_counts) {
295 output.per_vertex_count.emplace();
296 output.per_vertex_count->reserve(graph.vertex_count());
297 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
298 output.per_vertex_count->push_back(
299 {graph.external_id(vertex), materialized.per_vertex[vertex]});
300 }
301 }
302 }
303 const auto materialization_finished = std::chrono::steady_clock::now();
304 const auto finished = std::chrono::steady_clock::now();
305
306 ExecutionStatistics statistics;
307 statistics.backend_ms = backend.elapsed_ms;
308 statistics.output_materialization_ms =
309 std::chrono::duration<double, std::milli>(materialization_finished -
310 materialization_started)
311 .count();
312 statistics.end_to_end_ms =
313 std::chrono::duration<double, std::milli>(finished - started).count();
314 return ExecutionResult<KCliqueOutput>::success(
315 std::move(output), std::move(provenance), statistics,
316 std::move(warnings));
317}
