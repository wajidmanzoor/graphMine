16bool is_clique(const CsrGraph& graph,
17 const std::vector<VertexIndex>& vertices) {
18 for (std::size_t i = 0; i < vertices.size(); ++i) {
19 const auto begin = graph.neighbors.begin() +
20 static_cast<std::ptrdiff_t>(graph.offsets[vertices[i]]);
21 const auto end = graph.neighbors.begin() + static_cast<std::ptrdiff_t>(
22 graph.offsets[vertices[i] + 1]);
23 for (std::size_t j = i + 1; j < vertices.size(); ++j) {
24 if (!std::binary_search(begin, end, vertices[j])) {
25 return false;
26 }
27 }
28 }
29 return true;
30}

32std::uint32_t greedy_coloring_upper_bound(const CsrGraph& graph) {
33 if (graph.vertex_count() == 0) {
34 return 0;
35 }
36 std::vector<VertexIndex> order(graph.vertex_count());
37 std::iota(order.begin(), order.end(), VertexIndex{0});
38 std::sort(order.begin(), order.end(), [&](VertexIndex lhs, VertexIndex rhs) {
39 const auto lhs_degree = graph.offsets[lhs + 1] - graph.offsets[lhs];
40 const auto rhs_degree = graph.offsets[rhs + 1] - graph.offsets[rhs];
41 return lhs_degree == rhs_degree ? lhs < rhs : lhs_degree > rhs_degree;
42 });
43
44 std::vector<int> color(graph.vertex_count(), -1);
45 std::uint32_t color_count = 0;
46 for (const auto vertex : order) {
47 std::vector<bool> used(color_count, false);
48 for (auto offset = graph.offsets[vertex];
49 offset < graph.offsets[vertex + 1]; ++offset) {
50 const auto neighbor_color = color[graph.neighbors[offset]];
51 if (neighbor_color >= 0) {
52 used[static_cast<std::size_t>(neighbor_color)] = true;
53 }
54 }
55 std::uint32_t selected = 0;
56 while (selected < used.size() && used[selected]) {
57 ++selected;
58 }
59 if (selected == color_count) {
60 ++color_count;
61 }
62 color[vertex] = static_cast<int>(selected);
63 }
64 return color_count;
65}

116SupportReport MaximumClique::supports(const Graph& graph) const {
117 const auto selected = options_.backend == MaximumCliqueBackend::automatic
118 ? MaximumCliqueBackend::cuda_ms
119 : options_.backend;
120 if (graph.directed() && !options_.allow_directed_projection) {
121 return {false,
122 "directed input requires allow_directed_projection=true"};
123 }
124 if (options_.execution.device_ids.empty()) {
125 return {false, "at least one CUDA device id must be provided"};
126 }
127 if (options_.known_lower_bound &&
128 *options_.known_lower_bound > graph.vertex_count()) {
129 return {false, "known_lower_bound exceeds the graph vertex count"};
130 }
131 if (options_.return_all_ties &&
132 selected != MaximumCliqueBackend::gpu_maximum_clique) {
133 return {false,
134 "return_all_ties is only available with GPUMaximumClique"};
135 }
136 bool compiled = false;
137 switch (selected) {
138 case MaximumCliqueBackend::cuda_ms:
139 compiled = detail::cuda_ms_backend_compiled();
140 break;
141 case MaximumCliqueBackend::gpu_maximum_clique:
142 compiled = detail::gpu_maximum_clique_backend_compiled();
143 break;
144 case MaximumCliqueBackend::maximum_clique_on_gpu:
145 compiled = detail::maximum_clique_on_gpu_backend_compiled();
146 break;
147 case MaximumCliqueBackend::automatic:
148 break;
149 }
150 if (!compiled) {
151 return {false, std::string("the ") + to_string(selected) +
152 " backend is not compiled in this build"};
153 }
154 return {true, {}};
155}

157ExecutionResult<MaximumCliqueOutput> MaximumClique::run(
158 const Graph& graph) const {
159 const auto started = std::chrono::steady_clock::now();
160 const auto support = supports(graph);
161 const auto selected = options_.backend == MaximumCliqueBackend::automatic
162 ? MaximumCliqueBackend::cuda_ms
163 : options_.backend;
164 auto provenance = maximum_clique_provenance(selected);
165 if (!support.supported) {
166 return ExecutionResult<MaximumCliqueOutput>::failure(
167 {StatusCode::unsupported, support.reason}, provenance);
168 }
169
170 UndirectedGraphView normalized;
171 try {
172 normalized = graph.simple_undirected(
173 {options_.allow_directed_projection});
174 } catch (const std::exception& error) {
175 return ExecutionResult<MaximumCliqueOutput>::failure(
176 {StatusCode::invalid_argument, error.what()}, provenance);
177 }
178
179 std::vector<std::string> warnings;
180 if (normalized.normalization.directed_projection_applied) {
181 warnings.emplace_back("directed input was projected to an undirected graph");
182 }
183 if (normalized.normalization.self_loops_removed > 0) {
184 warnings.emplace_back("self-loops were removed during normalization");
185 }
186 if (normalized.normalization.parallel_edges_collapsed > 0) {
187 warnings.emplace_back("parallel edges were collapsed during normalization");
188 }
189
190 detail::MaximumCliqueBackendResult backend;
191 if (normalized.csr.vertex_count() == 0) {
192 backend.status = Status::success();
193 } else if (normalized.csr.undirected_edge_count() == 0) {
194 backend.status = Status::success();
195 backend.clique = {0};
196 provenance.execution_path += " (trivial edgeless-graph result)";
197 } else {
198 const auto lower_bound = options_.known_lower_bound.value_or(0U);
199 switch (selected) {
200 case MaximumCliqueBackend::cuda_ms:
201 backend = detail::run_cuda_ms(
202 normalized.csr, options_.execution.device_ids.front());
203 break;
204 case MaximumCliqueBackend::gpu_maximum_clique:
205 backend = detail::run_gpu_maximum_clique(
206 normalized.csr, options_.execution.device_ids.front(),
207 lower_bound, options_.return_all_ties);
208 break;
209 case MaximumCliqueBackend::maximum_clique_on_gpu:
210 backend = detail::run_maximum_clique_on_gpu(
211 normalized.csr, options_.execution.device_ids.front(),
212 lower_bound);
213 break;
214 case MaximumCliqueBackend::automatic:
215 break;
216 }
217 }
218 if (!backend.status.ok()) {
219 return ExecutionResult<MaximumCliqueOutput>::failure(
220 backend.status, provenance, std::move(warnings));
221 }
222 if (!is_clique(normalized.csr, backend.clique)) {
223 return ExecutionResult<MaximumCliqueOutput>::failure(
224 {StatusCode::correctness_mismatch,
225 std::string(to_string(selected)) +
226 " returned vertices that do not form a clique"},
227 provenance, std::move(warnings));
228 }
229 for (const auto& tied_clique : backend.tied_cliques) {
230 if (tied_clique.size() != backend.clique.size() ||
231 !is_clique(normalized.csr, tied_clique)) {
232 return ExecutionResult<MaximumCliqueOutput>::failure(
233 {StatusCode::correctness_mismatch,
234 std::string(to_string(selected)) +
235 " returned an invalid tied maximum clique"},
236 provenance, std::move(warnings));
237 }
238 }
239 if (options_.known_lower_bound &&
240 backend.clique.size() < *options_.known_lower_bound) {
241 return ExecutionResult<MaximumCliqueOutput>::failure(
242 {StatusCode::correctness_mismatch,
243 std::string(to_string(selected)) +
244 " result is smaller than the caller's valid lower bound"},
245 provenance, std::move(warnings));
246 }
247
248 auto certified_upper = greedy_coloring_upper_bound(normalized.csr);
249 if (backend.relaxation_upper_size >= backend.clique.size() &&
250 backend.relaxation_upper_size != 0U) {
251 certified_upper =
252 std::min(certified_upper, backend.relaxation_upper_size);
253 }
254 if (backend.clique.size() > certified_upper) {
255 return ExecutionResult<MaximumCliqueOutput>::failure(
256 {StatusCode::internal_error,
257 "computed coloring bound is below the feasible clique size"},
258 provenance, std::move(warnings));
259 }
260
261 MaximumCliqueOutput output;
262 output.maximum_size = static_cast<std::uint32_t>(backend.clique.size());
263 const auto append_clique = [&](const std::vector<VertexIndex>& clique) {
264 output.cliques.emplace_back();
265 output.cliques.back().reserve(clique.size());
266 for (const auto vertex : clique) {
267 output.cliques.back().push_back(graph.external_id(vertex));
268 }
269 };
270 if (backend.tied_cliques.empty()) {
271 append_clique(backend.clique);
272 } else {
273 for (const auto& clique : backend.tied_cliques) append_clique(clique);
274 }
275 output.optimal = selected != MaximumCliqueBackend::cuda_ms ||
276 output.maximum_size == certified_upper;
277 if (has_output(options_.optional_outputs,
278 MaximumCliqueOptionalOutput::upper_bound)) {
279 output.upper_bound = certified_upper;
280 }
281 if (!output.optimal) {
282 warnings.emplace_back(
283 std::string(to_string(selected)) +
284 " returned a feasible clique, but the common coloring bound "
285 "did not prove global optimality");
286 }
287
288 const auto finished = std::chrono::steady_clock::now();
289 ExecutionStatistics statistics;
290 statistics.backend_ms = backend.elapsed_ms;
291 statistics.end_to_end_ms =
292 std::chrono::duration<double, std::milli>(finished - started).count();
293 return ExecutionResult<MaximumCliqueOutput>::success(
294 std::move(output), std::move(provenance), statistics,
295 std::move(warnings));
296}
