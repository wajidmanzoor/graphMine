69BackendCountResult run_mce_gpu_count(const CsrGraph& graph, int device_id) {
70 if (graph.vertex_count() == 0) {
71 return {Status::success(), 0, 0.0};
72 }
73 if (graph.vertex_count() > static_cast<std::size_t>(UINT_MAX - 1U) ||
74 graph.neighbors.size() > static_cast<std::size_t>(UINT_MAX)) {
75 return {{StatusCode::unsupported,
76 "mce-gpu uses unsigned 32-bit graph indices"},
77 0, 0.0};
78 }
79
80 int device_count = 0;
81 if (device_id < 0 || cudaGetDeviceCount(&device_count) != cudaSuccess ||
82 device_id >= device_count || cudaSetDevice(device_id) != cudaSuccess) {
83 return {{StatusCode::backend_unavailable,
84 "requested CUDA device is unavailable for mce-gpu"},
85 0, 0.0};
86 }
87
88 try {
89 using UpstreamGraph =
90 graphmine_mce_gpu_upstream::COOCSRGraph_d<DataType>;
91 // The artifact's converter and validated inputs reserve vertex zero and
92 // store graph vertices from one. Preserve that convention in memory; its
93 // kernels ignore the isolated sentinel in the maximal-clique total.
94 const auto vertex_count =
95 static_cast<DataType>(graph.vertex_count() + 1U);
96 const auto directed_edges = static_cast<DataType>(graph.neighbors.size());
97
98 std::vector<DataType> offsets(vertex_count + 1U, 0);
99 for (std::size_t vertex = 0; vertex <= graph.vertex_count(); ++vertex) {
100 offsets[vertex + 1U] = static_cast<DataType>(graph.offsets[vertex]);
101 }
102 std::vector<DataType> neighbors;
103 neighbors.reserve(graph.neighbors.size());
104 for (const auto neighbor : graph.neighbors) {
105 neighbors.push_back(neighbor + 1U);
106 }
107 std::vector<DataType> row_indices(neighbors.size());
108 for (DataType vertex = 0; vertex < vertex_count; ++vertex) {
109 std::fill(row_indices.begin() + offsets[vertex],
110 row_indices.begin() + offsets[vertex + 1U], vertex);
111 }
112
113 UpstreamGraph original{};
114 original.numNodes = vertex_count;
115 original.numEdges = directed_edges;
116 original.capacity = directed_edges;
117 CUDA_RUNTIME(cudaMallocManaged(&original.rowPtr,
118 sizeof(DataType) * (vertex_count + 1ULL)));
119 CUDA_RUNTIME(cudaMallocManaged(&original.rowInd,
120 sizeof(DataType) * directed_edges));
121 CUDA_RUNTIME(cudaMallocManaged(&original.colInd,
122 sizeof(DataType) * directed_edges));
123 CUDA_RUNTIME(cudaMemcpy(original.rowPtr, offsets.data(),
124 sizeof(DataType) * (vertex_count + 1ULL),
125 cudaMemcpyHostToDevice));
126 CUDA_RUNTIME(cudaMemcpy(original.rowInd, row_indices.data(),
127 sizeof(DataType) * directed_edges,
128 cudaMemcpyHostToDevice));
129 CUDA_RUNTIME(cudaMemcpy(original.colInd, neighbors.data(),
130 sizeof(DataType) * directed_edges,
131 cudaMemcpyHostToDevice));
132
133 graphmine_mce_gpu_upstream::SingleGPU_Kcore<DataType, PeelType> kcore(
134 device_id);
135 kcore.findKcoreIncremental_async(original);
136
137 UpstreamGraph oriented{};
138 oriented.numNodes = vertex_count;
139 oriented.numEdges = directed_edges;
140 oriented.capacity = directed_edges;
141 CUDA_RUNTIME(cudaMallocManaged(&oriented.colInd,
142 sizeof(DataType) * directed_edges));
143 CUDA_RUNTIME(cudaMallocManaged(&oriented.splitPtr,
144 sizeof(DataType) * (vertex_count + 1ULL)));
145 CUDA_RUNTIME(cudaMallocManaged(&oriented.rowPtr,
146 sizeof(DataType) * (vertex_count + 1ULL)));
147 CUDA_RUNTIME(cudaMemcpy(oriented.rowPtr, original.rowPtr,
148 sizeof(DataType) * (vertex_count + 1ULL),
149 cudaMemcpyDefault));
150 CUDA_RUNTIME(cudaMallocManaged(&oriented.rowInd,
151 sizeof(DataType) * directed_edges));
152 CUDA_RUNTIME(cudaMemcpy(oriented.rowInd, original.rowInd,
153 sizeof(DataType) * directed_edges,
154 cudaMemcpyDefault));
155
156 constexpr std::size_t block_size = 1024;
157 graphmine_mce_gpu_upstream::GPUArray<DataType> tmp_block(
158 "Temp Block", AllocationTypeEnum::unified,
159 (directed_edges + block_size - 1U) / block_size, device_id);
160 graphmine_mce_gpu_upstream::GPUArray<DataType> split_ptr(
161 "Split Ptr", AllocationTypeEnum::unified, vertex_count + 1ULL,
162 device_id);
163 tmp_block.setAll(0, true);
164 split_ptr.setAll(0, true);
165 execKernel((graphmine_mce_gpu_set_priority<DataType>),
166 (directed_edges + block_size - 1U) / block_size, block_size,
167 device_id, false, original, directed_edges,
168 kcore.nodePriority.gdata(), tmp_block.gdata(),
169 split_ptr.gdata());
170 CUDA_RUNTIME(cudaMemcpy(oriented.splitPtr, split_ptr.gdata(),
171 sizeof(DataType) * (vertex_count + 1ULL),
172 cudaMemcpyDefault));
173 execKernel((graphmine_mce_gpu_split_pointer<DataType>),
174 (vertex_count + 1ULL + block_size - 1U) / block_size,
175 block_size, device_id, false, original, oriented.splitPtr);
176 CUBScanExclusive<DataType, DataType>(
177 split_ptr.gdata(), split_ptr.gdata(), vertex_count + 1U, device_id, 0);
178 CUBScanExclusive<DataType, DataType>(
179 tmp_block.gdata(), tmp_block.gdata(),
180 (directed_edges + block_size - 1U) / block_size, device_id, 0);
181 execKernel((graphmine_mce_gpu_split_data<DataType, block_size>),
182 (directed_edges + block_size - 1U) / block_size, block_size,
183 device_id, false, original, directed_edges,
184 kcore.nodePriority.gdata(), tmp_block.gdata(),
185 split_ptr.gdata(), oriented.splitPtr, oriented.colInd);
186 tmp_block.freeGPU();
187 split_ptr.freeGPU();
188
189 CUDA_RUNTIME(cudaFree(original.rowPtr));
190 CUDA_RUNTIME(cudaFree(original.rowInd));
191 CUDA_RUNTIME(cudaFree(original.colInd));
192
193 GraphMineMceGpuConfig config{};
194 config.mt = MAINTASK::MCE;
195 config.level = PARLEVEL::L2;
196 config.induced = INDUCEDSUBGRAPH::IPX;
197 config.workerlist = WORKERLIST::WL;
198 config.deviceId = device_id;
199 config.gpus = {device_id};
200
201 graphmine_mce_gpu_upstream::MultiGPU_MCE<DataType> counter(device_id, 0,
202 1);
203 const auto started = std::chrono::steady_clock::now();
204 if (kcore.count() <= 300U) {
205 counter.mce_count<1>(oriented, config);
206 } else {
207 counter.mce_count<8>(oriented, config);
208 }
209 counter.sync();
210 const auto count = counter.mce_counter.getSingle(0);
211 const auto finished = std::chrono::steady_clock::now();
212
213 CUDA_RUNTIME(cudaFree(oriented.colInd));
214 CUDA_RUNTIME(cudaFree(oriented.splitPtr));
215 CUDA_RUNTIME(cudaFree(oriented.rowPtr));
216 CUDA_RUNTIME(cudaFree(oriented.rowInd));
217
218 return {Status::success(), static_cast<std::uint64_t>(count),
219 std::chrono::duration<double, std::milli>(finished - started)
220 .count()};
221 } catch (const std::exception& error) {
222 return {{StatusCode::execution_failed,
223 std::string("mce-gpu execution failed: ") + error.what()},
224 0, 0.0};
225 }
