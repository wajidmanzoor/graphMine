38SupportReport BetweennessCentrality::supports(const Graph& graph) const {
39 if (graph.directed() && !options_.allow_directed_projection) {
40 return {false,
41 "directed input requires allow_directed_projection=true"};
42 }
43 if (options_.execution.device_ids.empty()) {
44 return {false, "at least one CUDA device id must be provided"};
45 }
46 if (!detail::turbobc_backend_compiled() && graph.vertex_count() > 0) {
47 return {false, "the TurboBC backend is not compiled in this build"};
48 }
49 return {true, {}};

52ExecutionResult<BetweennessCentralityOutput> BetweennessCentrality::run(
53 const Graph& graph) const {
54 const auto started = std::chrono::steady_clock::now();
55 auto provenance = turbobc_provenance();
56 const auto support = supports(graph);
57 if (!support.supported) {
58 return ExecutionResult<BetweennessCentralityOutput>::failure(
59 {StatusCode::unsupported, support.reason}, provenance);
60 }
61
62 UndirectedGraphView normalized;
63 try {
64 normalized = graph.simple_undirected(
65 {options_.allow_directed_projection});
66 } catch (const std::exception& error) {
67 return ExecutionResult<BetweennessCentralityOutput>::failure(
68 {StatusCode::invalid_argument, error.what()}, provenance);
69 }
70
71 std::vector<std::string> warnings;
72 if (normalized.normalization.directed_projection_applied) {
73 warnings.emplace_back("directed input was projected to an undirected graph");
74 }
75 if (normalized.normalization.self_loops_removed > 0) {
76 warnings.emplace_back("self-loops were removed during normalization");
77 }
78 if (normalized.normalization.parallel_edges_collapsed > 0) {
79 warnings.emplace_back("parallel edges were collapsed during normalization");
80 }
81
82 const auto backend = normalized.csr.vertex_count() == 0
83 ? detail::BetweennessCentralityBackendResult{
84 Status::success(), {}, 0.0}
85 : detail::run_turbobc(
86 normalized.csr,
87 options_.execution.device_ids.front());
88 if (!backend.status.ok()) {
89 return ExecutionResult<BetweennessCentralityOutput>::failure(
90 backend.status, provenance, std::move(warnings));
91 }
92 if (backend.scores.size() != graph.vertex_count()) {
93 return ExecutionResult<BetweennessCentralityOutput>::failure(
94 {StatusCode::correctness_mismatch,
95 "TurboBC did not return one score per input vertex"},
96 provenance, std::move(warnings));
97 }
98 if (std::any_of(backend.scores.begin(), backend.scores.end(),
99 [](double score) { return !std::isfinite(score); })) {
100 return ExecutionResult<BetweennessCentralityOutput>::failure(
101 {StatusCode::correctness_mismatch,
102 "TurboBC returned a non-finite centrality score"},
103 provenance, std::move(warnings));
104 }
105
106 const auto materialization_started = std::chrono::steady_clock::now();
107 BetweennessCentralityOutput output;
108 output.score_by_vertex.reserve(graph.vertex_count());
109 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
110 output.score_by_vertex.push_back(
111 {graph.external_id(vertex), backend.scores[vertex]});
112 }
113
114 if (has_output(options_.optional_outputs,
115 BetweennessCentralityOptionalOutput::normalized_scores)) {
116 output.normalized_score_by_vertex.emplace();
117 output.normalized_score_by_vertex->reserve(graph.vertex_count());
118 const auto n = graph.vertex_count();
119 const double scale = n > 2
120 ? 1.0 / (static_cast<double>(n - 1) *
121 static_cast<double>(n - 2))
122 : 0.0;
123 for (const auto& entry : output.score_by_vertex) {
124 output.normalized_score_by_vertex->push_back(
125 {entry.vertex, entry.score * scale});
126 }
127 }
128
129 if (has_output(options_.optional_outputs,
130 BetweennessCentralityOptionalOutput::ranking)) {
131 output.ranking = output.score_by_vertex;
132 std::stable_sort(output.ranking->begin(), output.ranking->end(),
133 [](const auto& lhs, const auto& rhs) {
134 if (lhs.score != rhs.score) {
135 return lhs.score > rhs.score;
136 }
137 return lhs.vertex < rhs.vertex;
138 });
139 }
140
141 if (has_output(options_.optional_outputs,
142 BetweennessCentralityOptionalOutput::maximum) &&
143 !output.score_by_vertex.empty()) {
144 output.maximum = *std::max_element(
145 output.score_by_vertex.begin(), output.score_by_vertex.end(),
146 [](const auto& lhs, const auto& rhs) {
147 if (lhs.score != rhs.score) {
148 return lhs.score < rhs.score;
149 }
150 return rhs.vertex < lhs.vertex;
151 });
152 }
153 const auto materialization_finished = std::chrono::steady_clock::now();
154 const auto finished = std::chrono::steady_clock::now();
155
156 ExecutionStatistics statistics;
157 statistics.backend_ms = backend.elapsed_ms;
158 statistics.output_materialization_ms =
159 std::chrono::duration<double, std::milli>(materialization_finished -
160 materialization_started)
161 .count();
162 statistics.end_to_end_ms =
163 std::chrono::duration<double, std::milli>(finished - started).count();
164 return ExecutionResult<BetweennessCentralityOutput>::success(
165 std::move(output), std::move(provenance), statistics,
166 std::move(warnings));
167}
