23PartitionView make_partitions(const Graph& graph,
24 const std::vector<ExternalId>& left_ids) {
25 PartitionView partition;
26 partition.is_left.assign(graph.vertex_count(), false);
27 partition.left.reserve(left_ids.size());
28 for (const auto& id : left_ids) {
29 const auto vertex = graph.dense_index(id);
30 if (partition.is_left[vertex]) {
31 throw std::invalid_argument(
32 "left_partition contains a duplicate vertex: " + id.to_string());
33 }
34 partition.is_left[vertex] = true;
35 partition.left.push_back(vertex);
36 }
37 std::sort(partition.left.begin(), partition.left.end());
38 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
39 if (!partition.is_left[vertex]) {
40 partition.right.push_back(vertex);
41 }
42 }
43 if (partition.left.empty() || partition.right.empty()) {
44 throw std::invalid_argument(
45 "left_partition and its right-partition complement must both be "
46 "non-empty");
47 }
48 return partition;
49}

51void validate_bipartite_edges(const Graph& graph,
52 const PartitionView& partition) {
53 for (const auto& edge : graph.canonical().edges) {
54 const auto source = graph.dense_index(edge.source);
55 const auto target = graph.dense_index(edge.target);
56 if (source == target ||
57 partition.is_left[source] == partition.is_left[target]) {
58 throw std::invalid_argument(
59 "every edge must connect the declared left partition to its "
60 "right-partition complement");
61 }
62 }

65detail::BipartiteCsrGraph make_backend_graph(
66 const CsrGraph& graph, const PartitionView& partition) {
67 detail::BipartiteCsrGraph output;
68 output.left_vertex_count =
69 static_cast<std::uint32_t>(partition.left.size());
70 output.right_vertex_count =
71 static_cast<std::uint32_t>(partition.right.size());
72 output.left_offsets.reserve(partition.left.size() + 1);
73 output.left_offsets.push_back(0);
74
75 std::vector<std::uint32_t> right_local(
76 graph.vertex_count(), std::numeric_limits<std::uint32_t>::max());
77 for (std::uint32_t local = 0; local < partition.right.size(); ++local) {
78 right_local[partition.right[local]] = local;
79 }
80 for (const auto left : partition.left) {
81 for (auto offset = graph.offsets[left]; offset < graph.offsets[left + 1];
82 ++offset) {
83 output.right_neighbors.push_back(right_local[graph.neighbors[offset]]);
84 }
85 output.left_offsets.push_back(output.right_neighbors.size());
86 }
87 return output;
88}

90CsrGraph make_completed_graph(const CsrGraph& graph,
91 const PartitionView& partition) {
92 CsrGraph completed;
93 completed.offsets.reserve(graph.vertex_count() + 1);
94 completed.offsets.push_back(0);
95 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
96 const auto& same_side = partition.is_left[vertex] ? partition.left
97 : partition.right;
98 for (const auto neighbor : same_side) {
99 if (neighbor != vertex) {
100 completed.neighbors.push_back(neighbor);
101 }
102 }
103 completed.neighbors.insert(
104 completed.neighbors.end(),
105 graph.neighbors.begin() +
106 static_cast<std::ptrdiff_t>(graph.offsets[vertex]),
107 graph.neighbors.begin() +
108 static_cast<std::ptrdiff_t>(graph.offsets[vertex + 1]));
109 std::sort(completed.neighbors.begin() +
110 static_cast<std::ptrdiff_t>(completed.offsets.back()),
111 completed.neighbors.end());
112 completed.offsets.push_back(completed.neighbors.size());
113 }
114 return completed;
115}

131bool adjacent(const CsrGraph& graph, VertexIndex lhs, VertexIndex rhs) {
132 const auto begin = graph.neighbors.begin() +
133 static_cast<std::ptrdiff_t>(graph.offsets[lhs]);
134 const auto end = graph.neighbors.begin() +
135 static_cast<std::ptrdiff_t>(graph.offsets[lhs + 1]);
136 return std::binary_search(begin, end, rhs);

139std::vector<VertexIndex> intersect_neighbors(
140 const CsrGraph& graph, const std::vector<VertexIndex>& vertices,
141 VertexIndex pivot) {
142 std::vector<VertexIndex> output;
143 const auto begin = graph.neighbors.begin() +
144 static_cast<std::ptrdiff_t>(graph.offsets[pivot]);
145 const auto end = graph.neighbors.begin() +
146 static_cast<std::ptrdiff_t>(graph.offsets[pivot + 1]);
147 std::set_intersection(vertices.begin(), vertices.end(), begin, end,
148 std::back_inserter(output));
149 return output;

152VertexIndex choose_pivot(const CsrGraph& graph,
153 const std::vector<VertexIndex>& possible,
154 const std::vector<VertexIndex>& excluded) {
155 VertexIndex best = possible.empty() ? excluded.front() : possible.front();
156 std::size_t best_score = 0;
157 const auto consider = [&](VertexIndex candidate) {
158 const auto begin = graph.neighbors.begin() +
159 static_cast<std::ptrdiff_t>(graph.offsets[candidate]);
160 const auto end = graph.neighbors.begin() +
161 static_cast<std::ptrdiff_t>(graph.offsets[candidate + 1]);
162 std::size_t score = 0;
163 auto possible_it = possible.begin();
164 auto neighbor_it = begin;
165 while (possible_it != possible.end() && neighbor_it != end) {
166 if (*possible_it < *neighbor_it) {
167 ++possible_it;
168 } else if (*neighbor_it < *possible_it) {
169 ++neighbor_it;
170 } else {
171 ++score;
172 ++possible_it;
173 ++neighbor_it;
174 }
175 }
176 return score;
177 };
178 for (const auto candidate : possible) {
179 const auto score = consider(candidate);
180 if (score > best_score) {
181 best = candidate;
182 best_score = score;
183 }
184 }
185 for (const auto candidate : excluded) {
186 const auto score = consider(candidate);
187 if (score > best_score) {
188 best = candidate;
189 best_score = score;
190 }
191 }
192 return best;
193}

195void collect_bicliques(const CsrGraph& graph,
196 std::vector<VertexIndex>& current,
197 std::vector<VertexIndex> possible,
198 std::vector<VertexIndex> excluded,
199 CollectionState& state) {
200 if (state.stopped_early) {
201 return;
202 }
203 if (possible.empty() && excluded.empty()) {
204 InternalBiclique biclique;
205 for (const auto vertex : current) {
206 (state.partition->is_left[vertex] ? biclique.first : biclique.second)
207 .push_back(vertex);
208 }
209 std::sort(biclique.first.begin(), biclique.first.end());
210 std::sort(biclique.second.begin(), biclique.second.end());
211 if (biclique.first.size() < state.minimum_left_size ||
212 biclique.second.size() < state.minimum_right_size) {
213 return;
214 }
215 if (state.qualifying_count == std::numeric_limits<std::uint64_t>::max()) {
216 throw std::overflow_error("maximal-biclique count exceeds uint64_t");
217 }
218 ++state.qualifying_count;
219 if (!state.result_limit ||
220 state.bicliques.size() < *state.result_limit) {
221 state.bicliques.push_back(std::move(biclique));
222 }
223 if (state.result_limit &&
224 state.bicliques.size() >= *state.result_limit &&
225 !state.must_finish) {
226 state.stopped_early = true;
227 }
228 return;
229 }
230
231 const auto pivot = choose_pivot(graph, possible, excluded);
232 std::vector<VertexIndex> candidates;
233 for (const auto vertex : possible) {
234 if (!adjacent(graph, pivot, vertex)) {
235 candidates.push_back(vertex);
236 }
237 }
238 for (const auto vertex : candidates) {
239 if (state.stopped_early) {
240 return;
241 }
242 const auto present =
243 std::lower_bound(possible.begin(), possible.end(), vertex);
244 if (present == possible.end() || *present != vertex) {
245 continue;
246 }
247 current.push_back(vertex);
248 auto next_possible = intersect_neighbors(graph, possible, vertex);
249 auto next_excluded = intersect_neighbors(graph, excluded, vertex);
250 collect_bicliques(graph, current, std::move(next_possible),
251 std::move(next_excluded), state);
252 current.pop_back();
253 possible.erase(std::lower_bound(possible.begin(), possible.end(), vertex));
254 excluded.insert(std::lower_bound(excluded.begin(), excluded.end(), vertex),
255 vertex);
256 }

259CollectionState materialize_bicliques(const CsrGraph& bipartite_graph,
260 const PartitionView& partition,
261 const MaximalBicliqueOptions& options) {
262 CollectionState state;
263 state.partition = &partition;
264 state.minimum_left_size = options.minimum_left_size;
265 state.minimum_right_size = options.minimum_right_size;
266 state.result_limit = options.result_limit;
267 state.must_finish =
268 has_output(options.optional_outputs,
269 MaximalBicliqueOptionalOutput::total_count) &&
270 (options.minimum_left_size != 1 || options.minimum_right_size != 1);
271
272 auto completed = make_completed_graph(bipartite_graph, partition);
273 std::vector<VertexIndex> possible(completed.vertex_count());
274 std::iota(possible.begin(), possible.end(), VertexIndex{0});
275 std::vector<VertexIndex> current;
276 collect_bicliques(completed, current, std::move(possible), {}, state);
277 std::sort(state.bicliques.begin(), state.bicliques.end(),
278 [](const auto& lhs, const auto& rhs) {
279 if (lhs.first.size() != rhs.first.size()) {
280 return lhs.first.size() < rhs.first.size();
281 }
282 if (lhs.first != rhs.first) {
283 return lhs.first < rhs.first;
284 }
285 if (lhs.second.size() != rhs.second.size()) {
286 return lhs.second.size() < rhs.second.size();
287 }
288 return lhs.second < rhs.second;
289 });
290 return state;
291}

316SupportReport MaximalBicliques::supports(const Graph& graph) const {
317 if (options_.minimum_left_size == 0 || options_.minimum_right_size == 0) {
318 return {false, "minimum partition sizes must both be at least one"};
319 }
320 if (options_.result_limit && *options_.result_limit == 0) {
321 return {false, "result_limit must be null or at least one"};
322 }
323 if (graph.directed() && !options_.allow_directed_projection) {
324 return {false,
325 "directed input requires allow_directed_projection=true"};
326 }
327 if (options_.execution.device_ids.empty()) {
328 return {false, "at least one CUDA device id must be provided"};
329 }
330 if (graph.vertex_count() >
331 static_cast<std::size_t>(std::numeric_limits<std::uint32_t>::max())) {
332 return {false, "cuMBE uses 32-bit vertex indices"};
333 }
334 try {
335 const auto partition = make_partitions(graph, options_.left_partition);
336 validate_bipartite_edges(graph, partition);
337 } catch (const std::exception& error) {
338 return {false, error.what()};
339 }
340 if (!detail::cumbe_backend_compiled() && graph.edge_count() > 0) {
341 return {false, "the cuMBE backend is not compiled in this build"};
342 }
343 return {true, {}};
344}

346ExecutionResult<MaximalBicliqueOutput> MaximalBicliques::run(
347 const Graph& graph) const {
348 const auto started = std::chrono::steady_clock::now();
349 auto provenance = cumbe_provenance();
350 const auto support = supports(graph);
351 if (!support.supported) {
352 return ExecutionResult<MaximalBicliqueOutput>::failure(
353 {StatusCode::unsupported, support.reason}, provenance);
354 }
355
356 UndirectedGraphView normalized;
357 PartitionView partition;
358 try {
359 normalized = graph.simple_undirected(
360 {options_.allow_directed_projection});
361 partition = make_partitions(graph, options_.left_partition);
362 } catch (const std::exception& error) {
363 return ExecutionResult<MaximalBicliqueOutput>::failure(
364 {StatusCode::invalid_argument, error.what()}, provenance);
365 }
366
367 std::vector<std::string> warnings;
368 if (normalized.normalization.directed_projection_applied) {
369 warnings.emplace_back("directed input was projected to an undirected graph");
370 }
371 if (normalized.normalization.parallel_edges_collapsed > 0) {
372 warnings.emplace_back("parallel edges were collapsed during normalization");
373 }
374
375 const auto backend_graph = make_backend_graph(normalized.csr, partition);
376 const auto backend = backend_graph.right_neighbors.empty()
377 ? detail::MaximalBicliqueBackendResult{
378 Status::success(), 0, 0.0}
379 : detail::run_cumbe(
380 backend_graph,
381 options_.execution.device_ids.front());
382 if (!backend.status.ok()) {
383 return ExecutionResult<MaximalBicliqueOutput>::failure(
384 backend.status, provenance, std::move(warnings));
385 }
386
387 const auto materialization_started = std::chrono::steady_clock::now();
388 CollectionState collected;
389 try {
390 collected = materialize_bicliques(normalized.csr, partition, options_);
391 } catch (const std::exception& error) {
392 return ExecutionResult<MaximalBicliqueOutput>::failure(
393 {StatusCode::execution_failed, error.what()}, provenance,
394 std::move(warnings));
395 }
396 const auto materialization_finished = std::chrono::steady_clock::now();
397
398 const bool unfiltered = options_.minimum_left_size == 1 &&
399 options_.minimum_right_size == 1;
400 const bool collector_finished = !collected.stopped_early;
401 if (unfiltered && collector_finished &&
402 collected.qualifying_count != backend.count) {
403 return ExecutionResult<MaximalBicliqueOutput>::failure(
404 {StatusCode::correctness_mismatch,
405 "cuMBE count disagreed with the exact output materializer"},
406 provenance, std::move(warnings));
407 }
408
409 MaximalBicliqueOutput output;
410 output.bicliques.reserve(collected.bicliques.size());
411 for (const auto& internal : collected.bicliques) {
412 MaximalBiclique biclique;
413 for (const auto vertex : internal.first) {
414 biclique.left.push_back(graph.external_id(vertex));
415 }
416 for (const auto vertex : internal.second) {
417 biclique.right.push_back(graph.external_id(vertex));
418 }
419 output.bicliques.push_back(std::move(biclique));
420 }
421 output.returned_count = output.bicliques.size();
422
423 std::uint64_t known_total = collected.qualifying_count;
424 bool total_known = collector_finished;
425 if (unfiltered) {
426 known_total = backend.count;
427 total_known = true;
428 }
429 output.complete = total_known && output.returned_count == known_total;
430 if (has_output(options_.optional_outputs,
431 MaximalBicliqueOptionalOutput::total_count)) {
432 output.total_count = known_total;
433 }
434 warnings.emplace_back(
435 "cuMBE supplies the validated GPU total; biclique sets are materialized "
436 "by GraphMine's exact completed-graph collector");
437
438 const auto finished = std::chrono::steady_clock::now();
439 ExecutionStatistics statistics;
440 statistics.backend_ms = backend.elapsed_ms;
441 statistics.output_materialization_ms =
442 std::chrono::duration<double, std::milli>(materialization_finished -
443 materialization_started)
444 .count();
445 statistics.end_to_end_ms =
446 std::chrono::duration<double, std::milli>(finished - started).count();
447 return ExecutionResult<MaximalBicliqueOutput>::success(
448 std::move(output), std::move(provenance), statistics,
449 std::move(warnings));
450}
