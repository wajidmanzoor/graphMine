101std::vector<VertexIndex> copy_original_vertices(
102 graphmine_maximum_clique_on_gpu_upstream::GPUArray<DataType>& values,
103 std::size_t count) {
104 std::vector<VertexIndex> result;
105 if (count == 0 || values.N < count) return result;
106 DataType* copied = values.copytocpu(0, count, true);
107 result.reserve(count);
108 for (std::size_t i = 0; i < count; ++i) {
109 // Vertex zero is the isolated sentinel used by the artifact's converter.
110 if (copied[i] != 0) result.push_back(copied[i] - 1U);
111 }
112 std::free(copied);
113 return result;
114}
