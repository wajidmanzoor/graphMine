24EdgeKey edge_key(VertexIndex source, VertexIndex target) {
25 if (target < source) std::swap(source, target);
26 return (static_cast<EdgeKey>(source) << 32U) |
27 static_cast<EdgeKey>(target);
28}

30std::set<EdgeKey> edge_set(const CsrGraph& graph) {
31 std::set<EdgeKey> edges;
32 for (VertexIndex source = 0; source < graph.vertex_count(); ++source) {
33 for (auto offset = graph.offsets[source];
34 offset < graph.offsets[source + 1]; ++offset) {
35 const auto target = graph.neighbors[offset];
36 if (source < target) edges.insert(edge_key(source, target));
37 }
38 }
39 return edges;
40}

42CsrGraph make_csr(std::size_t vertex_count, const std::set<EdgeKey>& edges) {
43 CsrGraph graph;
44 graph.offsets.assign(vertex_count + 1U, 0);
45 for (const auto key : edges) {
46 const auto source = static_cast<VertexIndex>(key >> 32U);
47 const auto target = static_cast<VertexIndex>(key & 0xffffffffULL);
48 ++graph.offsets[source + 1U];
49 ++graph.offsets[target + 1U];
50 }
51 for (std::size_t vertex = 1; vertex < graph.offsets.size(); ++vertex) {
52 graph.offsets[vertex] += graph.offsets[vertex - 1U];
53 }
54 graph.neighbors.resize(edges.size() * 2U);
55 auto cursor = graph.offsets;
56 for (const auto key : edges) {
57 const auto source = static_cast<VertexIndex>(key >> 32U);
58 const auto target = static_cast<VertexIndex>(key & 0xffffffffULL);
59 graph.neighbors[cursor[source]++] = target;
60 graph.neighbors[cursor[target]++] = source;
61 }
62 for (VertexIndex vertex = 0; vertex < vertex_count; ++vertex) {
63 std::sort(graph.neighbors.begin() +
64 static_cast<std::ptrdiff_t>(graph.offsets[vertex]),
65 graph.neighbors.begin() +
66 static_cast<std::ptrdiff_t>(graph.offsets[vertex + 1U]));
67 }
68 return graph;
69}

71template <typename Visitor>
72std::uint64_t visit_triangles(const CsrGraph& graph, Visitor&& visitor) {
73 std::uint64_t count = 0;
74 for (VertexIndex u = 0; u < graph.vertex_count(); ++u) {
75 const auto u_begin = graph.neighbors.begin() +
76 static_cast<std::ptrdiff_t>(graph.offsets[u]);
77 const auto u_end = graph.neighbors.begin() +
78 static_cast<std::ptrdiff_t>(graph.offsets[u + 1U]);
79 for (auto v_it = std::upper_bound(u_begin, u_end, u); v_it != u_end;
80 ++v_it) {
81 const auto v = *v_it;
82 const auto v_begin = graph.neighbors.begin() +
83 static_cast<std::ptrdiff_t>(graph.offsets[v]);
84 const auto v_end = graph.neighbors.begin() +
85 static_cast<std::ptrdiff_t>(graph.offsets[v + 1U]);
86 auto left = std::upper_bound(v_it + 1, u_end, v);
87 auto right = std::upper_bound(v_begin, v_end, v);
88 while (left != u_end && right != v_end) {
89 if (*left < *right) {
90 ++left;
91 } else if (*right < *left) {
92 ++right;
93 } else {
94 ++count;
95 visitor(Triangle{u, v, *left});
96 ++left;
97 ++right;
98 }
99 }
100 }
101 }
102 return count;
103}

127Status prepare_batch(const Graph& graph,
128 const std::vector<TriangleUpdate>& updates,
129 bool allow_directed_projection,
130 PreparedUpdateBatch& prepared) {
131 try {
132 prepared.initial =
133 graph.simple_undirected({allow_directed_projection});
134 } catch (const std::exception& error) {
135 return {StatusCode::invalid_argument, error.what()};
136 }
137
138 if (prepared.initial.normalization.directed_projection_applied) {
139 prepared.warnings.emplace_back(
140 "directed input and updates were projected to an undirected graph");
141 }
142 if (prepared.initial.normalization.self_loops_removed > 0) {
143 prepared.warnings.emplace_back(
144 "self-loops were removed from the initial graph during normalization");
145 }
146 if (prepared.initial.normalization.parallel_edges_collapsed > 0) {
147 prepared.warnings.emplace_back(
148 "parallel edges were collapsed in the initial graph during "
149 "normalization");
150 }
151
152 prepared.initial_edges = edge_set(prepared.initial.csr);
153 prepared.after_deletion_edges = prepared.initial_edges;
154 prepared.updates.reserve(updates.size());
155 std::set<EdgeKey> deletions;
156 std::set<EdgeKey> insertions;

158 struct ResolvedUpdate {
159 TriangleUpdateKind kind;
160 VertexIndex source;
161 VertexIndex target;
162 EdgeKey key;
163 };
164 std::vector<ResolvedUpdate> resolved;
165 resolved.reserve(updates.size());
166 try {
167 for (const auto& update : updates) {
168 auto source = graph.dense_index(update.source);
169 auto target = graph.dense_index(update.target);
170 if (source == target) {
171 return {StatusCode::invalid_argument,
172 "dynamic triangle updates cannot contain self-loops"};
173 }
174 if (target < source) std::swap(source, target);
175 const auto key = edge_key(source, target);
176 auto& same_kind = update.kind == TriangleUpdateKind::delete_edge
177 ? deletions
178 : insertions;
179 if (!same_kind.insert(key).second) {
180 return {StatusCode::invalid_argument,
181 "a dynamic triangle batch contains a duplicate " +
182 std::string(to_string(update.kind)) + " update"};
183 }
184 resolved.push_back({update.kind, source, target, key});
185 prepared.updates.push_back(
186 {update.kind == TriangleUpdateKind::insert_edge, source, target});
187 }
188 } catch (const std::exception& error) {
189 return {StatusCode::invalid_argument,
190 std::string("dynamic triangle update references an unknown "
191 "vertex: ") +
192 error.what()};
193 }
194
195 for (const auto& update : resolved) {
196 if (update.kind != TriangleUpdateKind::delete_edge) continue;
197 if (prepared.initial_edges.count(update.key) == 0U) {
198 return {StatusCode::invalid_argument,
199 "a dynamic triangle deletion references a missing edge"};
200 }
201 prepared.after_deletion_edges.erase(update.key);
202 }
203 prepared.final_edges = prepared.after_deletion_edges;
204 for (const auto& update : resolved) {
205 if (update.kind != TriangleUpdateKind::insert_edge) continue;
206 if (prepared.final_edges.count(update.key) != 0U) {
207 return {StatusCode::invalid_argument,
208 "a dynamic triangle insertion references an existing edge"};
209 }
210 prepared.final_edges.insert(update.key);
211 }
212
213 prepared.after_deletions =
214 make_csr(graph.vertex_count(), prepared.after_deletion_edges);
215 prepared.final = make_csr(graph.vertex_count(), prepared.final_edges);
216 return Status::success();
217}

280ExecutionResult<DynamicTriangleOutput> DynamicTriangleCounting::run(
281 const Graph& graph, const std::vector<TriangleUpdate>& updates) const {
282 const auto started = std::chrono::steady_clock::now();
283 auto provenance = edtc_provenance();
284 if (options_.execution.device_ids.empty()) {
285 return ExecutionResult<DynamicTriangleOutput>::failure(
286 {StatusCode::invalid_argument,
287 "at least one CUDA device id must be provided"},
288 provenance);
289 }
290 if (options_.result_limit && *options_.result_limit == 0) {
291 return ExecutionResult<DynamicTriangleOutput>::failure(
292 {StatusCode::invalid_argument,
293 "result_limit must be null or at least one"},
294 provenance);
295 }
296 if (options_.result_limit && !options_.list_changed_instances) {
297 return ExecutionResult<DynamicTriangleOutput>::failure(
298 {StatusCode::invalid_argument,
299 "result_limit is meaningful only when "
300 "list_changed_instances=true"},
301 provenance);
302 }
303 if (!detail::edtc_backend_compiled()) {
304 return ExecutionResult<DynamicTriangleOutput>::failure(
305 {StatusCode::backend_unavailable,
306 "EDTC was not compiled into this GraphMine build"},
307 provenance);
308 }
309
310 PreparedUpdateBatch prepared;
311 const auto prepare_status = prepare_batch(
312 graph, updates, options_.allow_directed_projection, prepared);
313 if (!prepare_status.ok()) {
314 return ExecutionResult<DynamicTriangleOutput>::failure(
315 prepare_status, provenance, std::move(prepared.warnings));
316 }
317
318 const auto backend = detail::run_edtc(
319 prepared.initial.csr, prepared.updates,
320 options_.execution.device_ids.front());
321 if (!backend.status.ok()) {
322 return ExecutionResult<DynamicTriangleOutput>::failure(
323 backend.status, provenance, std::move(prepared.warnings));
324 }
325
326 DynamicTriangleOutput output;
327 output.deleted_triangle_count = backend.deleted_count;
328 output.inserted_triangle_count = backend.inserted_count;
329 if (backend.inserted_count >= backend.deleted_count) {
330 const auto difference = backend.inserted_count - backend.deleted_count;
331 if (difference > static_cast<std::uint64_t>(INT64_MAX)) {
332 return ExecutionResult<DynamicTriangleOutput>::failure(
333 {StatusCode::unsupported,
334 "EDTC net triangle change exceeds signed 64-bit range"},
335 provenance, std::move(prepared.warnings));
336 }
337 output.net_triangle_change = static_cast<std::int64_t>(difference);
338 } else {
339 const auto difference = backend.deleted_count - backend.inserted_count;
340 if (difference > static_cast<std::uint64_t>(INT64_MAX)) {
341 return ExecutionResult<DynamicTriangleOutput>::failure(
342 {StatusCode::unsupported,
343 "EDTC net triangle change exceeds signed 64-bit range"},
344 provenance, std::move(prepared.warnings));
345 }
346 output.net_triangle_change = -static_cast<std::int64_t>(difference);
347 }
348
349 const auto materialization_started = std::chrono::steady_clock::now();
350 if (options_.include_global_counts || options_.list_changed_instances) {
351 const auto initial_count = count_triangles(prepared.initial.csr);
352 const auto after_delete_count = count_triangles(prepared.after_deletions);
353 const auto final_count = count_triangles(prepared.final);
354 const auto expected_deleted = initial_count - after_delete_count;
355 const auto expected_inserted = final_count - after_delete_count;
356 if (expected_deleted != backend.deleted_count ||
357 expected_inserted != backend.inserted_count) {
358 return ExecutionResult<DynamicTriangleOutput>::failure(
359 {StatusCode::correctness_mismatch,
360 "EDTC change counts disagreed with the exact optional "
361 "materializer"},
362 provenance, std::move(prepared.warnings));
363 }
364 if (options_.include_global_counts) {
365 output.initial_global_triangle_count = initial_count;
366 output.final_global_triangle_count = final_count;
367 }
368
369 if (options_.list_changed_instances) {
370 output.deleted_triangles.emplace();
371 output.inserted_triangles.emplace();
372 const auto limit = options_.result_limit.value_or(
373 std::numeric_limits<std::uint64_t>::max());
374 visit_triangles(prepared.initial.csr, [&](const Triangle& triangle) {
375 if (!triangle_exists(prepared.after_deletion_edges, triangle) &&
376 output.deleted_triangles->size() < limit) {
377 output.deleted_triangles->push_back(
378 {graph.external_id(triangle[0]), graph.external_id(triangle[1]),
379 graph.external_id(triangle[2])});
380 }
381 });
382 visit_triangles(prepared.final, [&](const Triangle& triangle) {
383 if (!triangle_exists(prepared.after_deletion_edges, triangle) &&
384 output.inserted_triangles->size() < limit) {
385 output.inserted_triangles->push_back(
386 {graph.external_id(triangle[0]), graph.external_id(triangle[1]),
387 graph.external_id(triangle[2])});
388 }
389 });
390 output.changed_instances_complete =
391 output.deleted_triangles->size() == backend.deleted_count &&
392 output.inserted_triangles->size() == backend.inserted_count;
393 }
394 }
395 const auto materialization_finished = std::chrono::steady_clock::now();
396 const auto finished = std::chrono::steady_clock::now();
397
398 ExecutionStatistics statistics;
399 statistics.backend_ms = backend.elapsed_ms;
400 statistics.output_materialization_ms =
401 std::chrono::duration<double, std::milli>(materialization_finished -
402 materialization_started)
403 .count();
404 statistics.end_to_end_ms =
405 std::chrono::duration<double, std::milli>(finished - started).count();
406 return ExecutionResult<DynamicTriangleOutput>::success(
407 std::move(output), std::move(provenance), statistics,
408 std::move(prepared.warnings));
409}
