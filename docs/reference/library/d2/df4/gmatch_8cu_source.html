48std::vector<std::vector<int>> make_adjacency(const CsrGraph& graph) {
49 std::vector<std::vector<int>> adjacency(graph.vertex_count());
50 for (std::size_t vertex = 0; vertex < graph.vertex_count(); ++vertex) {
51 auto& row = adjacency[vertex];
52 row.reserve(graph.offsets[vertex + 1] - graph.offsets[vertex]);
53 for (auto edge = graph.offsets[vertex]; edge < graph.offsets[vertex + 1];
54 ++edge) {
55 row.push_back(static_cast<int>(graph.neighbors[edge]));
56 }
57 }
58 return adjacency;
59}
