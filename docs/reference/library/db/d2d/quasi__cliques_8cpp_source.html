16bool is_quasi_clique(const CsrGraph& graph,
17 const std::vector<VertexIndex>& vertices,
18 double ratio) {
19 if (vertices.size() < 2) {
20 return false;
21 }
22 const auto minimum_degree = static_cast<std::size_t>(
23 std::ceil(ratio * static_cast<double>(vertices.size() - 1)));
24 for (const auto vertex : vertices) {
25 std::size_t degree = 0;
26 const auto begin = graph.neighbors.begin() +
27 static_cast<std::ptrdiff_t>(graph.offsets[vertex]);
28 const auto end = graph.neighbors.begin() +
29 static_cast<std::ptrdiff_t>(graph.offsets[vertex + 1]);
30 for (const auto candidate : vertices) {
31 if (candidate != vertex &&
32 std::binary_search(begin, end, candidate)) {
33 ++degree;
34 }
35 }
36 if (degree < minimum_degree) {
37 return false;
38 }
39 }
40 return true;
41}

43Status validate_backend_output(
44 const CsrGraph& graph,
45 const std::vector<std::vector<VertexIndex>>& quasi_cliques,
46 const QuasiCliqueOptions& options) {
47 std::set<std::vector<VertexIndex>> unique;
48 for (const auto& quasi_clique : quasi_cliques) {
49 if (quasi_clique.size() < options.minimum_size) {
50 return {StatusCode::correctness_mismatch,
51 "cuQC returned a result below minimum_size"};
52 }
53 if (!std::is_sorted(quasi_clique.begin(), quasi_clique.end()) ||
54 std::adjacent_find(quasi_clique.begin(), quasi_clique.end()) !=
55 quasi_clique.end() ||
56 (!quasi_clique.empty() &&
57 quasi_clique.back() >= graph.vertex_count())) {
58 return {StatusCode::correctness_mismatch,
59 "cuQC returned an invalid vertex set"};
60 }
61 if (!is_quasi_clique(graph, quasi_clique,
62 options.minimum_degree_ratio)) {
63 return {StatusCode::correctness_mismatch,
64 "cuQC returned a set that does not satisfy gamma"};
65 }
66 if (!unique.emplace(quasi_clique).second) {
67 return {StatusCode::correctness_mismatch,
68 "cuQC returned a duplicate maximal quasi-clique"};
69 }
70
71 std::vector<bool> member(graph.vertex_count(), false);
72 for (const auto vertex : quasi_clique) {
73 member[vertex] = true;
74 }
75 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
76 if (member[vertex]) {
77 continue;
78 }
79 auto extension = quasi_clique;
80 extension.insert(
81 std::lower_bound(extension.begin(), extension.end(), vertex),
82 vertex);
83 if (is_quasi_clique(graph, extension,
84 options.minimum_degree_ratio)) {
85 std::string members;
86 for (const auto member : quasi_clique) {
87 members += (members.empty() ? "" : ",") + std::to_string(member);
88 }
89 return {StatusCode::correctness_mismatch,
90 "cuQC returned a non-maximal quasi-clique; vertex " +
91 std::to_string(vertex) + " can extend a returned set of " +
92 std::to_string(quasi_clique.size()) + " vertices [" +
93 members + "]; backend returned " +
94 std::to_string(quasi_cliques.size()) +
95 " sets with maximum size " +
96 std::to_string(quasi_cliques.empty()
97 ? 0
98 : quasi_cliques.back().size())};
99 }
100 }
101 }
102 return Status::success();
103}

128SupportReport QuasiCliques::supports(const Graph& graph) const {
129 if (!std::isfinite(options_.minimum_degree_ratio) ||
130 options_.minimum_degree_ratio < 0.5 ||
131 options_.minimum_degree_ratio > 1.0) {
132 return {false, "minimum_degree_ratio must be in [0.5, 1.0]"};
133 }
134 if (options_.minimum_size <= 1) {
135 return {false, "minimum_size must be greater than one"};
136 }
137 if (options_.minimum_size >
138 static_cast<std::size_t>(std::numeric_limits<int>::max())) {
139 return {false, "cuQC uses a signed 32-bit minimum size"};
140 }
141 if (options_.result_limit && *options_.result_limit == 0) {
142 return {false, "result_limit must be null or at least one"};
143 }
144 if (graph.vertex_count() >
145 static_cast<std::size_t>(std::numeric_limits<int>::max()) ||
146 graph.edge_count() >
147 static_cast<std::size_t>(std::numeric_limits<int>::max() / 2)) {
148 return {false, "cuQC uses signed 32-bit graph sizes"};
149 }
150 if (graph.directed() && !options_.allow_directed_projection) {
151 return {false,
152 "directed input requires allow_directed_projection=true"};
153 }
154 if (options_.execution.device_ids.empty()) {
155 return {false, "at least one CUDA device id must be provided"};
156 }
157 if (!detail::cuqc_backend_compiled() &&
158 graph.vertex_count() >= options_.minimum_size) {
159 return {false, "the cuQC backend is not compiled in this build"};
160 }
161 return {true, {}};
162}

164ExecutionResult<QuasiCliqueOutput> QuasiCliques::run(
165 const Graph& graph) const {
166 const auto started = std::chrono::steady_clock::now();
167 auto provenance = cuqc_provenance();
168 const auto support = supports(graph);
169 if (!support.supported) {
170 return ExecutionResult<QuasiCliqueOutput>::failure(
171 {StatusCode::unsupported, support.reason}, provenance);
172 }
173
174 UndirectedGraphView normalized;
175 try {
176 normalized = graph.simple_undirected(
177 {options_.allow_directed_projection});
178 } catch (const std::exception& error) {
179 return ExecutionResult<QuasiCliqueOutput>::failure(
180 {StatusCode::invalid_argument, error.what()}, provenance);
181 }
182
183 std::vector<std::string> warnings;
184 if (normalized.normalization.directed_projection_applied) {
185 warnings.emplace_back("directed input was projected to an undirected graph");
186 }
187 if (normalized.normalization.self_loops_removed > 0) {
188 warnings.emplace_back("self-loops were removed during normalization");
189 }
190 if (normalized.normalization.parallel_edges_collapsed > 0) {
191 warnings.emplace_back("parallel edges were collapsed during normalization");
192 }
193
194 const auto backend =
195 graph.vertex_count() < options_.minimum_size
196 ? detail::QuasiCliqueBackendResult{Status::success(), {}, 0.0}
197 : detail::run_cuqc(
198 normalized.csr, options_.minimum_degree_ratio,
199 options_.minimum_size,
200 options_.scheduling == CuQCScheduling::static_schedule,
201 options_.execution.device_ids.front());
202 if (!backend.status.ok()) {
203 return ExecutionResult<QuasiCliqueOutput>::failure(
204 backend.status, provenance, std::move(warnings));
205 }
206 const auto validation =
207 validate_backend_output(normalized.csr, backend.quasi_cliques, options_);
208 if (!validation.ok()) {
209 return ExecutionResult<QuasiCliqueOutput>::failure(
210 validation, provenance, std::move(warnings));
211 }
212
213 const auto materialization_started = std::chrono::steady_clock::now();
214 QuasiCliqueOutput output;
215 const std::uint64_t total = backend.quasi_cliques.size();
216 const std::size_t returned = options_.result_limit
217 ? std::min<std::uint64_t>(
218 total, *options_.result_limit)
219 : static_cast<std::size_t>(total);
220 output.quasi_cliques.reserve(returned);
221 for (std::size_t index = 0; index < returned; ++index) {
222 std::vector<ExternalId> external;
223 external.reserve(backend.quasi_cliques[index].size());
224 for (const auto vertex : backend.quasi_cliques[index]) {
225 external.push_back(graph.external_id(vertex));
226 }
227 output.quasi_cliques.push_back(std::move(external));
228 }
229 output.returned_count = output.quasi_cliques.size();
230 output.complete = output.returned_count == total;
231 if (has_output(options_.optional_outputs,
232 QuasiCliqueOptionalOutput::total_count)) {
233 output.total_count = total;
234 }
235 const auto materialization_finished = std::chrono::steady_clock::now();
236 const auto finished = std::chrono::steady_clock::now();
237
238 ExecutionStatistics statistics;
239 statistics.backend_ms = backend.elapsed_ms;
240 statistics.output_materialization_ms =
241 std::chrono::duration<double, std::milli>(materialization_finished -
242 materialization_started)
243 .count();
244 statistics.end_to_end_ms =
245 std::chrono::duration<double, std::milli>(finished - started).count();
246 return ExecutionResult<QuasiCliqueOutput>::success(
247 std::move(output), std::move(provenance), statistics,
248 std::move(warnings));
249}
