52 void allocate(std::size_t count) {
53 if (count == 0) return;

55 size_ = count;
56 }

59 if (data_ != nullptr) {

65 void copy_from(const std::vector<T>& source) {
66 if (source.size() != size_) {
67 throw std::logic_error("CUDA buffer size does not match host input");
68 }
69 if (!source.empty()) {

71 cudaMemcpyHostToDevice),
72 "copying temporal motif input to CUDA");
73 }
74 }

109TemporalMotifBackendResult execute(const TemporalMotifBackendInput& input,
110 int device_id, int dispatch_blocks,
111 int expansion_blocks, int block_size,
112 const char* backend_name,
113 Runner&& runner) {
114 constexpr std::size_t kOffloadCapacity = 2092ULL * 96ULL;
115 try {
116 const auto started = std::chrono::steady_clock::now();
117
118 std::vector<RawTemporalEdge> events;
119 events.reserve(input.events.size());
120 std::vector<int> incoming_offsets(
121 static_cast<std::size_t>(input.vertex_count) + 1U, 0);
122 std::vector<int> outgoing_offsets(
123 static_cast<std::size_t>(input.vertex_count) + 1U, 0);
124 for (const auto& event : input.events) {
125 events.push_back({event.source, event.target, event.timestamp});
126 ++outgoing_offsets[static_cast<std::size_t>(event.source) + 1U];
127 ++incoming_offsets[static_cast<std::size_t>(event.target) + 1U];
128 }
129 for (std::size_t vertex = 1; vertex < incoming_offsets.size(); ++vertex) {
130 incoming_offsets[vertex] += incoming_offsets[vertex - 1U];
131 outgoing_offsets[vertex] += outgoing_offsets[vertex - 1U];
132 }
133
134 std::vector<int> incoming_events(events.size());
135 std::vector<int> outgoing_events(events.size());
136 auto incoming_cursor = incoming_offsets;
137 auto outgoing_cursor = outgoing_offsets;
138 for (std::size_t index = 0; index < events.size(); ++index) {
139 incoming_events[static_cast<std::size_t>(
140 incoming_cursor[static_cast<std::size_t>(events[index].v)]++)] =
141 static_cast<int>(index);
142 outgoing_events[static_cast<std::size_t>(
143 outgoing_cursor[static_cast<std::size_t>(events[index].u)]++)] =
144 static_cast<int>(index);
145 }
146
147 DeviceBuffer<RawTemporalEdge> events_device(events.size());
148 DeviceBuffer<int> incoming_offsets_device(incoming_offsets.size());
149 DeviceBuffer<int> outgoing_offsets_device(outgoing_offsets.size());
150 DeviceBuffer<int> incoming_events_device(incoming_events.size());
151 DeviceBuffer<int> outgoing_events_device(outgoing_events.size());
152 events_device.copy_from(events);
153 incoming_offsets_device.copy_from(incoming_offsets);
154 outgoing_offsets_device.copy_from(outgoing_offsets);
155 incoming_events_device.copy_from(incoming_events);
156 outgoing_events_device.copy_from(outgoing_events);
157
158 std::vector<RawMotifEdgeInfo> motif{
159 {1, -2, 2, outgoing_offsets_device.get(),
160 outgoing_events_device.get(), 1, true},
161 {2, 0, 3, nullptr, nullptr, -1, false},
162 };
163 DeviceBuffer<RawMotifEdgeInfo> motif_device(motif.size());
164 motif_device.copy_from(motif);
165
166 DeviceBuffer<int> yield_device(1);
167 DeviceBuffer<int> source_device(1);
168 DeviceBuffer<Context> offload_device(kOffloadCapacity);
169 DeviceBuffer<int> offload_top_device(1);
170 DeviceBuffer<int> next_offload_top_device(1);
171 DeviceBuffer<int> offload_width_device(kOffloadCapacity);
172 DeviceBuffer<unsigned long long> count_device(1);
173 yield_device.zero();
174 source_device.zero();
175 offload_device.zero();
176 offload_top_device.zero();
177 next_offload_top_device.zero();
178 offload_width_device.zero();
179 count_device.zero();
180
181 Invocation<Context> invocation;
182 invocation.dispatch_blocks = dispatch_blocks;
183 invocation.expansion_blocks = expansion_blocks;
184 invocation.block_size = block_size;
185 invocation.work = static_cast<int>(events.size());
186 invocation.delta = input.max_time_span;
187 invocation.events = events_device.get();
188 invocation.event_count = static_cast<int>(events.size());
189 invocation.incoming_events = incoming_events_device.get();
190 invocation.incoming_offsets = incoming_offsets_device.get();
191 invocation.outgoing_events = outgoing_events_device.get();
192 invocation.outgoing_offsets = outgoing_offsets_device.get();
193 invocation.motif = motif_device.get();
194 invocation.yield = yield_device.get();
195 invocation.source = source_device.get();
196 invocation.offload = offload_device.get();
197 invocation.offload_top = offload_top_device.get();
198 invocation.next_offload_top = next_offload_top_device.get();
199 invocation.offload_width = offload_width_device.get();
200 invocation.count = count_device.get();
201
202 const auto count = runner(invocation);
203 check(cudaDeviceSynchronize(),
204 std::string("synchronizing ") + backend_name);
205 const auto finished = std::chrono::steady_clock::now();
206 return {Status::success(), static_cast<std::uint64_t>(count),
207 std::chrono::duration<double, std::milli>(finished - started)
208 .count()};
209 } catch (const std::exception& error) {
210 return {{StatusCode::execution_failed,
211 std::string(backend_name) + " execution failed: " +
212 error.what()},
213 0, 0.0};
214 }
