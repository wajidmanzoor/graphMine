16std::vector<VertexIndex> make_peeling_order(const CsrGraph& graph) {
17 using QueueEntry = std::pair<std::uint32_t, VertexIndex>;
18 std::priority_queue<QueueEntry, std::vector<QueueEntry>,
19 std::greater<QueueEntry>>
20 queue;
21 std::vector<std::uint32_t> degree(graph.vertex_count());
22 std::vector<bool> removed(graph.vertex_count(), false);
23
24 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
25 degree[vertex] = static_cast<std::uint32_t>(
26 graph.offsets[vertex + 1] - graph.offsets[vertex]);
27 queue.emplace(degree[vertex], vertex);
28 }
29
30 std::vector<VertexIndex> order;
31 order.reserve(graph.vertex_count());
32 while (!queue.empty()) {
33 const auto [queued_degree, vertex] = queue.top();
34 queue.pop();
35 if (removed[vertex] || queued_degree != degree[vertex]) {
36 continue;
37 }
38 removed[vertex] = true;
39 order.push_back(vertex);
40 for (auto offset = graph.offsets[vertex];
41 offset < graph.offsets[vertex + 1]; ++offset) {
42 const auto neighbor = graph.neighbors[offset];
43 if (!removed[neighbor] && degree[neighbor] > 0) {
44 --degree[neighbor];
45 queue.emplace(degree[neighbor], neighbor);
46 }
47 }
48 }
49 return order;
50}

78 }

86 KCoreOptionalOutput::requested_core_edges);

97ExecutionResult<KCoreOutput> KCore::run(const Graph& graph) const {
98 const auto started = std::chrono::steady_clock::now();
99 const auto support = supports(graph);
100 auto provenance = kcore_provenance();
101 if (!support.supported) {
102 return ExecutionResult<KCoreOutput>::failure(
103 {StatusCode::unsupported, support.reason}, provenance);
104 }
105
106 UndirectedGraphView normalized;
107 try {
108 normalized = graph.simple_undirected(
109 {options_.allow_directed_projection});
110 } catch (const std::exception& error) {
111 return ExecutionResult<KCoreOutput>::failure(
112 {StatusCode::invalid_argument, error.what()}, provenance);
113 }
114
115 std::vector<std::string> warnings;
116 if (normalized.normalization.directed_projection_applied) {
117 warnings.emplace_back("directed input was projected to an undirected graph");
118 }
119 if (normalized.normalization.self_loops_removed > 0) {
120 warnings.emplace_back("self-loops were removed during normalization");
121 }
122 if (normalized.normalization.parallel_edges_collapsed > 0) {
123 warnings.emplace_back("parallel edges were collapsed during normalization");
124 }
125
126 const auto backend = detail::run_kcore_gpu(
127 normalized.csr, options_.execution.device_ids.front());
128 if (!backend.status.ok()) {
129 return ExecutionResult<KCoreOutput>::failure(
130 backend.status, provenance, std::move(warnings));
131 }
132 if (backend.core_numbers.size() != graph.vertex_count()) {
133 return ExecutionResult<KCoreOutput>::failure(
134 {StatusCode::correctness_mismatch,
135 "KCoreGPU did not return one core number per input vertex"},
136 provenance, std::move(warnings));
137 }
138
139 KCoreOutput output;
140 output.core_number_by_vertex.reserve(graph.vertex_count());
141 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
142 output.core_number_by_vertex.push_back(
143 {graph.external_id(vertex), backend.core_numbers[vertex]});
144 output.degeneracy =
145 std::max(output.degeneracy, backend.core_numbers[vertex]);
146 }
147
148 if (has_output(options_.optional_outputs,
149 KCoreOptionalOutput::requested_core_vertices)) {
150 output.requested_core_vertices.emplace();
151 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
152 if (backend.core_numbers[vertex] >= *options_.requested_k) {
153 output.requested_core_vertices->push_back(graph.external_id(vertex));
154 }
155 }
156 }
157
158 if (has_output(options_.optional_outputs,
159 KCoreOptionalOutput::requested_core_edges)) {
160 output.requested_core_edges.emplace();
161 std::set<std::pair<VertexIndex, VertexIndex>> emitted;
162 for (const auto& edge : graph.canonical().edges) {
163 auto source = graph.dense_index(edge.source);
164 auto target = graph.dense_index(edge.target);
165 if (source == target) {
166 continue;
167 }
168 if (target < source) {
169 std::swap(source, target);
170 }
171 if (backend.core_numbers[source] >= *options_.requested_k &&
172 backend.core_numbers[target] >= *options_.requested_k &&
173 emitted.emplace(source, target).second) {
174 output.requested_core_edges->push_back(edge.id);
175 }
176 }
177 }
178
179 const auto materialization_started = std::chrono::steady_clock::now();
180 if (has_output(options_.optional_outputs,
181 KCoreOptionalOutput::peeling_order)) {
182 output.peeling_order.emplace();
183 for (const auto vertex : make_peeling_order(normalized.csr)) {
184 output.peeling_order->push_back(graph.external_id(vertex));
185 }
186 warnings.emplace_back(
187 "peeling_order is a deterministic valid common-layer order; "
188 "KCoreGPU does not export its internal concurrent order");
189 }
190 const auto materialization_finished = std::chrono::steady_clock::now();
191
192 const auto finished = std::chrono::steady_clock::now();
193 ExecutionStatistics statistics;
194 statistics.backend_ms = backend.elapsed_ms;
195 statistics.output_materialization_ms =
196 std::chrono::duration<double, std::milli>(materialization_finished -
197 materialization_started)
198 .count();
199 statistics.end_to_end_ms =
200 std::chrono::duration<double, std::milli>(finished - started).count();
201 return ExecutionResult<KCoreOutput>::success(
202 std::move(output), std::move(provenance), statistics,
203 std::move(warnings));
204}
