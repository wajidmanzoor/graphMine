37KokkosRuntime& runtime() {
38 // Deliberately process-lived: Kokkos installs CUDA runtime teardown state
39 // during initialize(). Registering our finalizer afterwards makes it run
40 // before that state is destroyed.
41 static auto* state = new KokkosRuntime;
42 return *state;
43}

45void ensure_kokkos(int device_id) {
46 auto& state = runtime();
47 if (!Kokkos::is_initialized()) {
48 Kokkos::InitializationSettings settings;
49 settings.set_device_id(device_id);
50 Kokkos::initialize(settings);
51 state.owned = true;
52 state.device_id = device_id;
53 if (!state.finalizer_registered) {
54 std::atexit([] {
55 auto& final_state = runtime();
56 if (final_state.owned && Kokkos::is_initialized()) {
57 Kokkos::finalize();
58 }
59 });
60 state.finalizer_registered = true;
61 }
62 return;
63 }
64 if (state.device_id < 0) {
65 int active_device = -1;
66 if (cudaGetDevice(&active_device) == cudaSuccess) {
67 state.device_id = active_device;
68 }
69 }
70 if (state.device_id >= 0 && state.device_id != device_id) {
71 throw std::invalid_argument(
72 "the Kokkos CUDA runtime is already initialized on another device");
73 }
74}

76matrix_t make_matrix(const CsrGraph& graph) {
77 const auto vertex_count = static_cast<ordinal_t>(graph.vertex_count());
78 const auto arc_count = static_cast<edge_offset_t>(graph.neighbors.size());
79
80 PggcOffsetView offsets_device(
81 Kokkos::ViewAllocateWithoutInitializing("GraphMine row offsets"),
82 static_cast<std::size_t>(vertex_count) + 1U);
83 PggcVertexView neighbors_device(
84 Kokkos::ViewAllocateWithoutInitializing("GraphMine neighbors"),
85 static_cast<std::size_t>(arc_count));
86 auto offsets_host = Kokkos::create_mirror_view(offsets_device);
87 auto neighbors_host = Kokkos::create_mirror_view(neighbors_device);
88 for (ordinal_t vertex = 0; vertex <= vertex_count; ++vertex) {
89 offsets_host(vertex) = static_cast<edge_offset_t>(graph.offsets[vertex]);
90 }
91 for (edge_offset_t edge = 0; edge < arc_count; ++edge) {
92 neighbors_host(edge) = static_cast<ordinal_t>(graph.neighbors[edge]);
93 }
94 Kokkos::deep_copy(offsets_device, offsets_host);
95 Kokkos::deep_copy(neighbors_device, neighbors_host);
96
97 using GraphType = typename matrix_t::staticcrsgraph_type;
98 GraphType topology(neighbors_device, offsets_device);
99 PggcWeightView no_explicit_weights;
100 return matrix_t("GraphMine community graph", vertex_count,
101 no_explicit_weights, topology);
102}

104PggcWeightView make_degree_weights(const CsrGraph& graph) {
105 PggcWeightView weights(
106 Kokkos::ViewAllocateWithoutInitializing("GraphMine vertex degrees"),
107 graph.vertex_count());
108 auto host = Kokkos::create_mirror_view(weights);
109 for (std::size_t vertex = 0; vertex < graph.vertex_count(); ++vertex) {
110 host(static_cast<ordinal_t>(vertex)) = static_cast<value_t>(
111 graph.offsets[vertex + 1U] - graph.offsets[vertex]);
112 }
113 Kokkos::deep_copy(weights, host);
114 return weights;
115}

121CommunityDetectionBackendResult run_pggc(const CsrGraph& graph,
122 int device_id,
123 PggcAlgorithm algorithm) {
124 if (graph.vertex_count() == 0) {
125 return {Status::success(), {}, 0.0, 0.0};
126 }
127 if (graph.vertex_count() >
128 static_cast<std::size_t>(std::numeric_limits<ordinal_t>::max()) ||
129 graph.neighbors.size() > static_cast<std::size_t>(
130 std::numeric_limits<edge_offset_t>::max())) {
131 return {{StatusCode::unsupported,
132 "parallel multilevel clustering uses signed 32-bit graph "
133 "indices"},
134 {}, std::nullopt, 0.0};
135 }
136
137 int device_count = 0;
138 if (cudaGetDeviceCount(&device_count) != cudaSuccess || device_id < 0 ||
139 device_id >= device_count) {
140 return {{StatusCode::backend_unavailable,
141 "requested CUDA device is unavailable for parallel multilevel "
142 "clustering"},
143 {}, std::nullopt, 0.0};
144 }
145
146 try {
147 std::lock_guard<std::mutex> lock(pggc_mutex);
148 if (!Kokkos::is_initialized()) {
149 const auto code = cudaSetDevice(device_id);
150 if (code != cudaSuccess) {
151 throw std::runtime_error(cudaGetErrorString(code));
152 }
153 }
154 ensure_kokkos(device_id);
155
156 const auto started = std::chrono::steady_clock::now();
157 auto matrix = make_matrix(graph);
158 jet_community::weighted_graph weighted;
159 weighted.mtx = matrix;
160 weighted.vtx_w = make_degree_weights(graph);
161 weighted.edge_uniform = true;
162
163 modularity objective(matrix, weighted.vtx_w, 1.0, true);
164 memory_store memory(matrix, objective);
165 jet_community::ExperimentLoggerUtil<value_t> experiment;
166 PggcVertexView unused_constraint;
167 PggcVertexView partition;
168 switch (algorithm) {
169 case PggcAlgorithm::parallel_louvain:
170 partition = jet_community::clustering_methods::louvain_part<false>(
171 memory, weighted, objective, experiment, unused_constraint);
172 break;
173 case PggcAlgorithm::parallel_leiden:
174 partition = jet_community::clustering_methods::leiden_part<false,
175 false>(
176 memory, weighted, objective, experiment, unused_constraint);
177 break;
178 case PggcAlgorithm::parallel_leiden_plus:
179 partition = jet_community::clustering_methods::leiden_part<true,
180 false>(
181 memory, weighted, objective, experiment, unused_constraint);
182 break;
183 }
184 Kokkos::fence();
185 const auto partition_host = Kokkos::create_mirror_view(partition);
186 Kokkos::deep_copy(partition_host, partition);
187
188 std::vector<std::uint32_t> labels(graph.vertex_count());
189 for (std::size_t vertex = 0; vertex < labels.size(); ++vertex) {
190 const auto label = partition_host(static_cast<ordinal_t>(vertex));
191 if (label < 0 || label >= static_cast<ordinal_t>(graph.vertex_count())) {
192 return {{StatusCode::correctness_mismatch,
193 "parallel multilevel clustering returned an invalid label"},
194 {}, std::nullopt, 0.0};
195 }
196 labels[vertex] = static_cast<std::uint32_t>(label);
197 }
198 const auto native_modularity = objective.get_objective();
199 const auto finished = std::chrono::steady_clock::now();
200 return {Status::success(), std::move(labels), native_modularity,
201 std::chrono::duration<double, std::milli>(finished - started)
202 .count()};
203 } catch (const std::exception& error) {
204 return {{StatusCode::execution_failed,
205 std::string("parallel multilevel clustering failed: ") +
206 error.what()},
207 {}, std::nullopt, 0.0};
208 }
209}
