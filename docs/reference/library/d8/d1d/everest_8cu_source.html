41 int device_count = 0;

44 return {{StatusCode::backend_unavailable,
45 "requested CUDA device is unavailable for Everest"},
46 0, 0.0};
47 }
48
49 try {
50 temporal_cuda::check(cudaSetDevice(device_id),
51 "selecting the Everest CUDA device");
52 cudaDeviceProp properties{};
53 temporal_cuda::check(cudaGetDeviceProperties(&properties, device_id),
54 "reading Everest CUDA device properties");
55
56 constexpr int block_size = 96;
57 int blocks_per_sm = 0;
58 temporal_cuda::check(
59 cudaOccupancyMaxActiveBlocksPerMultiprocessor(
60 &blocks_per_sm, graphmine_everest_dispatch, block_size, 0),
61 "computing Everest dispatch occupancy");
62 const int dispatch_blocks = blocks_per_sm * properties.multiProcessorCount;
63 temporal_cuda::check(
64 cudaOccupancyMaxActiveBlocksPerMultiprocessor(
65 &blocks_per_sm, graphmine_everest_expand, block_size, 0),
66 "computing Everest expansion occupancy");
67 const int expansion_blocks = blocks_per_sm * properties.multiProcessorCount;
68
69 return temporal_cuda::execute<GraphMineEverestContext>(
70 input, device_id, dispatch_blocks, expansion_blocks, block_size,
71 "Everest", [](const auto& invocation) {

75 invocation.events, invocation.event_count,

77 invocation.outgoing_events, invocation.outgoing_offsets,
