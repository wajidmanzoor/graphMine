46 template <typename T>
47 T* allocate(std::size_t count) {
48 if (count == 0) {
49 return nullptr;
50 }
51 if (count > std::numeric_limits<std::size_t>::max() / sizeof(T)) {
52 throw std::overflow_error("cuMBE allocation size overflow");
53 }
54 T* pointer = nullptr;
55 check_cuda(cudaMallocManaged(reinterpret_cast<void**>(&pointer),
56 sizeof(T) * count),
57 "allocating cuMBE managed memory");
58 allocations_.push_back(pointer);
59 return pointer;
60 }

76HostBipartiteGraph prepare_graph(const BipartiteCsrGraph& input) {
77 const auto left_count = static_cast<std::size_t>(input.left_vertex_count);
78 const auto right_count = static_cast<std::size_t>(input.right_vertex_count);
79 if (input.left_offsets.size() != left_count + 1 ||
80 input.left_offsets.front() != 0 ||
81 input.left_offsets.back() != input.right_neighbors.size()) {
82 throw std::invalid_argument("invalid bipartite CSR offsets");
83 }
84 if (input.right_neighbors.size() >
85 static_cast<std::size_t>(std::numeric_limits<int>::max())) {
86 throw std::invalid_argument("cuMBE uses signed 32-bit edge indices");
87 }
88
89 std::vector<Node> left_nodes(left_count);
90 std::vector<int> left_edges;
91 left_edges.reserve(input.right_neighbors.size());
92 std::vector<std::uint32_t> right_degrees(right_count, 0);
93 for (std::size_t left = 0; left < left_count; ++left) {
94 const auto begin = input.left_offsets[left];
95 const auto end = input.left_offsets[left + 1];
96 if (end < begin || end > input.right_neighbors.size()) {
97 throw std::invalid_argument("invalid bipartite CSR row");
98 }
99 left_nodes[left] = {static_cast<int>(begin),
100 static_cast<int>(end - begin)};
101 for (auto offset = begin; offset < end; ++offset) {
102 const auto right = input.right_neighbors[offset];
103 if (right >= right_count) {
104 throw std::invalid_argument("bipartite neighbor is out of range");
105 }
106 left_edges.push_back(static_cast<int>(right));
107 ++right_degrees[right];
108 }
109 }
110
111 std::vector<Node> right_nodes(right_count);
112 std::vector<std::size_t> cursor(right_count, 0);
113 std::size_t offset = 0;
114 for (std::size_t right = 0; right < right_count; ++right) {
115 right_nodes[right] = {static_cast<int>(offset),
116 static_cast<int>(right_degrees[right])};
117 cursor[right] = offset;
118 offset += right_degrees[right];
119 }
120 std::vector<int> right_edges(input.right_neighbors.size());
121 for (std::size_t left = 0; left < left_count; ++left) {
122 for (auto edge = input.left_offsets[left];
123 edge < input.left_offsets[left + 1]; ++edge) {
124 const auto right = input.right_neighbors[edge];
125 right_edges[cursor[right]++] = static_cast<int>(left);
126 }
127 }
128
129 HostBipartiteGraph output;
130 output.num_edges = static_cast<int>(left_edges.size());
131 // The original executable transposes when its candidate side is larger.
132 // Building both directions in memory lets us make the same choice directly.
133 if (left_count <= right_count) {
134 output.num_r = static_cast<int>(left_count);
135 output.num_l = static_cast<int>(right_count);
136 output.node_r = std::move(left_nodes);
137 output.edge_r = std::move(left_edges);
138 output.node_l = std::move(right_nodes);
139 output.edge_l = std::move(right_edges);
140 } else {
141 output.num_r = static_cast<int>(right_count);
142 output.num_l = static_cast<int>(left_count);
143 output.node_r = std::move(right_nodes);
144 output.edge_r = std::move(right_edges);
145 output.node_l = std::move(left_nodes);
146 output.edge_l = std::move(left_edges);
147 }
148 return output;
149}

161 int device_id) {
162 if (input.left_vertex_count == 0 || input.right_vertex_count == 0 ||
163 input.right_neighbors.empty()) {
164 return {Status::success(), 0, 0.0};
165 }
166 if (input.left_vertex_count >
167 static_cast<std::uint32_t>(std::numeric_limits<int>::max()) ||
168 input.right_vertex_count >
169 static_cast<std::uint32_t>(std::numeric_limits<int>::max())) {
170 return {{StatusCode::unsupported,
171 "cuMBE uses signed 32-bit vertex indices"},
172 0, 0.0};
173 }

175 try {
176 const auto host = prepare_graph(input);
177 std::lock_guard<std::mutex> lock(cumbe_mutex);
178 check_cuda(cudaSetDevice(device_id), "selecting the cuMBE CUDA device");
179
180 cudaDeviceProp properties{};
181 check_cuda(cudaGetDeviceProperties(&properties, device_id),
182 "querying the cuMBE CUDA device");
183 if (!properties.cooperativeLaunch) {
184 return {{StatusCode::unsupported,
185 "cuMBE requires CUDA cooperative-kernel launch support"},
186 0, 0.0};
187 }
188 if (properties.maxThreadsPerBlock < NUM_THDS) {
189 return {{StatusCode::unsupported,
190 "cuMBE requires at least 512 threads per block"},
191 0, 0.0};
192 }

194 int blocks_per_sm = 0;
195 check_cuda(cudaOccupancyMaxActiveBlocksPerMultiprocessor(

197 "computing cuMBE cooperative occupancy");
198 const int block_count =
199 std::max(1, std::min(NUM_BLKS,
200 properties.multiProcessorCount * blocks_per_sm));
201 const std::size_t l = static_cast<std::size_t>(host.num_l);
202 const std::size_t r = static_cast<std::size_t>(host.num_r);
203 const std::size_t blocks = static_cast<std::size_t>(block_count);
204 if (l > std::numeric_limits<std::size_t>::max() / blocks ||
205 r > std::numeric_limits<std::size_t>::max() / blocks) {
206 return {{StatusCode::resource_exhausted,
207 "cuMBE workspace size overflow"},
208 0, 0.0};
209 }

322 return {Status::success(), static_cast<std::uint64_t>(*num_mb),

324 .count()};
325 } catch (const std::exception& error) {
326 return {{StatusCode::execution_failed,
327 std::string("cuMBE execution failed: ") + error.what()},
328 0, 0.0};
