52static double median(double array[], const int n)
53{
54 double median = 0;
55 std::sort(array, array + n);
56 if (n % 2 == 0) median = (array[(n - 1) / 2] + array[n / 2]) / 2.0;
57 else median = array[n / 2];
58 return median;
59}
