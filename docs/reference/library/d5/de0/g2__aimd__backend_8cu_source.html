73BackendCountResult run_g2_aimd_count(const CsrGraph& graph, int device_id) {
74 if (graph.vertex_count() == 0) {
75 return {Status::success(), 0, 0.0};
76 }
77 if (graph.vertex_count() > static_cast<std::size_t>(UINT_MAX) ||
78 graph.neighbors.size() >
79 static_cast<std::size_t>(std::numeric_limits<uintE>::max())) {
80 return {{StatusCode::unsupported,
81 "G2-AIMD uses unsigned 32-bit vertex indices"},
82 0, 0.0};
83 }
84 std::uint64_t maximum_degree = 0;
85 for (std::size_t vertex = 0; vertex < graph.vertex_count(); ++vertex) {
86 maximum_degree =
87 std::max(maximum_degree,
88 graph.offsets[vertex + 1U] - graph.offsets[vertex]);
89 }
90 if (maximum_degree >= TEMPSIZE) {
91 return {{StatusCode::unsupported,
92 "G2-AIMD requires maximum degree below its 100000-entry "
93 "per-warp temporary capacity"},
94 0, 0.0};
95 }
96
97 int device_count = 0;
98 if (device_id < 0 || device_id >= 8 ||
99 cudaGetDeviceCount(&device_count) != cudaSuccess ||
100 device_id >= device_count || cudaSetDevice(device_id) != cudaSuccess) {
101 return {{StatusCode::backend_unavailable,
102 "requested CUDA device is unavailable for G2-AIMD"},
103 0, 0.0};
104 }
105
106 try {
107 std::lock_guard<std::mutex> lock(g2_aimd_mutex);
108 std::vector<uintE> offsets(graph.offsets.begin(), graph.offsets.end());
109 std::vector<uintV> neighbors(graph.neighbors.begin(),
110 graph.neighbors.end());
111 std::vector<uintV> sources(graph.vertex_count());
112 std::iota(sources.begin(), sources.end(), uintV{0});
113
114 GraphMineG2AimdGraph original;
115 original.SetVertexCount(graph.vertex_count());
116 original.SetEdgeCount(graph.neighbors.size());
117 original.SetRowPtrs(offsets.data());
118 original.SetCols(neighbors.data());
119
120 GraphMineG2AimdBKExpandSequential application;
121 std::vector<StoreStrategy> strategy;
122 constexpr std::size_t maximum_buffer_entries = 16'000'000ULL;
123 const auto adjacency_bytes =
124 std::max<std::size_t>(sizeof(uintV),
125 neighbors.size() * sizeof(uintV));
126 GraphMineG2AimdPipelineExecutor<GraphMineG2AimdBKExpandSequential>
127 executor(static_cast<std::size_t>(device_id), &original,
128 graph.vertex_count(), application, adjacency_bytes, strategy,
129 maximum_buffer_entries);
130
131 executor.app.ctx->sources_num[0] = sources.size();
132 HToD(executor.app.ctx->sources, sources.data(), sources.size());
133 HToD(executor.app.ctx->d_row_ptrs, offsets.data(), offsets.size());
134 HToD(executor.app.ctx->d_cols, neighbors.data(), neighbors.size());
135
136 const auto started = std::chrono::steady_clock::now();
137 executor.Run();
138 CUDA_ERROR(cudaDeviceSynchronize());
139 const auto finished = std::chrono::steady_clock::now();
140 const auto count = executor.app.totalCliques[0];
141 return {Status::success(), static_cast<std::uint64_t>(count),
142 std::chrono::duration<double, std::milli>(finished - started)
143 .count()};
144 } catch (const std::exception& error) {
145 return {{StatusCode::execution_failed,
146 std::string("G2-AIMD execution failed: ") + error.what()},
147 0, 0.0};
148 }
149}
