21TriangleMaterialization materialize_triangles(
22 const Graph& original, const CsrGraph& graph,
23 const TriangleOptions& options) {
24 TriangleMaterialization result;
25 if (options.include_per_vertex_counts) {
26 result.vertex_counts.resize(graph.vertex_count(), 0);
27 }
28
29 std::map<std::pair<VertexIndex, VertexIndex>, std::size_t> edge_positions;
30 if (options.include_per_edge_counts) {
31 for (const auto& edge : original.canonical().edges) {
32 auto source = original.dense_index(edge.source);
33 auto target = original.dense_index(edge.target);
34 if (source == target) {
35 continue;
36 }
37 if (target < source) {
38 std::swap(source, target);
39 }
40 const auto [it, inserted] = edge_positions.emplace(
41 std::make_pair(source, target), result.edge_counts.size());
42 if (inserted) {
43 result.edge_counts.push_back({edge.id, 0});
44 }
45 }
46 }
47
48 for (VertexIndex u = 0; u < graph.vertex_count(); ++u) {
49 const auto u_begin = graph.neighbors.begin() +
50 static_cast<std::ptrdiff_t>(graph.offsets[u]);
51 const auto u_end = graph.neighbors.begin() +
52 static_cast<std::ptrdiff_t>(graph.offsets[u + 1]);
53 for (auto v_it = std::upper_bound(u_begin, u_end, u); v_it != u_end;
54 ++v_it) {
55 const auto v = *v_it;
56 const auto v_begin = graph.neighbors.begin() +
57 static_cast<std::ptrdiff_t>(graph.offsets[v]);
58 const auto v_end = graph.neighbors.begin() +
59 static_cast<std::ptrdiff_t>(graph.offsets[v + 1]);
60 auto left = std::upper_bound(v_it + 1, u_end, v);
61 auto right = std::upper_bound(v_begin, v_end, v);
62 while (left != u_end && right != v_end) {
63 if (*left < *right) {
64 ++left;
65 } else if (*right < *left) {
66 ++right;
67 } else {
68 const auto w = *left;
69 ++result.count;
70 if (options.list_instances &&
71 (!options.result_limit ||
72 result.instances.size() < *options.result_limit)) {
73 result.instances.push_back({u, v, w});
74 }
75 if (options.include_per_vertex_counts) {
76 ++result.vertex_counts[u];
77 ++result.vertex_counts[v];
78 ++result.vertex_counts[w];
79 }
80 if (options.include_per_edge_counts) {
81 ++result.edge_counts[edge_positions.at({u, v})].count;
82 ++result.edge_counts[edge_positions.at({u, w})].count;
83 ++result.edge_counts[edge_positions.at({v, w})].count;
84 }
85 ++left;
86 ++right;
87 }
88 }
89 }
90 }
91 return result;
92}

94Provenance triangle_provenance(TriangleBackend backend) {
95 Provenance provenance;
96 provenance.problem = "triangle_counting_listing";
97 provenance.backend = to_string(backend);
98 switch (backend) {
99 case TriangleBackend::tot:
100 provenance.backend_version = "ToT tensor-core artifact";
101 provenance.source_commit = "5f9cde83a528";
102 provenance.execution_path =
103 "original ToT bitmap/tensor count + optional common materializer";
104 break;
105 case TriangleBackend::wetric:
106 provenance.backend_version = "WeTriC artifact";
107 provenance.source_commit =
108 "f0c80447dc428ead5f8a41f2f1d9c4c4cbc41c2e";
109 provenance.execution_path =
110 "original WeTriC wedge-parallel count + optional common materializer";
111 break;
112 case TriangleBackend::automatic:
113 break;
114 }
115 return provenance;

135SupportReport TriangleCounting::supports(const Graph& graph) const {
136 if (graph.directed() && !options_.allow_directed_projection) {
137 return {false,
138 "directed input requires allow_directed_projection=true"};
139 }
140 if (options_.execution.device_ids.empty()) {
141 return {false, "at least one CUDA device id must be provided"};
142 }
143 if (options_.result_limit && *options_.result_limit == 0) {
144 return {false, "result_limit must be null or at least one"};
145 }
146 if (options_.result_limit && !options_.list_instances) {
147 return {false, "result_limit is meaningful only when list_instances=true"};
148 }
149 if (options_.wetric_spread == 0) {
150 return {false, "wetric_spread must be at least one"};
151 }
152 if (options_.wetric_adjacency_matrix_length == 0 ||
153 options_.wetric_adjacency_matrix_length % 64 != 0) {
154 return {false,
155 "wetric_adjacency_matrix_length must be a positive multiple of 64"};
156 }
157 const auto selected = options_.backend == TriangleBackend::automatic
158 ? TriangleBackend::tot
159 : options_.backend;
160 if (selected == TriangleBackend::tot &&
161 !detail::tot_backend_compiled()) {
162 return {false, "the ToT backend is not compiled in this build"};
163 }
164 if (selected == TriangleBackend::wetric &&
165 !detail::wetric_backend_compiled()) {
166 return {false, "the WeTriC backend is not compiled in this build"};
167 }
168 return {true, {}};
169}

171ExecutionResult<TriangleOutput> TriangleCounting::run(
172 const Graph& graph) const {
173 const auto started = std::chrono::steady_clock::now();
174 const auto support = supports(graph);
175 const auto selected = options_.backend == TriangleBackend::automatic
176 ? TriangleBackend::tot
177 : options_.backend;
178 auto provenance = triangle_provenance(selected);
179 if (!support.supported) {
180 return ExecutionResult<TriangleOutput>::failure(
181 {StatusCode::unsupported, support.reason}, provenance);
182 }
183
184 UndirectedGraphView normalized;
185 try {
186 normalized = graph.simple_undirected(
187 {options_.allow_directed_projection});
188 } catch (const std::exception& error) {
189 return ExecutionResult<TriangleOutput>::failure(
190 {StatusCode::invalid_argument, error.what()}, provenance);
191 }
192
193 std::vector<std::string> warnings;
194 if (normalized.normalization.directed_projection_applied) {
195 warnings.emplace_back("directed input was projected to an undirected graph");
196 }
197 if (normalized.normalization.self_loops_removed > 0) {
198 warnings.emplace_back("self-loops were removed during normalization");
199 }
200 if (normalized.normalization.parallel_edges_collapsed > 0) {
201 warnings.emplace_back("parallel edges were collapsed during normalization");
202 }
203
204 detail::TriangleBackendResult backend;
205 switch (selected) {
206 case TriangleBackend::tot:
207 backend = detail::run_tot(normalized.csr,
208 options_.execution.device_ids.front());
209 break;
210 case TriangleBackend::wetric:
211 backend = detail::run_wetric(
212 normalized.csr, options_.execution.device_ids.front(),
213 options_.wetric_spread,
214 options_.wetric_adjacency_matrix_length);
215 break;
216 case TriangleBackend::automatic:
217 return ExecutionResult<TriangleOutput>::failure(
218 {StatusCode::internal_error,
219 "automatic triangle backend was not resolved"},
220 provenance, std::move(warnings));
221 }
222 if (!backend.status.ok()) {
223 return ExecutionResult<TriangleOutput>::failure(
224 backend.status, provenance, std::move(warnings));
225 }
226
227 TriangleOutput output;
228 output.global_triangle_count = backend.count;
229 const bool needs_materialization = options_.list_instances ||
230 options_.include_per_vertex_counts ||
231 options_.include_per_edge_counts;
232 const auto materialization_started = std::chrono::steady_clock::now();
233 if (needs_materialization) {
234 auto materialized =
235 materialize_triangles(graph, normalized.csr, options_);
236 if (materialized.count != backend.count) {
237 return ExecutionResult<TriangleOutput>::failure(
238 {StatusCode::correctness_mismatch,
239 std::string(to_string(selected)) +
240 " count " + std::to_string(backend.count) +
241 " disagreed with exact materializer count " +
242 std::to_string(materialized.count)},
243 provenance, std::move(warnings));
244 }
245 if (options_.list_instances) {
246 output.triangles.emplace();
247 output.triangles->reserve(materialized.instances.size());
248 for (const auto& triangle : materialized.instances) {
249 output.triangles->push_back({graph.external_id(triangle[0]),
250 graph.external_id(triangle[1]),
251 graph.external_id(triangle[2])});
252 }
253 output.instances_complete =
254 output.triangles->size() == output.global_triangle_count;
255 }
256 if (options_.include_per_vertex_counts) {
257 output.per_vertex_count.emplace();
258 output.per_vertex_count->reserve(graph.vertex_count());
259 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
260 output.per_vertex_count->push_back(
261 {graph.external_id(vertex), materialized.vertex_counts[vertex]});
262 }
263 }
264 if (options_.include_per_edge_counts) {
265 output.per_edge_count = std::move(materialized.edge_counts);
266 if (normalized.normalization.parallel_edges_collapsed > 0) {
267 warnings.emplace_back(
268 "per-edge counts use the first input edge ID as the representative "
269 "of each collapsed parallel-edge group");
270 }
271 }
272 }
273 const auto materialization_finished = std::chrono::steady_clock::now();
274
275 const auto finished = std::chrono::steady_clock::now();
276 ExecutionStatistics statistics;
277 statistics.backend_ms = backend.elapsed_ms;
278 statistics.output_materialization_ms =
279 std::chrono::duration<double, std::milli>(materialization_finished -
280 materialization_started)
281 .count();
282 statistics.end_to_end_ms =
283 std::chrono::duration<double, std::milli>(finished - started).count();
284 return ExecutionResult<TriangleOutput>::success(
285 std::move(output), std::move(provenance), statistics,
286 std::move(warnings));
287}
