42KCliqueBackendResult run_kcgpu(const CsrGraph& graph, std::uint32_t k,
43 int device_id) {
44 if (k < 3 || k > 5) {
45 return {{StatusCode::unsupported,
46 "the validated KCGPU path supports 3 <= k <= 5"},
47 0, 0.0};
48 }
49 if (graph.vertex_count() < k) {
50 return {Status::success(), 0, 0.0};
51 }
52 if (graph.vertex_count() > static_cast<std::size_t>(UINT_MAX) ||
53 graph.undirected_edge_count() > static_cast<std::uint64_t>(UINT_MAX)) {
54 return {{StatusCode::unsupported, "KCGPU uses unsigned 32-bit indices"},
55 0, 0.0};
56 }
57
58 int device_count = 0;
59 if (device_id < 0 || cudaGetDeviceCount(&device_count) != cudaSuccess ||
60 device_id >= device_count || cudaSetDevice(device_id) != cudaSuccess) {
61 return {{StatusCode::backend_unavailable,
62 "requested CUDA device is unavailable for KCGPU"},
63 0, 0.0};
64 }
65
66 try {
67 std::vector<std::uint32_t> degrees(graph.vertex_count());
68 for (VertexIndex vertex = 0; vertex < graph.vertex_count(); ++vertex) {
69 degrees[vertex] = static_cast<std::uint32_t>(
70 graph.offsets[vertex + 1] - graph.offsets[vertex]);
71 }
72
73 // KCGPU's validated MatrixMarket path retains the format's one-based
74 // vertex numbering (vertex zero is an empty sentinel). Preserve that
75 // internal layout while the public GraphMine graph remains zero-based.
76 const auto upstream_vertex_count = graph.vertex_count() + 1;
77 std::vector<std::uint32_t> offsets(upstream_vertex_count + 1, 0);
78 std::vector<std::uint32_t> rows;
79 std::vector<std::uint32_t> columns;
80 rows.reserve(graph.undirected_edge_count());
81 columns.reserve(graph.undirected_edge_count());
82 for (VertexIndex source = 0; source < graph.vertex_count(); ++source) {
83 for (auto offset = graph.offsets[source];
84 offset < graph.offsets[source + 1]; ++offset) {
85 const auto target = graph.neighbors[offset];
86 // This is the exact Degree mode used by the validated executable:
87 // lower degree to higher degree, then lower ID to higher ID.
88 if (degrees[source] < degrees[target] ||
89 (degrees[target] == degrees[source] && source < target)) {
90 rows.push_back(source + 1);
91 columns.push_back(target + 1);
92 ++offsets[source + 2];
93 }
94 }
95 }
96 for (std::size_t index = 1; index < offsets.size(); ++index) {
97 offsets[index] += offsets[index - 1];
98 }
99
100 DeviceGraph device;
101 device.value.numNodes = static_cast<uint>(upstream_vertex_count);
102 device.value.numEdges = static_cast<uint>(columns.size());
103 device.value.capacity = device.value.numEdges;
104 if (cudaMalloc(&device.value.rowPtr,
105 offsets.size() * sizeof(std::uint32_t)) != cudaSuccess ||
106 cudaMalloc(&device.value.rowInd,
107 rows.size() * sizeof(std::uint32_t)) != cudaSuccess ||
108 cudaMalloc(&device.value.colInd,
109 columns.size() * sizeof(std::uint32_t)) != cudaSuccess) {
110 return {{StatusCode::resource_exhausted,
111 "KCGPU device graph allocation failed"},
112 0, 0.0};
113 }
114 if (cudaMemcpy(device.value.rowPtr, offsets.data(),
115 offsets.size() * sizeof(std::uint32_t),
116 cudaMemcpyHostToDevice) != cudaSuccess ||
117 cudaMemcpy(device.value.rowInd, rows.data(),
118 rows.size() * sizeof(std::uint32_t),
119 cudaMemcpyHostToDevice) != cudaSuccess ||
120 cudaMemcpy(device.value.colInd, columns.data(),
121 columns.size() * sizeof(std::uint32_t),
122 cudaMemcpyHostToDevice) != cudaSuccess) {
123 return {{StatusCode::execution_failed,
124 "KCGPU device graph transfer failed"},
125 0, 0.0};
126 }
127
128 const std::lock_guard<std::mutex> lock(kcgpu_mutex);
129 graph::SingleGPU_Kclique<uint> algorithm(device_id, device.value);
130 const auto started = std::chrono::steady_clock::now();
131 // PIV-8-BE is the exact KCGPU configuration that passed validation.
132 algorithm.findKclqueIncremental_node_pivot_async<8>(
133 static_cast<int>(k), device.value, BlockWarp);
134 algorithm.sync();
135 const auto count = algorithm.result_count();
136 const auto cuda_result = cudaGetLastError();
137 const auto finished = std::chrono::steady_clock::now();
138 algorithm.free();
139 cudaStreamDestroy(algorithm.stream());
140 if (cuda_result != cudaSuccess) {
141 return {{StatusCode::execution_failed,
142 std::string("KCGPU failed: ") +
143 cudaGetErrorString(cuda_result)},
144 0, 0.0};
145 }
146 return {Status::success(), count,
147 std::chrono::duration<double, std::milli>(finished - started)
148 .count()};
149 } catch (const std::exception& error) {
150 return {{StatusCode::execution_failed,
151 std::string("KCGPU execution failed: ") + error.what()},
152 0, 0.0};
153 }
154}
