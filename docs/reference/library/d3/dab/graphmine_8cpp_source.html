273graphmine::Graph parse_graph(const json::value& value,
274 std::string_view source_name) {
275 const auto& root = require_object(value, source_name);
276 reject_unknown_fields(root, {"graph", "vertices", "edges"}, source_name);
277
278 graphmine::CanonicalGraphInput input;
279 const auto& descriptor = require_object(
280 require_field(root, "graph", source_name), "graph");
281 reject_unknown_fields(descriptor,
282 {"id", "directed", "allows_self_loops",
283 "allows_parallel_edges", "attributes"},
284 "graph");
285 input.graph.id = require_string(require_field(descriptor, "id", "graph"),
286 "graph.id");
287 input.graph.directed = require_bool(
288 require_field(descriptor, "directed", "graph"), "graph.directed");
289 input.graph.allows_self_loops = require_bool(
290 require_field(descriptor, "allows_self_loops", "graph"),
291 "graph.allows_self_loops");
292 input.graph.allows_parallel_edges = require_bool(
293 require_field(descriptor, "allows_parallel_edges", "graph"),
294 "graph.allows_parallel_edges");
295 input.graph.attributes = parse_attributes(descriptor, "graph.attributes");
296
297 const auto& vertices = require_array(
298 require_field(root, "vertices", source_name), "vertices");
299 input.vertices.reserve(vertices.size());
300 for (std::size_t index = 0; index < vertices.size(); ++index) {
301 const auto context = "vertices[" + std::to_string(index) + "]";
302 const auto& vertex = require_object(vertices[index], context);
303 reject_unknown_fields(vertex, {"id", "label", "type", "attributes"},
304 context);
305 graphmine::VertexRecord record;
306 record.id =
307 parse_external_id(require_field(vertex, "id", context), context + ".id");
308 if (const auto* label = vertex.if_contains("label")) {
309 record.label = require_string(*label, context + ".label");
310 }
311 if (const auto* type = vertex.if_contains("type")) {
312 record.type = require_string(*type, context + ".type");
313 }
314 record.attributes = parse_attributes(vertex, context + ".attributes");
315 input.vertices.push_back(std::move(record));
316 }
317
318 const auto& edges =
319 require_array(require_field(root, "edges", source_name), "edges");
320 input.edges.reserve(edges.size());
321 for (std::size_t index = 0; index < edges.size(); ++index) {
322 const auto context = "edges[" + std::to_string(index) + "]";
323 const auto& edge = require_object(edges[index], context);
324 reject_unknown_fields(edge,
325 {"id", "source", "target", "weight", "timestamp",
326 "type", "attributes"},
327 context);
328 graphmine::EdgeRecord record;
329 record.id =
330 parse_external_id(require_field(edge, "id", context), context + ".id");
331 record.source = parse_external_id(
332 require_field(edge, "source", context), context + ".source");
333 record.target = parse_external_id(
334 require_field(edge, "target", context), context + ".target");
335 if (const auto* weight = edge.if_contains("weight")) {
336 record.weight = require_number(*weight, context + ".weight");
337 if (!std::isfinite(*record.weight)) {
338 throw UsageError(context + ".weight must be finite");
339 }
340 }
341 if (const auto* timestamp = edge.if_contains("timestamp")) {
342 if (timestamp->is_string()) {
343 record.timestamp = std::string(timestamp->as_string());
344 } else {
345 record.timestamp = require_int64(*timestamp, context + ".timestamp");
346 }
347 }
348 if (const auto* type = edge.if_contains("type")) {
349 record.type = require_string(*type, context + ".type");
350 }
351 record.attributes = parse_attributes(edge, context + ".attributes");
352 input.edges.push_back(std::move(record));
353 }
354
355 try {
356 return graphmine::Graph(std::move(input));
357 } catch (const std::exception& error) {
358 throw UsageError(std::string(source_name) + ": " + error.what());
359 }
360}

456void write_pretty(std::ostream& output, const json::value& value,
457 unsigned indent = 0) {
458 if (value.is_array()) {
459 const auto& array = value.as_array();
460 if (array.empty()) {
461 output << "[]";
462 return;
463 }
464 output << "[\n";
465 for (std::size_t i = 0; i < array.size(); ++i) {
466 write_indent(output, indent + 2);
467 write_pretty(output, array[i], indent + 2);
468 output << (i + 1 == array.size() ? "\n" : ",\n");
469 }
470 write_indent(output, indent);
471 output << ']';
472 return;
473 }
474 if (value.is_object()) {
475 const auto& object = value.as_object();
476 if (object.empty()) {
477 output << "{}";
478 return;
479 }
480 output << "{\n";
481 std::size_t index = 0;
482 for (const auto& member : object) {
483 write_indent(output, indent + 2);
484 output << json::serialize(member.key()) << ": ";
485 write_pretty(output, member.value(), indent + 2);
486 output << (++index == object.size() ? "\n" : ",\n");
487 }
488 write_indent(output, indent);
489 output << '}';
490 return;
491 }
492 output << json::serialize(value);
493}

513class Arguments {
514 public:
515 Arguments(int argc, char** argv, int first) {
516 for (int i = first; i < argc; ++i) {
517 std::string token(argv[i]);
518 if (token.rfind("--", 0) != 0 || token.size() == 2) {
519 throw UsageError("unexpected positional argument: " + token);
520 }
521 token.erase(0, 2);
522 std::optional<std::string> value;
523 const auto equals = token.find('=');
524 if (equals != std::string::npos) {
525 value = token.substr(equals + 1);
526 token.resize(equals);
527 } else if (i + 1 < argc &&
528 std::string_view(argv[i + 1]).rfind("--", 0) != 0) {
529 value = std::string(argv[++i]);
530 }
531 entries_.push_back({std::move(token), std::move(value), false});
532 }
533 }

535 bool flag(std::string_view name) {
536 auto matches = find(name);
537 if (matches.size() > 1) {
538 throw UsageError("--" + std::string(name) + " may be specified once");
539 }
540 if (matches.empty()) return false;
541 auto& entry = entries_[matches.front()];
542 entry.consumed = true;
543 if (entry.value) {
544 throw UsageError("--" + std::string(name) + " is a flag and takes no value");
545 }
546 return true;

550 auto matches = find(name);
551 if (matches.size() > 1) {
552 throw UsageError("--" + std::string(name) + " may be specified once");
553 }
554 if (matches.empty()) return std::nullopt;
555 auto& entry = entries_[matches.front()];
556 entry.consumed = true;
557 if (!entry.value) {
558 throw UsageError("--" + std::string(name) + " requires a value");
559 }

571 std::vector<std::string> multiple(std::string_view name) {
572 std::vector<std::string> result;

575 entry.consumed = true;

578 }

648std::vector<int> parse_devices(const std::string& text) {
649 std::vector<int> result;

651 while (start <= text.size()) {
652 const auto comma = text.find(',', start);

654 if (part.empty()) throw UsageError("--device contains an empty device id");

656 if (comma == std::string::npos) break;
657 start = comma + 1;
658 }
659 if (result.empty()) throw UsageError("--device requires at least one id");
660 return result;

724std::vector<graphmine::TriangleUpdate> read_updates(const std::string& path) {
725 const auto value = read_json(path);
726 const json::array* source = nullptr;
727 if (value.is_array()) {
728 source = &value.as_array();
729 } else {
730 const auto& object = require_object(value, path);
731 source = &require_array(require_field(object, "updates", path), "updates");
732 }
733 std::vector<graphmine::TriangleUpdate> result;
734 result.reserve(source->size());
735 for (std::size_t i = 0; i < source->size(); ++i) {
736 const auto context = "updates[" + std::to_string(i) + "]";
737 const auto& update = require_object((*source)[i], context);
738 reject_unknown_fields(update, {"kind", "source", "target"}, context);
739 const auto kind = normalized_name(require_string(
740 require_field(update, "kind", context), context + ".kind"));
741 graphmine::TriangleUpdate record;
742 if (kind == "insert-edge" || kind == "insert") {
743 record.kind = graphmine::TriangleUpdateKind::insert_edge;
744 } else if (kind == "delete-edge" || kind == "delete") {
745 record.kind = graphmine::TriangleUpdateKind::delete_edge;
746 } else {
747 throw UsageError(context +
748 ".kind must be insert_edge or delete_edge");
749 }
750 record.source = parse_external_id(
751 require_field(update, "source", context), context + ".source");
752 record.target = parse_external_id(
753 require_field(update, "target", context), context + ".target");
754 result.push_back(std::move(record));
755 }
756 return result;
757}
