23bool adjacent(const CsrGraph& graph, VertexIndex lhs, VertexIndex rhs) {
24 const auto begin = graph.neighbors.begin() +
25 static_cast<std::ptrdiff_t>(graph.offsets[lhs]);
26 const auto end = graph.neighbors.begin() +
27 static_cast<std::ptrdiff_t>(graph.offsets[lhs + 1]);
28 return std::binary_search(begin, end, rhs);
29}

31bool connected(const CsrGraph& graph) {
32 if (graph.vertex_count() == 0) {
33 return false;
34 }
35 std::vector<bool> visited(graph.vertex_count(), false);
36 std::queue<VertexIndex> pending;
37 pending.push(0);
38 visited[0] = true;
39 std::size_t reached = 1;
40 while (!pending.empty()) {
41 const auto vertex = pending.front();
42 pending.pop();
43 for (auto edge = graph.offsets[vertex]; edge < graph.offsets[vertex + 1];
44 ++edge) {
45 const auto neighbor = graph.neighbors[edge];
46 if (!visited[neighbor]) {
47 visited[neighbor] = true;
48 ++reached;
49 pending.push(neighbor);
50 }
51 }
52 }
53 return reached == graph.vertex_count();
54}

56int graphminer_pattern_index(const CsrGraph& motif) {
57 const auto size = motif.vertex_count();
58 const auto edges = motif.undirected_edge_count();
59 std::vector<std::uint64_t> degrees(size);
60 for (VertexIndex vertex = 0; vertex < size; ++vertex) {
61 degrees[vertex] = motif.offsets[vertex + 1] - motif.offsets[vertex];
62 }
63 std::sort(degrees.begin(), degrees.end());
64 if (size == 3) {
65 if (edges == 3) return 0; // triangle
66 if (edges == 2) return 1; // wedge
67 } else if (size == 4) {
68 if (edges == 3 && degrees == std::vector<std::uint64_t>{1, 1, 1, 3})
69 return 0; // 3-star
70 if (edges == 3 && degrees == std::vector<std::uint64_t>{1, 1, 2, 2})
71 return 1; // 4-path
72 if (edges == 4 && degrees == std::vector<std::uint64_t>{1, 2, 2, 3})
73 return 2; // tailed triangle
74 if (edges == 4 && degrees == std::vector<std::uint64_t>{2, 2, 2, 2})
75 return 3; // 4-cycle
76 if (edges == 5) return 4; // diamond
77 if (edges == 6) return 5; // 4-clique
78 }
79 return -1;
80}

82std::uint64_t automorphism_count(const CsrGraph& motif) {
83 std::vector<VertexIndex> permutation(motif.vertex_count());
84 std::iota(permutation.begin(), permutation.end(), VertexIndex{0});
85 std::uint64_t count = 0;
86 do {
87 bool valid = true;
88 for (VertexIndex lhs = 0; lhs < motif.vertex_count() && valid; ++lhs) {
89 for (VertexIndex rhs = lhs + 1; rhs < motif.vertex_count(); ++rhs) {
90 if (adjacent(motif, lhs, rhs) !=
91 adjacent(motif, permutation[lhs], permutation[rhs])) {
92 valid = false;
93 break;
94 }
95 }
96 }
97 if (valid) ++count;
98 } while (std::next_permutation(permutation.begin(), permutation.end()));
99 return count;
100}

119OccurrenceKey occurrence_key(const MatchState& state) {
120 OccurrenceKey key;
121 for (VertexIndex source = 0; source < state.motif->vertex_count(); ++source) {
122 for (auto edge = state.motif->offsets[source];
123 edge < state.motif->offsets[source + 1]; ++edge) {
124 const auto target = state.motif->neighbors[edge];
125 if (source >= target) continue;
126 auto mapped_source = state.mapping[source];
127 auto mapped_target = state.mapping[target];
128 if (mapped_target < mapped_source) {
129 std::swap(mapped_source, mapped_target);
130 }
131 key.emplace_back(mapped_source, mapped_target);
132 }
133 }
134 std::sort(key.begin(), key.end());
135 return key;
136}

161void enumerate_matches(std::size_t depth, MatchState& state) {
162 if (depth == state.order.size()) {
163 record_match(state);
164 return;
165 }
166 const auto query_vertex = state.order[depth];
167 const auto query_degree = state.motif->offsets[query_vertex + 1] -
168 state.motif->offsets[query_vertex];
169 for (VertexIndex data_vertex = 0;
170 data_vertex < state.data->vertex_count(); ++data_vertex) {
171 if (state.used[data_vertex] ||
172 state.data->offsets[data_vertex + 1] -
173 state.data->offsets[data_vertex] <
174 query_degree) {
175 continue;
176 }
177 bool compatible = true;
178 for (VertexIndex other = 0; other < state.motif->vertex_count(); ++other) {
179 if (state.mapping[other] == std::numeric_limits<VertexIndex>::max()) {
180 continue;
181 }
182 const bool query_edge = adjacent(*state.motif, query_vertex, other);
183 const bool data_edge =
184 adjacent(*state.data, data_vertex, state.mapping[other]);
185 if ((query_edge && !data_edge) ||
186 (state.options->induced && query_edge != data_edge)) {
187 compatible = false;
188 break;
189 }
190 }
191 if (!compatible) continue;
192 state.mapping[query_vertex] = data_vertex;
193 state.used[data_vertex] = true;
194 enumerate_matches(depth + 1, state);
195 state.used[data_vertex] = false;
196 state.mapping[query_vertex] = std::numeric_limits<VertexIndex>::max();
197 }

200MaterializedMotif materialize(const CsrGraph& data, const CsrGraph& motif,
201 const GraphMotifOptions& options) {
202 MatchState state;
203 state.data = &data;
204 state.motif = &motif;
205 state.options = &options;
206 state.order.resize(motif.vertex_count());
207 std::iota(state.order.begin(), state.order.end(), VertexIndex{0});
208 std::stable_sort(state.order.begin(), state.order.end(),
209 [&](VertexIndex lhs, VertexIndex rhs) {
210 const auto lhs_degree =
211 motif.offsets[lhs + 1] - motif.offsets[lhs];
212 const auto rhs_degree =
213 motif.offsets[rhs + 1] - motif.offsets[rhs];
214 return lhs_degree == rhs_degree ? lhs < rhs
215 : lhs_degree > rhs_degree;
216 });
217 state.mapping.assign(motif.vertex_count(),
218 std::numeric_limits<VertexIndex>::max());
219 state.used.assign(data.vertex_count(), false);
220 if (has_output(options.optional_outputs,
221 GraphMotifOptionalOutput::per_vertex_participation)) {
222 state.output.participation.assign(data.vertex_count(), 0);
223 }
224 enumerate_matches(0, state);
225 return std::move(state.output);
226}

307SupportReport GraphMotifs::supports(
308 const Graph& data_graph, const std::vector<Graph>& motifs) const {
309 if (motifs.empty()) return {false, "at least one motif is required"};
310 if (options_.execution.device_ids.empty()) {
311 return {false, "at least one CUDA device id must be provided"};
312 }
313 if (options_.result_limit_per_motif &&
314 *options_.result_limit_per_motif == 0) {
315 return {false, "result_limit_per_motif must be null or at least one"};
316 }
317 if (options_.result_limit_per_motif &&
318 !has_output(options_.optional_outputs,
319 GraphMotifOptionalOutput::instances)) {
320 return {false, "result_limit_per_motif requires the instances flag"};
321 }
322 if (data_graph.directed() && !options_.allow_directed_projection) {
323 return {false,
324 "the validated motif backends require undirected data; set "
325 "allow_directed_projection=true to project explicitly"};
326 }
327
328 const auto selected = resolve_backend(options_);
329 if ((selected == GraphMotifBackend::graphminer ||
330 selected == GraphMotifBackend::dumato) &&
331 !options_.induced) {
332 return {false, std::string(to_string(selected)) +
333 " was validated for induced motifs only"};
334 }
335 if (selected == GraphMotifBackend::graphset && options_.induced) {
336 return {false, "GraphSet was validated for non-induced motifs only"};
337 }
338 if (!backend_compiled(selected)) {
339 return {false, std::string(to_string(selected)) +
340 " motif backend is not compiled in this build"};
341 }
342
343 std::set<std::string> ids;
344 try {
345 for (const auto& motif_graph : motifs) {
346 if (motif_graph.directed() && !options_.allow_directed_projection) {
347 return {false,
348 "the validated motif backends require undirected motifs"};
349 }
350 if (motif_graph.canonical().graph.id.empty() ||
351 !ids.insert(motif_graph.canonical().graph.id).second) {
352 return {false, "motif graph IDs must be nonempty and unique"};
353 }
354 const auto motif = motif_graph.simple_undirected(
355 {options_.allow_directed_projection});
356 if ((motif.csr.vertex_count() != 3 &&
357 motif.csr.vertex_count() != 4) ||
358 !connected(motif.csr) || graphminer_pattern_index(motif.csr) < 0) {
359 return {false,
360 "validated motif backends support connected undirected "
361 "3- and 4-vertex simple motifs"};
362 }
363 }
364 } catch (const std::exception& error) {
365 return {false, error.what()};
366 }
367 return {true, {}};
368}

370ExecutionResult<GraphMotifOutput> GraphMotifs::run(
371 const Graph& data_graph, const std::vector<Graph>& motifs) const {
372 const auto started = std::chrono::steady_clock::now();
373 const auto selected = resolve_backend(options_);
374 auto provenance = motif_provenance(selected);
375 const auto support = supports(data_graph, motifs);
376 if (!support.supported) {
377 return ExecutionResult<GraphMotifOutput>::failure(
378 {StatusCode::unsupported, support.reason}, provenance);
379 }
380
381 UndirectedGraphView data;
382 std::vector<UndirectedGraphView> normalized_motifs;
383 try {
384 data = data_graph.simple_undirected(
385 {options_.allow_directed_projection});
386 normalized_motifs.reserve(motifs.size());
387 for (const auto& motif : motifs) {
388 normalized_motifs.push_back(motif.simple_undirected(
389 {options_.allow_directed_projection}));
390 }
391 } catch (const std::exception& error) {
392 return ExecutionResult<GraphMotifOutput>::failure(
393 {StatusCode::invalid_argument, error.what()}, provenance);
394 }
395
396 std::vector<std::string> warnings;
397 if (data.normalization.directed_projection_applied) {
398 warnings.emplace_back("directed data was projected to an undirected graph");
399 }
400 if (data.normalization.self_loops_removed > 0) {
401 warnings.emplace_back("data self-loops were removed during normalization");
402 }
403 if (data.normalization.parallel_edges_collapsed > 0) {
404 warnings.emplace_back(
405 "parallel data edges were collapsed during normalization");
406 }
407
408 std::map<std::uint32_t, detail::GraphMotifBackendResult> backend_results;
409 double backend_ms = 0.0;
410 for (const auto& motif : normalized_motifs) {
411 const auto size = static_cast<std::uint32_t>(motif.csr.vertex_count());
412 if (backend_results.count(size) != 0) continue;
413 detail::GraphMotifBackendResult backend;
414 switch (selected) {
415 case GraphMotifBackend::graphminer:
416 backend = detail::run_graphminer_motifs(
417 data.csr, size, options_.execution.device_ids.front());
418 break;
419 case GraphMotifBackend::graphset:
420 backend = detail::run_graphset_motifs(
421 data.csr, size, options_.execution.device_ids.front());
422 break;
423 case GraphMotifBackend::dumato:
424 backend = detail::run_dumato_motifs(
425 data.csr, size, options_.execution.device_ids.front());
426 break;
427 case GraphMotifBackend::automatic:
428 return ExecutionResult<GraphMotifOutput>::failure(
429 {StatusCode::internal_error,
430 "automatic motif backend was not resolved"},
431 provenance, std::move(warnings));
432 }
433 if (!backend.status.ok()) {
434 return ExecutionResult<GraphMotifOutput>::failure(
435 backend.status, provenance, std::move(warnings));
436 }
437 backend_ms += backend.elapsed_ms;
438 backend_results.emplace(size, std::move(backend));
439 }
440
441 GraphMotifOutput output;
442 output.motifs.reserve(motifs.size());
443 const auto materialization_started = std::chrono::steady_clock::now();
444 for (std::size_t index = 0; index < motifs.size(); ++index) {
445 const auto& motif = normalized_motifs[index].csr;
446 const auto pattern_index = graphminer_pattern_index(motif);
447 const auto& counts =
448 backend_results.at(static_cast<std::uint32_t>(motif.vertex_count()))
449 .counts;
450 if (pattern_index < 0 ||
451 static_cast<std::size_t>(pattern_index) >= counts.size()) {
452 return ExecutionResult<GraphMotifOutput>::failure(
453 {StatusCode::internal_error,
454 "motif classifier did not match the backend count vector"},
455 provenance, std::move(warnings));
456 }
457
458 GraphMotifResult result;
459 result.motif_id = motifs[index].canonical().graph.id;
460 result.count = counts[pattern_index];
461 if (options_.occurrence_identity ==
462 MotifOccurrenceIdentity::all_embeddings) {
463 result.count *= automorphism_count(motif);
464 }
465
466 const bool needs_materialization =
467 options_.optional_outputs != GraphMotifOptionalOutput::none;
468 if (needs_materialization) {
469 auto exact = materialize(data.csr, motif, options_);
470 if (exact.count != result.count) {
471 return ExecutionResult<GraphMotifOutput>::failure(
472 {StatusCode::correctness_mismatch,
473 std::string(to_string(selected)) + " count for motif " +
474 result.motif_id +
475 " disagreed with the exact optional-output materializer"},
476 provenance, std::move(warnings));
477 }
478 if (has_output(options_.optional_outputs,
479 GraphMotifOptionalOutput::instances)) {
480 result.instances.emplace();
481 result.instances->reserve(exact.mappings.size());
482 for (const auto& mapping : exact.mappings) {
483 MotifInstance instance;
484 instance.reserve(mapping.size());
485 for (VertexIndex query_vertex = 0;
486 query_vertex < mapping.size(); ++query_vertex) {
487 instance.push_back(
488 {motifs[index].external_id(query_vertex),
489 data_graph.external_id(mapping[query_vertex])});
490 }
491 result.instances->push_back(std::move(instance));
492 }
493 result.instances_complete = result.instances->size() == result.count;
494 }
495 if (has_output(options_.optional_outputs,
496 GraphMotifOptionalOutput::per_vertex_participation)) {
497 result.per_vertex_participation.emplace();
498 result.per_vertex_participation->reserve(data_graph.vertex_count());
499 for (VertexIndex vertex = 0; vertex < data_graph.vertex_count();
500 ++vertex) {
501 result.per_vertex_participation->push_back(
502 {data_graph.external_id(vertex), exact.participation[vertex]});
503 }
504 }
505 }
506 output.motifs.push_back(std::move(result));
507 }
508 const auto materialization_finished = std::chrono::steady_clock::now();
509 const auto finished = std::chrono::steady_clock::now();
510
511 ExecutionStatistics statistics;
512 statistics.backend_ms = backend_ms;
513 statistics.output_materialization_ms =
514 std::chrono::duration<double, std::milli>(materialization_finished -
515 materialization_started)
516 .count();
517 statistics.end_to_end_ms =
518 std::chrono::duration<double, std::milli>(finished - started).count();
519 return ExecutionResult<GraphMotifOutput>::success(
520 std::move(output), std::move(provenance), statistics,
521 std::move(warnings));
522}
