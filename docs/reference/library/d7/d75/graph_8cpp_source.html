37Graph::Graph(CanonicalGraphInput input) : input_(std::move(input)) {
38 if (input_.graph.id.empty()) {
39 throw std::invalid_argument("canonical graph id must not be empty");
40 }
41 if (input_.vertices.size() >
42 static_cast<std::size_t>(std::numeric_limits<VertexIndex>::max())) {
43 throw std::invalid_argument("graph has more vertices than the dense ID type supports");
44 }
45
46 dense_ids_.reserve(input_.vertices.size());
47 for (std::size_t i = 0; i < input_.vertices.size(); ++i) {
48 const auto inserted = dense_ids_.emplace(
49 input_.vertices[i].id, static_cast<VertexIndex>(i));
50 if (!inserted.second) {
51 throw std::invalid_argument("duplicate vertex id: " +
52 input_.vertices[i].id.to_string());
53 }
54 }
55
56 std::unordered_set<ExternalId, ExternalIdHash> edge_ids;
57 edge_ids.reserve(input_.edges.size());
58 std::unordered_set<std::uint64_t> logical_edges;
59 logical_edges.reserve(input_.edges.size());
60
61 for (const auto& edge : input_.edges) {
62 if (!edge_ids.emplace(edge.id).second) {
63 throw std::invalid_argument("duplicate edge id: " + edge.id.to_string());
64 }
65 const auto source_it = dense_ids_.find(edge.source);
66 const auto target_it = dense_ids_.find(edge.target);
67 if (source_it == dense_ids_.end() || target_it == dense_ids_.end()) {
68 throw std::invalid_argument("edge " + edge.id.to_string() +
69 " references an unknown vertex");
70 }
71
72 auto source = source_it->second;
73 auto target = target_it->second;
74 if (source == target && !input_.graph.allows_self_loops) {
75 throw std::invalid_argument("self-loop is present but graph disallows self-loops");
76 }
77 if (!input_.graph.directed && target < source) {
78 std::swap(source, target);
79 }
80 if (!input_.graph.allows_parallel_edges &&
81 !logical_edges.emplace(edge_key(source, target)).second) {
82 throw std::invalid_argument(
83 "parallel edge is present but graph disallows parallel edges");
84 }
85 }
86}

88Graph Graph::from_edges(
89 std::string graph_id, std::vector<ExternalId> vertices,
90 const std::vector<std::pair<ExternalId, ExternalId>>& edges,
91 bool directed) {
92 CanonicalGraphInput input;
93 input.graph.id = std::move(graph_id);
94 input.graph.directed = directed;
95 input.graph.allows_self_loops = false;
96 input.graph.allows_parallel_edges = false;
97 input.vertices.reserve(vertices.size());
98 for (auto& id : vertices) {
99 input.vertices.push_back(VertexRecord{std::move(id), std::nullopt,
100 std::nullopt, {}});
101 }
102 input.edges.reserve(edges.size());
103 for (std::size_t i = 0; i < edges.size(); ++i) {
104 input.edges.push_back(EdgeRecord{ExternalId{static_cast<std::int64_t>(i)},
105 edges[i].first, edges[i].second,
106 std::nullopt, std::nullopt, std::nullopt,
107 {}});
108 }
109 return Graph(std::move(input));
110}

127UndirectedGraphView Graph::simple_undirected(
128 UndirectedProjectionOptions options) const {
129 if (input_.graph.directed && !options.allow_directed_projection) {
130 throw std::invalid_argument(
131 "undirected algorithm received directed input without an explicit projection");
132 }
133
134 UndirectedGraphView view;
135 view.normalization.directed_projection_applied = input_.graph.directed;
136
137 std::set<std::pair<VertexIndex, VertexIndex>> unique_edges;
138 for (const auto& edge : input_.edges) {
139 auto source = dense_index(edge.source);
140 auto target = dense_index(edge.target);
141 if (source == target) {
142 ++view.normalization.self_loops_removed;
143 continue;
144 }
145 if (target < source) {
146 std::swap(source, target);
147 }
148 if (!unique_edges.emplace(source, target).second) {
149 ++view.normalization.parallel_edges_collapsed;
150 }
151 }
152
153 std::vector<std::vector<VertexIndex>> adjacency(vertex_count());
154 for (const auto& [source, target] : unique_edges) {
155 adjacency[source].push_back(target);
156 adjacency[target].push_back(source);
157 }
158
159 view.csr.offsets.resize(vertex_count() + 1, 0);
160 for (std::size_t vertex = 0; vertex < adjacency.size(); ++vertex) {
161 std::sort(adjacency[vertex].begin(), adjacency[vertex].end());
162 view.csr.offsets[vertex + 1] =
163 view.csr.offsets[vertex] + adjacency[vertex].size();
164 view.csr.neighbors.insert(view.csr.neighbors.end(),
165 adjacency[vertex].begin(),
166 adjacency[vertex].end());
167 }
168 return view;
169}
