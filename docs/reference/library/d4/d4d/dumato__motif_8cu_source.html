38std::vector<std::uint64_t> common_order(
39 std::uint32_t motif_size,
40 const std::vector<std::uint64_t>& canonical_graph_counts) {
41 if (motif_size == 3 && canonical_graph_counts.size() >= 3) {
42 // DuMato canonical graph ids: 1=wedge, 2=triangle.
43 return {canonical_graph_counts[2], canonical_graph_counts[1]};
44 }
45 if (motif_size == 4 && canonical_graph_counts.size() >= 10) {
46 // DuMato canonical graph ids: 3=path, 4=cycle, 6=star,
47 // 7=tailed triangle, 8=diamond, 9=clique.
48 return {canonical_graph_counts[6], canonical_graph_counts[3],
49 canonical_graph_counts[7], canonical_graph_counts[4],
50 canonical_graph_counts[8], canonical_graph_counts[9]};
51 }
52 return {};
53}

60 std::uint32_t motif_size,
61 int device_id) {
62 const auto result_size = motif_size == 3 ? 2U : 6U;
63 if (motif_size < 3 || motif_size > 4) {
64 return {{StatusCode::unsupported,
65 "the validated DuMato motif path supports sizes 3 and 4"},
66 {}, 0.0};
67 }
68 if (graph.vertex_count() < motif_size || graph.neighbors.empty()) {
69 return {Status::success(), std::vector<std::uint64_t>(result_size, 0),
70 0.0};
71 }
72 if (graph.vertex_count() > static_cast<std::size_t>(INT_MAX) ||
73 graph.neighbors.size() > static_cast<std::size_t>(INT_MAX)) {
74 return {{StatusCode::unsupported,
75 "DuMato uses signed 32-bit graph indices"},
76 {}, 0.0};
77 }

82 device_id >= device_count || cudaSetDevice(device_id) != cudaSuccess ||
83 cudaGetDeviceProperties(&properties, device_id) != cudaSuccess) {
84 return {{StatusCode::backend_unavailable,
85 "requested CUDA device is unavailable for DuMato"},
86 {}, 0.0};
87 }
88

92 std::vector<std::int32_t> neighbours(graph.neighbors.begin(),
93 graph.neighbors.end());

96 const auto started = std::chrono::steady_clock::now();
97 auto* original =

99 GraphMineDumatoMotifCpu engine(
100 original, static_cast<int>(motif_size), 102400, 256,
101 properties.multiProcessorCount, 16, graphmine_dumato_motif_kernel,
102 100, true, false);
103 engine.runKernel();
104 engine.waitKernel();
105
106 const auto number_of_canonical_graphs =
107 static_cast<std::size_t>(engine.quickMapping->numberOfCgs);
108 const auto number_of_warps =
109 static_cast<std::size_t>(engine.dataCPU->h_numberOfWarps);
110 const auto copied = cudaMemcpy(

112 number_of_warps * number_of_canonical_graphs *

114 cudaMemcpyDeviceToHost);
115 if (copied != cudaSuccess) {

119 {}, 0.0};
120 }

123 number_of_canonical_graphs, 0);

125 for (std::size_t canonical = 0;
126 canonical < number_of_canonical_graphs; ++canonical) {
127 canonical_counts[canonical] +=
128 engine.dataCPU
129 ->h_hashPerWarp[warp * number_of_canonical_graphs + canonical];
130 }
131 }
132 auto counts = common_order(motif_size, canonical_counts);
133 const auto finished = std::chrono::steady_clock::now();
134 if (counts.size() != result_size) {
135 return {{StatusCode::internal_error,
136 "DuMato returned an unexpected canonical-graph vector"},
137 {}, 0.0};
138 }
139 return {Status::success(), std::move(counts),
140 std::chrono::duration<double, std::milli>(finished - started)
141 .count()};
142 } catch (const std::bad_alloc&) {
143 return {{StatusCode::resource_exhausted,
144 "DuMato motif host allocation failed"},
145 {}, 0.0};
146 } catch (const std::exception& error) {
147 return {{StatusCode::execution_failed,
148 std::string("DuMato motif execution failed: ") + error.what()},
149 {}, 0.0};
150 }
