58 offsets.push_back(static_cast<int>(offset));
59 }

61 neighbors.reserve(graph.neighbors.size());
62 for (const auto neighbor : graph.neighbors) {
63 neighbors.push_back(static_cast<int>(neighbor));
64 }

66 const auto vertex_count = static_cast<int>(graph.vertex_count());
67 std::vector<int> levels(graph.vertex_count(), 0);

69 std::vector<float> raw_scores(graph.vertex_count(), 0.0F);
