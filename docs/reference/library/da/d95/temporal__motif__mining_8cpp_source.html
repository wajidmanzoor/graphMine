75Status validate(const Graph& graph, const TemporalMotifOptions& options,
76 TemporalMotifBackend backend) {
77 if (!graph.directed()) {
78 return {StatusCode::unsupported,
79 "temporal motif mining requires a directed canonical graph"};
80 }
81 if (options.pattern != TemporalMotifPattern::feed_forward_triangle) {
82 return {StatusCode::unsupported,
83 "the validated kernels only support feed_forward_triangle"};
84 }
85 if (options.max_time_span < 0 ||
86 options.max_time_span > std::numeric_limits<int>::max()) {
87 return {StatusCode::invalid_argument,
88 "max_time_span must fit a non-negative signed 32-bit integer"};
89 }
90 if (options.execution.device_ids.empty()) {
91 return {StatusCode::invalid_argument,
92 "at least one CUDA device id must be provided"};
93 }
94 if (graph.vertex_count() >
95 static_cast<std::size_t>(std::numeric_limits<int>::max()) ||
96 graph.edge_count() >
97 static_cast<std::size_t>(std::numeric_limits<int>::max())) {
98 return {StatusCode::unsupported,
99 "Everest and Mayura use signed 32-bit graph indices"};
100 }
101 for (const auto& edge : graph.canonical().edges) {
102 if (!edge.timestamp.has_value()) {
103 return {StatusCode::invalid_argument,
104 "every temporal edge must have a timestamp"};
105 }
106 if (!std::holds_alternative<std::int64_t>(*edge.timestamp)) {
107 return {StatusCode::unsupported,
108 "Everest and Mayura require integer edge timestamps"};
109 }
110 }
111 if (!backend_compiled(backend) && graph.edge_count() != 0) {
112 return {StatusCode::backend_unavailable,
113 std::string("the ") + to_string(backend) +
114 " backend is not compiled in this build"};
115 }
116 return Status::success();
117}

119Status prepare(const Graph& graph, const TemporalMotifOptions& options,
120 PreparedTemporalGraph& prepared) {
121 prepared.events.reserve(graph.edge_count());
122 const auto& canonical = graph.canonical();
123 for (std::size_t index = 0; index < canonical.edges.size(); ++index) {
124 const auto& edge = canonical.edges[index];
125 prepared.events.push_back(
126 {graph.dense_index(edge.source), graph.dense_index(edge.target),
127 std::get<std::int64_t>(*edge.timestamp), index});
128 }
129 std::stable_sort(prepared.events.begin(), prepared.events.end(),
130 [](const auto& lhs, const auto& rhs) {
131 return lhs.timestamp < rhs.timestamp;
132 });
133
134 prepared.backend.vertex_count = static_cast<int>(graph.vertex_count());
135 prepared.backend.max_time_span =
136 static_cast<int>(options.max_time_span);
137 prepared.backend.events.reserve(prepared.events.size());
138 if (prepared.events.empty()) return Status::success();
139
140 const auto first_timestamp = prepared.events.front().timestamp;
141 for (const auto& event : prepared.events) {
142 const auto relative = static_cast<__int128>(event.timestamp) -
143 static_cast<__int128>(first_timestamp);
144 if (relative < 0 ||
145 relative + static_cast<__int128>(options.max_time_span) >
146 std::numeric_limits<int>::max()) {
147 return {StatusCode::unsupported,
148 "timestamp span plus max_time_span exceeds the kernels' "
149 "signed 32-bit time representation"};
150 }
151 prepared.backend.events.push_back(
152 {static_cast<int>(event.source), static_cast<int>(event.target),
153 static_cast<int>(relative)});
154 }
155 return Status::success();
156}

163MaterializedInstances materialize_instances(
164 const Graph& graph, const PreparedTemporalGraph& prepared,
165 std::optional<std::uint64_t> limit) {
166 MaterializedInstances result;
167 std::vector<std::vector<std::size_t>> outgoing(graph.vertex_count());
168 for (std::size_t index = 0; index < prepared.events.size(); ++index) {
169 outgoing[prepared.events[index].source].push_back(index);
170 }
171
172 const auto should_store = [&]() {
173 return !limit || result.values.size() < *limit;
174 };
175 const auto& canonical_edges = graph.canonical().edges;
176 for (std::size_t first = 0; first < prepared.events.size(); ++first) {
177 const auto& first_event = prepared.events[first];
178 if (first_event.source == first_event.target) continue;
179 const auto deadline = static_cast<__int128>(first_event.timestamp) +
180 static_cast<__int128>(
181 prepared.backend.max_time_span);
182
183 for (const auto second : outgoing[first_event.target]) {
184 if (second <= first) continue;
185 const auto& second_event = prepared.events[second];
186 if (static_cast<__int128>(second_event.timestamp) > deadline) break;
187 const auto third_vertex = second_event.target;
188 if (third_vertex == first_event.source ||
189 third_vertex == first_event.target) {
190 continue;
191 }
192
193 for (const auto third : outgoing[first_event.source]) {
194 if (third <= second) continue;
195 const auto& third_event = prepared.events[third];
196 if (static_cast<__int128>(third_event.timestamp) > deadline) break;
197 if (third_event.target != third_vertex) continue;
198
199 if (result.count == std::numeric_limits<std::uint64_t>::max()) {
200 throw std::overflow_error("temporal motif instance count overflow");
201 }
202 ++result.count;
203 if (should_store()) {
204 result.values.push_back(
205 {{{graph.external_id(first_event.source),
206 graph.external_id(first_event.target),
207 graph.external_id(third_vertex)}},
208 {{canonical_edges[first_event.original_index].id,
209 canonical_edges[second_event.original_index].id,
210 canonical_edges[third_event.original_index].id}}});
211 }
212 }
213 }
214 }
215 return result;
216}

248ExecutionResult<TemporalMotifOutput> TemporalMotifMining::run(
249 const Graph& graph) const {
250 const auto started = std::chrono::steady_clock::now();
251 const auto backend = selected_backend(options_);
252 auto provenance = provenance_for(backend);
253 const auto validation = validate(graph, options_, backend);
254 if (!validation.ok()) {
255 return ExecutionResult<TemporalMotifOutput>::failure(validation,
256 provenance);
257 }
258
259 PreparedTemporalGraph prepared;
260 const auto prepared_status = prepare(graph, options_, prepared);
261 if (!prepared_status.ok()) {
262 return ExecutionResult<TemporalMotifOutput>::failure(prepared_status,
263 provenance);
264 }
265
266 detail::TemporalMotifBackendResult backend_result;
267 if (prepared.backend.events.empty()) {
268 backend_result.status = Status::success();
269 provenance.execution_path += " (trivial empty-event result)";
270 } else if (backend == TemporalMotifBackend::everest) {
271 backend_result = detail::run_everest(
272 prepared.backend, options_.execution.device_ids.front());
273 } else {
274 backend_result = detail::run_mayura(
275 prepared.backend, options_.execution.device_ids.front());
276 }
277 if (!backend_result.status.ok()) {
278 return ExecutionResult<TemporalMotifOutput>::failure(
279 backend_result.status, provenance);
280 }
281
282 const auto materialization_started = std::chrono::steady_clock::now();
283 TemporalMotifOutput output;
284 output.count = backend_result.count;
285 if (has_output(options_.optional_outputs,
286 TemporalMotifOptionalOutput::instances)) {
287 try {
288 auto instances =
289 materialize_instances(graph, prepared, options_.result_limit);
290 if (instances.count != output.count) {
291 return ExecutionResult<TemporalMotifOutput>::failure(
292 {StatusCode::correctness_mismatch,
293 std::string(to_string(backend)) + " returned " +
294 std::to_string(output.count) +
295 " matches, but facade materialization found " +
296 std::to_string(instances.count)},
297 provenance);
298 }
299 output.instances_complete = instances.values.size() == output.count;
300 output.instances = std::move(instances.values);
301 } catch (const std::exception& error) {
302 return ExecutionResult<TemporalMotifOutput>::failure(
303 {StatusCode::execution_failed,
304 std::string("temporal instance materialization failed: ") +
305 error.what()},
306 provenance);
307 }
308 }
309 const auto materialization_finished = std::chrono::steady_clock::now();
310 const auto finished = std::chrono::steady_clock::now();
311
312 ExecutionStatistics statistics;
313 statistics.backend_ms = backend_result.elapsed_ms;
314 statistics.output_materialization_ms =
315 std::chrono::duration<double, std::milli>(materialization_finished -
316 materialization_started)
317 .count();
318 statistics.end_to_end_ms =
319 std::chrono::duration<double, std::milli>(finished - started).count();
320 return ExecutionResult<TemporalMotifOutput>::success(
321 std::move(output), std::move(provenance), statistics);
322}
