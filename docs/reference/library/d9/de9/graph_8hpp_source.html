38
39 /// Compares external IDs for inequality.
40 friend bool operator!=(const ExternalId& lhs, const ExternalId& rhs) {
41 return !(lhs == rhs);
42 }
43
44 /// Orders numeric IDs before string IDs and then orders by stored value.
45 friend bool operator<(const ExternalId& lhs, const ExternalId& rhs) {
46 if (lhs.value_.index() != rhs.value_.index()) {
47 return lhs.value_.index() < rhs.value_.index();
48 }
49 if (std::holds_alternative<std::int64_t>(lhs.value_)) {
50 return std::get<std::int64_t>(lhs.value_) <
51 std::get<std::int64_t>(rhs.value_);
52 }
53 return std::get<std::string>(lhs.value_) <
54 std::get<std::string>(rhs.value_);
55 }

57 private:
