51bool adjacent(const CsrGraph& graph, VertexIndex lhs, VertexIndex rhs) {
52 const auto begin = graph.neighbors.begin() +
53 static_cast<std::ptrdiff_t>(graph.offsets[lhs]);
54 const auto end = graph.neighbors.begin() +
55 static_cast<std::ptrdiff_t>(graph.offsets[lhs + 1]);
56 return std::binary_search(begin, end, rhs);
57}

59bool connected(const CsrGraph& graph) {
60 if (graph.vertex_count() <= 1) {
61 return true;
62 }
63 std::vector<bool> visited(graph.vertex_count(), false);
64 std::queue<VertexIndex> queue;
65 queue.push(0);
66 visited[0] = true;
67 std::size_t count = 1;
68 while (!queue.empty()) {
69 const auto vertex = queue.front();
70 queue.pop();
71 for (auto edge = graph.offsets[vertex]; edge < graph.offsets[vertex + 1];
72 ++edge) {
73 const auto neighbor = graph.neighbors[edge];
74 if (!visited[neighbor]) {
75 visited[neighbor] = true;
76 ++count;
77 queue.push(neighbor);
78 }
79 }
80 }
81 return count == graph.vertex_count();
82}

97void collect_embeddings(std::size_t depth, EmbeddingCollector& state) {
98 if (state.stopped) {
99 return;
100 }
101 if (depth == state.order.size()) {
102 state.embeddings.push_back(state.mapping);
103 if (state.limit && state.embeddings.size() >= *state.limit) {
104 state.stopped = true;
105 }
106 return;
107 }
108
109 const auto query_vertex = state.order[depth];
110 const auto query_degree = state.query->offsets[query_vertex + 1] -
111 state.query->offsets[query_vertex];
112 for (VertexIndex data_vertex = 0;
113 data_vertex < state.data->vertex_count(); ++data_vertex) {
114 if (state.used[data_vertex] ||
115 (*state.data_labels)[data_vertex] !=
116 (*state.query_labels)[query_vertex] ||
117 state.data->offsets[data_vertex + 1] -
118 state.data->offsets[data_vertex] <
119 query_degree) {
120 continue;
121 }
122 bool compatible = true;
123 for (VertexIndex other = 0; other < state.query->vertex_count(); ++other) {
124 if (state.mapping[other] != std::numeric_limits<VertexIndex>::max() &&
125 adjacent(*state.query, query_vertex, other) &&
126 !adjacent(*state.data, data_vertex, state.mapping[other])) {
127 compatible = false;
128 break;
129 }
130 }
131 if (!compatible) {
132 continue;
133 }
134 state.mapping[query_vertex] = data_vertex;
135 state.used[data_vertex] = true;
136 collect_embeddings(depth + 1, state);
137 state.used[data_vertex] = false;
138 state.mapping[query_vertex] = std::numeric_limits<VertexIndex>::max();
139 if (state.stopped) {
140 return;
141 }
142 }
143}

145EmbeddingCollector materialize_embeddings(
146 const CsrGraph& data, const CsrGraph& query,
147 const std::vector<int>& data_labels,
148 const std::vector<int>& query_labels,
149 std::optional<std::uint64_t> limit) {
150 EmbeddingCollector state;
151 state.data = &data;
152 state.query = &query;
153 state.data_labels = &data_labels;
154 state.query_labels = &query_labels;
155 state.limit = limit;
156 state.order.resize(query.vertex_count());
157 for (VertexIndex vertex = 0; vertex < query.vertex_count(); ++vertex) {
158 state.order[vertex] = vertex;
159 }
160 std::stable_sort(state.order.begin(), state.order.end(),
161 [&](VertexIndex lhs, VertexIndex rhs) {
162 const auto lhs_degree =
163 query.offsets[lhs + 1] - query.offsets[lhs];
164 const auto rhs_degree =
165 query.offsets[rhs + 1] - query.offsets[rhs];
166 if (lhs_degree != rhs_degree) {
167 return lhs_degree > rhs_degree;
168 }
169 return lhs < rhs;
170 });
171 state.mapping.assign(query.vertex_count(),
172 std::numeric_limits<VertexIndex>::max());
173 state.used.assign(data.vertex_count(), false);
174 collect_embeddings(0, state);
175 return state;
176}

202SupportReport SubgraphIsomorphism::supports(
203 const Graph& data_graph, const Graph& query_graph) const {
204 if ((data_graph.directed() || query_graph.directed()) &&
205 !options_.allow_directed_projection) {
206 return {false,
207 "directed input requires allow_directed_projection=true"};
208 }
209 if (query_graph.vertex_count() > 16) {
210 return {false, "the validated gMatch bitmap path supports at most 16 query vertices"};
211 }
212 if (data_graph.vertex_count() > SHRT_MAX) {
213 return {false,
214 "the validated gMatch short-candidate path supports at most "
215 "32767 data vertices"};
216 }
217 if (data_graph.edge_count() > static_cast<std::size_t>(INT_MAX) ||
218 query_graph.edge_count() > static_cast<std::size_t>(INT_MAX)) {
219 return {false, "gMatch uses signed 32-bit graph sizes"};
220 }
221 if (options_.initial_match_capacity == 0 ||
222 options_.initial_match_capacity > static_cast<std::size_t>(INT_MAX)) {
223 return {false, "initial_match_capacity must fit a positive int"};
224 }
225 if (options_.result_limit && *options_.result_limit == 0) {
226 return {false, "result_limit must be null or at least one"};
227 }
228 if (options_.result_limit &&
229 !has_output(options_.optional_outputs,
230 SubgraphIsomorphismOptionalOutput::embeddings)) {
231 return {false, "result_limit requires the embeddings output flag"};
232 }
233 if (options_.execution.device_ids.empty()) {
234 return {false, "at least one CUDA device id must be provided"};
235 }
236 try {
237 const auto query = query_graph.simple_undirected(
238 {options_.allow_directed_projection});
239 if (!connected(query.csr)) {
240 return {false, "the validated gMatch path requires a connected query"};
241 }
242 } catch (const std::exception& error) {
243 return {false, error.what()};
244 }
245 if (!detail::gmatch_backend_compiled() &&
246 query_graph.vertex_count() > 1 &&
247 query_graph.vertex_count() <= data_graph.vertex_count()) {
248 return {false, "the gMatch backend is not compiled in this build"};
249 }
250 return {true, {}};
251}

253ExecutionResult<SubgraphIsomorphismOutput> SubgraphIsomorphism::run(
254 const Graph& data_graph, const Graph& query_graph) const {
255 const auto started = std::chrono::steady_clock::now();
256 auto provenance = gmatch_provenance();
257 const auto support = supports(data_graph, query_graph);
258 if (!support.supported) {
259 return ExecutionResult<SubgraphIsomorphismOutput>::failure(
260 {StatusCode::unsupported, support.reason}, provenance);
261 }
262
263 UndirectedGraphView data;
264 UndirectedGraphView query;
265 try {
266 data = data_graph.simple_undirected(
267 {options_.allow_directed_projection});
268 query = query_graph.simple_undirected(
269 {options_.allow_directed_projection});
270 } catch (const std::exception& error) {
271 return ExecutionResult<SubgraphIsomorphismOutput>::failure(
272 {StatusCode::invalid_argument, error.what()}, provenance);
273 }
274 const auto labels = encode_labels(data_graph, query_graph,
275 options_.respect_vertex_labels);
276
277 std::vector<std::string> warnings;
278 if (data.normalization.directed_projection_applied ||
279 query.normalization.directed_projection_applied) {
280 warnings.emplace_back("directed input was projected to undirected graphs");
281 }
282 if (data.normalization.self_loops_removed > 0 ||
283 query.normalization.self_loops_removed > 0) {
284 warnings.emplace_back("self-loops were removed during normalization");
285 }
286 if (data.normalization.parallel_edges_collapsed > 0 ||
287 query.normalization.parallel_edges_collapsed > 0) {
288 warnings.emplace_back("parallel edges were collapsed during normalization");
289 }
290
291 const auto backend = detail::run_gmatch(
292 data.csr, labels.data, query.csr, labels.query,
293 options_.initial_match_capacity, options_.execution.device_ids.front());
294 if (!backend.status.ok()) {
295 return ExecutionResult<SubgraphIsomorphismOutput>::failure(
296 backend.status, provenance, std::move(warnings));
297 }
298
299 SubgraphIsomorphismOutput output;
300 output.count = backend.count;
301 const auto materialization_started = std::chrono::steady_clock::now();
302 if (has_output(options_.optional_outputs,
303 SubgraphIsomorphismOptionalOutput::embeddings)) {
304 const auto collected = materialize_embeddings(
305 data.csr, query.csr, labels.data, labels.query, options_.result_limit);
306 const bool complete = !collected.stopped;
307 if (complete && collected.embeddings.size() != backend.count) {
308 return ExecutionResult<SubgraphIsomorphismOutput>::failure(
309 {StatusCode::correctness_mismatch,
310 "gMatch count disagreed with the exact embedding materializer"},
311 provenance, std::move(warnings));
312 }
313 output.embeddings.emplace();
314 output.embeddings->reserve(collected.embeddings.size());
315 for (const auto& internal : collected.embeddings) {
316 SubgraphEmbedding embedding;
317 embedding.reserve(internal.size());
318 for (VertexIndex query_vertex = 0;
319 query_vertex < internal.size(); ++query_vertex) {
320 embedding.push_back({query_graph.external_id(query_vertex),
321 data_graph.external_id(internal[query_vertex])});
322 }
323 output.embeddings->push_back(std::move(embedding));
324 }
325 output.embeddings_complete =
326 output.embeddings->size() == output.count;
327 }
328 const auto materialization_finished = std::chrono::steady_clock::now();
329 const auto finished = std::chrono::steady_clock::now();
330
331 ExecutionStatistics statistics;
332 statistics.backend_ms = backend.elapsed_ms;
333 statistics.output_materialization_ms =
334 std::chrono::duration<double, std::milli>(materialization_finished -
335 materialization_started)
336 .count();
337 statistics.end_to_end_ms =
338 std::chrono::duration<double, std::milli>(finished - started).count();
339 return ExecutionResult<SubgraphIsomorphismOutput>::success(
340 std::move(output), std::move(provenance), statistics,
341 std::move(warnings));
342}
